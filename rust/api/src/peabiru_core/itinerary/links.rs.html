1//! The `osm_link_ids = TRUE` columns: the OpenStreetMap ways and the
2//! street edges of each street leg, and the stops of each ride.
3//!
4//! A street leg's ids come from R5's `StreetSegment.streetEdges` (direct
5//! trips, access legs and walks between stops) or from
6//! `EgressPath.build` (the leg from the last stop): one entry per edge
7//! traversed, the egress leg's partial edge included when it has a length.
8//! `StreetSegment` merges entries first (`compactEdges`): an entry joins the
9//! previous group when both are "similar", and the group keeps its first
10//! edge. Similar means the same mode, and either both "continue" on a street
11//! of the same name, or both turn on a roundabout. The network has no
12//! street names (R5 drops its OSM database once the network is built, and
13//! `StreetLayer.getNameEdgeIdx` then returns null for every edge), so only
14//! the roundabout rule applies: an edge of a `junction=roundabout` way,
15//! after the first edge of the leg, joins a group that starts on such an
16//! edge with the same mode. Kept then are, in order and once each, the
17//! OSM way id of each group's edge and that edge's index, when the way id
18//! is positive (connectors have none).
19
20use crate::street::flags::ROUNDABOUT;
21use crate::street::p2p::Router;
22use crate::street::{Mode, StreetNetwork};

24/// The ids of one leg (empty lists and stop ids for most legs).
25#[derive(Debug, Clone, Default, PartialEq)]
26pub(crate) struct Links {
27    pub(crate) osm: Vec<i64>,
28    pub(crate) edges: Vec<u32>,
29    pub(crate) board_stop: String,
30    pub(crate) alight_stop: String,
31}
32
33/// The edges of the path ending at state `last`, with the mode each was
34/// traversed by (`StreetPath.getStates()`, states with a back edge).
35pub(crate) fn path_edges(r: &Router, last: u32) -> Vec<(u32, Mode)> {
36    r.path(last)
37        .into_iter()
38        .filter_map(|id| {
39            let s = &r.states[id as usize];
40            (s.back_edge >= 0).then_some((s.back_edge as u32, s.mode))
41        })
42        .collect()
43}
44
45/// `StreetSegment` (`compactEdges`) then `TripLeg`'s lists.
46pub(crate) fn compacted(net: &StreetNetwork, edges: &[(u32, Mode)]) -> Links {
47    let mut heads = Vec::with_capacity(edges.len());
48    // the group's first edge, its mode and whether it turns on a roundabout
49    let mut head: Option<(u32, Mode, bool)> = None;
50    for (i, &(e, mode)) in edges.iter().enumerate() {
51        let circle = i > 0 && net.edge_flags[e as usize] & ROUNDABOUT != 0;
52        if let Some((h, hm, hc)) = head {
53            if hc && circle && hm == mode {
54                continue;
55            }
56            heads.push(h);
57        }
58        head = Some((e, mode, circle));
59    }
60    if let Some((h, _, _)) = head {
61        heads.push(h);
62    }
63    lists(net, heads)
64}
65
66/// `TripLeg`'s lists from every edge given, in order.
67pub(crate) fn lists(net: &StreetNetwork, edges: impl IntoIterator<Item = u32>) -> Links {
68    let mut l = Links::default();
69    for e in edges {
70        let osm = net.pair_way[e as usize / 2];
71        if osm > 0 {
72            // LinkedHashSet.add: the first time only
73            if !l.osm.contains(&osm) {
74                l.osm.push(osm);
75            }
76            if !l.edges.contains(&e) {
77                l.edges.push(e);
78            }
79        }
80    }
81    l
82}

84/// `LinkedHashSet.toString()`: `[a, b, c]`, `[]` when empty.
85pub(crate) fn list_text<T: std::fmt::Display>(v: &[T]) -> String {
86    let mut s = String::from("[");
87    for (i, x) in v.iter().enumerate() {
88        if i > 0 {
89            s.push_str(", ");
90        }
91        s.push_str(&x.to_string());
92    }
93    s.push(']');
94    s
95}
