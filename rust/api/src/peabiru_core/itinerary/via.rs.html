1//! Itineraries whose result could depend on the order R5 has left the via
2//! edges of turn restrictions in.
3//!
4//! R5's `canTurnFrom` reverses a restriction's array of via edges in place
5//! (`TurnRestriction.reverse`) every time a reverse car search checks a
6//! restriction with two or more of them. The array belongs to the network
7//! that every search shares, so the order a check sees, in a reverse search
8//! or a forward one, depends on how many checks of that restriction came
9//! before it: in the same search, in the other pairs of the call (which R5's
10//! callers route in parallel, in an order that is not fixed) and in earlier calls
11//! of the session. In `detailed_itineraries()` this happens with
12//! `mode_egress = "CAR"`, whose egress search runs backwards; the car
13//! searches of such a call (egress, and car access and direct trips) may
14//! then see either order.
15//!
16//! peabiru plans each pair as R5 does on a network on which no reverse car
17//! search has run before it (a freshly built network, the pair alone, one
18//! thread): every array as built, reversed by each check of the pair's own
19//! reverse searches, in the order its searches run. To say when R5 could
20//! give something else, a pair whose searches meet an "ambiguous" check
21//! (one whose outcome changes with the order: the edge taken is the via
22//! edge expected in one order and not in the other) is planned again for
23//! every combination of orders its ambiguous checks can see (each one on its
24//! own, as threads can interleave anywhere); the pair "depends on the
25//! history" when some combination gives other itineraries.
26
27use crate::street::p2p::Router;
28use crate::RouteError;
29use std::collections::HashMap;
30
31/// Most plans of one pair when looking for other itineraries; a pair that
32/// needs more counts as depending on the history.
33const MAX_RUNS: usize = 256;
34
35/// The orders the restriction checks of one plan take, shared by its car
36/// searches in the order they run.
37#[derive(Default)]
38pub(crate) struct ViaOrders {
39    /// `None` with `fresh` `None`: the network as built (no reverse car
40    /// search in the call).
41    choices: Option<Vec<bool>>,
42    /// The orders of a network on which only this pair's searches ran
43    /// (see `Router::via_fresh`).
44    fresh: Option<HashMap<u64, bool>>,
45    used: Vec<bool>,
46    /// Set by the plan when the pair is to be planned again with McRAPTOR.
47    pub(crate) mcraptor_fallback: bool,
48}
49
50impl ViaOrders {
51    /// Every check sees the via edges in the order of its own search (as
52    /// built in a forward search, reversed in a reverse one): the turn
53    /// restrictions as OpenStreetMap states them, whatever ran before. For
54    /// peabiru's own functions, which do not reproduce R5's in-place
55    /// reversal.
56    pub(super) fn as_stated() -> Self {
57        ViaOrders {
58            choices: Some(Vec::new()),
59            ..ViaOrders::default()
60        }
61    }
62
63    /// Give `r` the orders of the checks still to come.
64    pub(crate) fn prepare(&self, r: &mut Router) {
65        r.via_choices = self
66            .choices
67            .as_ref()
68            .map(|c| c.get(self.used.len()..).unwrap_or(&[]).to_vec());
69        r.via_fresh = self.fresh.clone();
70    }
71
72    /// Record the orders `r`'s search took.
73    pub(crate) fn collect(&mut self, r: &mut Router) {
74        self.used.append(&mut r.via_used);
75        if let Some(f) = r.via_fresh.take() {
76            self.fresh = Some(f);
77        }
78    }
79}
80
81/// Run `plan` once with the network as built when `free` is false.
82/// Otherwise run it with the orders of a fresh network (see the module),
83/// and then for every other order its ambiguous checks can see. Returns the
84/// first result, and whether some order gave another one (or there were too
85/// many orders to try).
86pub(crate) fn settle<R: PartialEq>(
87    free: bool,
88    mut plan: impl FnMut(&mut ViaOrders) -> Result<R, RouteError>,
89) -> Result<(R, bool), RouteError> {
90    if !free {
91        return Ok((plan(&mut ViaOrders::default())?, false));
92    }
93    let mut fresh = ViaOrders {
94        fresh: Some(HashMap::new()),
95        ..ViaOrders::default()
96    };
97    let first = plan(&mut fresh)?;
98    let mut todo = children(0, &fresh.used);
99    let mut runs = 1;
100    while let Some(prefix) = todo.pop() {
101        runs += 1;
102        if runs > MAX_RUNS {
103            return Ok((first, true));
104        }
105        let n = prefix.len();
106        let mut v = ViaOrders {
107            choices: Some(prefix),
108            ..ViaOrders::default()
109        };
110        let r = plan(&mut v)?;
111        if r != first {
112            return Ok((first, true));
113        }
114        todo.extend(children(n, &v.used));
115    }
116    Ok((first, false))
117}
118
119/// The prefixes that take the other order at one of the checks after the
120/// first `from`.
121fn children(from: usize, used: &[bool]) -> Vec<Vec<bool>> {
122    (from..used.len())
123        .map(|i| {
124            let mut c = used[..i].to_vec();
125            c.push(!used[i]);
126            c
127        })
128        .collect()
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134
135    /// A fake plan: `n` ambiguous checks, which take `true` in a fresh plan
136    /// when `fresh_true`; the result counts the reversed orders when
137    /// `count`, else is constant.
138    fn fake(
139        n: usize,
140        count: bool,
141        fresh_true: bool,
142    ) -> impl FnMut(&mut ViaOrders) -> Result<usize, RouteError> {
143        move |v| {
144            v.used = match (&v.fresh, &v.choices) {
145                (Some(_), _) => vec![fresh_true; n],
146                (None, Some(ch)) => (0..n).map(|i| ch.get(i).copied().unwrap_or(true)).collect(),
147                (None, None) => vec![false; n],
148            };
149            Ok(if count {
150                v.used.iter().filter(|&&b| b).count()
151            } else {
152                7
153            })
154        }
155    }
156
157    #[test]
158    fn every_order_is_tried() {
159        let mut seen = std::collections::HashSet::new();
160        let mut f = fake(5, false, false);
161        let r = settle(true, |v| {
162            let r = f(v);
163            seen.insert(v.used.clone());
164            r
165        });
166        assert_eq!(r.ok(), Some((7, false)));
167        assert_eq!(seen.len(), 32);
168        // the fresh plan's result is returned, and the pair depends
169        assert_eq!(settle(true, fake(3, true, false)).ok(), Some((0, true)));
170        assert_eq!(settle(true, fake(3, true, true)).ok(), Some((3, true)));
171        // 512 orders > MAX_RUNS: counted as depending
172        assert_eq!(settle(true, fake(9, false, false)).ok(), Some((7, true)));
173        assert_eq!(settle(false, fake(3, true, true)).ok(), Some((0, false)));
174        assert_eq!(settle(true, fake(0, true, true)).ok(), Some((0, false)));
175    }
