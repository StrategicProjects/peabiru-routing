1//! The public transport trips of the ported `TripPlanner` with McRAPTOR: with
2//! `shortest_path = FALSE` (the alternatives, with `suboptimal_minutes`),
3//! with a fare structure, and when FastRaptor's path cannot be read back
4//! (the pair is then routed again with McRAPTOR). The search is R5's
5//! `McRaptorSuboptimalPathProfileRouter` from the stops of the access
6//! search to those of the egress search; every state at the destination of
7//! every draw becomes a trip (`Trip(state, departureTime, ...)`), one per
8//! sequence of routes (`Trip.getKey`), the fastest of each.

24/// One leg of a trip as `Trip.loadTransitLegs` reads it from the states.
25#[derive(Debug, Clone)]
26pub(super) enum RawLeg {
27    /// A walk between two stops (`newStopTransferLeg`).
28    Walk {
29        from: u32,
30        to: u32,
31        duration: i32,
32        fare: i32,
33    },
34    /// A ride (`newTransitLeg` with `setPatternData`).
35    Ride {
36        pattern: usize,
37        board_pos: usize,
38        alight_pos: usize,
39        duration: i32,
40        wait: i32,
41        fare: i32,
42    },
43}
44
45/// A trip before `augment`.
46#[derive(Debug, Clone)]
47pub(super) struct RawTrip {
48    pub(super) direct: Option<Trip>,
49    pub(super) departure: i32,
50    /// `state.time - departureTime`.
51    pub(super) duration: i32,
52    pub(super) fare: i32,
53    pub(super) legs: Vec<RawLeg>,
54}
55
56impl RawTrip {
57    pub(super) fn n_legs(&self) -> usize {
58        match &self.direct {
59            Some(t) => t.sort_legs,
60            None => self.legs.len(),
61        }
62    }
63}

65/// The fare setup of a McRAPTOR itinerary search.
66pub(crate) struct McFares<'a> {
67    pub(crate) calc: &'a FareCalculator,
68    /// `request.maxFare`, cents.
69    pub(crate) max_fare: i32,

72/// Read the legs of the chain ending at `s` (`Trip.loadTransitLegs`).
73fn raw_trip(
74    arena: &[McState],
75    s: u32,
76    departure: i32,
77    layer: &crate::TransitLayer,
78) -> Result<(RawTrip, String), String> {
79    let st = &arena[s as usize];
80    let fare = st.fare.map_or(0, |f| f.paid);
81    let mut legs = Vec::new();
82    let mut routes = Vec::new();
83    let mut k = s;
84    while k != NONE {
85        let st = &arena[k as usize];
86        if st.stop != -1 && (st.pattern != -1 || st.back != NONE) {
87            let fare = st.fare.map_or(0, |f| f.paid);
88            let back = &arena[st.back as usize];
89            if st.pattern == -1 {
90                legs.push(RawLeg::Walk {
91                    from: back.stop as u32,
92                    to: st.stop as u32,
93                    duration: st.time - back.time,
94                    fare,
95                });
96            } else {
97                let pattern = st.pattern as usize;
98                routes.push(layer.patterns[pattern].route_id.clone());
99                legs.push(RawLeg::Ride {
100                    pattern,
101                    board_pos: st.board_pos as usize,
102                    alight_pos: st.alight_pos as usize,
103                    duration: st.time - st.board_time,
104                    wait: st.board_time - back.time,
105                    fare,
106                });
107            }
108        }
109        k = st.back;
110    }
111    legs.reverse();
112    routes.reverse();
113    let key = routes
114        .into_iter()
115        .filter(|r| !r.is_empty())
116        .collect::<Vec<_>>()
117        .join("|");
118    Ok((
119        RawTrip {
120            direct: None,
121            departure,
122            duration: st.time - departure,
123            fare,
124            legs,
125        },
126        key,
127    ))
128}

130impl Planner<'_> {
131    /// `TripPlanner.plan()` with McRAPTOR for one pair: the direct trip (if
132    /// any) and the transit trips, filtered, sorted and (with
133    /// `shortest_path`) reduced to the fastest, then augmented.
134    pub(super) fn mc_plan(
135        &self,
136        o: (f64, f64),
137        d: (f64, f64),
138        direct: Option<Trip>,
139        fares: Option<&McFares>,
140        scratch: &mut Vec<u32>,
141        via: &mut ViaOrders,
142    ) -> Result<Vec<Trip>, RouteError> {
143        let (net, p, layer) = (self.net, self.p, self.layer);
144        let max_t = p.max_trip_duration * 60;
145        // trips: Map<String, Trip>, the direct trip under its mode's name
146        let mut keys: JavaHashMap<String> = JavaHashMap::new();
147        let mut values: Vec<RawTrip> = Vec::new();
148        let put =
149            |key: String, t: RawTrip, keys: &mut JavaHashMap<String>, values: &mut Vec<RawTrip>| {
150                let h = string_hash(Some(&key));
151                let i = keys.insert(key, h).map_err(RouteError::Invalid)?;
152                if i == values.len() {
153                    values.push(t);
154                } else if values[i].duration > t.duration {
155                    values[i] = t;
156                }
157                Ok::<_, RouteError>(())
158            };
159        if let Some(t) = &direct {
160            let key = p.mode.r5_name().to_string();
161            let h = string_hash(Some(&key));
162            keys.insert(key, h).map_err(RouteError::Invalid)?;
163            values.push(RawTrip {
164                direct: Some(t.clone()),
165                departure: t.departure,
166                duration: t.sort_duration,
167                fare: 0,
168                legs: Vec::new(),
169            });
170        }
171        // access and egress stop searches (findAccessPaths, findEgressPaths)
172        let mut access = Router::new(net, p.mode, p.req(), false);
173        access.stop_search(&self.vertex_stop);
174        access.time_limit = p.max_time_s(p.mode);
175        via.prepare(&mut access);
176        let access_split = find_split(net, o.0, o.1, p.mode, scratch);
177        if let Some(os) = access_split {
178            access.set_origin_split(os);
179            access.route();
180        }
181        via.collect(&mut access);
182        let mut egress = Router::new(net, p.egress_mode, p.req(), true);
183        egress.stop_search(&self.vertex_stop);
184        egress.time_limit = p.max_time_s(p.egress_mode);
185        via.prepare(&mut egress);
186        let egress_split: Option<Split> = find_split(net, d.0, d.1, p.egress_mode, scratch);
187        let mut egress_ok = false;
188        if let Some(ds) = egress_split {
189            if egress.set_egress_origin(ds) {
190                egress.route();
191                via.collect(&mut egress);
192                egress_ok = true;
193                if let Some(why) = egress.unsupported {
194                    return Err(RouteError::Invalid(format!("not supported: {why}")));
195                }
196            }
197        }
198        let access_times: Vec<(u32, i32)> = if access_split.is_some() {
199            access
200                .stops_order
201                .iter()
202                .map(|s| (*s, access.reached_stops()[s]))
203                .collect()
204        } else {
205            Vec::new()
206        };
207        // the egress stops in the iteration order of R5's TIntIntHashMap
208        let egress_times: Vec<(u32, i32)> = if egress_ok {
209            let mut t = TroveIntKeys::new();
210            for &s in &egress.stops_order {
211                t.insert(s as i32);
212            }
213            t.iteration_order()
214                .map(|s| (s as u32, egress.reached_stops()[&(s as u32)]))
215                .collect()
216        } else {
217            Vec::new()
218        };
219        let max_fare = fares.map_or(-1, |f| f.max_fare);
220        let req = McRequest {
221            from_time: p.departure_time,
222            to_time: p.departure_time + p.time_window * 60,
223            max_trip_s: max_t,
224            max_rides: p.max_rides,
225            suboptimal_minutes: p.suboptimal_minutes,
226            max_fare,
227            monte_carlo_draws: p.time_window,
228            walk_mms: f64::from(p.req().walk * 1000.0f32),
229            seed: None,
230            origin: o,
231        };
232        let dom = match fares {
233            Some(f) => Dominance::Fare(f.calc),
234            None => {
235                let mut max_duration = max_t;
236                if p.shortest_path {
237                    if let Some(t) = &direct {
238                        max_duration = max_duration.min(t.sort_duration);
239                    }
240                }
241                Dominance::Suboptimal {
242                    max_duration,
243                    shortest: p.shortest_path,
244                }
245            }
246        };
247        let allowed = &self.mc_allowed;
248        let active = &self.mc_active;
249        let schedules = self
250            .mc_schedules
251            .get_or_init(|| crate::transit::mcraptor::Schedules::new(layer, allowed, active));
252        let mut router = McRaptor::new(
253            layer,
254            &self.data.transfers,
255            allowed,
256            active,
257            schedules,
258            &req,
259            dom,
260            Some(&egress_times),
261            McRaptor::departure_seed(None, o),
262        );
263        // finalStatesByDepartureTime: per draw, its trips and keys
264        let mut by_departure: Vec<(i32, Vec<(RawTrip, String)>)> = Vec::new();
265        let mut failure = None;
266        router
267            .route(&access_times, &p.cancel, |dep, out| {
268                if let DrawOut::Final(arena, states) = out {
269                    let mut v = Vec::with_capacity(states.len());
270                    for &s in states {
271                        match raw_trip(arena, s, dep, layer) {
272                            Ok(t) => v.push(t),
273                            Err(e) => failure = Some(e),
274                        }
275                    }
276                    by_departure.push((dep, v));
277                }
278            })
279            .map_err(RouteError::Invalid)?;
280        p.cancel.check()?;
281        if let Some(e) = failure {
282            return Err(RouteError::Invalid(e));
283        }
284        let mut order = TroveIntKeys::new();
285        for (dep, _) in &by_departure {
286            order.insert(*dep);
287        }
288        let mut by_dep: std::collections::HashMap<i32, Vec<(RawTrip, String)>> =
289            by_departure.into_iter().collect();
290        for dep in order.iteration_order().collect::<Vec<_>>() {
291            for (t, key) in by_dep.remove(&dep).unwrap_or_default() {
292                put(key, t, &mut keys, &mut values)?;
293            }
294        }
295        // values in the map's order, filtered, sorted (stable)
296        let mut list: Vec<RawTrip> = keys
297            .order()
298            .map(|i| values[i].clone())
299            .filter(|t| t.duration <= max_t && t.fare <= fares.map_or(i32::MAX, |f| f.max_fare))
300            .collect();
301        list.sort_by_key(|t| (i32::from(t.direct.is_none()), t.n_legs(), t.duration));
302        if p.shortest_path {
303            if let Some(best) = list.iter().map(|t| t.duration).min() {
304                let k = list.iter().position(|t| t.duration == best).expect("min");
305                list = vec![list.swap_remove(k)];
306            }
307        }
308        let mut out = Vec::with_capacity(list.len());
309        for t in list {
310            if let Some(mut d) = t.direct {
311                if fares.is_some() {
312                    d.fares = Some((0, vec![0; d.legs.len()]));
313                }
314                out.push(d);
315                continue;
316            }
317            if let Some(t) = self.augment(
318                t,
319                &access,
320                &egress,
321                egress_split.as_ref(),
322                d,
323                fares.is_some(),
324                scratch,
325            )? {
326                out.push(t);
327            }
328        }
329        Ok(out)
330    }

332    /// `Trip.augment` of a McRAPTOR trip: the rides drawn, the walks between
333    /// stops routed, the access and egress legs added. `None` when the trip is
334    /// skipped (`EgressPath.InvalidOriginException`).
335    #[allow(clippy::too_many_arguments)]
336    pub(super) fn augment(
337        &self,
338        t: RawTrip,
339        access: &Router,
340        egress: &Router,
341        egress_split: Option<&Split>,
342        d: (f64, f64),
343        with_fares: bool,
344        scratch: &mut Vec<u32>,
345    ) -> Result<Option<Trip>, RouteError> {
346        let (net, layer, p) = (self.net, self.layer, self.p);
347        let n_legs = t.legs.len();
348        let mut out: Vec<Leg> = Vec::with_capacity(n_legs + 2);
349        let mut fares: Vec<i32> = Vec::with_capacity(n_legs + 2);
350        let mut first_board = None;
351        let mut last_alight = None;
352        for l in &t.legs {
353            match *l {
354                RawLeg::Walk {
355                    from,
356                    to,
357                    duration,
358                    fare,
359                } => {
360                    let mut leg = self.stop_transfer_leg(from, to, duration);
361                    self.route_transfer(&mut leg, from, to, scratch);
362                    out.push(leg);
363                    fares.push(fare);
364                    if first_board.is_none() {
365                        first_board = Some(from);
366                    }
367                    last_alight = Some(to);
368                }
369                RawLeg::Ride {
370                    pattern,
371                    board_pos,
372                    alight_pos,
373                    duration,
374                    wait,
375                    fare,
376                } => {
377                    let pat = &layer.patterns[pattern];
378                    let mode = match route_type_mode(pat.route_type) {
379                        Some(Some(m)) => m.name().to_string(),
380                        Some(None) => "AIR".to_string(),
381                        None => {
382                            return Err(RouteError::Invalid(format!(
383                                "unsupported GTFS route_type {}",
384                                pat.route_type
385                            )))
386                        }
387                    };
388                    let line = ride_line(net, layer, pattern, board_pos, alight_pos)
389                        .map_err(|why| RouteError::Invalid(format!("not supported: {why}")))?;
390                    let distance = line_length_m(&line);
391                    let mut links = Links::default();
392                    if p.osm_link_ids {
393                        links.board_stop = layer.stop_ids[pat.stops[board_pos] as usize].clone();
394                        links.alight_stop = layer.stop_ids[pat.stops[alight_pos] as usize].clone();
395                    }
396                    out.push(Leg {
397                        mode,
398                        duration,
399                        wait,
400                        route: pat.route_id.clone(),
401                        line,
402                        distance,
403                        links,
404                    });
405                    fares.push(fare);
406                    if first_board.is_none() {
407                        first_board = Some(pat.stops[board_pos]);
408                    }
409                    last_alight = Some(pat.stops[alight_pos]);
410                }
411            }
412        }
413        let last_fare = fares.last().copied().unwrap_or(0);
414        if let Some(first) = first_board {
415            let v = layer.stop_vertex[first as usize];
416            if let Some(st) = (v >= 0).then(|| access.state_at_vertex(v as u32)).flatten() {
417                let line = street_path_line(access, st);
418                let distance = line_length_m(&line);
419                let links = if p.osm_link_ids {
420                    links::compacted(net, &links::path_edges(access, st))
421                } else {
422                    Links::default()
423                };
424                out.insert(
425                    0,
426                    Leg {
427                        mode: p.mode.r5_name().to_string(),
428                        duration: access.states[st as usize].dur,
429                        wait: 0,
430                        route: String::new(),
431                        line,
432                        distance,
433                        links,
434                    },
435                );
436                fares.insert(0, 0);
437            }
438        }
439        if let (Some(last), Some(ds)) = (last_alight, egress_split) {
440            let v = layer.stop_vertex[last as usize];
441            if let Some(st) = (v >= 0).then(|| egress.state_at_vertex(v as u32)).flatten() {
442                match egress_line(egress, st, ds, d) {
443                    Some((line, edges)) => {
444                        let distance = line_length_m(&line);
445                        let links = if p.osm_link_ids {
446                            links::lists(net, edges)
447                        } else {
448                            Links::default()
449                        };
450                        out.push(Leg {
451                            mode: p.egress_mode.r5_name().to_string(),
452                            duration: egress.states[st as usize].dur,
453                            wait: 0,
454                            route: String::new(),
455                            line,
456                            distance,
457                            links,
458                        });
459                        fares.push(last_fare);
460                    }
461                    None => return Ok(None),
462                }
463            }
464        }
465        let duration = out.iter().map(|l| l.duration + l.wait).sum();
466        Ok(Some(Trip {
467            direct: false,
468            departure: t.departure,
469            sort_duration: t.duration,
470            sort_legs: n_legs,
471            duration,
472            legs: out,
473            fares: with_fares.then_some((t.fare, fares)),
474        }))
475    }
