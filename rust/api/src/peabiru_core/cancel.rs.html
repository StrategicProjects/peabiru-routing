1//! Cancelling a long computation from another thread.
2
3use std::sync::atomic::{AtomicBool, Ordering};
4use std::sync::Arc;
5
6/// A flag that stops a running computation. Every routing parameter
7/// struct carries one ([`RouteParams::cancel`](crate::RouteParams::cancel),
8/// [`TransitParams::cancel`](crate::TransitParams::cancel)); keep
9/// a clone, hand the parameters to the call, and call
10/// [`cancel`](CancelToken::cancel) from any thread. The workers notice
11/// between origins (between batches of eight in a street matrix computed
12/// by contraction hierarchy, and between the rounds of the contraction)
13/// and, from a transit origin, at every departure minute and Monte Carlo
14/// draw and between destinations; the call then returns
15/// [`RouteError::Cancelled`] instead of a partial result. The check is one
16/// relaxed atomic load.
17///
18/// Clones share the flag. Once cancelled, a token stays cancelled: every
19/// later call made with it returns `Cancelled` without routing a single
20/// origin, so make a new one (the default) for the next call. The setup
21/// before the first origin (validating the parameters, linking the
22/// destinations to the streets) is not interrupted.
23#[derive(Debug, Clone, Default)]
24pub struct CancelToken(Arc<AtomicBool>);
25
26impl CancelToken {
27    /// A token that has not been cancelled.
28    pub fn new() -> Self {
29        Self::default()
30    }
31
32    /// Ask the computations that hold this token to stop.
33    pub fn cancel(&self) {
34        self.0.store(true, Ordering::Relaxed);
35    }
36
37    /// Whether [`cancel`](CancelToken::cancel) was called on this token or
38    /// a clone of it.
39    #[inline]
40    pub fn is_cancelled(&self) -> bool {
41        self.0.load(Ordering::Relaxed)
42    }
43
44    /// `Err(Cancelled)` once cancelled.
45    pub(crate) fn check(&self) -> Result<(), RouteError> {
46        if self.is_cancelled() {
47            Err(RouteError::Cancelled)
48        } else {
49            Ok(())
50        }
51    }
52}
53
54/// Why a routing computation returned no result.
55#[derive(Debug, Clone, PartialEq, Eq)]
56#[non_exhaustive]
57pub enum RouteError {
58    /// The parameters or the inputs were rejected; the message says why.
59    Invalid(String),
60    /// The call's [`CancelToken`] was cancelled before it finished.
61    Cancelled,
62}
63
64impl std::fmt::Display for RouteError {
65    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
66        match self {
67            RouteError::Invalid(msg) => f.write_str(msg),
68            RouteError::Cancelled => f.write_str("the computation was cancelled"),
69        }
70    }
71}
72
73impl std::error::Error for RouteError {}

75#[cfg(test)]
76mod tests {
77    use super::*;
78
79    #[test]
80    fn clones_share_the_flag() {
81        let a = CancelToken::new();
82        let b = a.clone();
83        assert!(!b.is_cancelled());
84        assert_eq!(b.check(), Ok(()));
85        a.cancel();
86        assert!(b.is_cancelled());
87        assert_eq!(b.check(), Err(RouteError::Cancelled));
88        assert!(!CancelToken::default().is_cancelled());
89    }
90}
