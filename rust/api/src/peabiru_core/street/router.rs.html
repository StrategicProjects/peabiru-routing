114/// Check a speed in km/h as R5 does: finite, and at least 1 mm/s once
115/// converted to `(int)(m/s as float * 1000)`, which is a divisor
116/// (`TravelTimeComputer`: "Speed of access mode must be greater than 0").
117pub(crate) fn check_speed(name: &str, v: f64) -> Result<(), String> {
118    if !v.is_finite() || ((v * 5.0 / 18.0) as f32 * 1000.0f32) < 1.0 {
119        return Err(format!(
120            "{name} must be at least 0.0036 km/h (1 mm/s), got {}",
121            compact(v)
122        ));
123    }
124    Ok(())
125}

127impl RouteParams {
128    /// Check the parameters: speeds must give at least 1 mm/s,
129    /// `max_lts` must be 1 to 4, the trip limit must
130    /// be a positive number of minutes that fits 32-bit seconds. Every
131    /// street routing function calls it first; [`RouteError::Invalid`]
132    /// says what is wrong.
133    pub fn validate(&self) -> Result<(), RouteError> {
134        check_speed("walk_speed", self.walk_speed).map_err(RouteError::Invalid)?;
135        check_speed("bike_speed", self.bike_speed).map_err(RouteError::Invalid)?;
136        // 1 to 4 (R5 treats anything else as 4)
137        if !(1..=4).contains(&self.max_lts) {
138            return Err(RouteError::Invalid(format!(
139                "max_lts must be from 1 to 4 (the highest level of traffic stress), got {}",
140                self.max_lts
141            )));
142        }
143        let max_minutes = i32::MAX / 60;
144        if !(1..=max_minutes).contains(&self.max_trip_duration) {
145            return Err(RouteError::Invalid(format!(
146                "max_trip_duration must be from 1 to {max_minutes} minutes, got {}",
147                self.max_trip_duration
148            )));
149        }
150        self.scenario.check().map_err(RouteError::Invalid)
151    }
152}

169/// Worker threads for `n_items` origins when `requested` were asked for
170/// (0 = all cores): never more than the cores available, since each worker
171/// holds its own search state and more threads than cores only add memory,
172/// and never more than the origins. Results never depend on it: origins
173/// are handed out one at a time and stored by index.
174pub(crate) fn worker_threads(requested: usize, n_items: usize) -> usize {
175    let cores = crate::sys::cores();
176    let n = if requested == 0 {
177        cores
178    } else {
179        requested.min(cores)
180    };
181    n.clamp(1, n_items.max(1))
182}

213/// Binary min-heap on time with exactly the sift rules of
214/// `java.util.PriorityQueue`, so states with equal times leave the queue in
215/// the same order as in R5. That order matters: among states carrying the
216/// same turn-restriction progress on an edge, R5 keeps the first one created.
217#[derive(Default)]
218struct JavaHeap {
219    a: Vec<(i32, u32, u32)>,
220}
221
222impl JavaHeap {
223    fn clear(&mut self) {
224        self.a.clear();
225    }
226
227    #[inline]
228    fn offer(&mut self, x: (i32, u32, u32)) {
229        let mut k = self.a.len();
230        self.a.push(x);
231        while k > 0 {
232            let parent = (k - 1) >> 1;
233            let e = self.a[parent];
234            if x.0 >= e.0 {
235                break;
236            }
237            self.a[k] = e;
238            k = parent;
239        }
240        self.a[k] = x;
241    }
242
243    #[inline]
244    fn poll(&mut self) -> Option<(i32, u32, u32)> {
245        let last = self.a.pop()?;
246        if self.a.is_empty() {
247            return Some(last);
248        }
249        let result = self.a[0];
250        let n = self.a.len();
251        let half = n >> 1;
252        let mut k = 0;
253        while k < half {
254            let mut child = 2 * k + 1;
255            let mut c = self.a[child];
256            let right = child + 1;
257            if right < n && c.0 > self.a[right].0 {
258                child = right;
259                c = self.a[child];
260            }
261            if last.0 <= c.0 {
262                break;
263            }
264            self.a[k] = c;
265            k = child;
266        }
267        self.a[k] = last;
268        Some(result)
269    }
270}

272/// Monotone integer priority queue (a radix heap) for searches whose
273/// states carry no turn-restriction progress.
274///
275/// Such a search keeps, per directed edge, the least time of any state
276/// reaching it, and every step adds a non-negative cost: the times it ends
277/// with are shortest-path times, whatever order equal-time states leave the
278/// queue in. Only the restricted car search depends on that order (see
279/// [`JavaHeap`]), so only it needs the exact `java.util.PriorityQueue`
280/// replica; every other search uses this cheaper queue and finds the same
281/// times.
282///
283/// Keys must not decrease below the last one popped (true for a search
284/// with non-negative steps). Keys are `i32` mapped to `u32` in order.
285pub(crate) struct RadixHeap {
286    /// Image of the last popped key; bucket `i > 0` holds keys whose highest
287    /// bit differing from it is bit `i - 1`, bucket 0 keys equal to it.
288    last: u32,
289    len: usize,
290    buckets: [Vec<(u32, u32)>; 33],
291}
292
293impl Default for RadixHeap {
294    fn default() -> Self {
295        RadixHeap {
296            last: 0,
297            len: 0,
298            buckets: std::array::from_fn(|_| Vec::new()),
299        }
300    }
301}
302
303impl RadixHeap {
304    #[inline]
305    fn key(t: i32) -> u32 {
306        (t as u32) ^ 0x8000_0000
307    }
308
309    #[inline]
310    fn bucket(&self, k: u32) -> usize {
311        (32 - (k ^ self.last).leading_zeros()) as usize
312    }
313
314    pub(crate) fn clear(&mut self) {
315        for b in &mut self.buckets {
316            b.clear();
317        }
318        self.last = 0;
319        self.len = 0;
320    }
321
322    #[inline]
323    pub(crate) fn push(&mut self, t: i32, e: u32) {
324        // A key below the last popped one cannot come from a search with
325        // non-negative steps; clamping keeps the queue consistent anyway.
326        let k = Self::key(t).max(self.last);
327        let b = self.bucket(k);
328        self.buckets[b].push((k, e));
329        self.len += 1;
330    }
331
332    #[inline]
333    pub(crate) fn pop(&mut self) -> Option<(i32, u32)> {
334        if self.len == 0 {
335            return None;
336        }
337        if self.buckets[0].is_empty() {
338            let i = (1..33)
339                .find(|&i| !self.buckets[i].is_empty())
340                .expect("a non-empty bucket");
341            let mut moved = std::mem::take(&mut self.buckets[i]);
342            self.last = moved.iter().map(|&(k, _)| k).min().expect("non-empty");
343            for &(k, e) in &moved {
344                let b = self.bucket(k);
345                self.buckets[b].push((k, e));
346            }
347            moved.clear();
348            self.buckets[i] = moved; // keep the allocation
349        }
350        let (k, e) = self.buckets[0].pop().expect("bucket 0 is non-empty");
351        self.len -= 1;
352        Some(((k ^ 0x8000_0000) as i32, e))
353    }
354}

376/// The street graph as one mode's search reads it: every vertex's outgoing
377/// edges with their traversal times precomputed and laid out contiguously
378/// (`arcs[out_offsets[v]..out_offsets[v + 1]]`, in `out_edges` order), so
379/// the inner loop of the search reads one array. Built once per query.
380pub(crate) struct ModeGraph {
381    pub(super) mode: Mode,
382    pub(super) sp: Speeds,
383    arcs: Vec<Arc>,
384    /// The search must pop equal times in `java.util.PriorityQueue` order:
385    /// car searches on a network with turn restrictions.
386    pub(super) java_order: bool,
387}
388
389impl ModeGraph {
390    pub(super) fn new(net: &StreetNetwork, mode: Mode, sp: Speeds, max_lts: i32) -> Self {
391        // slope costs on walking and cycling edges (R5 applies none to cars)
392        let sloped = |f: usize, t: i32| -> i32 {
393            match net.edge_slope.get(f) {
394                Some(&k) => crate::elevation::sloped_seconds(t, k),
395                None => t,
396            }
397        };
398        let step_of = |f: usize| -> i32 {
399            let flags = net.edge_flags[f];
400            let len = net.pair_length_mm[f / 2];
401            match mode {
402                Mode::Walk => {
403                    if flags & ALLOWS_PEDESTRIAN == 0 {
404                        return -1;
405                    }
406                    sloped(f, traversal_seconds(len, sp.walk))
407                }
408                Mode::Bicycle => {
409                    let mut walk_bike = flags & ALLOWS_BIKE == 0;
410                    if max_lts > 0
411                        && max_lts < 4
412                        && (flags & BIKE_LTS_4 != 0
413                            || (max_lts < 3 && flags & BIKE_LTS_3 != 0)
414                            || (max_lts < 2 && flags & BIKE_LTS_2 != 0))
415                    {
416                        walk_bike = true;
417                    }
418                    if walk_bike {
419                        if flags & ALLOWS_PEDESTRIAN == 0 {
420                            return -1;
421                        }
422                        sloped(f, traversal_seconds(len, sp.walk))
423                    } else {
424                        sloped(f, traversal_seconds(len, sp.bike))
425                    }
426                }
427                Mode::Car => {
428                    if flags & ALLOWS_CAR == 0 {
429                        return -1;
430                    }
431                    traversal_seconds(len, cms_to_ms(net.edge_speed_cms[f]))
432                }
433            }
434        };
435        let arcs = net
436            .out_edges
437            .iter()
438            .map(|&f| {
439                let fu = f as usize;
440                Arc {
441                    edge: f,
442                    to: net.edge_to(fu) as u32,
443                    // a step never comes out negative for a usable edge
444                    // (lengths >= 0, speeds > 0), so -1 is free as a marker
445                    step: step_of(fu).max(-1),
446                    in_deg: net.in_angle(fu) as i16,
447                    link: net.edge_flags[fu] & LINK != 0,
448                }
449            })
450            .collect();
451        ModeGraph {
452            mode,
453            sp,
454            arcs,
455            java_order: mode == Mode::Car && !net.restriction_starts.is_empty(),
456        }
457    }
458
459    #[inline]
460    pub(super) fn arcs(&self, net: &StreetNetwork, v: usize) -> &[Arc] {
461        &self.arcs[net.out_offsets[v] as usize..net.out_offsets[v + 1] as usize]
462    }
463}

465/// Reusable per-thread search state.
466pub(crate) struct Search {
467    /// Best time at the end of each directed edge, over states without
468    /// restriction progress.
469    edge_time: Vec<i32>,
470    /// Best time at each vertex (min over all incoming edge states).
471    pub(super) vertex_time: Vec<i32>,
472    touched_edges: Vec<u32>,
473    touched_vertices: Vec<u32>,
474    /// States in the middle of turn restrictions: `(edge, progress)`. R5
475    /// keeps them apart: they neither dominate nor are dominated by plain
476    /// states, and the first state with a given progress on an edge wins.
477    restricted: Vec<(u32, RMap)>,
478    /// Parallel to `restricted`: the state's time, and whether it was
479    /// made on foot by [`continue_on_foot`] (R5's `State.streetMode` is
480    /// then `WALK`, and its restriction progress is never carried on).
481    restricted_time: Vec<i32>,
482    restricted_walk: Vec<bool>,
483    restricted_at: std::collections::HashMap<u32, Vec<u32>>,
484    /// Queue of `(time, edge, restricted state id or CLEAN)` for searches
485    /// that need R5's exact order of equal times.
486    heap: JavaHeap,
487    /// Queue of `(time, edge)` for the others.
488    radix: RadixHeap,
489}
490
491impl Search {
492    pub(crate) fn new(net: &StreetNetwork) -> Self {
493        Search {
494            edge_time: vec![i32::MAX; net.n_edges()],
495            vertex_time: vec![i32::MAX; net.n_vertices()],
496            touched_edges: Vec::new(),
497            touched_vertices: Vec::new(),
498            restricted: Vec::new(),
499            restricted_time: Vec::new(),
500            restricted_walk: Vec::new(),
501            restricted_at: std::collections::HashMap::new(),
502            heap: JavaHeap::default(),
503            radix: RadixHeap::default(),
504        }
505    }
506
507    /// Best time at each vertex after the last search (`i32::MAX` where
508    /// unreached).
509    pub(crate) fn vertex_time(&self) -> &[i32] {
510        &self.vertex_time
511    }
512
513    fn reset(&mut self) {
514        for &e in &self.touched_edges {
515            self.edge_time[e as usize] = i32::MAX;
516        }
517        for &v in &self.touched_vertices {
518            self.vertex_time[v as usize] = i32::MAX;
519        }
520        self.touched_edges.clear();
521        self.touched_vertices.clear();
522        self.restricted.clear();
523        self.restricted_time.clear();
524        self.restricted_walk.clear();
525        self.restricted_at.clear();
526        self.heap.clear();
527        self.radix.clear();
528    }
529
530    #[inline]
531    fn push<const JAVA: bool>(&mut self, t: i32, e: u32, tag: u32) {
532        if JAVA {
533            self.heap.offer((t, e, tag));
534        } else {
535            debug_assert_eq!(tag, CLEAN);
536            self.radix.push(t, e);
537        }
538    }
539
540    #[inline]
541    fn pop<const JAVA: bool>(&mut self) -> Option<(i32, u32, u32)> {
542        if JAVA {
543            self.heap.poll()
544        } else {
545            self.radix.pop().map(|(t, e)| (t, e, CLEAN))
546        }
547    }
548
549    #[inline]
550    fn touch_vertex(&mut self, v: usize, t: i32) {
551        if t < self.vertex_time[v] {
552            if self.vertex_time[v] == i32::MAX {
553                self.touched_vertices.push(v as u32);
554            }
555            self.vertex_time[v] = t;
556        }
557    }
558
559    /// [`Search::add_state`] for a state without restriction progress:
560    /// whether it was kept.
561    #[inline(always)]
562    fn add_plain(&mut self, e: usize, to: usize, t: i32) -> bool {
563        let cur = self.edge_time[e];
564        if t >= cur {
565            return false; // ties keep the existing state
566        }
567        if cur == i32::MAX {
568            self.touched_edges.push(e as u32);
569        }
570        self.edge_time[e] = t;
571        self.touch_vertex(to, t);
572        true
573    }
574
575    /// Record a state at the end of edge `e` (whose end vertex is `to`);
576    /// returns its heap tag, or `None` when an existing state dominates it
577    /// (R5 `isDominated`).
578    #[inline]
579    fn add_state(&mut self, e: usize, to: usize, t: i32, map: RMap, walk: bool) -> Option<u32> {
580        if map.is_empty() {
581            return self.add_plain(e, to, t).then_some(CLEAN);
582        }
583        let ids = self.restricted_at.entry(e as u32).or_default();
584        if ids.iter().any(|&id| self.restricted[id as usize].1 == map) {
585            return None;
586        }
587        let id = self.restricted.len() as u32;
588        ids.push(id);
589        self.restricted.push((e as u32, map));
590        self.restricted_time.push(t);
591        self.restricted_walk.push(walk);
592        self.touch_vertex(to, t);
593        Some(id)
594    }
595}

597/// Carry restriction progress from the current edge onto edge `f`
598/// (R5 `canTurnFrom`); `None` when the turn is forbidden.
599fn advance_restrictions(net: &StreetNetwork, map: &RMap, f: u32) -> Option<RMap> {
600    let mut out = Vec::with_capacity(map.len());
601    for &(r, pos) in map {
602        let rest = &net.restrictions[r as usize];
603        let p = (pos - 1) as usize;
604        if p < rest.via_edges.len() {
605            if f != rest.via_edges[p] {
606                if rest.only {
607                    return None;
608                }
609                // left a "no" restriction: forget it
610            } else {
611                out.push((r, pos + 1));
612            }
613        } else if f != rest.to_edge {
614            if rest.only {
615                return None;
616            }
617        } else if !rest.only {
618            return None;
619        }
620        // reached the end of an "only" restriction: done with it
621    }
622    Some(out)
623}

625/// Start the restrictions whose from edge is `f` (R5 `startTurnRestriction`).
626#[inline]
627fn start_restrictions(net: &StreetNetwork, map: &mut RMap, f: u32) {
628    if net.edge_flags[f as usize] & RESTRICTION_START == 0 {
629        return;
630    }
631    for r in net.restrictions_starting_at(f) {
632        match map.binary_search_by_key(&r, |&(k, _)| k) {
633            Ok(i) => map[i].1 = 1,
634            Err(i) => map.insert(i, (r, 1)),
635        }
636    }
637}

639/// Turn cost for a car leaving an edge at bearing `out` (whole degrees) onto
640/// one entered at bearing `inn` (R5 `turnTimeSeconds`).
641#[inline]
642pub(super) fn car_turn_cost(out: i32, inn: i32) -> i32 {
643    let mut out = out;
644    if out < inn {
645        out += 360;
646    }
647    let a = out - inn;
648    if a < 27 {
649        STRAIGHT_ON
650    } else if a < 153 {
651        LEFT_TURN
652    } else if a < 207 {
653        U_TURN
654    } else if a < 333 {
655        RIGHT_TURN
656    } else {
657        STRAIGHT_ON
658    }
659}

682/// The two start states of a search from an origin split, as `(edge,
683/// seconds)`: the backward edge, which ends at the split's first vertex,
684/// then the forward edge. They carry the split edge as back edge so turn
685/// costs (and turn restrictions) apply when leaving it, and their times are
686/// not checked against the limit.
687pub(super) fn start_states(
688    net: &StreetNetwork,
689    g: &ModeGraph,
690    origin: &Split,
691) -> [(usize, i32); 2] {
692    let (mode, sp) = (g.mode, &g.sp);
693    let fwd = origin.edge as usize;
694    let bwd = fwd + 1;
695    let off = origin.distance_to_edge_mm / sp.walk_mms;
696    let start_speed = |e: usize| match mode {
697        Mode::Walk => sp.walk,
698        Mode::Bicycle => sp.bike,
699        Mode::Car => cms_to_ms(net.edge_speed_cms[e]),
700    };
701    let t1 = ((origin.distance1_mm / 1000) as f32 / start_speed(fwd)) as i32 + off;
702    let t0 = ((origin.distance0_mm / 1000) as f32 / start_speed(bwd)) as i32 + off;
703    [(bwd, t0), (fwd, t1)]
704}

706/// [`route`] with the queue chosen at compile time: `JAVA` for the car
707/// search with turn restrictions (states may carry restriction progress),
708/// otherwise every state is plain.
709fn route_impl<const JAVA: bool>(
710    net: &StreetNetwork,
711    g: &ModeGraph,
712    origin: &Split,
713    limit: i32,
714    s: &mut Search,
715) {
716    s.reset();
717    let mode = g.mode;
718    for (e, t) in start_states(net, g, origin) {
719        let mut map = RMap::new();
720        if JAVA {
721            start_restrictions(net, &mut map, e as u32);
722        }
723        if let Some(tag) = s.add_state(e, net.edge_to(e), t, map, false) {
724            s.push::<JAVA>(t, e as u32, tag);
725        }
726    }
727    let car = mode == Mode::Car;
728    let mut map = RMap::new();
729    while let Some((t, e, tag)) = s.pop::<JAVA>() {
730        let e = e as usize;
731        map.clear();
732        if tag == CLEAN {
733            if t > s.edge_time[e] {
734                continue; // dominated since it was queued
735            }
736        } else {
737            map.extend_from_slice(&s.restricted[tag as usize].1);
738        }
739        let e_link = net.edge_flags[e] & LINK != 0;
740        let out_deg = if car { net.out_angle(e) } else { 0 };
741        for a in g.arcs(net, net.edge_to(e)) {
742            if a.step < 0 {
743                continue; // the mode cannot use this edge
744            }
745            if a.link && e_link {
746                continue; // two connectors in a row would be a shortcut
747            }
748            // Only the restricted car search carries restriction progress.
749            let next_map = if JAVA {
750                let mut m = if map.is_empty() {
751                    RMap::new()
752                } else {
753                    match advance_restrictions(net, &map, a.edge) {
754                        Some(m) => m,
755                        None => continue,
756                    }
757                };
758                start_restrictions(net, &mut m, a.edge);
759                m
760            } else {
761                RMap::new()
762            };
763            let step = if car {
764                a.step + car_turn_cost(out_deg, i32::from(a.in_deg))
765            } else {
766                a.step
767            };
768            let mut t1 = t.saturating_add(step);
769            if t1 == t {
770                t1 += 1;
771            }
772            if t1 < limit {
773                if !JAVA {
774                    if s.add_plain(a.edge as usize, a.to as usize, t1) {
775                        s.push::<JAVA>(t1, a.edge, CLEAN);
776                    }
777                } else if let Some(tag) =
778                    s.add_state(a.edge as usize, a.to as usize, t1, next_map, false)
779                {
780                    s.push::<JAVA>(t1, a.edge, tag);
781                }
782            }
783        }
784    }
785}

787/// Heap tag of a plain state made on foot by [`continue_on_foot`] (plain
788/// states of the first search are [`CLEAN`]).
789const CLEAN_WALK: u32 = u32::MAX - 1;
790
791/// Carry a finished bicycle or car search on on foot, as R5's
792/// `StreetRouter.keepRoutingOnFoot` does before reading the stops reached
793/// for public transport access: every state the search kept goes back into
794/// the queue, and the search goes on by `foot` (the walking graph) within
795/// the same `limit`, adding states to the same tables. The vertex times
796/// after it are the least over states by the first mode and on foot.
797///
798/// R5 keeps the mode of each state: a step on foot out of a car state still
799/// checks that car state's turn restrictions and starts those of the edge
800/// it walks onto (`canTurnFrom` and `startTurnRestriction` look at the
801/// mode of the state they leave), so such a walking state can carry
802/// restriction progress, which makes it incomparable to plain states and
803/// beaten by an earlier state with the same progress on its edge. A step
804/// out of a walking state carries none. Equal times may leave the queue in
805/// any order here: the states on one edge with the same progress are made
806/// by steps of the same length out of states that leave the queue in order
807/// of time, so the first one made is also the earliest.
808pub(crate) fn continue_on_foot(
809    net: &StreetNetwork,
810    first: &ModeGraph,
811    foot: &ModeGraph,
812    limit: i32,
813    s: &mut Search,
814) {
815    debug_assert_eq!(foot.mode, Mode::Walk);
816    if !first.java_order {
817        // no state of the first search is a car state on a network with
818        // turn restrictions
819        continue_on_foot_plain(net, foot, limit, s);
820        return;
821    }
822    s.heap.clear();
823    for i in 0..s.touched_edges.len() {
824        let e = s.touched_edges[i];
825        let t = s.edge_time[e as usize];
826        if t < i32::MAX {
827            s.heap.offer((t, e, CLEAN));
828        }
829    }
830    for id in 0..s.restricted.len() {
831        s.heap
832            .offer((s.restricted_time[id], s.restricted[id].0, id as u32));
833    }
834    let mut map = RMap::new();
835    while let Some((t, e, tag)) = s.heap.poll() {
836        let e = e as usize;
837        map.clear();
838        // whether the state left is a car state, whose restrictions apply
839        let by_car = match tag {
840            CLEAN | CLEAN_WALK => {
841                if t > s.edge_time[e] {
842                    continue; // dominated since it was queued
843                }
844                tag == CLEAN
845            }
846            id => {
847                let id = id as usize;
848                if s.restricted_walk[id] {
849                    false
850                } else {
851                    map.extend_from_slice(&s.restricted[id].1);
852                    true
853                }
854            }
855        };
856        let e_link = net.edge_flags[e] & LINK != 0;
857        for a in foot.arcs(net, net.edge_to(e)) {
858            if a.step < 0 || (a.link && e_link) {
859                continue;
860            }
861            let next_map = if by_car {
862                let mut m = if map.is_empty() {
863                    RMap::new()
864                } else {
865                    match advance_restrictions(net, &map, a.edge) {
866                        Some(m) => m,
867                        None => continue,
868                    }
869                };
870                start_restrictions(net, &mut m, a.edge);
871                m
872            } else {
873                RMap::new()
874            };
875            let mut t1 = t.saturating_add(a.step);
876            if t1 == t {
877                t1 += 1;
878            }
879            if t1 < limit {
880                if let Some(tag) = s.add_state(a.edge as usize, a.to as usize, t1, next_map, true) {
881                    let tag = if tag == CLEAN { CLEAN_WALK } else { tag };
882                    s.heap.offer((t1, a.edge, tag));
883                }
884            }
885        }
886    }
887}

889/// [`continue_on_foot`] when no step can carry restriction progress (after
890/// a bicycle search, or a car search on a network without restrictions):
891/// every state is plain and the mode of the state left changes nothing.
892fn continue_on_foot_plain(net: &StreetNetwork, foot: &ModeGraph, limit: i32, s: &mut Search) {
893    s.radix.clear();
894    for i in 0..s.touched_edges.len() {
895        let e = s.touched_edges[i];
896        let t = s.edge_time[e as usize];
897        if t < i32::MAX {
898            s.radix.push(t, e);
899        }
900    }
901    while let Some((t, e)) = s.radix.pop() {
902        let e = e as usize;
903        if t > s.edge_time[e] {
904            continue;
905        }
906        let e_link = net.edge_flags[e] & LINK != 0;
907        for a in foot.arcs(net, net.edge_to(e)) {
908            if a.step < 0 || (a.link && e_link) {
909                continue;
910            }
911            let mut t1 = t.saturating_add(a.step);
912            if t1 == t {
913                t1 += 1;
914            }
915            if t1 < limit && s.add_plain(a.edge as usize, a.to as usize, t1) {
916                s.radix.push(t1, a.edge);
917            }
918        }
919    }
920}

922/// Time in seconds from the searched origin to a linked destination
923/// (R5 `LinkedPointSet.eval`), or `None` when unreachable.
924pub(super) fn time_to_point(
925    net: &StreetNetwork,
926    g: &ModeGraph,
927    origin: &Split,
928    dest: &Split,
929    s: &Search,
930) -> Option<i64> {
931    let sp = &g.sp;
932    let on = match g.mode {
933        Mode::Car => (cms_to_ms(net.edge_speed_cms[dest.edge as usize]) * 1000.0f32) as i32,
934        _ => sp.on_street_mms,
935    };
936    let off = sp.walk_mms;
937    if dest.edge == origin.edge {
938        let along = (origin.distance0_mm - dest.distance0_mm).abs();
939        return Some(i64::from(
940            origin.distance_to_edge_mm / off + along / on + dest.distance_to_edge_mm / off,
941        ));
942    }
943    let t0 = s.vertex_time[dest.vertex0 as usize];
944    let t1 = s.vertex_time[dest.vertex1 as usize];
945    if t0 == i32::MAX && t1 == i32::MAX {
946        return None;
947    }
948    let off_t = i64::from(dest.distance_to_edge_mm / off);
949    let via = |t: i32, d: i32| {
950        if t == i32::MAX {
951            i64::MAX
952        } else {
953            i64::from(t) + i64::from(d / on) + off_t
954        }
955    };
956    Some(via(t0, dest.distance0_mm).min(via(t1, dest.distance1_mm)))
957}

1174    }

1179    }

1181    /// The whole matrix as one slice, row by row.

1184    }

1186    /// The whole matrix as one vector, row by row, without copying:
1187    /// the value for `(origin, destination)` is at
1188    /// `origin * n_destinations + destination`.
1189    pub fn into_vec(self) -> Vec<Option<u32>> {
1190        self.minutes
1191    }

1210/// The minutes of [`street_travel_time_matrix`], row by row.
1211fn matrix_minutes(
1212    net: &StreetNetwork,
1213    origins: &[(f64, f64)],
1214    destinations: &[(f64, f64)],
1215    mode: Mode,
1216    params: &RouteParams,
1217) -> Result<Vec<Option<u32>>, RouteError> {
1218    params.validate()?;
1219    // a scenario: the searches and the origins' links on its network, the
1220    // destinations linked on `net` (R5 copies the base linkage)
1221    let scen = super::scenario::scenario_network(net, &params.scenario, mode == Mode::Car)?;
1222    let street = scen.as_ref().unwrap_or(net);
1223    let g = ModeGraph::new(street, mode, Speeds::new(params, mode), params.max_lts);
1224    let m = params.max_trip_duration;
1225    if scen.is_none() {
1226        if let Some(m) = super::ch::travel_time_matrix(net, &g, origins, destinations, params) {
1227            return m;
1228        }
1229    }
1230    let limit = m * 60;
1231    let dest_splits = link_points(
1232        net,
1233        destinations.len(),
1234        |j| destinations[j],
1235        mode,
1236        params.n_threads,
1237        &params.cancel,
1238    )
1239    .ok_or(RouteError::Cancelled)?;
1240    let nd = destinations.len();
1241    let mut out = vec![None; origins.len() * nd];
1242    if out.is_empty() {
1243        params.cancel.check()?;
1244        return Ok(out);
1245    }
1246    let cancel = &params.cancel;
1247    let n_threads = worker_threads(params.n_threads, origins.len());
1248    // Each worker takes the next origin's row of `out` and writes it in
1249    // place: the rows are disjoint, so the matrix is never held twice.
1250    let rows = Mutex::new(out.chunks_mut(nd).enumerate());
1251    crate::sys::scoped(0..n_threads, |_| {
1252        let mut search = Search::new(street);
1253        let mut scratch = Vec::new();
1254        while !cancel.is_cancelled() {
1255            let next = rows.lock().expect("row queue").next();
1256            let Some((i, row)) = next else { break };
1257            let (lat, lon) = origins[i];
1258            if let Some(o) = find_split(street, lat, lon, mode, &mut scratch) {
1259                route(street, &g, &o, limit, &mut search);
1260                for (j, d) in dest_splits.iter().enumerate() {
1261                    let Some(d) = d else { continue };
1262                    if let Some(secs) = time_to_point(street, &g, &o, d, &search) {
1263                        let minutes = secs / 60;
1264                        if minutes <= i64::from(m) {
1265                            row[j] = Some(minutes as u32);
1266                        }
1267                    }
1268                }
1269            }
1270        }
1271    });
1272    cancel.check()?;
1273    Ok(out)
1274}

1276/// Street-only travel times in seconds, before any rounding, reduced per
1277/// origin: `reduce(i, Some(times))` gets origin `i`'s time to every
1278/// destination (`i32::MAX` where the destination is unlinked or
1279/// unreachable), and `reduce(i, None)` an origin that cannot be linked.
1280/// The search is the one behind [`street_travel_time_matrix`], so a time can
1281/// exceed `max_trip_duration` by the partial edges at both ends, as in
1282/// R5's `LinkedPointSet.eval`. Results are in origin order;
1283/// [`RouteError::Cancelled`] when `params.cancel` is cancelled before every
1284/// origin is done.
1285pub(crate) fn travel_time_seconds<T: Send>(
1286    net: &StreetNetwork,
1287    origins: &[(f64, f64)],
1288    destinations: &[(f64, f64)],
1289    mode: Mode,
1290    params: &RouteParams,
1291    reduce: impl Fn(usize, Option<&[i32]>) -> T + Sync,
1292) -> Result<Vec<T>, RouteError> {
1293    params.validate()?;
1294    let dest_splits = link_points(
1295        net,
1296        destinations.len(),
1297        |j| destinations[j],
1298        mode,
1299        params.n_threads,
1300        &params.cancel,
1301    )
1302    .ok_or(RouteError::Cancelled)?;
1303    travel_time_seconds_linked(net, origins, &dest_splits, mode, params, reduce)
1304}

1306/// [`travel_time_seconds`] to destinations already linked by `mode`
1307/// (`dest_splits`, `None` for one that cannot be linked) on `net`; with a
1308/// scenario, the searches run on the scenario's network.
1309pub(crate) fn travel_time_seconds_linked<T: Send>(
1310    net: &StreetNetwork,
1311    origins: &[(f64, f64)],
1312    dest_splits: &[Option<Split>],
1313    mode: Mode,
1314    params: &RouteParams,
1315    reduce: impl Fn(usize, Option<&[i32]>) -> T + Sync,
1316) -> Result<Vec<T>, RouteError> {
1317    params.validate()?;
1318    let scen = super::scenario::scenario_network(net, &params.scenario, mode == Mode::Car)?;
1319    let net = scen.as_ref().unwrap_or(net);
1320    let g = ModeGraph::new(net, mode, Speeds::new(params, mode), params.max_lts);
1321    let limit = params.max_trip_duration * 60;
1322    let n_threads = worker_threads(params.n_threads, origins.len());
1323    let cancel = &params.cancel;
1324    let next = AtomicUsize::new(0);
1325    let chunks: Vec<Vec<(usize, T)>> = crate::sys::scoped(0..n_threads, |_| {
1326        let mut search = Search::new(net);
1327        let mut scratch = Vec::new();
1328        let mut times = Vec::with_capacity(dest_splits.len());
1329        let mut rows = Vec::new();
1330        while !cancel.is_cancelled() {
1331            let i = next.fetch_add(1, Ordering::Relaxed);
1332            if i >= origins.len() {
1333                break;
1334            }
1335            let (lat, lon) = origins[i];
1336            let Some(o) = find_split(net, lat, lon, mode, &mut scratch) else {
1337                rows.push((i, reduce(i, None)));
1338                continue;
1339            };
1340            search_from(net, &g, &o, dest_splits, limit, &mut search, &mut times);
1341            rows.push((i, reduce(i, Some(&times))));
1342        }
1343        rows
1344    });
1345    cancel.check()?;
1346    let mut slots: Vec<Option<T>> = (0..origins.len()).map(|_| None).collect();
1347    for (i, r) in chunks.into_iter().flatten() {
1348        slots[i] = Some(r);
1349    }
1350    Ok(slots
1351        .into_iter()
1352        .map(|r| r.expect("every origin is routed"))
1353        .collect())
1354}

1605    /// The walk continuation of a bicycle or car search (R5

1607    /// from every edge state the first search kept: same vertex times, and

1610    fn continuing_on_foot_matches_relaxation() {
1611        let net = jittered_grid();
1612        let foot = ModeGraph::with_speeds(&net, Mode::Walk, 3.6, 12.0, 2);
1613        let mut scratch = Vec::new();
1614        let (mut checked, mut improved) = (0, 0);
1615        for mode in [Mode::Bicycle, Mode::Car] {
1616            let g = ModeGraph::with_speeds(&net, mode, 3.6, 12.0, 1);
1617            assert!(!g.java_order);
1618            for k in 0..40 {
1619                let (lat, lon) = (f64::from(k % 11) * 0.001 + 0.0003, f64::from(k / 4) * 0.001);
1620                let Some(o) = find_split(&net, lat, lon, mode, &mut scratch) else {
1621                    continue;
1622                };
1623                for limit in [120, 400, 3600] {
1624                    let mut s = Search::new(&net);
1625                    route(&net, &g, &o, limit, &mut s);
1626                    let first = s.vertex_time.clone();
1627                    let mut best = s.edge_time.clone();
1628                    continue_on_foot(&net, &g, &foot, limit, &mut s);
1629                    let mut changed = true;
1630                    while changed {
1631                        changed = false;
1632                        for e in 0..net.n_edges() {
1633                            let t = best[e];
1634                            if t == i32::MAX {
1635                                continue;
1636                            }
1637                            let e_link = net.edge_flags[e] & LINK != 0;
1638                            for a in foot.arcs(&net, net.edge_to(e)) {
1639                                if a.step < 0 || (a.link && e_link) {
1640                                    continue;
1641                                }
1642                                let mut t1 = t + a.step;
1643                                if t1 == t {
1644                                    t1 += 1;
1645                                }
1646                                if t1 < limit && t1 < best[a.edge as usize] {
1647                                    best[a.edge as usize] = t1;
1648                                    changed = true;
1649                                }
1650                            }
1651                        }
1652                    }
1653                    let mut want = vec![i32::MAX; net.n_vertices()];
1654                    for (e, &t) in best.iter().enumerate() {
1655                        let v = net.edge_to(e);
1656                        want[v] = want[v].min(t);
1657                    }
1658                    assert_eq!(s.vertex_time, want, "{mode:?} origin {k} limit {limit}");
1659                    assert!(first.iter().zip(&want).all(|(a, b)| b <= a));
1660                    checked += 1;
1661                    improved += usize::from(first != want);
1662                }
1663            }
1664        }
1665        assert!(checked > 100, "{checked}");
1666        // walking reaches footways and goes against one-way streets
1667        assert!(improved > checked / 5, "{improved} of {checked}");
1668    }
