1//! Scenarios: the car speeds and levels of traffic stress of a street
2//! network changed for one routing call (`new_carspeeds`, `carspeed_scale`
3//! and `new_lts` in the packages).
4//!
5//! They are applied to a scenario copy of the network (as R5 copies it:
6//! `getTransportNetworkForScenario`) with four modifications, outside R5's
7//! scenario machinery, so that nothing is rebuilt afterwards:
8//!
9//! - car speeds by OSM way (`RoadCongestionOSM`): each directed edge whose
10//!   way is listed gets its value, the others `carspeed_scale`; a value of
11//!   0 forbids cars on the edge, a speed in km/h replaces the speed, a
12//!   factor scales it (`(short)` of a `float` product);
13//! - car speeds by polygon (R5 `RoadCongestion`): each directed edge takes
14//!   the factor of the polygon it intersects with the highest priority
15//!   (ties: the longest overlap), `carspeed_scale` outside every polygon
16//!   (`(short)` of a `double` product; a factor of 0 leaves cars allowed);
17//! - levels of traffic stress by OSM way (`SetLtsOsm`) and by line
18//!   (R5's `ShapefileMatcher`: each edge pair takes the level
19//!   of the line closest to it by discrete Hausdorff distance among those
20//!   whose bounding box meets the edge's).
21//!
22//! Connectors (transit stops, park-and-ride) are edges of OSM way -1, as
23//! in R5. The scenario network is then used for the street
24//! searches from the origins, the links of the origins, and the speed of
25//! the destination's edge. The destinations stay linked as on the base
26//! network (R5 copies the base linkage), the egress tables from the stops
27//! are the base network's (R5 copies them too) and so are the walking
28//! trees and transfers.

39/// New car speeds (`new_carspeeds`).
40#[derive(Debug, Clone, Default, PartialEq)]
41#[non_exhaustive]
42pub enum CarSpeeds {
43    /// No change (`new_carspeeds = NULL`): `carspeed_scale` alone scales
44    /// every edge.
45    #[default]
46    Unchanged,
47    /// A data frame with `speed_type = "scale"`: per OSM way id, a factor
48    /// for the car speed of its edges (0 forbids cars). A repeated id
49    /// keeps its last value.
50    ScaleByWay(Vec<(i64, f32)>),
51    /// A data frame with `speed_type = "km/h"`: per OSM way id, the car
52    /// speed of its edges in km/h (0 forbids cars). A repeated id keeps its
53    /// last value.
54    KmhByWay(Vec<(i64, f32)>),
55    /// An `sf` data frame of polygons: each edge takes the factor of the
56    /// polygon it meets with the highest priority.
57    ByPolygon(Vec<SpeedPolygon>),
58}
59
60/// A polygon of [`CarSpeeds::ByPolygon`]: a
61/// multipolygon in `[lon, lat]` degrees, a factor for the car speeds of
62/// the edges it meets and a priority. The coordinates are used as given.
63/// The R, Python and JavaScript packages first round them as GDAL writes
64/// GeoJSON (15 decimals of each coordinate, a tail that looks like
65/// round-off trimmed); do the same to get their results.
66#[derive(Debug, Clone, PartialEq)]
67#[non_exhaustive]
68pub struct SpeedPolygon {
69    /// The polygons (each a shell then its holes, every ring closed), in
70    /// `[lon, lat]` degrees.
71    pub polygons: Vec<Polygon>,
72    /// The factor (`scale`).
73    pub scale: f64,
74    /// The priority (`priority`): the highest wins.
75    pub priority: f64,
76}
77
78impl SpeedPolygon {
79    /// A polygon of `new_carspeeds`.
80    pub fn new(polygons: Vec<Polygon>, scale: f64, priority: f64) -> Self {
81        SpeedPolygon {
82            polygons,
83            scale,
84            priority,
85        }
86    }
87}

89/// New levels of traffic stress (`new_lts`).
90#[derive(Debug, Clone, Default, PartialEq)]
91#[non_exhaustive]
92pub enum LtsChange {
93    /// No change (`new_lts = NULL`).
94    #[default]
95    Unchanged,
96    /// A data frame: per OSM way id, the level of traffic stress of its
97    /// edges (below 2 is 1, above 4 is 4, 0 leaves the edge as it is). A
98    /// repeated id keeps its last value.
99    ByWay(Vec<(i64, i32)>),
100    /// An `sf` data frame of lines: each edge pair takes the level of the
101    /// line closest to it.
102    ByLine(Vec<LtsLine>),
103}
104
105/// A line of [`LtsChange::ByLine`], in `[lon, lat]`
106/// degrees, and its level of traffic stress.
107#[derive(Debug, Clone, PartialEq)]
108#[non_exhaustive]
109pub struct LtsLine {
110    /// The points of the line, `[lon, lat]` degrees.
111    pub points: Vec<[f64; 2]>,
112    /// The level of traffic stress (below 2 is 1, above 4 is 4).
113    pub lts: i32,
114}
115
116impl LtsLine {
117    /// A line of `new_lts`.
118    pub fn new(points: Vec<[f64; 2]>, lts: i32) -> Self {
119        LtsLine { points, lts }
120    }
121}

123/// Changes to the street network for one routing call: new car speeds, a
124/// factor for them and new levels of traffic stress. The default changes
125/// nothing. Fill it in by changing the fields of [`Scenario::default()`].
126///
127/// ```
128/// use peabiru_core::{CarSpeeds, Scenario};
129/// let mut s = Scenario::default();
130/// s.car_speeds = CarSpeeds::ScaleByWay(vec![(27184648, 0.5)]);
131/// s.car_speed_scale = 0.8; // every other edge
132/// let mut p = peabiru_core::RouteParams::default();
133/// p.scenario = s;
134/// ```
135#[derive(Debug, Clone, PartialEq)]
136#[non_exhaustive]
137pub struct Scenario {
138    /// New car speeds (`new_carspeeds`).
139    pub car_speeds: CarSpeeds,
140    /// The factor for the car speeds of the edges `car_speeds` does not
141    /// change (`carspeed_scale`, default 1).
142    pub car_speed_scale: f32,
143    /// New levels of traffic stress (`new_lts`).
144    pub lts: LtsChange,
145}
146
147impl Default for Scenario {
148    fn default() -> Self {
149        Scenario {
150            car_speeds: CarSpeeds::Unchanged,
151            car_speed_scale: 1.0,
152            lts: LtsChange::Unchanged,
153        }
154    }
155}
156
157impl Scenario {
158    /// Whether the scenario changes nothing, so that routing is on the
159    /// network as built (no `new_carspeeds`, `carspeed_scale` 1, no
160    /// `new_lts`).
161    pub fn is_unchanged(&self) -> bool {
162        !self.changes_car_speeds() && self.lts == LtsChange::Unchanged
163    }
164
165    /// Whether car speeds change: new speeds, or a factor other than 1.
166    fn changes_car_speeds(&self) -> bool {
167        self.car_speeds != CarSpeeds::Unchanged || self.car_speed_scale != 1.0
168    }
169
170    /// Check the values: the
171    /// factor finite and at least 0, no NaN, rings of at least 4 points
172    /// and closed, lines of at least 2 points, finite coordinates.
173    pub(crate) fn check(&self) -> Result<(), String> {
174        let s = self.car_speed_scale;
175        if s == f32::INFINITY {
176            // a finite double above the float range: R5 reads the factor
177            // as a Java float, which is then Infinity
178            return Err(
179                "carspeed_scale is too large: it is read as a 32-bit float, in which it \
180                        becomes infinite; use a smaller factor"
181                    .into(),
182            );
183        }
184        if !s.is_finite() || s < 0.0 {
185            return Err(format!(
186                "carspeed_scale must be a finite number of at least 0, got {s}"
187            ));
188        }
189        match &self.car_speeds {
190            CarSpeeds::Unchanged => {}
191            CarSpeeds::ScaleByWay(v) | CarSpeeds::KmhByWay(v) => {
192                if let Some((id, _)) = v.iter().find(|(_, x)| x.is_nan()) {
193                    return Err(format!(
194                        "new_carspeeds: the max_speed of OSM way {id} is NaN"
195                    ));
196                }
197            }
198            CarSpeeds::ByPolygon(polys) => {
199                for (i, p) in polys.iter().enumerate() {
200                    if p.scale.is_nan() || p.priority.is_nan() {
201                        return Err(format!(
202                            "new_carspeeds: the scale or priority of polygon {} is NaN",
203                            i + 1
204                        ));
205                    }
206                    let finite = p
207                        .polygons
208                        .iter()
209                        .flatten()
210                        .flatten()
211                        .all(|c| c[0].is_finite() && c[1].is_finite());
212                    let rings_ok = p
213                        .polygons
214                        .iter()
215                        .flatten()
216                        .all(|r| r.len() >= 4 && r.first() == r.last());
217                    if !finite || !rings_ok {
218                        return Err(format!(
219                            "new_carspeeds: polygon {} must have finite coordinates and closed \
220                             rings of at least 4 points",
221                            i + 1
222                        ));
223                    }
224                    if !p.polygons.iter().all(|poly| jts::polygon_is_simple(poly)) {
225                        return Err(format!(
226                            "new_carspeeds: polygon {} is not valid (a ring crosses or touches \
227                             itself or another ring, or a hole is outside its shell); the \
228                             ways such a polygon covers cannot be told reliably, so it is \
229                             refused (sf::st_make_valid() can repair it)",
230                            i + 1
231                        ));
232                    }
233                }
234            }
235        }
236        if let LtsChange::ByLine(lines) = &self.lts {
237            for (i, l) in lines.iter().enumerate() {
238                if l.points.len() < 2
239                    || !l
240                        .points
241                        .iter()
242                        .all(|c| c[0].is_finite() && c[1].is_finite())
243                {
244                    return Err(format!(
245                        "new_lts: line {} must have at least 2 points, with finite coordinates",
246                        i + 1
247                    ));
248                }
249            }
250        }
251        Ok(())
252    }
253
254    /// The scenario network: a copy of `base` with the car speeds and
255    /// levels of traffic stress changed, or `None` when the scenario
256    /// changes nothing. The copy keeps no contraction hierarchy or grid
257    /// linkage of `base`.
258    pub(crate) fn apply(&self, base: &StreetNetwork) -> Result<Option<StreetNetwork>, RouteError> {
259        self.check().map_err(RouteError::Invalid)?;
260        if self.is_unchanged() {
261            return Ok(None);
262        }
263        let mut net = base.clone();
264        net.forget_kept();
265        // R5's scenario copy of the edges (`EdgeStore.extendOnlyCopy`, made
266        // by `getTransportNetworkForScenario`) leaves out the cost
267        // fields: walking and cycling lose their slope costs (the packages
268        // warn of this).
269        net.edge_slope = Vec::new();
270        net.elevation = crate::Elevation::None;
271        if self.changes_car_speeds() {
272            match &self.car_speeds {
273                CarSpeeds::ByPolygon(polys) => {
274                    speeds_by_polygon(&mut net, polys, f64::from(self.car_speed_scale))?
275                }
276                CarSpeeds::Unchanged => speeds_by_way(&mut net, &[], false, self.car_speed_scale),
277                CarSpeeds::ScaleByWay(v) => speeds_by_way(&mut net, v, false, self.car_speed_scale),
278                CarSpeeds::KmhByWay(v) => speeds_by_way(&mut net, v, true, self.car_speed_scale),
279            }
280        }
281        match &self.lts {
282            LtsChange::Unchanged => {}
283            LtsChange::ByWay(v) => lts_by_way(&mut net, v),
284            LtsChange::ByLine(lines) => lts_by_line(&mut net, lines),
285        }
286        Ok(Some(net))
287    }
288}

290/// Java's `Long.parseLong`: an optional sign and decimal digits, nothing
291/// else (the ids reach R5 as text, so a number written as `1e+05` is
292/// refused).
293pub fn java_parse_long(s: &str) -> Option<i64> {
294    let digits = s.strip_prefix(['-', '+']).unwrap_or(s);
295    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) {
296        return None;
297    }
298    s.parse().ok()
299}
300
301/// Java's `Integer.parseInt`.
302pub fn java_parse_int(s: &str) -> Option<i32> {
303    java_parse_long(s).and_then(|v| i32::try_from(v).ok())
304}

306/// Java's `Float.parseFloat` for decimal text: blanks around it trimmed,
307/// an optional sign, digits with an optional point and exponent, an
308/// optional `f`/`d` suffix; `NaN` and `Infinity` as Java writes them (R's
309/// `Inf` is refused, as in Java). Correctly rounded, as Java's.
310pub fn java_parse_float(s: &str) -> Option<f32> {
311    let t = s.trim_matches(|c: char| c <= ' ');
312    let body = t.strip_prefix(['-', '+']).unwrap_or(t);
313    match body {
314        "NaN" => return Some(f32::NAN),
315        "Infinity" => {
316            return Some(if t.starts_with('-') {
317                f32::NEG_INFINITY
318            } else {
319                f32::INFINITY
320            })
321        }
322        _ => {}
323    }
324    let t = t.strip_suffix(['f', 'F', 'd', 'D']).unwrap_or(t);
325    let body = t.strip_prefix(['-', '+']).unwrap_or(t);
326    let (mantissa, exponent) = match body.find(['e', 'E']) {
327        Some(i) => (&body[..i], Some(&body[i + 1..])),
328        None => (body, None),
329    };
330    let digits = mantissa.bytes().filter(u8::is_ascii_digit).count();
331    let mantissa_ok = digits > 0
332        && mantissa.bytes().all(|b| b.is_ascii_digit() || b == b'.')
333        && mantissa.bytes().filter(|&b| b == b'.').count() <= 1;
334    let exponent_ok = match exponent {
335        None => true,
336        Some(e) => {
337            let d = e.strip_prefix(['-', '+']).unwrap_or(e);
338            !d.is_empty() && d.bytes().all(|b| b.is_ascii_digit())
339        }
340    };
341    if !mantissa_ok || !exponent_ok {
342        return None;
343    }
344    t.parse().ok()
345}

347/// The OSM way id R5 gives directed edge `e`: -1 for a connector.
348fn osm_id(net: &StreetNetwork, e: usize) -> i64 {
349    if net.edge_flags[e] & LINK != 0 {
350        -1
351    } else {
352        net.pair_way[e / 2]
353    }
354}

356/// R5 `intToLts`: below 2 is 1, above 4 is 4.
357fn lts_flag(lts: i32) -> u32 {
358    match lts {
359        i32::MIN..=1 => BIKE_LTS_1,
360        2 => BIKE_LTS_2,
361        3 => BIKE_LTS_3,
362        _ => BIKE_LTS_4,
363    }
364}

366/// `Edge.setFlag(BIKE_LTS_EXPLICIT)` and `Edge.setLts(lts)`.
367fn set_lts(net: &mut StreetNetwork, e: usize, lts: i32) {
368    let f = &mut net.edge_flags[e];
369    *f = (*f & !LTS_MASK) | BIKE_LTS_EXPLICIT | lts_flag(lts);
370}

372/// Car speeds by OSM way (the ported `RoadCongestionOSM.apply`).
373fn speeds_by_way(net: &mut StreetNetwork, values: &[(i64, f32)], kmh: bool, default: f32) {
374    let map: HashMap<i64, f32> = values.iter().copied().collect();
375    for e in 0..net.n_edges() {
376        let value = map.get(&osm_id(net, e)).copied();
377        let scaling = value.unwrap_or(default);
378        if scaling == 0.0 {
379            net.edge_flags[e] &= !ALLOWS_CAR;
380        } else if let (Some(v), true) = (value, kmh) {
381            // Edge.setSpeedKph(double): (short)(kph / 3.6 * 100)
382            net.edge_speed_cms[e] = (f64::from(v) / 3.6 * 100.0) as i32 as i16;
383        } else {
384            // (short)(edge.getSpeed() * scaling), a float product
385            net.edge_speed_cms[e] = (f32::from(net.edge_speed_cms[e]) * scaling) as i32 as i16;
386        }
387    }
388}

390/// The points of directed edge `e`, `[lon, lat]` degrees in its direction
391/// (R5 `Edge.getGeometry`).
392fn edge_points(net: &StreetNetwork, e: usize, out: &mut Vec<Pt>) {
393    out.clear();
394    out.extend(
395        net.pair_points(e / 2)
396            .map(|(lat, lon)| [f64::from(lon) / FIXED_FACTOR, f64::from(lat) / FIXED_FACTOR]),
397    );
398    if e % 2 == 1 {
399        out.reverse();
400    }
401}

403/// R5 `RoadCongestion.apply`, with the polygons' attributes
404/// (`scale`, `priority`) and `default` (`carspeed_scale`) outside them.
405fn speeds_by_polygon(
406    net: &mut StreetNetwork,
407    polys: &[SpeedPolygon],
408    default: f64,
409) -> Result<(), RouteError> {
410    let envs: Vec<Option<Env>> = polys
411        .iter()
412        .map(|p| {
413            let pts: Vec<Pt> = p.polygons.iter().flatten().flatten().copied().collect();
414            Env::of(&pts)
415        })
416        .collect();
417    let tree = StrTree::build(&envs);
418    let mut line = Vec::new();
419    let mut found = Vec::new();
420    for e in 0..net.n_edges() {
421        edge_points(net, e, &mut line);
422        let env = Env::of(&line).expect("an edge has two points");
423        tree.query(&env, &mut found);
424        let mut winner: Option<usize> = None;
425        for &c in &found {
426            let cand = &polys[c];
427            if !jts::polygon_intersects_line(&cand.polygons, &line) {
428                continue;
429            }
430            match winner {
431                None => winner = Some(c),
432                Some(w) if cand.priority > polys[w].priority => winner = Some(c),
433                Some(w) if cand.priority == polys[w].priority && cand.scale != polys[w].scale => {
434                    // both hold the whole edge: the same length, the first wins
435                    if jts::line_in_interior(&polys[w].polygons, &line)
436                        && jts::line_in_interior(&cand.polygons, &line)
437                    {
438                        continue;
439                    }
440                    let lw = jts::intersection_length(&polys[w].polygons, &line);
441                    let lc = jts::intersection_length(&cand.polygons, &line);
442                    if (lw - lc).abs() <= 1e-9 * lw.max(lc) {
443                        return Err(RouteError::Invalid(format!(
444                            "new_carspeeds: polygons {} and {} have the same priority, different \
445                             scales and overlap OSM way {} by the same length; such a tie \
446                             has no well-defined winner, so the \
447                             scenario is not supported (give one of them a higher priority)",
448                            w + 1,
449                            c + 1,
450                            osm_id(net, e)
451                        )));
452                    }
453                    if lc > lw {
454                        winner = Some(c);
455                    }
456                }
457                Some(_) => {}
458            }
459        }
460        let scale = winner.map_or(default, |w| polys[w].scale);
461        // (short)(edge.getSpeed() * winner.scale), a double product
462        net.edge_speed_cms[e] = (f64::from(net.edge_speed_cms[e]) * scale) as i32 as i16;
463    }
464    Ok(())
465}

467/// Levels of traffic stress by OSM way (the ported `SetLtsOsm.apply`).
468fn lts_by_way(net: &mut StreetNetwork, values: &[(i64, i32)]) {
469    let map: HashMap<i64, i32> = values.iter().copied().collect();
470    for e in 0..net.n_edges() {
471        let lts = map.get(&osm_id(net, e)).copied().unwrap_or(0);
472        if lts != 0 {
473            set_lts(net, e, lts);
474        }
475    }
476}

478/// Levels of traffic stress by line: R5 `ShapefileMatcher.match`.
479fn lts_by_line(net: &mut StreetNetwork, lines: &[LtsLine]) {
480    let envs: Vec<Option<Env>> = lines.iter().map(|l| Env::of(&l.points)).collect();
481    let tree = StrTree::build(&envs);
482    let mut edge = Vec::new();
483    let mut found = Vec::new();
484    for k in 0..net.n_edges() / 2 {
485        edge_points(net, 2 * k, &mut edge);
486        let env = Env::of(&edge).expect("an edge has two points");
487        tree.query(&env, &mut found);
488        // ShapefileMatcher.findBestMatch: the first of equal distances wins
489        let (mut best, mut best_distance) = (None, f64::INFINITY);
490        for &c in &found {
491            let d = jts::discrete_hausdorff(&lines[c].points, &edge);
492            if best_distance > d {
493                best_distance = d;
494                best = Some(c);
495            }
496        }
497        if let Some(c) = best {
498            set_lts(net, 2 * k, lines[c].lts);
499            set_lts(net, 2 * k + 1, lines[c].lts);
500        }
501    }
502}

504/// The network the street searches of a call with scenario `s` run on
505/// ([`Scenario::apply`]), checked for cars ([`check_car_speeds`]) when
506/// `by_car`; `None` when the scenario changes nothing.
507pub(crate) fn scenario_network(
508    base: &StreetNetwork,
509    s: &Scenario,
510    by_car: bool,
511) -> Result<Option<StreetNetwork>, RouteError> {
512    let net = s.apply(base)?;
513    if let (Some(n), true) = (&net, by_car) {
514        check_car_speeds(n)?;
515    }
516    Ok(net)
517}
518
519/// When a car search runs on a scenario network, the car speeds R5 would
520/// divide by: an edge cars may use with a speed of 0 or less (a factor too
521/// small, or a speed beyond the 327 m/s R5 stores, which wraps around)
522/// makes R5 fail or route with negative times. That is an error here.
523pub(crate) fn check_car_speeds(net: &StreetNetwork) -> Result<(), RouteError> {
524    let bad: Vec<usize> = (0..net.n_edges())
525        .filter(|&e| net.edge_flags[e] & ALLOWS_CAR != 0 && net.edge_speed_cms[e] <= 0)
526        .collect();
527    if let Some(&e) = bad.first() {
528        return Err(RouteError::Invalid(format!(
529            "the scenario leaves {} street edges open to cars with a speed of 0 or less (such \
530             as an edge of OSM way {}, at {} cm/s): car speeds are kept in cm/s as 16-bit \
531             integers, so a factor or speed that brings a speed below 1 cm/s, or above \
532             32767 cm/s (about 1180 km/h), cannot be represented and would give wrong or \
533             negative travel times; use \
534             another factor or speed, or a factor or speed of 0 to close the road",
535            bad.len(),
536            osm_id(net, e),
537            net.edge_speed_cms[e]
538        )));
539    }
540    Ok(())
541}

572        let (base, _) = build_from_dir(spo, &BuildOptions::default()).unwrap();
573        let e = (0..base.n_edges())
574            .find(|&e| base.edge_flags[e] & ALLOWS_CAR != 0 && base.edge_flags[e] & LINK == 0)
575            .unwrap();
576        let way = base.pair_way[e / 2];
577        let speed = base.edge_speed_cms[e];
578        let mut s = Scenario::default();
579        assert!(s.is_unchanged() && s.apply(&base).unwrap().is_none());
580        // a factor: a float product, truncated
581        s.car_speeds = CarSpeeds::ScaleByWay(vec![(way, 2.0), (way, 0.7)]);
582        let net = s.apply(&base).unwrap().unwrap();
583        assert_eq!(net.edge_speed_cms[e], (f32::from(speed) * 0.7f32) as i16);
584        // carspeed_scale for the others, connectors included (way -1)
585        s.car_speed_scale = 0.5;
586        let net = s.apply(&base).unwrap().unwrap();
587        let link = (0..base.n_edges())
588            .find(|&e| base.edge_flags[e] & LINK != 0)
589            .unwrap();
590        assert_eq!(osm_id(&base, link), -1);
591        assert_eq!(net.edge_speed_cms[link], base.edge_speed_cms[link] / 2);
592        // km/h replaces; 0 closes to cars, speed unchanged
593        s.car_speeds = CarSpeeds::KmhByWay(vec![(way, 36.0)]);
594        assert_eq!(s.apply(&base).unwrap().unwrap().edge_speed_cms[e], 1000);
595        s.car_speeds = CarSpeeds::KmhByWay(vec![(way, 0.0)]);
596        let net = s.apply(&base).unwrap().unwrap();
597        assert_eq!(net.edge_flags[e] & ALLOWS_CAR, 0);
598        assert_eq!(net.edge_speed_cms[e], speed);
599        // a speed beyond 32767 cm/s wraps around and is refused for cars
600        s.car_speeds = CarSpeeds::KmhByWay(vec![(way, 2000.0)]);
601        let net = s.apply(&base).unwrap().unwrap();
602        assert!(net.edge_speed_cms[e] < 0);
603        assert!(check_car_speeds(&net).is_err());
604        // levels: below 2 is 1, above 4 is 4, 0 leaves the edge
605        let mut s = Scenario::default();
606        for (lts, flag) in [(-3, BIKE_LTS_1), (3, BIKE_LTS_3), (9, BIKE_LTS_4)] {
607            s.lts = LtsChange::ByWay(vec![(way, lts)]);
608            let f = s.apply(&base).unwrap().unwrap().edge_flags[e];
609            assert_eq!(f & LTS_MASK, flag);
610            assert_ne!(f & BIKE_LTS_EXPLICIT, 0);
611        }
612        s.lts = LtsChange::ByWay(vec![(way, 0)]);
613        assert_eq!(
614            s.apply(&base).unwrap().unwrap().edge_flags[e],
615            base.edge_flags[e]
616        );
617        // the base network is never changed
618        assert_eq!(base.edge_speed_cms[e], speed);
619    }
