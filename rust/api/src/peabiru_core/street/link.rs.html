8/// First and fallback search radii in metres (R5 `INITIAL_LINK_RADIUS_METERS`,
9/// `LINK_RADIUS_METERS`).
10pub const INITIAL_LINK_RADIUS_M: f64 = 300.0;
11pub const LINK_RADIUS_M: f64 = 1600.0;
12const METERS_PER_DEGREE_LAT: f64 = 111_111.111;

14/// Where a point meets the network: a position along a forward edge.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub struct Split {
17    /// Forward (even) edge index.
18    pub edge: u32,
19    /// From and to vertex of that edge.
20    pub vertex0: u32,
21    pub vertex1: u32,
22    /// Distance along the edge from `vertex0` and to `vertex1`, millimetres.
23    pub distance0_mm: i32,
24    pub distance1_mm: i32,
25    /// Off-street distance from the point to the edge, millimetres.
26    pub distance_to_edge_mm: i32,
27    /// Segment of the edge geometry holding the split, and the split point
28    /// (fixed-point), used when a split cuts the edge in two.
29    pub seg: u32,
30    pub fixed_lat: i32,
31    pub fixed_lon: i32,
32}

34/// Uniform grid over forward-edge bounding boxes, in fixed-point degrees,
35/// storing only the cells that hold a pair (memory follows the network, not
36/// the area of its bounding box).
37#[derive(Debug, Clone, Default)]
38pub struct EdgeIndex {
39    min_lat: i32,
40    min_lon: i32,
41    n_rows: usize,
42    n_cols: usize,
43    /// Occupied cells, as `row * n_cols + col`, ascending.
44    cells: Vec<u64>,
45    /// Pairs in `cells[i]`: `pairs[offsets[i]..offsets[i + 1]]`.
46    offsets: Vec<u32>,
47    pairs: Vec<u32>,
48    /// Pairs added after the grid was built (edge splits), by cell (row and
49    /// column on the same grid, unbounded).
50    extra: HashMap<(i64, i64), Vec<u32>>,
51}
52
53/// Cell side in fixed-point degrees (0.005 degrees, about 550 m).
54const CELL: i64 = 50_000;
55
56/// Margin around a query envelope when looking up split pairs. A pair can
57/// be within the link radius of a point and still lie a few fixed-point
58/// units outside the query envelope, whose longitude half-width is
59/// truncated to an integer and whose distances are measured after
60/// truncating the scaled longitude difference (the overshoot is below
61/// `2 + 1 / cos(lat)` units; one cell covers it up to 89.998 degrees).
62/// The margin keeps every split pair a query could select.
63const EXTRA_MARGIN: i64 = CELL;
64
65/// Bounding box `(min_lat, min_lon, max_lat, max_lon)` of pair `k`.
66pub fn pair_bbox(net: &StreetNetwork, k: usize) -> (i32, i32, i32, i32) {
67    let mut b = (i32::MAX, i32::MAX, i32::MIN, i32::MIN);
68    for (lat, lon) in net.pair_points(k) {
69        b = (b.0.min(lat), b.1.min(lon), b.2.max(lat), b.3.max(lon));
70    }
71    b
72}

74impl EdgeIndex {
75    pub fn build(net: &StreetNetwork) -> Self {
76        let n_pairs = net.pair_from.len();
77        if n_pairs == 0 {
78            return EdgeIndex::default();
79        }
80        let boxes: Vec<_> = (0..n_pairs).map(|k| pair_bbox(net, k)).collect();
81        let min_lat = boxes.iter().map(|b| b.0).min().unwrap();
82        let min_lon = boxes.iter().map(|b| b.1).min().unwrap();
83        let max_lat = boxes.iter().map(|b| b.2).max().unwrap();
84        let max_lon = boxes.iter().map(|b| b.3).max().unwrap();
85        let n_rows = ((i64::from(max_lat) - i64::from(min_lat)) / CELL + 1) as usize;
86        let n_cols = ((i64::from(max_lon) - i64::from(min_lon)) / CELL + 1) as usize;
87        let cell_range = |b: &(i32, i32, i32, i32)| {
88            let r0 = ((i64::from(b.0) - i64::from(min_lat)) / CELL) as u64;
89            let r1 = ((i64::from(b.2) - i64::from(min_lat)) / CELL) as u64;
90            let c0 = ((i64::from(b.1) - i64::from(min_lon)) / CELL) as u64;
91            let c1 = ((i64::from(b.3) - i64::from(min_lon)) / CELL) as u64;
92            (r0..=r1).flat_map(move |r| (c0..=c1).map(move |c| r * n_cols as u64 + c))
93        };
94        let mut members: Vec<(u64, u32)> = Vec::new();
95        for (k, b) in boxes.iter().enumerate() {
96            members.extend(cell_range(b).map(|c| (c, k as u32)));
97        }
98        members.sort_unstable();
99        let mut idx = EdgeIndex {
100            min_lat,
101            min_lon,
102            n_rows,
103            n_cols,
104            cells: Vec::new(),
105            offsets: Vec::new(),
106            pairs: Vec::with_capacity(members.len()),
107            extra: HashMap::new(),
108        };
109        for (c, k) in members {
110            if idx.cells.last() != Some(&c) {
111                idx.cells.push(c);
112                idx.offsets.push(idx.pairs.len() as u32);
113            }
114            idx.pairs.push(k);
115        }
116        idx.offsets.push(idx.pairs.len() as u32);
117        idx
118    }
119
120    /// Grid row and column of a fixed-point position (unbounded).
121    fn cell_of(&self, lat: i64, lon: i64) -> (i64, i64) {
122        (
123            (lat - i64::from(self.min_lat)).div_euclid(CELL),
124            (lon - i64::from(self.min_lon)).div_euclid(CELL),
125        )
126    }
127
128    /// Register a pair created after the grid was built, with its bounding
129    /// box. The pair's geometry must stay inside that box (later splits of
130    /// the pair cut it at points on its geometry, so it does).
131    pub fn insert_extra(&mut self, pair: u32, bbox: (i32, i32, i32, i32)) {
132        let (r0, c0) = self.cell_of(i64::from(bbox.0), i64::from(bbox.1));
133        let (r1, c1) = self.cell_of(i64::from(bbox.2), i64::from(bbox.3));
134        for r in r0..=r1 {
135            for c in c0..=c1 {
136                self.extra.entry((r, c)).or_default().push(pair);
137            }
138        }
139    }
140
141    /// Pairs whose cells overlap an envelope, fixed-point, in no particular
142    /// order and with repeats: every pair whose bounding box meets the
143    /// envelope, and possibly others. Split pairs are looked up with
144    /// [`EXTRA_MARGIN`] around the envelope.
145    pub fn query(
146        &self,
147        min_lat: i64,
148        min_lon: i64,
149        max_lat: i64,
150        max_lon: i64,
151        out: &mut Vec<u32>,
152    ) {
153        out.clear();
154        if !self.extra.is_empty() {
155            let (r0, c0) = self.cell_of(min_lat - EXTRA_MARGIN, min_lon - EXTRA_MARGIN);
156            let (r1, c1) = self.cell_of(max_lat + EXTRA_MARGIN, max_lon + EXTRA_MARGIN);
157            let n_cells = (r1 - r0 + 1).saturating_mul(c1 - c0 + 1);
158            if n_cells > self.extra.len() as i64 {
159                for (&(r, c), pairs) in &self.extra {
160                    if (r0..=r1).contains(&r) && (c0..=c1).contains(&c) {
161                        out.extend_from_slice(pairs);
162                    }
163                }
164            } else {
165                for r in r0..=r1 {
166                    for c in c0..=c1 {
167                        if let Some(pairs) = self.extra.get(&(r, c)) {
168                            out.extend_from_slice(pairs);
169                        }
170                    }
171                }
172            }
173        }
174        if self.n_rows == 0 {
175            return;
176        }
177        let clamp_row = |lat: i64| {
178            ((lat - i64::from(self.min_lat)).div_euclid(CELL)).clamp(0, self.n_rows as i64 - 1)
179                as usize
180        };
181        let clamp_col = |lon: i64| {
182            ((lon - i64::from(self.min_lon)).div_euclid(CELL)).clamp(0, self.n_cols as i64 - 1)
183                as usize
184        };
185        if max_lat < i64::from(self.min_lat) || max_lon < i64::from(self.min_lon) {
186            return;
187        }
188        let max_lat_grid = i64::from(self.min_lat) + self.n_rows as i64 * CELL;
189        let max_lon_grid = i64::from(self.min_lon) + self.n_cols as i64 * CELL;
190        if min_lat >= max_lat_grid || min_lon >= max_lon_grid {
191            return;
192        }
193        let (c0, c1) = (clamp_col(min_lon) as u64, clamp_col(max_lon) as u64);
194        for r in clamp_row(min_lat)..=clamp_row(max_lat) {
195            let row = r as u64 * self.n_cols as u64;
196            let mut i = self.cells.partition_point(|&cell| cell < row + c0);
197            while i < self.cells.len() && self.cells[i] <= row + c1 {
198                out.extend_from_slice(
199                    &self.pairs[self.offsets[i] as usize..self.offsets[i + 1] as usize],
200                );
201                i += 1;
202            }
203        }
204    }
205}

216/// Link a point to the nearest eligible edge, first within 300 m then within
217/// 1600 m (R5 `StreetLayer.findSplit`). `scratch` is a reusable buffer.
218pub fn find_split(
219    net: &StreetNetwork,
220    lat: f64,
221    lon: f64,
222    mode: super::Mode,
223    scratch: &mut Vec<u32>,
224) -> Option<Split> {
225    find_split_radius(net, lat, lon, INITIAL_LINK_RADIUS_M, mode, scratch)
226        .or_else(|| find_split_radius(net, lat, lon, LINK_RADIUS_M, mode, scratch))
227}
228
229/// Points linked per batch by [`link_points`].
230const LINK_BATCH: usize = 512;
231
232/// [`find_split`] for `n` points, `point(i)` giving point `i` as `(lat,
233/// lon)`, shared out in batches among up to `n_threads` threads (0 = all
234/// cores): the splits in point order, the same whatever the number of
235/// threads. `None` when `cancel` is cancelled, which the workers check
236/// between batches.
237pub(crate) fn link_points(
238    net: &StreetNetwork,
239    n: usize,
240    point: impl Fn(usize) -> (f64, f64) + Sync,
241    mode: super::Mode,
242    n_threads: usize,
243    cancel: &crate::CancelToken,
244) -> Option<Vec<Option<Split>>> {
245    let mut out = vec![None; n];
246    let n_threads = super::router::worker_threads(n_threads, n.div_ceil(LINK_BATCH));
247    let batches = std::sync::Mutex::new(out.chunks_mut(LINK_BATCH).enumerate());
248    crate::sys::scoped(0..n_threads, |_| {
249        let mut scratch = Vec::new();
250        while !cancel.is_cancelled() {
251            // nothing panics while the lock is held
252            let next = batches.lock().unwrap_or_else(|e| e.into_inner()).next();
253            let Some((b, slots)) = next else { break };
254            for (k, slot) in slots.iter_mut().enumerate() {
255                let (lat, lon) = point(b * LINK_BATCH + k);
256                *slot = find_split(net, lat, lon, mode, &mut scratch);
257            }
258        }
259    });
260    (!cancel.is_cancelled()).then_some(out)
261}

263/// R5 `Split.find` within one radius.
264pub fn find_split_radius(
265    net: &StreetNetwork,
266    lat: f64,
267    lon: f64,
268    radius_m: f64,
269    mode: super::Mode,
270    scratch: &mut Vec<u32>,
271) -> Option<Split> {
272    let fixed_lat = java_int(lat * FIXED_FACTOR);
273    let fixed_lon = java_int(lon * FIXED_FACTOR);
274    let cos_lat = lat.to_radians().cos();
275    let radius_lat = i64::from(java_int(radius_m / METERS_PER_DEGREE_LAT * FIXED_FACTOR));
276    let radius_lon = i64::from(java_int(radius_lat as f64 / cos_lat));
277    net.index.query(
278        i64::from(fixed_lat) - radius_lat,
279        i64::from(fixed_lon) - radius_lon,
280        i64::from(fixed_lat) + radius_lat,
281        i64::from(fixed_lon) + radius_lon,
282        scratch,
283    );
284    scratch.sort_unstable();
285    scratch.dedup();
286    select_split(net, lat, lon, radius_m, mode, scratch)
287}

289/// The nearest eligible pair among `candidates` within the radius, ties to
290/// the lowest pair. The result does not depend on the order of the
291/// candidates, nor on extra candidates beyond the radius.
292fn select_split(
293    net: &StreetNetwork,
294    lat: f64,
295    lon: f64,
296    radius_m: f64,
297    mode: super::Mode,
298    candidates: &[u32],
299) -> Option<Split> {
300    let fixed_lat = java_int(lat * FIXED_FACTOR);
301    let fixed_lon = java_int(lon * FIXED_FACTOR);
302    let cos_lat = lat.to_radians().cos();
303    let radius_lat = i64::from(java_int(radius_m / METERS_PER_DEGREE_LAT * FIXED_FACTOR));
304    let sq_radius = radius_lat * radius_lat;
305    let flag = link_flag(mode) | LINKABLE;
306
307    // best: (squared distance, pair, segment, fraction, split lat, split lon)
308    let mut best: Option<(i64, usize, usize, f64, i32, i32)> = None;
309    for &k in candidates {
310        let k = k as usize;
311        if net.edge_flags[2 * k] & flag != flag || net.edge_flags[2 * k + 1] & flag != flag {
312            continue;
313        }
314        let mut prev: Option<(i32, i32)> = None;
315        for (seg_plus_one, (plat, plon)) in net.pair_points(k).enumerate() {
316            let Some((lat0, lon0)) = prev.replace((plat, plon)) else {
317                continue;
318            };
319            let seg = seg_plus_one - 1;
320            let (lat1, lon1) = (plat, plon);
321            let frac = segment_fraction(lon0, lat0, lon1, lat1, fixed_lon, fixed_lat, cos_lat);
322            let p_lon = java_int(f64::from(lon0) + frac * f64::from(lon1 - lon0));
323            let p_lat = java_int(f64::from(lat0) + frac * f64::from(lat1 - lat0));
324            let dx = (f64::from(p_lon.wrapping_sub(fixed_lon)) * cos_lat) as i64;
325            let dy = i64::from(p_lat) - i64::from(fixed_lat);
326            let d2 = dx * dx + dy * dy;
327            if d2 < sq_radius {
328                let better = match best {
329                    None => true,
330                    Some((bd2, bk, ..)) => d2 < bd2 || (d2 == bd2 && k < bk),
331                };
332                if better {
333                    best = Some((d2, k, seg, frac, p_lat, p_lon));
334                }
335            }
336        }
337    }
338    let (d2, k, best_seg, best_frac, split_lat, split_lon) = best?;
339
340    // Length along the edge up to the split, planar at the point's latitude.
341    let mut before = 0.0f64;
342    let mut prev: Option<(i32, i32)> = None;
343    for (seg_plus_one, (plat, plon)) in net.pair_points(k).enumerate() {
344        let Some((lat0, lon0)) = prev.replace((plat, plon)) else {
345            continue;
346        };
347        let seg = seg_plus_one - 1;
348        if seg <= best_seg {
349            let dx = f64::from(plon - lon0) * cos_lat;
350            let dy = f64::from(plat - lat0);
351            let mut len = (dx * dx + dy * dy).sqrt();
352            if seg == best_seg {
353                len *= best_frac;
354            }
355            before += len;
356        }
357    }
358    let length_mm = net.pair_length_mm[k];
359    let before_deg = f64::from(java_int(before)) / FIXED_FACTOR;
360    let d0 = java_int(before_deg * METERS_PER_DEGREE_LAT * 1000.0).clamp(0, length_mm);
361    let to_edge_deg = (d2 as f64).sqrt() / FIXED_FACTOR;
362    Some(Split {
363        edge: (2 * k) as u32,
364        vertex0: net.pair_from[k],
365        vertex1: net.pair_to[k],
366        distance0_mm: d0,
367        distance1_mm: length_mm - d0,
368        distance_to_edge_mm: java_int(to_edge_deg * METERS_PER_DEGREE_LAT * 1000.0),
369        seg: best_seg as u32,
370        fixed_lat: split_lat,
371        fixed_lon: split_lon,
372    })
373}

383    /// Deterministic pseudo-random numbers for the tests (xorshift).
384    pub(crate) struct Rng(pub u64);
385    impl Rng {
386        pub fn next(&mut self) -> u64 {
387            self.0 ^= self.0 << 13;
388            self.0 ^= self.0 >> 7;
389            self.0 ^= self.0 << 17;
390            self.0
391        }
392        /// Uniform in [0, 1).
393        pub fn unit(&mut self) -> f64 {
394            (self.next() >> 11) as f64 / (1u64 << 53) as f64
395        }
396    }

398    /// An `n` x `n` grid of residential streets from (`lat0`, `lon0`), `step`
399    /// degrees between intersections, with two shape points on every block
400    /// side; every position is jittered so edges cross cell boundaries at
401    /// arbitrary places.
402    pub(crate) fn jittered_grid(
403        rng: &mut Rng,
404        n: usize,
405        lat0: f64,
406        lon0: f64,
407        step: f64,
408    ) -> Extract {
409        let fixed = |d: f64| (d * 1e7).round() as i32;
410        let jitter = |rng: &mut Rng| (rng.unit() - 0.5) * step * 0.3;
411        let mut pos = vec![(0.0, 0.0); n * n];
412        for r in 0..n {
413            for c in 0..n {
414                pos[r * n + c] = (
415                    lat0 + r as f64 * step + jitter(rng),
416                    lon0 + c as f64 * step + jitter(rng),
417                );
418            }
419        }
420        let mut ex = Extract::default();
421        for (i, &(lat, lon)) in pos.iter().enumerate() {
422            ex.node_ids.push(1 + i as i64);
423            ex.lat.push(fixed(lat));
424            ex.lon.push(fixed(lon));
425            ex.ref_counts.push(2);
426        }
427        let mut next_id = 1 + (n * n) as i64;
428        let tags = || Tags(vec![("highway".into(), "residential".into())]);
429        for vertical in [false, true] {
430            for line in 0..n {
431                let at = |i: usize| if vertical { i * n + line } else { line * n + i };
432                let mut nodes = Vec::new();
433                for i in 0..n {
434                    nodes.push(1 + at(i) as i64);
435                    if i + 1 == n {
436                        break;
437                    }
438                    let (a, b) = (pos[at(i)], pos[at(i + 1)]);
439                    for t in [1.0 / 3.0, 2.0 / 3.0] {
440                        ex.node_ids.push(next_id);
441                        ex.lat
442                            .push(fixed(a.0 + t * (b.0 - a.0) + jitter(rng) * 0.3));
443                        ex.lon
444                            .push(fixed(a.1 + t * (b.1 - a.1) + jitter(rng) * 0.3));
445                        ex.ref_counts.push(1);
446                        nodes.push(next_id);
447                        next_id += 1;
448                    }
449                }
450                ex.ways.push(Way {
451                    id: 1 + (usize::from(vertical) * n + line) as i64,
452                    tags: tags(),
453                    nodes,
454                });
455            }
456        }
457        ex
458    }
