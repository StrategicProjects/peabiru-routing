1//! Street network for walk, bike and car routing, built from OpenStreetMap
2//! with the semantics of Conveyal R5 7.6, so travel times are R5's.
3//!
4//! The network stores edge *pairs*: pair `k` has the forward edge `2k`
5//! (along the OSM way) and the backward edge `2k + 1`. Coordinates are
6//! fixed-point degrees (1e-7), lengths millimetres, times whole seconds.
7

24/// Edge flags (bit set per directed edge).
25#[allow(missing_docs)]
26pub mod flags {
27    pub const ALLOWS_PEDESTRIAN: u32 = 1 << 0;
28    pub const ALLOWS_BIKE: u32 = 1 << 1;
29    pub const ALLOWS_CAR: u32 = 1 << 2;
30    pub const NO_THRU_TRAFFIC_PEDESTRIAN: u32 = 1 << 3;
31    pub const NO_THRU_TRAFFIC_BIKE: u32 = 1 << 4;
32    pub const NO_THRU_TRAFFIC_CAR: u32 = 1 << 5;
33    pub const LINKABLE: u32 = 1 << 6;
34    pub const BIKE_LTS_1: u32 = 1 << 7;
35    pub const BIKE_LTS_2: u32 = 1 << 8;
36    pub const BIKE_LTS_3: u32 = 1 << 9;
37    pub const BIKE_LTS_4: u32 = 1 << 10;
38    pub const BIKE_LTS_EXPLICIT: u32 = 1 << 11;
39    /// Some turn restriction starts when a car enters this edge.
40    pub const RESTRICTION_START: u32 = 1 << 12;
41    /// A connector (park-and-ride, later transit stops): every mode may use
42    /// it, points never link to it, and two in a row are not allowed.
43    pub const LINK: u32 = 1 << 13;
44    /// A `junction=roundabout` way (R5's `ROUNDABOUT`; only the
45    /// `osm_link_ids` of itineraries read it).
46    pub const ROUNDABOUT: u32 = 1 << 14;
47
48    pub const ALL_ALLOWS: u32 = ALLOWS_PEDESTRIAN | ALLOWS_BIKE | ALLOWS_CAR;
49    /// Flags that make an edge worth creating (R5 `ALL_PERMISSIONS`).
50    pub const ALL_PERMISSIONS: u32 =
51        ALL_ALLOWS | NO_THRU_TRAFFIC_PEDESTRIAN | NO_THRU_TRAFFIC_BIKE | NO_THRU_TRAFFIC_CAR;
52    pub const LTS_MASK: u32 = BIKE_LTS_1 | BIKE_LTS_2 | BIKE_LTS_3 | BIKE_LTS_4;
53}

63/// A routable street network for walking, cycling and driving, built by
64/// [`build_network`](crate::build_network) or, with its transit stops linked to the streets, by
65/// [`build_from_dir`](crate::build_from_dir).
66///
67/// A network is read-only once built: share it between threads by
68/// reference, as every routing function takes `&StreetNetwork`. Street
69/// matrices may build contraction hierarchies and keep them in the network
70/// for later calls (a few megabytes for a city; see [`StreetEngine`]), and
71/// travel-time surfaces keep the links of their grid's cells to the
72/// streets (40 bytes a cell, at most 64 MiB in all but for the last one
73/// used; see [`street_travel_time_surface`](crate::street_travel_time_surface));
74/// a clone shares them.
75#[derive(Debug, Clone, Default)]
76pub struct StreetNetwork {
77    // Invariant: the arrays below never change once a network has been
78    // handed out. They are written only while building, by
79    // `build::network_from_extract` and, in `build_from_dir`, by
80    // `TransitLayer::link_stops`, both before any street matrix can have
81    // run, so the hierarchies kept in `hierarchies` and the grid linkages
82    // kept in `grid_links` (built from these arrays) are never stale.
83    // `link_stops` drops them all the same. Any new code that changes the
84    // arrays of a network that may have been routed on must drop them too
85    // (`forget_kept`).
86    /// Vertex latitude, fixed-point 1e-7 degrees.
87    pub(crate) v_lat: Vec<i32>,
88    /// Vertex longitude, fixed-point 1e-7 degrees.
89    pub(crate) v_lon: Vec<i32>,
90    pub(crate) v_flags: Vec<u8>,
91
92    /// Per pair: from and to vertex of the forward edge.
93    pub(crate) pair_from: Vec<u32>,
94    pub(crate) pair_to: Vec<u32>,
95    /// Per pair: length in millimetres (haversine over the OSM nodes).
96    pub(crate) pair_length_mm: Vec<i32>,
97    /// Per pair: OSM way id.
98    pub(crate) pair_way: Vec<i64>,
99    /// Per pair: R5 street class code (`StreetClass.forWay`: 0 motorway,
100    /// 1 primary, 2 secondary, 3 tertiary, 4 other), copied by splits.
101    pub(crate) pair_class: Vec<u8>,
102    /// R5 `StreetLayer.envelope`: the OSM nodes of every way segment an edge
103    /// was attempted for, as `(min_lat, max_lat, min_lon, max_lon)` in node
104    /// fixed-point degrees; `None` when no node was seen.
105    pub(crate) envelope: Option<(i32, i32, i32, i32)>,
106    /// Per pair: intermediate geometry `geom[a..b]` for `pair_geom[k] = (a, b)`,
107    /// as `(lat, lon)` fixed-point, in forward order. A split appends the new
108    /// geometry and moves the range.
109    pub(crate) pair_geom: Vec<(u32, u32)>,
110    pub(crate) geom: Vec<(i32, i32)>,
111    /// Per pair: bearing leaving the first point and entering the last one,
112    /// in binary radians (R5 brads).
113    pub(crate) pair_in_brad: Vec<i8>,
114    pub(crate) pair_out_brad: Vec<i8>,
115
116    /// Per directed edge: flags and car speed in cm/s.
117    pub(crate) edge_flags: Vec<u32>,
118    pub(crate) edge_speed_cms: Vec<i16>,
119
120    /// Outgoing edges of each vertex: `out_edges[out_offsets[v]..out_offsets[v + 1]]`.
121    pub(crate) out_offsets: Vec<u32>,
122    pub(crate) out_edges: Vec<u32>,
123    /// Incoming edges of each vertex.
124    pub(crate) in_offsets: Vec<u32>,
125    pub(crate) in_edges: Vec<u32>,
126
127    /// Spatial index over the forward edges, for linking.
128    pub(crate) index: link::EdgeIndex,
129
130    /// Car turn restrictions.
131    pub(crate) restrictions: Vec<TurnRestriction>,
132    /// The restrictions of reverse car searches: R5 turns each `only_`
133    /// restriction into `no_` ones for them (`addReverseTurnRestriction`).
134    pub(crate) reverse_restrictions: Vec<TurnRestriction>,
135    /// `(from_edge, restriction index)`, sorted: restrictions that start when
136    /// a car enters an edge.
137    pub(crate) restriction_starts: Vec<(u32, u32)>,
138
139    /// The slope cost function the network was built with, and per
140    /// directed edge its slope factor (empty without elevation): see
141    /// [`crate::Elevation`].
142    pub(crate) elevation: crate::Elevation,
143    pub(crate) edge_slope: Vec<f32>,
144    /// What looked wrong in the elevation raster (see
145    /// [`StreetNetwork::elevation_warnings`]).
146    pub(crate) elevation_warnings: Vec<String>,
147
148    /// Contraction hierarchies built by street matrices, kept for later
149    /// ones (see [`StreetEngine`]); a clone shares them.
150    hierarchies: ch::Cache,
151    /// Links of the cells of travel-time surface grids, kept for later
152    /// surfaces; a clone shares them.
153    pub(crate) grid_links: crate::isochrone::linkage::Cache,
154}
155
156impl StreetNetwork {
157    /// Number of vertices: street intersections and ends, split points,
158    /// and the vertices of transit stops and park-and-ride connectors.
159    pub fn n_vertices(&self) -> usize {
160        self.v_lat.len()
161    }
162
163    /// Number of directed edges (two per street segment, one each way,
164    /// whether or not a mode may use it).
165    pub fn n_edges(&self) -> usize {
166        self.edge_flags.len()
167    }
168
169    /// The slope cost function walking and cycling times use on this
170    /// network: [`Elevation::None`](crate::Elevation::None) unless it was
171    /// built with an elevation raster and a model (see
172    /// [`BuildOptions::elevation`]).
173    pub fn elevation(&self) -> crate::Elevation {
174        self.elevation
175    }
176
177    /// Warnings about the elevation raster the network was built with
178    /// (empty without one): samples whose value is stored wrapped around
179    /// in 16-bit decimetres
180    /// (beyond +/-3276.7 m, such as SRTM's -32768 voids, which become 0 m,
181    /// and a no-data value of -9999, which becomes 3108.2 m) or equal to
182    /// the raster's declared no-data value.
183    /// And, in a raster not in EPSG:4326, the samples within nanometres of
184    /// a boundary between pixels of different values, where the pixel read
185    /// depends on the last bits of the projection (peabiru reads the same
186    /// one on every machine).
187    /// The core never prints them; the packages turn them into warnings.
188    pub fn elevation_warnings(&self) -> &[String] {
189        &self.elevation_warnings
190    }
191
192    /// Number of car turn restrictions read from OpenStreetMap and kept.
193    pub fn n_restrictions(&self) -> usize {
194        self.restrictions.len()
195    }
196
197    /// Drop the contraction hierarchies kept for street matrices and the
198    /// grid linkages kept for surfaces: to be called by anything that
199    /// changes the network's arrays (see the invariant on the fields).
200    pub(crate) fn forget_kept(&mut self) {
201        self.hierarchies = ch::Cache::default();
202        self.grid_links = Default::default();
203    }
204
205    /// Indices of the restrictions that start on edge `e`.
206    #[inline]
207    pub(crate) fn restrictions_starting_at(&self, e: u32) -> impl Iterator<Item = u32> + '_ {
208        let lo = self.restriction_starts.partition_point(|&(f, _)| f < e);
209        self.restriction_starts[lo..]
210            .iter()
211            .take_while(move |&&(f, _)| f == e)
212            .map(|&(_, r)| r)
213    }
214
215    #[inline]
216    pub(crate) fn edge_from(&self, e: usize) -> usize {
217        let k = e / 2;
218        (if e % 2 == 0 {
219            self.pair_from[k]
220        } else {
221            self.pair_to[k]
222        }) as usize
223    }
224
225    #[inline]
226    pub(crate) fn edge_to(&self, e: usize) -> usize {
227        let k = e / 2;
228        (if e % 2 == 0 {
229            self.pair_to[k]
230        } else {
231            self.pair_from[k]
232        }) as usize
233    }
234
235    #[inline]
236    pub(crate) fn out_edges(&self, v: usize) -> &[u32] {
237        &self.out_edges[self.out_offsets[v] as usize..self.out_offsets[v + 1] as usize]
238    }
239
240    #[inline]
241    pub(crate) fn in_edges(&self, v: usize) -> &[u32] {
242        &self.in_edges[self.in_offsets[v] as usize..self.in_offsets[v + 1] as usize]
243    }
244
245    /// All points of a pair's forward geometry, vertices included, as
246    /// `(lat, lon)` fixed-point.
247    pub(crate) fn pair_points(&self, k: usize) -> impl Iterator<Item = (i32, i32)> + '_ {
248        let v0 = self.pair_from[k] as usize;
249        let v1 = self.pair_to[k] as usize;
250        let (a, b) = self.pair_geom[k];
251        let mid = &self.geom[a as usize..b as usize];
252        std::iter::once((self.v_lat[v0], self.v_lon[v0]))
253            .chain(mid.iter().copied())
254            .chain(std::iter::once((self.v_lat[v1], self.v_lon[v1])))
255    }
256
257    /// Bearing in whole degrees entering a directed edge (R5 `getInAngle`).
258    #[inline]
259    pub(crate) fn in_angle(&self, e: usize) -> i32 {
260        let k = e / 2;
261        if e % 2 == 0 {
262            geom::brads_to_degrees(self.pair_in_brad[k])
263        } else {
264            geom::brads_to_degrees((self.pair_out_brad[k] as i32 + 128) as i8)
265        }
266    }
267
268    /// Bearing in whole degrees leaving a directed edge (R5 `getOutAngle`).
269    #[inline]
270    pub(crate) fn out_angle(&self, e: usize) -> i32 {
271        let k = e / 2;
272        if e % 2 == 0 {
273            geom::brads_to_degrees(self.pair_out_brad[k])
274        } else {
275            geom::brads_to_degrees((self.pair_in_brad[k] as i32 + 128) as i8)
276        }
277    }
