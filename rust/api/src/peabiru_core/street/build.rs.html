19/// Options for [`build_network`](crate::build_network) and [`build_from_dir`](crate::build_from_dir).
20/// Fill it in by changing the fields of [`BuildOptions::default()`].
21#[derive(Debug, Clone, Default)]
22#[non_exhaustive]
23pub struct BuildOptions {
24    /// Worker threads for reading the PBF; 0 (the default) means all cores,
25    /// and more than the cores available are not used.
26    pub n_threads: usize,
27    /// Build the transit layer even from GTFS feeds with errors of
28    /// priority `HIGH`. By default (`false`) each feed is first checked as
29    /// [`gtfs_errors`](crate::gtfs_errors) checks it, and
30    /// the build fails when any feed has an error
31    /// of priority `HIGH`, or is one gtfs-lib fails to load at all (see
32    /// [`NetworkError::high_priority_gtfs_errors`](crate::NetworkError::high_priority_gtfs_errors)).
33    /// With `true` the feeds are not checked: peabiru builds what it can
34    /// read, leaving out trips that refer to stops or routes that do not
35    /// exist, and [`TransitLayer::feed_warnings`](crate::TransitLayer::feed_warnings)
36    /// says which feeds would have been refused, and why.
37    pub ignore_gtfs_errors: bool,
38    /// The slope cost function for walking and cycling, applied from an
39    /// elevation raster: [`build_from_dir`](crate::build_from_dir) reads the
40    /// directory's `.tif` file, and
41    /// [`build_from_bytes_with_elevation`](crate::build_from_bytes_with_elevation)
42    /// takes its bytes. [`Elevation::None`](crate::Elevation::None) (the
43    /// default) ignores any raster; so does a build with no raster, and so
44    /// does [`build_network`](crate::build_network), which reads a `.pbf` file only.
45    pub elevation: crate::Elevation,
46}

48/// The work of [`crate::build_network`].
49pub(crate) fn build_network(
50    path: impl AsRef<std::path::Path>,
51    opts: &BuildOptions,
52) -> Result<StreetNetwork, NetworkError> {
53    let path = path.as_ref();
54    let bytes = std::fs::read(path).map_err(|e| {
55        NetworkError::osm(PbfError::InFile {
56            path: path.to_path_buf(),
57            error: Box::new(PbfError::Io(e)),
58        })
59    })?;
60    build_network_from_bytes(path, &bytes, opts)
61}

63/// [`build_network`] from the bytes of a `.pbf` file; errors name the
64/// file `name`.
65pub(crate) fn build_network_from_bytes(
66    name: &std::path::Path,
67    bytes: &[u8],
68    opts: &BuildOptions,
69) -> Result<StreetNetwork, NetworkError> {
70    let in_file = |error| {
71        NetworkError::osm(PbfError::InFile {
72            path: name.to_path_buf(),
73            error: Box::new(error),
74        })
75    };
76    let not_building = |t: &[(&str, &str)]| !t.iter().any(|(k, _)| *k == "building");
77    let node_tags = |t: &[(&str, &str)]| {
78        t.iter()
79            .any(|(k, _)| matches!(*k, "entrance" | "access" | "highway"))
80    };
81    let park_ride = |t: &[(&str, &str)]| {
82        t.iter()
83            .any(|(k, v)| *k == "park_ride" && super::parkride::is_park_and_ride_value(v))
84    };
85    // Routable ways become edges; park-and-ride ways (routable or not) are
86    // linked to the streets afterwards.
87    let ways = |t: &[(&str, &str)]| is_way_routable(t) || park_ride(t);
88    let read = ReadOptions {
89        way_filter: &ways,
90        relation_filter: Some(&super::restrictions::is_restriction),
91        node_tag_filter: Some(&node_tags),
92        ref_count_filter: Some(&not_building),
93        poi_filter: Some(&park_ride),
94        n_threads: opts.n_threads,
95    };
96    let ex = read_pbf(bytes, &read).map_err(in_file)?;
97    let net = network_from_extract(ex);
98    // Connectors (park-and-ride links) are the only pairs that do not come
99    // from a way, and splits only exist where a street pair does.
100    if net.edge_flags.iter().step_by(2).all(|&f| f & LINK != 0) {
101        return Err(in_file(PbfError::NoStreets));
102    }
103    Ok(net)
104}

106/// References to each extract node from the extract's ways without a
107/// `building` tag, repeats within a way included (what the reader counts
108/// with [`build_network`]'s `ref_count_filter`).
109fn ref_counts_of_ways(ex: &Extract) -> Vec<u32> {
110    let mut counts = vec![0u32; ex.node_ids.len()];
111    for w in &ex.ways {
112        if w.tags.0.iter().any(|(k, _)| k == "building") {
113            continue;
114        }
115        for &id in &w.nodes {
116            if let Some(i) = ex.node_index(id) {
117                counts[i] = counts[i].saturating_add(1);
118            }
119        }
120    }
121    counts
122}

137/// Build the network from an already-read extract. For R5's network the
138/// extract must come from [`build_network`]'s read options (routable ways,
139/// reference counts over non-building ways, node tags).
140///
141/// An extract read without reference counts (`ref_count_filter: None`) gets
142/// them from its own non-building ways; ways the read filtered out then do
143/// not count, so a node shared only with such a way is not an intersection.
144/// A node without a latitude or longitude (`lat` or `lon` shorter than
145/// `node_ids`) is treated as absent from the file, like `i32::MIN`.
146pub fn network_from_extract(mut ex: Extract) -> StreetNetwork {
147    let n = ex.node_ids.len();
148    if ex.lat.len() != n || ex.lon.len() != n {
149        ex.lat.resize(n, i32::MIN);
150        ex.lon.resize(n, i32::MIN);
151        for i in 0..n {
152            if ex.lat[i] == i32::MIN || ex.lon[i] == i32::MIN {
153                (ex.lat[i], ex.lon[i]) = (i32::MIN, i32::MIN);
154            }
155        }
156    }
157    if ex.ref_counts.len() != n {
158        ex.ref_counts = ref_counts_of_ways(&ex);
159    }
160    ex.ways.sort_by_key(|w| w.id); // R5 iterates ways in ascending id
161    let mut impassable = vec![false; n];
162    let mut signal = vec![false; n];
163    for (id, tags) in &ex.node_tags {
164        if let Some(i) = ex.node_index(*id) {
165            let t = &tags.0;
166            signal[i] = tag(t, "highway") == Some("traffic_signals");
167            impassable[i] = is_impassable(t);
168        }
169    }
170    let node_lat = ex
171        .lat
172        .iter()
173        .map(|&v| if v == i32::MIN { v } else { node_fixed(v) })
174        .collect();
175    let node_lon = ex
176        .lon
177        .iter()
178        .map(|&v| if v == i32::MIN { v } else { node_fixed(v) })
179        .collect();
180    let mut b = Builder {
181        ex: &ex,
182        node_lat,
183        node_lon,
184        vertex_of: vec![u32::MAX; n],
185        impassable,
186        signal,
187        net: StreetNetwork::default(),
188    };
189    fn borrowed(w: &Way) -> Vec<(&str, &str)> {
190        w.tags
191            .0
192            .iter()
193            .map(|(k, v)| (k.as_str(), v.as_str()))
194            .collect()
195    }
196    for w in &ex.ways {
197        if is_way_routable(&borrowed(w)) {
198            b.add_way(w);
199        }
200    }
201    let vertex_of = std::mem::take(&mut b.vertex_of);
202    let (node_lat, node_lon) = (
203        std::mem::take(&mut b.node_lat),
204        std::mem::take(&mut b.node_lon),
205    );
206    let mut net = b.net;
207    build_adjacency(&mut net);
208    apply_intersection_lts(&mut net);
209    for (mode_flag, walk_counts) in [
210        (ALLOWS_CAR, false),
211        (ALLOWS_PEDESTRIAN, false),
212        (ALLOWS_BIKE, true),
213    ] {
214        prune_islands(&mut net, mode_flag, walk_counts);
215    }
216    net.index = link::EdgeIndex::build(&net);
217
218    // Park-and-ride: areas (ways, in id order) then nodes (in id order).
219    let is_pr_way = |w: &Way| {
220        w.tags
221            .get("park_ride")
222            .is_some_and(super::parkride::is_park_and_ride_value)
223    };
224    let areas: Vec<Vec<(i32, i32)>> = ex
225        .ways
226        .iter()
227        .filter(|w| is_pr_way(w))
228        .filter_map(|w| {
229            // R5 cannot handle a park-and-ride way with a missing node; skip it
230            w.nodes
231                .iter()
232                .map(|&id| {
233                    ex.node_index(id)
234                        .filter(|&i| node_lat[i] != i32::MIN)
235                        .map(|i| (node_lat[i], node_lon[i]))
236                })
237                .collect::<Option<Vec<_>>>()
238        })
239        .collect();
240    let pr_nodes: Vec<(f64, f64)> = ex
241        .pois
242        .iter()
243        .map(|&(_, lat, lon, _)| {
244            (
245                f64::from(node_fixed(lat)) / FIXED_FACTOR,
246                f64::from(node_fixed(lon)) / FIXED_FACTOR,
247            )
248        })
249        .collect();
250    super::parkride::apply_park_and_ride(&mut net, &areas, &pr_nodes);
251
252    // R5 builds turn restrictions with the edge lists from before the
253    // park-and-ride splits (they are rebuilt only afterwards), so do the same.
254    let vertex_for_node = |id: i64| {
255        ex.node_index(id)
256            .map(|i| vertex_of[i])
257            .filter(|&v| v != u32::MAX)
258    };
259    super::restrictions::apply_turn_restrictions(&mut net, &ex, &vertex_for_node);
260    build_adjacency(&mut net);
261    net
262}

264/// R5 `isImpassable` (step-free routing off).
265fn is_impassable(t: &[(String, String)]) -> bool {
266    if t.is_empty() {
267        return false;
268    }
269    if tag(t, "entrance") == Some("emergency") {
270        return true;
271    }
272    if tag(t, "access") == Some("no") {
273        let null_or_no = |k: &str| matches!(tag(t, k), None | Some("no"));
274        return null_or_no("foot") && null_or_no("bicycle");
275    }
276    false
277}

279impl Builder<'_> {
280    fn node(&self, id: i64) -> Option<usize> {
281        self.ex
282            .node_index(id)
283            .filter(|&i| self.node_lat[i] != i32::MIN)
284    }
285
286    /// R5 `getVertexIndexForOsmNode`: one vertex per node, except that an
287    /// impassable node gets a fresh vertex at every reference.
288    fn vertex(&mut self, i: usize) -> u32 {
289        let v = self.vertex_of[i];
290        if v != u32::MAX && self.net.v_flags[v as usize] & IMPASSABLE == 0 {
291            return v;
292        }
293        let v = self.net.v_lat.len() as u32;
294        self.net.v_lat.push(vertex_fixed(self.node_lat[i]));
295        self.net.v_lon.push(vertex_fixed(self.node_lon[i]));
296        let mut f = 0;
297        if self.signal[i] {
298            f |= TRAFFIC_SIGNAL;
299        }
300        if self.impassable[i] {
301            f |= IMPASSABLE;
302        }
303        self.net.v_flags.push(f);
304        self.vertex_of[i] = v;
305        v
306    }
307
308    fn add_way(&mut self, w: &Way) {
309        let mut begin = 0usize;
310        for n in 1..w.nodes.len() {
311            let Some(i) = self.node(w.nodes[n]) else {
312                break; // R5 bails out of a way with an undefined node
313            };
314            let intersection = self.ex.ref_counts[i] >= 2;
315            let last = n == w.nodes.len() - 1;
316            if intersection || last || self.impassable[i] {
317                self.make_edge_pair(w, begin, n);
318                begin = n;
319            }
320        }
321    }
322
323    fn make_edge_pair(&mut self, w: &Way, begin: usize, end: usize) {
324        // R5 grows the street layer's envelope with every node of the
325        // segment, up to the first undefined one, before any check can
326        // drop the segment.
327        for &id in &w.nodes[begin..=end] {
328            let Some(i) = self.node(id) else { break };
329            let (lat, lon) = (self.node_lat[i], self.node_lon[i]);
330            let env = self.net.envelope.get_or_insert((lat, lat, lon, lon));
331            env.0 = env.0.min(lat);
332            env.1 = env.1.max(lat);
333            env.2 = env.2.min(lon);
334            env.3 = env.3.max(lon);
335        }
336        // Vertices are created first, even if the segment is dropped below.
337        let (n0, n1) = (self.node(w.nodes[begin]), self.node(w.nodes[end]));
338        let (Some(i0), Some(i1)) = (n0, n1) else {
339            for i in [n0, n1].into_iter().flatten() {
340                self.vertex(i);
341            }
342            return;
343        };
344        let v0 = self.vertex(i0);
345        let v1 = self.vertex(i1);
346        let mut idx = Vec::with_capacity(end - begin + 1);
347        for &id in &w.nodes[begin..=end] {
348            match self.node(id) {
349                Some(i) => idx.push(i),
350                None => return,
351            }
352        }
353        let deg = |fixed: i32| f64::from(fixed) / FIXED_FACTOR;
354        let mut length_m = 0.0;
355        for p in idx.windows(2) {
356            length_m += haversine_m(
357                deg(self.node_lat[p[0]]),
358                deg(self.node_lon[p[0]]),
359                deg(self.node_lat[p[1]]),
360                deg(self.node_lon[p[1]]),
361            );
362        }
363        if length_m * 1000.0 > f64::from(i32::MAX) {
364            return;
365        }
366        let length_mm = (length_m * 1000.0) as i32;
367
368        let tags = &w.tags.0;
369        let fwd_speed = speed_to_cms(car_speed_ms(tags, false));
370        let bwd_speed = speed_to_cms(car_speed_ms(tags, true));
371        let (mut fwd, mut bwd) = permissions(tags);
372        if (fwd | bwd) & ALL_PERMISSIONS == 0 {
373            return;
374        }
375        label_lts(tags, &mut fwd, &mut bwd);
376        if linkable(tags) {
377            fwd |= LINKABLE;
378            bwd |= LINKABLE;
379        }
380        // TypeOfEdgeLabeler
381        if tag(tags, "junction") == Some("roundabout") {
382            fwd |= ROUNDABOUT;
383            bwd |= ROUNDABOUT;
384        }
385
386        let net = &mut self.net;
387        net.pair_from.push(v0);
388        net.pair_to.push(v1);
389        net.pair_length_mm.push(length_mm);
390        net.pair_way.push(w.id);
391        // R5 StreetClass.forWay
392        net.pair_class.push(match tag(tags, "highway") {
393            Some("motorway") => 0,
394            Some("primary") => 1,
395            Some("secondary") => 2,
396            Some("tertiary") => 3,
397            _ => 4,
398        });
399        let start = net.geom.len() as u32;
400        for &i in &idx[1..idx.len() - 1] {
401            net.geom.push((self.node_lat[i], self.node_lon[i]));
402        }
403        net.pair_geom.push((start, net.geom.len() as u32));
404
405        // Angles use the vertex coordinates at the ends and the node
406        // coordinates in between (R5 getGeometry), as (lon, lat) degrees.
407        let mut pts = Vec::with_capacity(idx.len());
408        pts.push((deg(net.v_lon[v0 as usize]), deg(net.v_lat[v0 as usize])));
409        for &i in &idx[1..idx.len() - 1] {
410            pts.push((deg(self.node_lon[i]), deg(self.node_lat[i])));
411        }
412        pts.push((deg(net.v_lon[v1 as usize]), deg(net.v_lat[v1 as usize])));
413        let (first, last) = first_last_angles(&pts);
414        net.pair_in_brad.push(radians_to_brads(first));
415        net.pair_out_brad.push(radians_to_brads(last));
416
417        net.edge_flags.push(fwd);
418        net.edge_flags.push(bwd);
419        net.edge_speed_cms.push(fwd_speed);
420        net.edge_speed_cms.push(bwd_speed);
421    }
422}

424pub(crate) fn build_adjacency(net: &mut StreetNetwork) {
425    let nv = net.n_vertices();
426    let ne = net.n_edges();
427    let mut out_deg = vec![0u32; nv + 1];
428    let mut in_deg = vec![0u32; nv + 1];
429    for e in 0..ne {
430        out_deg[net.edge_from(e) + 1] += 1;
431        in_deg[net.edge_to(e) + 1] += 1;
432    }
433    for v in 0..nv {
434        out_deg[v + 1] += out_deg[v];
435        in_deg[v + 1] += in_deg[v];
436    }
437    let mut out_edges = vec![0u32; ne];
438    let mut in_edges = vec![0u32; ne];
439    let mut oc = out_deg.clone();
440    let mut ic = in_deg.clone();
441    for e in 0..ne {
442        let f = net.edge_from(e);
443        out_edges[oc[f] as usize] = e as u32;
444        oc[f] += 1;
445        let t = net.edge_to(e);
446        in_edges[ic[t] as usize] = e as u32;
447        ic[t] += 1;
448    }
449    net.out_offsets = out_deg;
450    net.out_edges = out_edges;
451    net.in_offsets = in_deg;
452    net.in_edges = in_edges;
453}

455/// Spread the highest LTS at each unsignalled vertex to every edge touching
456/// it (R5 `applyIntersectionCosts`).
457fn apply_intersection_lts(net: &mut StreetNetwork) {
458    let nv = net.n_vertices();
459    let mut stress = vec![0i32; nv]; // 0 = signalled, skipped
460    for (v, s) in stress.iter_mut().enumerate() {
461        if net.v_flags[v] & TRAFFIC_SIGNAL != 0 {
462            continue;
463        }
464        let mut max_lts = 1;
465        for &e in net.in_edges(v).iter().chain(net.out_edges(v)) {
466            let f = net.edge_flags[e as usize];
467            if f & BIKE_LTS_2 != 0 {
468                max_lts = max_lts.max(2);
469            }
470            if f & BIKE_LTS_3 != 0 {
471                max_lts = max_lts.max(3);
472            }
473            if f & BIKE_LTS_4 != 0 {
474                max_lts = max_lts.max(4);
475            }
476        }
477        *s = max_lts;
478    }
479    for (v, &s) in stress.iter().enumerate() {
480        if s == 0 {
481            continue;
482        }
483        let incident: Vec<u32> = net
484            .in_edges(v)
485            .iter()
486            .chain(net.out_edges(v))
487            .copied()
488            .collect();
489        for e in incident {
490            let f = &mut net.edge_flags[e as usize];
491            if *f & BIKE_LTS_EXPLICIT != 0 {
492                continue;
493            }
494            let own = lts_of(*f & (BIKE_LTS_2 | BIKE_LTS_3 | BIKE_LTS_4));
495            let lts = s.max(own);
496            *f &= !LTS_MASK;
497            *f |= match lts {
498                1 => BIKE_LTS_1,
499                2 => BIKE_LTS_2,
500                3 => BIKE_LTS_3,
501                _ => BIKE_LTS_4,
502            };
503        }
504    }
505}

507/// Remove `mode_flag` around every vertex in a strongly connected component
508/// smaller than [`MIN_SUBGRAPH_SIZE`] (R5 `TarjanIslandPruner`). For bikes the
509/// component graph also follows walkable edges (`walk_counts`).
510fn prune_islands(net: &mut StreetNetwork, mode_flag: u32, walk_counts: bool) {
511    let usable = |f: u32| f & mode_flag != 0 || (walk_counts && f & ALLOWS_PEDESTRIAN != 0);
512    let comp = strongly_connected_components(net, &usable);
513    let mut size = vec![0usize; comp.iter().copied().max().map_or(0, |m| m as usize + 1)];
514    for &c in &comp {
515        size[c as usize] += 1;
516    }
517    for v in 0..net.n_vertices() {
518        if size[comp[v] as usize] < MIN_SUBGRAPH_SIZE {
519            let incident: Vec<u32> = net
520                .out_edges(v)
521                .iter()
522                .chain(net.in_edges(v))
523                .copied()
524                .collect();
525            for e in incident {
526                net.edge_flags[e as usize] &= !mode_flag;
527            }
528        }
529    }
530}

532/// Iterative Tarjan: component id of every vertex over the edges `usable`
533/// accepts.
534fn strongly_connected_components(net: &StreetNetwork, usable: &dyn Fn(u32) -> bool) -> Vec<u32> {
535    const UNVISITED: u32 = u32::MAX;
536    let nv = net.n_vertices();
537    let mut index = vec![UNVISITED; nv];
538    let mut low = vec![0u32; nv];
539    let mut on_stack = vec![false; nv];
540    let mut comp = vec![UNVISITED; nv];
541    let mut stack: Vec<u32> = Vec::new();
542    let mut next_index = 0u32;
543    let mut next_comp = 0u32;
544    // call stack of (vertex, position in its out-edge list)
545    let mut calls: Vec<(u32, u32)> = Vec::new();
546    for root in 0..nv {
547        if index[root] != UNVISITED {
548            continue;
549        }
550        calls.push((root as u32, 0));
551        index[root] = next_index;
552        low[root] = next_index;
553        next_index += 1;
554        stack.push(root as u32);
555        on_stack[root] = true;
556        while let Some(&(v, pos0)) = calls.last() {
557            let v = v as usize;
558            let edges = net.out_edges(v);
559            let mut pos = pos0 as usize;
560            let mut child = None;
561            while pos < edges.len() {
562                let e = edges[pos] as usize;
563                pos += 1;
564                if !usable(net.edge_flags[e]) {
565                    continue;
566                }
567                let w = net.edge_to(e);
568                if index[w] == UNVISITED {
569                    child = Some(w);
570                    break;
571                } else if on_stack[w] {
572                    low[v] = low[v].min(index[w]);
573                }
574            }
575            calls.last_mut().expect("call stack").1 = pos as u32;
576            if let Some(w) = child {
577                index[w] = next_index;
578                low[w] = next_index;
579                next_index += 1;
580                stack.push(w as u32);
581                on_stack[w] = true;
582                calls.push((w as u32, 0));
583                continue;
584            }
585            calls.pop();
586            if let Some(&(p, _)) = calls.last() {
587                let p = p as usize;
588                low[p] = low[p].min(low[v]);
589            }
590            if low[v] == index[v] {
591                loop {
592                    let w = stack.pop().expect("Tarjan stack underflow") as usize;
593                    on_stack[w] = false;
594                    comp[w] = next_comp;
595                    if w == v {
596                        break;
597                    }
598                }
599                next_comp += 1;
600            }
601        }
602    }
603    comp
604}

641        short.lon.pop();
