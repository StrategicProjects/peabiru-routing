1//! R5's `StreetRouter` with its states kept, for the street legs of
2//! `detailed_itineraries()`.
3//!
4//! The travel-time searches of [`super::router`] keep one time per edge and
5//! may pop equal times in any order, which gives the right times but not
6//! R5's paths: which of two equally fast paths R5 reports depends on the
7//! order in which `java.util.PriorityQueue` hands out equal keys, on the A*
8//! heuristic toward a destination, on target pruning, and on the order of
9//! the states R5 keeps per edge. This module follows `StreetRouter.route()`
10//! (R5 7.6) step by step: a state arena with back pointers, the per-edge
11//! state lists of `bestStatesAtEdge` in insertion order, the exact queue,
12//! the `StopVisitor` of transit stop searches (at most 200 stops, none
13//! closer than five minutes by bicycle or car), reverse searches (for
14//! egress legs), and `getState(Split)` / `getStateAtVertex` with their
15//! tie rules.
16

59/// `java.util.PriorityQueue` with a comparator on an `int` key: the same
60/// sift rules, so equal keys leave in R5's order.
61#[derive(Default)]
62struct JavaQueue {
63    a: Vec<(i32, u32)>,
64}
65
66impl JavaQueue {
67    fn clear(&mut self) {
68        self.a.clear();
69    }
70
71    fn is_empty(&self) -> bool {
72        self.a.is_empty()
73    }
74
75    fn offer(&mut self, x: (i32, u32)) {
76        let mut k = self.a.len();
77        self.a.push(x);
78        while k > 0 {
79            let parent = (k - 1) >> 1;
80            let e = self.a[parent];
81            if x.0 >= e.0 {
82                break;
83            }
84            self.a[k] = e;
85            k = parent;
86        }
87        self.a[k] = x;
88    }
89
90    fn poll(&mut self) -> Option<(i32, u32)> {
91        let last = self.a.pop()?;
92        if self.a.is_empty() {
93            return Some(last);
94        }
95        let result = self.a[0];
96        let n = self.a.len();
97        let half = n >> 1;
98        let mut k = 0;
99        while k < half {
100            let mut child = 2 * k + 1;
101            let mut c = self.a[child];
102            let right = child + 1;
103            if right < n && c.0 > self.a[right].0 {
104                child = right;
105                c = self.a[child];
106            }
107            if last.0 <= c.0 {
108                break;
109            }
110            self.a[k] = c;
111            k = child;
112        }
113        self.a[k] = last;
114        Some(result)
115    }
116}

147/// A street search, as one `StreetRouter` instance.
148pub(crate) struct Router<'a> {
149    pub(crate) net: &'a StreetNetwork,
150    pub(crate) mode: Mode,
151    pub(crate) reverse: bool,
152    pub(crate) req: Req,
153    /// `timeLimitSeconds` (0: none).
154    pub(crate) time_limit: i32,
155    /// `distanceLimitMeters` (0: none).
156    pub(crate) distance_limit_m: i32,
157    /// Transit stop search: the stop of each vertex (`-1` for none) and the
158    /// `StopVisitor`'s minimum time.
159    stop_search: Option<(&'a [i32], i32)>,
160    /// `StopVisitor.stops`: stop -> seconds.
161    pub(crate) stops: HashMap<u32, i32>,
162    /// The stops of `stops` in the order they were first put (the order
163    /// of R5's `TIntIntHashMap` follows from it).
164    pub(crate) stops_order: Vec<u32>,
165    pub(crate) states: Vec<St>,
166    maps: Vec<RMap>,
167    /// `bestStatesAtEdge`: per edge, its live states in insertion order.
168    at_edge: HashMap<u32, Vec<u32>>,
169    queue: JavaQueue,
170    dest: Option<Split>,
171    best_at_dest: i32,
172    pub(crate) origin_split: Option<Split>,
173    max_abs_origin_lat: i32,
174    mm_per_lon_fixed: f64,
175    max_speed_s_per_mm: f64,
176    /// Restrictions by their last edge, for reverse searches (indices into
177    /// `reverse_restrictions`).
178    reverse_starts: Vec<(u32, u32)>,
179    /// Set when the search met something R5 does not reproduce (see
180    /// [`Router::unsupported`]).
181    pub(crate) unsupported: Option<&'static str>,
182    /// The via edges of a turn restriction with two or more of them may
183    /// be in either order: R5 reverses that array in place, in the network
184    /// every search shares, on each check of a reverse car search
185    /// (`TurnRestriction.reverse` in `canTurnFrom`), so the order a check
186    /// sees depends on every earlier check, in this call or an earlier one,
187    /// on any thread. The arrays reversed are those of the `no_`
188    /// restrictions (forward and reverse searches share them) and of the
189    /// `no_` ones R5 derives from `only_` restrictions for reverse searches;
190    /// no search reverses the array of a forward `only_` restriction. A
191    /// check whose outcome depends on the order is "ambiguous".
192    ///
193    /// With `via_fresh`, the orders are those of a network on which no
194    /// reverse car search ran before the pair, each array reversed by every
195    /// check of the pair's own reverse searches, in the order they run:
196    /// `via_fresh` holds, per array ([`Router::via_key`]), whether it is
197    /// reversed now. With `via_choices`, the `i`-th ambiguous check met
198    /// takes the order `choices[i]` (`true`: reversed), or, past the end,
199    /// reversed in a reverse search and as built in a forward one. Both
200    /// record the orders their ambiguous checks took in `via_used`. With
201    /// neither, the order is the one built, and a reverse search that
202    /// checks such a restriction sets `unsupported`.
203    pub(crate) via_choices: Option<Vec<bool>>,
204    pub(crate) via_fresh: Option<HashMap<u64, bool>>,
205    pub(crate) via_used: Vec<bool>,
206    /// Relations of the network's `no_` restrictions with two or more via
207    /// edges, found on the first reverse check that needs them.
208    shared_via: Option<HashSet<i64>>,
209}

288    /// `setOrigin(lat, lon)` once the split is found: the two start states,
289    /// queued and recorded, without the limit check.
290    pub(crate) fn set_origin_split(&mut self, split: Split) {
291        self.origin_split = Some(split);
292        let fwd = split.edge as usize;
293        let off = split.distance_to_edge_mm / ((self.req.walk * 1000.0f32) as i32);
294        let sp1 = self.speed_on(fwd, self.mode);
295        let d1 = ((split.distance1_mm / 1000) as f32 / sp1) as i32 + off;
296        let sp0 = self.speed_on(fwd + 1, self.mode);
297        let d0 = ((split.distance0_mm / 1000) as f32 / sp0) as i32 + off;
298        let (mut v0, mut v1) = (split.vertex0, split.vertex1);
299        if self.reverse {
300            std::mem::swap(&mut v0, &mut v1);
301        }
302        let mk = |vertex, edge: usize, dur, dist| St {
303            vertex,
304            back_edge: edge as i32,
305            dur,
306            dist,
307            mode: self.mode,
308            back: NONE,
309            tr: NONE,
310            heuristic: 0,
311        };
312        let mut s0 = mk(
313            v0,
314            fwd + 1,
315            d0,
316            split.distance0_mm + split.distance_to_edge_mm,
317        );
318        let mut s1 = mk(v1, fwd, d1, split.distance1_mm + split.distance_to_edge_mm);
319        self.start_turn_restriction(self.mode, &mut s0);
320        self.start_turn_restriction(self.mode, &mut s1);
321        let a = self.push_state(s0);
322        let b = self.push_state(s1);
323        self.queue.offer((self.key(a), a));
324        self.queue.offer((self.key(b), b));
325        self.put_at_edge(fwd as u32 + 1, a);
326        self.put_at_edge(fwd as u32, b);
327        self.max_abs_origin_lat = split.fixed_lat;
328    }

369    fn seed(&mut self, vertex: u32, edge: u32, partial_mm: i32, off_mm: i32) {
370        let e = edge as usize;
371        let probe_start = St {
372            vertex: self.net.edge_to(e) as u32,
373            back_edge: -1,
374            dur: 0,
375            dist: 0,
376            mode: self.mode,
377            back: NONE,
378            tr: NONE,
379            heuristic: 0,
380        };
381        let mut probe = self.traverse_with(&probe_start, NONE, e, self.req);
382        if probe.is_none() && self.mode == Mode::Bicycle && self.req.lts > 0 && self.req.lts < 4 {
383            let mut relaxed = self.req;
384            relaxed.lts = 4;
385            probe = self.traverse_with(&probe_start, NONE, e, relaxed);
386        }
387        let Some(probe) = probe else { return };
388        let len = self.net.pair_length_mm[e / 2];
389        let partial_s = if partial_mm == 0 {
390            0
391        } else {
392            (f64::from(probe.dur) * (f64::from(partial_mm) / f64::from(len))) as i32
393        };
394        let off_s = off_mm / ((1000.0f32 * self.req.walk) as i32);
395        let dur = (i64::from(partial_s) + i64::from(off_s)) as i32;
396        if self.time_limit > 0 && dur >= self.time_limit {
397            return;
398        }
399        let mut st = St {
400            vertex,
401            back_edge: edge as i32,
402            dur,
403            dist: partial_mm + off_mm,
404            mode: probe.mode,
405            back: NONE,
406            tr: NONE,
407            heuristic: 0,
408        };
409        self.start_turn_restriction(probe.mode, &mut st);
410        let id = self.push_state(st);
411        self.queue.offer((self.key(id), id));
412        self.put_at_edge(edge, id);
413    }

415    /// `EdgeStore.startTurnRestriction(streetMode, reverseSearch, s1)`:
416    /// `prev_mode` is the mode of the state left.
417    fn start_turn_restriction(&mut self, prev_mode: Mode, s: &mut St) {
418        if prev_mode != Mode::Car || s.back_edge < 0 {
419            return;
420        }
421        let e = s.back_edge as u32;
422        let starting: Vec<u32> = if self.reverse {
423            let lo = self.reverse_starts.partition_point(|&(f, _)| f < e);
424            self.reverse_starts[lo..]
425                .iter()
426                .take_while(|&&(f, _)| f == e)
427                .map(|&(_, r)| r)
428                .collect()
429        } else {
430            if self.net.edge_flags[e as usize] & RESTRICTION_START == 0 {
431                return;
432            }
433            self.net.restrictions_starting_at(e).collect()
434        };
435        if starting.is_empty() {
436            return;
437        }
438        let mut map = if s.tr == NONE {
439            RMap::new()
440        } else {
441            self.maps[s.tr as usize].clone()
442        };
443        for r in starting {
444            match map.binary_search_by_key(&r, |&(k, _)| k) {
445                Ok(i) => map[i].1 = 1,
446                Err(i) => map.insert(i, (r, 1)),
447            }
448        }
449        self.maps.push(map);
450        s.tr = (self.maps.len() - 1) as u32;
451    }

453    /// `canTurnFrom(s0, s1)` for a step onto `edge`: `None` when forbidden,
454    /// else the restriction progress the new state carries (`None` inside
455    /// for no map).
456    fn can_turn_from(&mut self, s0: &St, edge: u32) -> Option<Option<RMap>> {
457        if s0.tr == NONE || s0.mode != Mode::Car {
458            return Some(None);
459        }
460        let mut out = RMap::new();
461        for &(r, pos) in &self.maps[s0.tr as usize] {
462            let rest = if self.reverse {
463                &self.net.reverse_restrictions[r as usize]
464            } else {
465                &self.net.restrictions[r as usize]
466            };
467            let via = &rest.via_edges;
468            let k = via.len();
469            let to_edge = if self.reverse {
470                rest.from_edge
471            } else {
472                rest.to_edge
473            };
474            let p = (pos - 1) as usize;
475            // the order of the via edges R5 compares with: as built, or
476            // reversed (see `via_fresh`)
477            let mut reversed_now = false;
478            if k > 1 {
479                if let Some(key) = Self::via_key(self.net, &mut self.shared_via, self.reverse, r) {
480                    // whether the step's outcome depends on the order
481                    let ambiguous = p < k && {
482                        let (built, reversed) = (via[p], via[k - 1 - p]);
483                        built != reversed && ((edge == built) != (edge == reversed))
484                    };
485                    if let Some(fresh) = &mut self.via_fresh {
486                        let state = fresh.entry(key).or_insert(false);
487                        if self.reverse {
488                            *state = !*state; // TurnRestriction.reverse
489                        }
490                        reversed_now = *state;
491                        if ambiguous {
492                            self.via_used.push(reversed_now);
493                        }
494                    } else if let Some(choices) = &self.via_choices {
495                        if ambiguous {
496                            let i = self.via_used.len();
497                            reversed_now = choices.get(i).copied().unwrap_or(self.reverse);
498                            self.via_used.push(reversed_now);
499                        }
500                    } else if self.reverse {
501                        // the reversal changes the network for every later
502                        // search: see `via_fresh`
503                        self.unsupported = Some(
504                            "a car search backwards through a turn restriction with two or more \
505                             via ways",
506                        );
507                    }
508                }
509            }
510            if p < k {
511                let expected = if reversed_now { via[k - 1 - p] } else { via[p] };
512                if edge != expected {
513                    if rest.only {
514                        return None;
515                    }
516                } else {
517                    out.push((r, pos + 1));
518                }
519            } else if to_edge != edge {
520                if rest.only {
521                    return None;
522                }
523            } else if !rest.only {
524                return None;
525            }
526        }
527        Some(Some(out))
528    }

530    /// The via edge array R5 may have reversed that restriction `r` of a
531    /// search (reverse or forward) checks, as a key of
532    /// [`Router::via_fresh`]: a `no_` restriction's relation (forward and
533    /// reverse searches share its array), or, with the top bit set, the
534    /// index of a reverse restriction derived from an `only_` one (its own
535    /// array). `None` for a forward `only_` restriction, whose array no
536    /// search reverses. `shared` caches the relations of the network's `no_`
537    /// restrictions with two or more via edges.
538    fn via_key(
539        net: &StreetNetwork,
540        shared: &mut Option<HashSet<i64>>,
541        reverse: bool,
542        r: u32,
543    ) -> Option<u64> {
544        if !reverse {
545            let rest = &net.restrictions[r as usize];
546            return (!rest.only).then_some(rest.relation as u64);
547        }
548        let rest = &net.reverse_restrictions[r as usize];
549        let shared = shared.get_or_insert_with(|| {
550            net.restrictions
551                .iter()
552                .filter(|t| !t.only && t.via_edges.len() > 1)
553                .map(|t| t.relation)
554                .collect()
555        });
556        Some(if shared.contains(&rest.relation) {
557            rest.relation as u64
558        } else {
559            (1 << 63) | u64::from(r)
560        })
561    }

563    /// `Edge.traverse(s0, streetMode, req, timeCalculator)` from state
564    /// `s0` (id `s0_id`, or `NONE` for a probe) onto directed edge `e`.
565    fn traverse_with(&mut self, s0: &St, s0_id: u32, e: usize, req: Req) -> Option<St> {
566        let net = self.net;
567        let vertex = if self.reverse {
568            net.edge_from(e)
569        } else {
570            net.edge_to(e)
571        } as u32;
572        let tr = self.can_turn_from(s0, e as u32)?;
573        let mut s1 = St {
574            vertex,
575            back_edge: e as i32,
576            dur: s0.dur,
577            dist: s0.dist,
578            mode: self.mode,
579            back: s0_id,
580            tr: NONE,
581            heuristic: 0,
582        };
583        // the progress carried on (an empty map is dropped), then the
584        // restrictions starting on this edge
585        if let Some(m) = tr {
586            if !m.is_empty() {
587                self.maps.push(m);
588                s1.tr = (self.maps.len() - 1) as u32;
589            }
590        }
591        self.start_turn_restriction(s0.mode, &mut s1);
592        let flags = net.edge_flags[e];
593        if s0.back_edge >= 0
594            && s0.back != NONE
595            && flags & LINK != 0
596            && net.edge_flags[s0.back_edge as usize] & LINK != 0
597        {
598            return None;
599        }
600        let mut mode = self.mode;
601        match mode {
602            Mode::Walk => {
603                if flags & ALLOWS_PEDESTRIAN == 0 {
604                    return None;
605                }
606            }
607            Mode::Bicycle => {
608                let mut walk = flags & ALLOWS_BIKE == 0;
609                if req.lts > 0 && req.lts < 4 {
610                    if flags & BIKE_LTS_4 != 0 {
611                        walk = true;
612                    }
613                    if req.lts < 3 && flags & BIKE_LTS_3 != 0 {
614                        walk = true;
615                    }
616                    if req.lts < 2 && flags & BIKE_LTS_2 != 0 {
617                        walk = true;
618                    }
619                }
620                if walk {
621                    if flags & ALLOWS_PEDESTRIAN == 0 {
622                        return None;
623                    }
624                    mode = Mode::Walk;
625                }
626            }
627            Mode::Car => {
628                if flags & ALLOWS_CAR == 0 {
629                    return None;
630                }
631            }
632        }
633        s1.mode = mode;
634        let len = net.pair_length_mm[e / 2];
635        let speed = match mode {
636            Mode::Car => cms_to_ms(net.edge_speed_cms[e]),
637            m => req.speed(m),
638        };
639        let mut trav = (f64::from(len) / 1000.0 / f64::from(speed)).ceil() as i32;
640        if mode != Mode::Car {
641            if let Some(&k) = net.edge_slope.get(e) {
642                // `MultistageTraversalTimeCalculator` with the slope costs
643                trav = crate::elevation::sloped_seconds(trav, k);
644            }
645        }
646        let turn = if s0.back_edge >= 0 && mode == Mode::Car {
647            let (from, to) = if self.reverse {
648                (e, s0.back_edge as usize)
649            } else {
650                (s0.back_edge as usize, e)
651            };
652            car_turn_cost(net.out_angle(from), net.in_angle(to))
653        } else {
654            0
655        };
656        s1.dur = s1.dur.wrapping_add(trav + turn);
657        s1.dist = s1.dist.wrapping_add(len);
658        if s1.dur == s0.dur {
659            s1.dur += 1;
660        }
661        if s1.dist == s0.dist {
662            s1.dist += 1;
663        }
664        Some(s1)
665    }

667    /// `dominates(s1, s2)`.
668    fn dominates(&self, a: &St, b: &St) -> bool {
669        if a.tr == NONE && b.tr == NONE {
670            return a.dur <= b.dur;
671        }
672        if a.tr != NONE && b.tr != NONE {
673            return self.maps[a.tr as usize] == self.maps[b.tr as usize];
674        }
675        false
676    }
677
678    /// `isDominated(newState)`: drops the existing states it dominates.
679    fn is_dominated(&mut self, s: &St) -> bool {
680        let e = s.back_edge as u32;
681        let Some(list) = self.at_edge.get(&e) else {
682            return false;
683        };
684        let mut keep = Vec::with_capacity(list.len());
685        let mut dominated = false;
686        for (i, &id) in list.iter().enumerate() {
687            let ex = &self.states[id as usize];
688            if self.dominates(ex, s) {
689                dominated = true;
690                keep.extend_from_slice(&list[i..]);
691                break;
692            } else if !self.dominates(s, ex) {
693                keep.push(id);
694            }
695        }
696        let list = self.at_edge.get_mut(&e).expect("present");
697        *list = keep;
698        dominated
699    }

701    fn calc_heuristic(&self, s: &St) -> i32 {
702        let Some(d) = self.dest else { return 0 };
703        let v = s.vertex as usize;
704        let dlat = d.fixed_lat - self.net.v_lat[v];
705        let dlon = d.fixed_lon - self.net.v_lon[v];
706        let x = self.mm_per_lon_fixed * f64::from(dlon);
707        let y = mm_per_unit_lat_fixed() * f64::from(dlat);
708        let dist = (x * x + y * y).sqrt();
709        (dist * self.max_speed_s_per_mm) as i32
710    }

712    /// `StreetRouter.route()`.
713    pub(crate) fn route(&mut self) {
714        if let Some(d) = self.dest {
715            let max_abs = d
716                .fixed_lat
717                .abs()
718                .max(self.max_abs_origin_lat.wrapping_abs());
719            let rad = (f64::from(max_abs) / 1e7).to_radians();
720            self.mm_per_lon_fixed = mm_per_unit_lat_fixed() * rad.cos();
721            let mut speed = f64::from(self.req.speed(self.mode));
722            if speed == 0.0 {
723                speed = 36.11;
724            }
725            self.max_speed_s_per_mm = 1.0 / (speed * 1000.0);
726        }
727        let dist_limit_mm = if self.distance_limit_m > 0 {
728            self.distance_limit_m.wrapping_mul(1000)
729        } else {
730            i32::MAX
731        };
732        let time_limit = if self.time_limit > 0 {
733            self.time_limit
734        } else {
735            i32::MAX
736        };
737        let net = self.net;
738        while let Some((_, id)) = self.queue.poll() {
739            let s0 = self.states[id as usize].clone();
740            if s0.back_edge >= 0
741                && !self
742                    .at_edge
743                    .get(&(s0.back_edge as u32))
744                    .is_some_and(|l| l.contains(&id))
745            {
746                continue;
747            }
748            if s0.dur > self.best_at_dest {
749                break;
750            }
751            if let Some((vertex_stop, min_time)) = self.stop_search {
752                let stop = vertex_stop[s0.vertex as usize];
753                if stop >= 0 && s0.dur >= min_time {
754                    let e = self.stops.entry(stop as u32).or_insert_with(|| {
755                        self.stops_order.push(stop as u32);
756                        i32::MAX
757                    });
758                    if *e > s0.dur {
759                        *e = s0.dur;
760                    }
761                }
762                if self.stops.len() >= MAX_ACCESS_STOPS {
763                    self.queue.clear();
764                    break;
765                }
766            }
767            if let Some(d) = self.dest {
768                if s0.vertex == d.vertex0 || s0.vertex == d.vertex1 {
769                    if let Some(at) = self.state_at_split(&d) {
770                        if self.best_at_dest > at.dur {
771                            self.best_at_dest = at.dur;
772                        }
773                    }
774                }
775            }
776            let v = s0.vertex as usize;
777            let edges = if self.reverse {
778                net.in_edges(v)
779            } else {
780                net.out_edges(v)
781            };
782            for &e in edges {
783                let Some(mut s1) = self.traverse_with(&s0, id, e as usize, self.req) else {
784                    continue;
785                };
786                if s1.dist <= dist_limit_mm && s1.dur < time_limit && !self.is_dominated(&s1) {
787                    s1.heuristic = self.calc_heuristic(&s1);
788                    let nid = self.push_state(s1);
789                    self.put_at_edge(e, nid);
790                    self.queue.offer((self.key(nid), nid));
791                }
792            }
793        }
794    }

796    /// The best state at the end of edge `e` (`getStateAtEdge`: the least
797    /// time, ties to the later state).
798    fn state_at_edge(&self, e: u32) -> Option<u32> {
799        let list = self.at_edge.get(&e)?;
800        let mut best: Option<u32> = None;
801        for &id in list {
802            best = match best {
803                Some(b) if self.states[b as usize].dur < self.states[id as usize].dur => Some(b),
804                _ => Some(id),
805            };
806        }
807        best
808    }

810    /// `getStateAtVertex(v)`: over the edges into `v` (out of it in a
811    /// reverse search), the least time, ties to the first edge.
812    pub(crate) fn state_at_vertex(&self, v: u32) -> Option<u32> {
813        let net = self.net;
814        let edges = if self.reverse {
815            net.out_edges(v as usize)
816        } else {
817            net.in_edges(v as usize)
818        };
819        let mut ret: Option<u32> = None;
820        for &e in edges {
821            let Some(id) = self.state_at_edge(e) else {
822                continue;
823            };
824            ret = match ret {
825                Some(r) if self.states[r as usize].dur <= self.states[id as usize].dur => Some(r),
826                _ => Some(id),
827            };
828        }
829        ret
830    }

832    /// The candidate states of `getState(split)` (forward search), in R5's
833    /// order: from the edges into `vertex0` onto the split edge, then from
834    /// the edges into `vertex1` onto its reverse.
835    fn split_candidates(&mut self, split: &Split) -> Vec<St> {
836        let net = self.net;
837        let mut out = Vec::new();
838        for (edge, from_v, dist) in [
839            (split.edge, split.vertex0, split.distance0_mm),
840            (split.edge + 1, split.vertex1, split.distance1_mm),
841        ] {
842            for &inc in net.in_edges(from_v as usize) {
843                let Some(list) = self.at_edge.get(&inc).cloned() else {
844                    continue;
845                };
846                for id in list {
847                    let s = self.states[id as usize].clone();
848                    if self.can_turn_from(&s, edge).is_none() {
849                        continue;
850                    }
851                    let turn = if s.mode == Mode::Car && s.back_edge >= 0 {
852                        car_turn_cost(
853                            net.out_angle(s.back_edge as usize),
854                            net.in_angle(edge as usize),
855                        )
856                    } else {
857                        0
858                    };
859                    let speed = self.speed_on(edge as usize, s.mode);
860                    let trav = (f64::from(dist) / 1000.0 / f64::from(speed) + 0.5).floor() as i32;
861                    out.push(St {
862                        vertex: u32::MAX,
863                        back_edge: edge as i32,
864                        dur: s.dur.wrapping_add(turn + trav),
865                        dist: s.dist.wrapping_add(dist),
866                        mode: s.mode,
867                        back: id,
868                        tr: NONE,
869                        heuristic: 0,
870                    });
871                }
872            }
873        }
874        out
875    }

877    /// `getState(split)` without keeping the state.
878    fn state_at_split(&mut self, split: &Split) -> Option<St> {
879        let mut best: Option<St> = None;
880        for c in self.split_candidates(split) {
881            best = match best {
882                Some(b) if b.dur < c.dur => Some(b),
883                _ => Some(c),
884            };
885        }
886        best
887    }

889    /// `getState(split)` after the search, kept in the arena: the id of
890    /// the state at the destination, or `None` when unreachable.
891    pub(crate) fn final_state(&mut self, split: &Split) -> Option<u32> {
892        let s = self.state_at_split(split)?;
893        Some(self.push_state(s))
894    }

896    /// The states of the path ending at `id`, first to last.
897    pub(crate) fn path(&self, id: u32) -> Vec<u32> {
898        let mut out = Vec::new();
899        let mut cur = id;
900        while cur != NONE {
901            out.push(cur);
902            cur = self.states[cur as usize].back;
903        }
904        out.reverse();
905        out
906    }
