1//! Geometry helpers reproducing R5's arithmetic (Java `double`, `(int)` casts).
2
3/// Radius and correction used by R5's equirectangular `fastDistance`.
4const FAST_RADIUS_M: f64 = 6371.01 * 1000.0;
5const MAX_ERR_INV: f64 = 0.999462;
6/// Fixed-point factor for degrees.
7pub const FIXED_FACTOR: f64 = 1e7;
8
9/// Java `(int) x` for a double: truncation toward zero, saturating, NaN to 0.
10#[inline]
11pub fn java_int(x: f64) -> i32 {
12    x as i32
13}
14
15/// R5 node coordinate from OSM 1e-7 degrees. osmlib decodes a PBF coordinate
16/// as `nanodegrees * 1e-9` and truncates `lat * 1e7`, which can land one unit
17/// below the exact value.
18#[inline]
19pub fn node_fixed(e7: i32) -> i32 {
20    let deg = (i64::from(e7) * 100) as f64 * 1e-9;
21    java_int(deg * FIXED_FACTOR)
22}
23
24/// R5 vertex coordinate from a node's fixed coordinate: the vertex store
25/// converts `fixed / 1e7` back with `(int)(deg * 1e7)`.
26#[inline]
27pub fn vertex_fixed(node_fixed: i32) -> i32 {
28    java_int((f64::from(node_fixed) / FIXED_FACTOR) * FIXED_FACTOR)
29}

31/// Haversine distance in metres (R5 `GeometryUtils.distance`), with
32/// commons-math's `FastMath` as R5 computes it (the same bits on every
33/// platform; the system's `sin`, `cos` and `asin` differ in the last bit
34/// from one to another, which could move a length truncated to the
35/// millimetre).
36pub fn haversine_m(lat0: f64, lon0: f64, lat1: f64, lon1: f64) -> f64 {
37    crate::fastmath_trig::haversine_m(lat0, lon0, lat1, lon1)
38}
39
40/// Equirectangular distance in metres (R5 `SphericalDistanceLibrary.fastDistance`),
41/// points as `(lon, lat)` degrees.
42fn fast_distance(a: (f64, f64), b: (f64, f64)) -> f64 {
43    let dlat = (b.1 - a.1).to_radians();
44    let dlon = (b.0 - a.0).to_radians() * ((a.1 + b.1) / 2.0).to_radians().cos();
45    FAST_RADIUS_M * (dlat * dlat + dlon * dlon).sqrt() * MAX_ERR_INV
46}

48/// Initial bearing from `a` to `b`, radians, points as `(lon, lat)` degrees
49/// (R5 `DirectionUtils.getAzimuth`).
50fn azimuth(a: (f64, f64), b: (f64, f64)) -> f64 {
51    let lat1 = a.1.to_radians();
52    let lat2 = b.1.to_radians();
53    let dlng = (b.0 - a.0).to_radians();
54    let y = dlng.sin() * lat2.cos();
55    let x = lat1.cos() * lat2.sin() - lat1.sin() * lat2.cos() * dlng.cos();
56    if x.abs() < 1e-10 && y.abs() < 1e-10 {
57        return std::f64::consts::PI;
58    }
59    y.atan2(x)
60}

62/// Radians to binary radians (`(byte) Math.round(rad * 128 / PI)`).
63#[inline]
64pub fn radians_to_brads(rad: f64) -> i8 {
65    ((rad * 128.0 / std::f64::consts::PI + 0.5).floor() as i64) as i8
66}
67
68/// Binary radians to whole degrees (Java int division, toward zero).
69#[inline]
70pub fn brads_to_degrees(b: i8) -> i32 {
71    i32::from(b) * 180 / 128
72}

74const ANGLE_DIST_M: f64 = 10.0;
75
76/// First and last bearings of a line (R5 `getFirstAngle` / `getLastAngle`),
77/// points as `(lon, lat)` degrees.
78pub fn first_last_angles(pts: &[(f64, f64)]) -> (f64, f64) {
79    let n = pts.len();
80    if n < 2 {
81        return (0.0, 0.0);
82    }
83    if n == 2 {
84        let a = azimuth(pts[0], pts[1]);
85        return (a, a);
86    }
87    let first = 'f: {
88        let mut d = 0.0;
89        let mut i = 0;
90        while d < ANGLE_DIST_M && i < n - 1 {
91            let seg = fast_distance(pts[i], pts[i + 1]);
92            if d + seg > ANGLE_DIST_M {
93                break 'f azimuth(pts[0], pts[i + 1]);
94            }
95            d += seg;
96            i += 1;
97        }
98        azimuth(pts[0], pts[n - 1])
99    };
100    let last = 'l: {
101        let mut d = 0.0;
102        let mut i = n as isize - 2;
103        while d < ANGLE_DIST_M && i >= 0 {
104            let iu = i as usize;
105            let seg = fast_distance(pts[iu], pts[iu + 1]);
106            if d + seg > ANGLE_DIST_M {
107                break 'l azimuth(pts[iu], pts[n - 1]);
108            }
109            d += seg;
110            i -= 1;
111        }
112        azimuth(pts[0], pts[n - 1])
113    };
114    (first, last)
115}

117/// Fraction along a segment of the projection of a point, all fixed-point,
118/// with R5's stretched projection: longitudes are *divided* by `cos_lat`
119/// (`GeometryUtils.segmentFraction` + JTS `LineSegment.segmentFraction`).
120pub fn segment_fraction(
121    lon0: i32,
122    lat0: i32,
123    lon1: i32,
124    lat1: i32,
125    lon: i32,
126    lat: i32,
127    cos_lat: f64,
128) -> f64 {
129    let x0 = f64::from(java_int(f64::from(lon0) / cos_lat));
130    let x1 = f64::from(java_int(f64::from(lon1) / cos_lat));
131    let x = f64::from(java_int(f64::from(lon) / cos_lat));
132    let (y0, y1, y) = (f64::from(lat0), f64::from(lat1), f64::from(lat));
133    let r = if x == x0 && y == y0 {
134        0.0
135    } else if x == x1 && y == y1 {
136        1.0
137    } else {
138        let dx = x1 - x0;
139        let dy = y1 - y0;
140        let len = dx * dx + dy * dy;
141        if len <= 0.0 {
142            f64::NAN
143        } else {
144            ((x - x0) * dx + (y - y0) * dy) / len
145        }
146    };
147    if r < 0.0 {
148        0.0
149    } else if r > 1.0 || r.is_nan() {
150        1.0
151    } else {
152        r
153    }
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    #[test]
161    fn brads_roundtrip_like_java() {
162        assert_eq!(radians_to_brads(std::f64::consts::PI), -128); // (byte) 128
163        assert_eq!(radians_to_brads(-std::f64::consts::PI), -128);
164        assert_eq!(radians_to_brads(std::f64::consts::FRAC_PI_2), 64);
165        assert_eq!(brads_to_degrees(64), 90);
166        assert_eq!(brads_to_degrees(-64), -90);
167        assert_eq!(brads_to_degrees(-128), -180);
168        assert_eq!(brads_to_degrees(1), 1); // 180/128 truncated
169    }
170
171    #[test]
172    fn haversine_one_degree_of_latitude() {
173        let d = haversine_m(0.0, 0.0, 1.0, 0.0);
174        assert!((d - 6_367_450.0 * std::f64::consts::PI / 180.0).abs() < 1e-6);
175    }
176
177    #[test]
178    fn fixed_point_conversions_truncate() {
179        // -30.0538460 deg: 1e-9 * nanodegrees then (int)(x * 1e7)
180        let f = node_fixed(-300_538_460);
181        assert!((f - -300_538_460).abs() <= 1);
182        assert!((vertex_fixed(f) - f).abs() <= 1);
183    }
184
185    #[test]
186    fn segment_fraction_clamps() {
187        let c = 1.0;
188        assert_eq!(segment_fraction(0, 0, 10, 0, 5, 3, c), 0.5);
189        assert_eq!(segment_fraction(0, 0, 10, 0, -5, 3, c), 0.0);
190        assert_eq!(segment_fraction(0, 0, 10, 0, 50, 3, c), 1.0);
191        assert_eq!(segment_fraction(0, 0, 0, 0, 5, 3, c), 1.0); // degenerate: NaN -> 1
192    }
