1//! Contraction of a graph with integer arc times into a hierarchy.
2
3use super::NONE;
4use crate::CancelToken;
5use std::cmp::Reverse;
6use std::collections::BinaryHeap;
7use std::sync::atomic::{AtomicUsize, Ordering};

9/// A contraction hierarchy with nodes renumbered by rank, highest first
10/// (`id[node]`), so a sweep from the top runs through memory in order.
11pub(super) struct Hierarchy {
12    pub(super) id: Vec<u32>,
13    /// Upward arcs `(to, seconds)` of each node, to lower ids.
14    pub(super) up_off: Vec<u32>,
15    pub(super) up: Vec<(u32, u32)>,
16    /// Downward arcs into each node `(from, seconds)`, from lower ids.
17    pub(super) down_off: Vec<u32>,
18    pub(super) down: Vec<(u32, u32)>,
19    pub(super) shortcuts: usize,
20    pub(super) rounds: usize,
21}
22
23/// Dijkstra scratch for witness searches.
24struct Witness {
25    dist: Vec<u32>,
26    touched: Vec<u32>,
27    heap: BinaryHeap<Reverse<(u32, u32)>>,
28    /// Run stamp of the nodes a run looks for, so it stops once all are
29    /// settled.
30    target: Vec<u32>,
31    stamp: u32,
32    /// The shortcuts of the last [`Witness::shortcuts_of`].
33    found: Vec<(u32, u32, u32)>,
34    settle: usize,
35}
36
37/// Nodes a witness search may settle before it gives up (a shortcut is
38/// then added; more shortcuts than needed never change a time).
39const WITNESS_SETTLE: usize = 32;
40/// The same when only estimating a node's priority: a rough count is
41/// enough, and these searches are most of the contraction's work.
42const ESTIMATE_SETTLE: usize = 2;
43/// Fewer items than this are not worth spawning threads for.
44const PAR_MIN_ITEMS: usize = 64;
45/// Items a worker takes at a time: their costs vary a lot (the last nodes
46/// have many neighbours), so the workers share them out as they go rather
47/// than in fixed shares.
48const PAR_GRAIN: usize = 16;
49
50impl Witness {
51    fn new(n: usize) -> Self {
52        Witness {
53            dist: vec![NONE; n],
54            touched: Vec::new(),
55            heap: BinaryHeap::new(),
56            target: vec![0; n],
57            stamp: 0,
58            found: Vec::new(),
59            settle: WITNESS_SETTLE,
60        }
61    }
62
63    /// Least time from `s` in the remaining graph without `skip` and the
64    /// `blocked` nodes, as far as `bound` and the settle limit allow, until
65    /// the `targets` marked with the current stamp are settled.
66    fn run(
67        &mut self,
68        out: &[Vec<(u32, u32)>],
69        blocked: &[bool],
70        skip: u32,
71        s: u32,
72        bound: u32,
73        mut targets: usize,
74    ) {
75        for &v in &self.touched {
76            self.dist[v as usize] = NONE;
77        }
78        self.touched.clear();
79        self.heap.clear();
80        self.dist[s as usize] = 0;
81        self.touched.push(s);
82        self.heap.push(Reverse((0, s)));
83        let mut settled = 0;
84        while let Some(Reverse((d, u))) = self.heap.pop() {
85            if d > self.dist[u as usize] {
86                continue;
87            }
88            if d > bound || settled >= self.settle {
89                break;
90            }
91            settled += 1;
92            if self.target[u as usize] == self.stamp {
93                targets -= 1;
94                if targets == 0 {
95                    break;
96                }
97            }
98            for &(v, w) in &out[u as usize] {
99                if v == skip || blocked[v as usize] {
100                    continue;
101                }
102                let nd = d.saturating_add(w);
103                if nd < self.dist[v as usize] {
104                    if self.dist[v as usize] == NONE {
105                        self.touched.push(v);
106                    }
107                    self.dist[v as usize] = nd;
108                    self.heap.push(Reverse((nd, v)));
109                }
110            }
111        }
112    }
113
114    /// Into `self.found`, the shortcuts `(u, v, seconds)` that contracting
115    /// `x` needs: one per pair of neighbours `u -> x -> v` without a path
116    /// of no more time that avoids `x` and the `blocked` nodes.
117    fn shortcuts_of(
118        &mut self,
119        out: &[Vec<(u32, u32)>],
120        inn: &[Vec<(u32, u32)>],
121        blocked: &[bool],
122        x: u32,
123    ) {
124        self.found.clear();
125        let outs = &out[x as usize];
126        for &(u, wu) in &inn[x as usize] {
127            self.stamp = self.stamp.wrapping_add(1);
128            if self.stamp == 0 {
129                self.target.iter_mut().for_each(|t| *t = 0);
130                self.stamp = 1;
131            }
132            let mut bound = 0u32;
133            let mut targets = 0;
134            for &(v, wv) in outs {
135                if v != u {
136                    bound = bound.max(wu.saturating_add(wv));
137                    self.target[v as usize] = self.stamp;
138                    targets += 1;
139                }
140            }
141            if targets == 0 {
142                continue;
143            }
144            self.run(out, blocked, x, u, bound, targets);
145            for &(v, wv) in outs {
146                let via = wu.saturating_add(wv);
147                if v != u && self.dist[v as usize] > via {
148                    self.found.push((u, v, via));
149                }
150            }
151        }
152    }
153}

155/// Add arc `u -> v` or lower its time.
156fn add_arc(list: &mut Vec<(u32, u32)>, v: u32, w: u32) {
157    match list.iter_mut().find(|a| a.0 == v) {
158        Some(a) => a.1 = a.1.min(w),
159        None => list.push((v, w)),
160    }
161}

163/// `f(worker scratch, item)` over `items`, over the workers when there are
164/// enough items to pay for the threads; results in item order. A result
165/// does not depend on the worker that computes it (a witness search starts
166/// from clean scratch), so neither does the hierarchy.
167fn par_map<T: Send>(
168    items: &[u32],
169    workers: &mut [Witness],
170    f: impl Fn(&mut Witness, u32) -> T + Sync,
171) -> Vec<T> {
172    if workers.len() == 1 || items.len() < PAR_MIN_ITEMS {
173        let w = &mut workers[0];
174        return items.iter().map(|&x| f(w, x)).collect();
175    }
176    let next = AtomicUsize::new(0);
177    let parts: Vec<Vec<(usize, T)>> = crate::sys::scoped(workers.iter_mut(), |w| {
178        let mut part = Vec::new();
179        loop {
180            let a = next.fetch_add(PAR_GRAIN, Ordering::Relaxed);
181            if a >= items.len() {
182                break;
183            }
184            let b = (a + PAR_GRAIN).min(items.len());
185            part.extend((a..b).map(|i| (i, f(w, items[i]))));
186        }
187        part
188    });
189    let mut slots: Vec<Option<T>> = (0..items.len()).map(|_| None).collect();
190    for (i, r) in parts.into_iter().flatten() {
191        slots[i] = Some(r);
192    }
193    slots
194        .into_iter()
195        .map(|r| r.expect("every item is mapped"))
196        .collect()
197}

199/// Shortcuts added against arcs removed, plus the neighbours already
200/// contracted and the depth, which spread the contraction evenly.
201fn priority(
202    out: &[Vec<(u32, u32)>],
203    inn: &[Vec<(u32, u32)>],
204    deleted: &[i32],
205    level: &[i32],
206    none: &[bool],
207    ws: &mut Witness,
208    x: u32,
209) -> i32 {
210    ws.settle = ESTIMATE_SETTLE;
211    ws.shortcuts_of(out, inn, none, x);
212    let xu = x as usize;
213    let degree = (out[xu].len() + inn[xu].len()) as i32;
214    4 * ws.found.len() as i32 - 2 * degree + 2 * deleted[xu] + level[xu]
215}
216
217impl Hierarchy {
218    /// Contract the graph of `n` nodes and arcs `(from, to, seconds)` in
219    /// rounds. Each round takes the remaining nodes whose priority is below
220    /// that of every remaining neighbour (so no two are adjacent), finds
221    /// their shortcuts in parallel with witness searches that avoid all of
222    /// them, then removes them. A path through a node `x` of the round runs
223    /// `u -> x -> v` with `u` and `v` outside the round, so it is replaced
224    /// by a shortcut or by a witness that avoids the round: every time
225    /// between the remaining nodes is kept. The result does not depend on
226    /// the thread count. `None` when `cancel` is cancelled first (checked
227    /// before every round).
228    pub(super) fn build(
229        n: usize,
230        arcs: &[(u32, u32, u32)],
231        n_threads: usize,
232        cancel: &CancelToken,
233    ) -> Option<Self> {
234        if cancel.is_cancelled() {
235            return None;
236        }
237        let mut out: Vec<Vec<(u32, u32)>> = vec![Vec::new(); n];
238        let mut inn: Vec<Vec<(u32, u32)>> = vec![Vec::new(); n];
239        for &(u, v, w) in arcs {
240            if u != v {
241                // a loop never shortens a path whose steps all cost >= 1
242                add_arc(&mut out[u as usize], v, w);
243                add_arc(&mut inn[v as usize], u, w);
244            }
245        }
246        let mut workers: Vec<Witness> = (0..n_threads.max(1)).map(|_| Witness::new(n)).collect();
247        let mut deleted = vec![0i32; n];
248        let mut level = vec![0i32; n];
249        let mut blocked = vec![false; n];
250        let mut done = vec![false; n];
251        let mut remaining: Vec<u32> = (0..n as u32).collect();
252        let mut prio = par_map(&remaining, &mut workers, |ws, x| {
253            priority(&out, &inn, &deleted, &level, &blocked, ws, x)
254        });
255        // Each node's arcs to the nodes still there when it goes: those all
256        // rank higher.
257        let mut up_arcs: Vec<Vec<(u32, u32)>> = vec![Vec::new(); n];
258        let mut down_arcs: Vec<Vec<(u32, u32)>> = vec![Vec::new(); n];
259        let mut order = Vec::with_capacity(n);
260        let mut shortcuts = 0;
261        let mut rounds = 0;
262        let mut round: Vec<u32> = Vec::new();
263        let mut dirty: Vec<u32> = Vec::new();
264        let mut is_dirty = vec![false; n];
265        while !remaining.is_empty() {
266            if cancel.is_cancelled() {
267                return None;
268            }
269            rounds += 1;
270            let key = |x: u32| (prio[x as usize], x);
271            round.clear();
272            round.extend(remaining.iter().copied().filter(|&x| {
273                let k = key(x);
274                out[x as usize]
275                    .iter()
276                    .chain(&inn[x as usize])
277                    .all(|&(y, _)| k < key(y))
278            }));
279            for &x in &round {
280                blocked[x as usize] = true;
281            }
282            let found = par_map(&round, &mut workers, |ws, x| {
283                ws.settle = WITNESS_SETTLE;
284                ws.shortcuts_of(&out, &inn, &blocked, x);
285                ws.found.clone()
286            });
287            dirty.clear();
288            for (&x, sc) in round.iter().zip(found) {
289                let xu = x as usize;
290                blocked[xu] = false;
291                done[xu] = true;
292                order.push(x);
293                up_arcs[xu] = std::mem::take(&mut out[xu]);
294                down_arcs[xu] = std::mem::take(&mut inn[xu]);
295                for &(v, _) in &up_arcs[xu] {
296                    inn[v as usize].retain(|a| a.0 != x);
297                }
298                for &(u, _) in &down_arcs[xu] {
299                    out[u as usize].retain(|a| a.0 != x);
300                }
301                for &(y, _) in up_arcs[xu].iter().chain(&down_arcs[xu]) {
302                    let yu = y as usize;
303                    deleted[yu] += 1;
304                    level[yu] = level[yu].max(level[xu] + 1);
305                    if !is_dirty[yu] {
306                        is_dirty[yu] = true;
307                        dirty.push(y);
308                    }
309                }
310                for &(u, v, w) in &sc {
311                    add_arc(&mut out[u as usize], v, w);
312                    add_arc(&mut inn[v as usize], u, w);
313                }
314                shortcuts += sc.len();
315            }
316            remaining.retain(|&x| !done[x as usize]);
317            let fresh = par_map(&dirty, &mut workers, |ws, y| {
318                priority(&out, &inn, &deleted, &level, &blocked, ws, y)
319            });
320            for (&y, p) in dirty.iter().zip(fresh) {
321                prio[y as usize] = p;
322                is_dirty[y as usize] = false;
323            }
324        }
325        // highest rank first: the last round's nodes get the lowest ids
326        let mut id = vec![0u32; n];
327        for (r, &x) in order.iter().rev().enumerate() {
328            id[x as usize] = r as u32;
329        }
330        let mut up_off = Vec::with_capacity(n + 1);
331        let mut down_off = Vec::with_capacity(n + 1);
332        up_off.push(0);
333        down_off.push(0);
334        let mut up = Vec::new();
335        let mut down = Vec::new();
336        for &x in order.iter().rev() {
337            let xu = x as usize;
338            up.extend(up_arcs[xu].iter().map(|&(v, w)| (id[v as usize], w)));
339            down.extend(down_arcs[xu].iter().map(|&(u, w)| (id[u as usize], w)));
340            up_off.push(up.len() as u32);
341            down_off.push(down.len() as u32);
342        }
343        Some(Hierarchy {
344            id,
345            up_off,
346            up,
347            down_off,
348            down,
349            shortcuts,
350            rounds,
351        })
352    }
353
354    pub(super) fn n(&self) -> usize {
355        self.id.len()
356    }
357
358    /// Memory held, in bytes.
359    pub(super) fn bytes(&self) -> usize {
360        4 * (self.id.len() + self.up_off.len() + self.down_off.len())
361            + 8 * (self.up.len() + self.down.len())
362    }
