8/// A turn restriction over directed edges: from `from_edge`, through
9/// `via_edges` in order, onto `to_edge`. A `no_` restriction forbids reaching
10/// `to_edge` that way; an `only_` one forbids everything else.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct TurnRestriction {
13    /// OSM relation id.
14    pub relation: i64,
15    pub only: bool,
16    pub from_edge: u32,
17    pub via_edges: Vec<u32>,
18    pub to_edge: u32,
19}
20
21/// Most partial paths the via-way search keeps. R5 enumerates every
22/// ordering of the via ways that chains from the from-way, which grows
23/// factorially when via ways share nodes; a relation that needs more paths
24/// than this is dropped, as R5 drops one it cannot resolve. Real
25/// restrictions (one to a few via ways in a chain) need a handful.
26const MAX_VIA_PATHS: usize = 1_000;
27
28/// Build the network's restrictions. `vertex_of` maps an OSM node id to the
29/// last vertex created for it (R5 `vertexIndexForOsmNode`).
30pub(crate) fn apply_turn_restrictions(
31    net: &mut StreetNetwork,
32    ex: &Extract,
33    vertex_of: &dyn Fn(i64) -> Option<u32>,
34) {
35    let ways: HashMap<i64, &Way> = ex.ways.iter().map(|w| (w.id, w)).collect();
36    let mut relations: Vec<_> = ex.relations.iter().collect();
37    relations.sort_by_key(|r| r.id);
38    for rel in relations {
39        if let Some(r) = restriction_from_relation(net, ex, &ways, vertex_of, rel) {
40            // StreetLayer.addReverseTurnRestriction
41            if r.only {
42                let remapped = remap(net, &r);
43                net.reverse_restrictions.extend(remapped);
44            } else {
45                net.reverse_restrictions.push(r.clone());
46            }
47            net.restrictions.push(r);
48        }
49    }
50    net.restriction_starts = net
51        .restrictions
52        .iter()
53        .enumerate()
54        .map(|(i, r)| (r.from_edge, i as u32))
55        .collect();
56    net.restriction_starts.sort_unstable();
57    for &(e, _) in &net.restriction_starts {
58        net.edge_flags[e as usize] |= super::flags::RESTRICTION_START;
59    }
60}

62/// `TurnRestriction.remap`: an `only_` restriction as the `no_` ones that
63/// R5's reverse searches use instead. From the end of the from edge and of
64/// each via edge, every edge out of it that cars or bicycles may use and
65/// that is not the next one of the restriction gets a `no_` restriction
66/// from the from edge, through the via edges so far, onto that edge. The
67/// edges out of a vertex are the lists as they are when restrictions are
68/// built (R5's `outgoingEdges` before they are rebuilt).
69fn remap(net: &StreetNetwork, r: &TurnRestriction) -> Vec<TurnRestriction> {
70    use super::flags::{ALLOWS_BIKE, ALLOWS_CAR};
71    let mut out = Vec::new();
72    let n = r.via_edges.len();
73    for pos in 0..=n {
74        // pos 0 is R5's -1: the from edge
75        let current = if pos == 0 {
76            r.from_edge
77        } else {
78            r.via_edges[pos - 1]
79        };
80        let next = if pos < n { r.via_edges[pos] } else { r.to_edge };
81        let v = net.edge_to(current as usize);
82        for &e in net.out_edges(v) {
83            if e == next {
84                continue;
85            }
86            if net.edge_flags[e as usize] & (ALLOWS_CAR | ALLOWS_BIKE) != 0 {
87                out.push(TurnRestriction {
88                    relation: r.relation,
89                    only: false,
90                    from_edge: r.from_edge,
91                    via_edges: r.via_edges[..pos].to_vec(),
92                    to_edge: e,
93                });
94            }
95        }
96    }
97    out
98}

100/// The unique edge among `edges` whose OSM way is `way`; `Err` if there are
101/// two (R5 treats that as ambiguous), `Ok(None)` if there is none.
102fn unique_edge_of_way(net: &StreetNetwork, edges: &[u32], way: i64) -> Result<Option<u32>, ()> {
103    let mut found = None;
104    for &e in edges {
105        if net.pair_way[e as usize / 2] == way {
106            if found.is_some() {
107                return Err(());
108            }
109            found = Some(e);
110        }
111    }
112    Ok(found)
113}

115fn restriction_from_relation(
116    net: &StreetNetwork,
117    ex: &Extract,
118    ways: &HashMap<i64, &Way>,
119    vertex_of: &dyn Fn(i64) -> Option<u32>,
120    rel: &crate::osm::pbf::Relation,
121) -> Option<TurnRestriction> {
122    let kind = rel.tags.get("restriction")?;
123    let only = if kind.starts_with("no_") {
124        false
125    } else if kind.starts_with("only_") {
126        true
127    } else {
128        return None;
129    };
130    let mut from = None;
131    let mut to = None;
132    let mut via = Vec::new();
133    for m in &rel.members {
134        match m.role.as_str() {
135            "from" => {
136                if from.is_some() || m.kind != MemberKind::Way {
137                    return None;
138                }
139                from = Some(m);
140            }
141            "to" => {
142                if to.is_some() || m.kind != MemberKind::Way {
143                    return None;
144                }
145                to = Some(m);
146            }
147            "via" => via.push(m),
148            _ => {}
149        }
150        // every way and node member must exist in the file
151        let present = match m.kind {
152            MemberKind::Way => ex.member_ways_present.binary_search(&m.id).is_ok(),
153            MemberKind::Node => ex.member_nodes_present.binary_search(&m.id).is_ok(),
154            MemberKind::Relation => true,
155        };
156        if !present {
157            return None;
158        }
159    }
160    let (from, to) = (from?, to?);
161    if via.is_empty() {
162        return None;
163    }
164    let via_ways = via.iter().any(|m| m.kind == MemberKind::Way);
165    let via_nodes = via.iter().any(|m| m.kind == MemberKind::Node);
166    if via.iter().any(|m| m.kind == MemberKind::Relation)
167        || (via_ways && via_nodes)
168        || (via_nodes && via.len() > 1)
169    {
170        return None;
171    }
172
173    if via_nodes {
174        let v = vertex_of(via[0].id)? as usize;
175        let from_edge = unique_edge_of_way(net, net.in_edges(v), from.id).ok()?;
176        let to_edge = unique_edge_of_way(net, net.out_edges(v), to.id).ok()?;
177        return Some(TurnRestriction {
178            relation: rel.id,
179            only,
180            from_edge: from_edge?,
181            via_edges: Vec::new(),
182            to_edge: to_edge?,
183        });
184    }
185
186    // Via ways: find the unique node path from the from-way through the via
187    // ways to the to-way. Ways that are not routable have no edges, so the
188    // restriction is dropped as R5 eventually does. The search gives up past
189    // MAX_VIA_PATHS partial paths.
190    let from_way = ways.get(&from.id)?;
191    let to_way = ways.get(&to.id)?;
192    let via_ids: Vec<i64> = via.iter().map(|m| m.id).collect();
193    let via_way_nodes: Vec<&[i64]> = via_ids
194        .iter()
195        .map(|id| ways.get(id).map(|w| w.nodes.as_slice()))
196        .collect::<Option<_>>()?;
197
198    let mut nodes: Vec<Vec<i64>> = Vec::new();
199    let mut paths: Vec<Vec<i64>> = Vec::new();
200    for &node in &from_way.nodes {
201        for (pos, vn) in via_way_nodes.iter().enumerate() {
202            for &via_node in *vn {
203                if node == via_node {
204                    if nodes.len() == MAX_VIA_PATHS {
205                        return None;
206                    }
207                    nodes.push(vec![node]);
208                    paths.push(vec![via_ids[pos]]);
209                }
210            }
211        }
212    }
213    for round in 0..via.len() - 1 {
214        let (prev_nodes, prev_paths) = (std::mem::take(&mut nodes), std::mem::take(&mut paths));
215        for (pn, pw) in prev_nodes.iter().zip(&prev_paths) {
216            let way = ways.get(&pw[round])?;
217            for &node in &way.nodes {
218                for (pos, &via_id) in via_ids.iter().enumerate() {
219                    if pw.contains(&via_id) {
220                        continue;
221                    }
222                    for &via_node in via_way_nodes[pos] {
223                        if via_node == node {
224                            if nodes.len() == MAX_VIA_PATHS {
225                                return None;
226                            }
227                            let mut nn = pn.clone();
228                            nn.push(node);
229                            let mut nw = pw.clone();
230                            nw.push(via_id);
231                            nodes.push(nn);
232                            paths.push(nw);
233                        }
234                    }
235                }
236            }
237        }
238    }
239    let mut path_nodes: Option<Vec<i64>> = None;
240    let mut path_ways: &[i64] = &[];
241    for (pn, pw) in nodes.iter().zip(&paths) {
242        let final_way = ways.get(pw.last()?)?;
243        for &node in &final_way.nodes {
244            for &to_node in &to_way.nodes {
245                if node == to_node {
246                    if path_nodes.is_some() {
247                        return None; // ambiguous
248                    }
249                    let mut p = pn.clone();
250                    p.push(node);
251                    path_nodes = Some(p);
252                    path_ways = pw;
253                }
254            }
255        }
256    }
257    let path_nodes = path_nodes?;
258
259    let from_vertex = vertex_of(path_nodes[0])? as usize;
260    let from_edge = unique_edge_of_way(net, net.in_edges(from_vertex), from.id).ok()?;
261    let to_vertex = vertex_of(*path_nodes.last()?)? as usize;
262    let to_edge = unique_edge_of_way(net, net.out_edges(to_vertex), to.id).ok()?;
263    let (from_edge, to_edge) = (from_edge?, to_edge?);
264    let mut via_edges = Vec::new();
265    for nidx in (1..path_nodes.len()).rev() {
266        let way_id = path_ways[nidx - 1];
267        let v = vertex_of(path_nodes[nidx])? as usize;
268        via_edges.push(unique_edge_of_way(net, net.in_edges(v), way_id).ok()??);
269    }
270    via_edges.reverse();
271    Some(TurnRestriction {
272        relation: rel.id,
273        only,
274        from_edge,
275        via_edges,
276        to_edge,
277    })
278}

299    /// A 7 x 7 grid of two-way residential streets, nodes 1e-3 degrees apart
300    /// near the equator. Row `r` is way `1000 + r` and column `c` way
301    /// `2000 + c`, except the central row and column, which are split at the
302    /// centre as OSM usually does around a restricted junction: row 3 is
303    /// `1003` (west half) and `1103` (east half), column 3 is `2003` (rows
304    /// 0-3) and `2103` (rows 3-6).
305    fn grid(relations: Vec<Relation>) -> Extract {
306        let tags = || Tags(vec![("highway".into(), "residential".into())]);
307        let way = |id: i64, nodes: Vec<i64>| Way {
308            id,
309            tags: tags(),
310            nodes,
311        };
312        let mut ways = Vec::new();
313        for r in 0..N {
314            let row: Vec<i64> = (0..N).map(|c| node(r, c)).collect();
315            if r == C {
316                ways.push(way(1003, row[..=C].to_vec()));
317                ways.push(way(1103, row[C..].to_vec()));
318            } else {
319                ways.push(way(1000 + r as i64, row));
320            }
321        }
322        for c in 0..N {
323            let col: Vec<i64> = (0..N).map(|r| node(r, c)).collect();
324            if c == C {
325                ways.push(way(2003, col[..=C].to_vec()));
326                ways.push(way(2103, col[C..].to_vec()));
327            } else {
328                ways.push(way(2000 + c as i64, col));
329            }
330        }
331        let node_ids: Vec<i64> = (0..N * N).map(|i| 1 + i as i64).collect();
332        let lat = node_ids
333            .iter()
334            .map(|&i| (((i - 1) as usize / N) * 10_000) as i32)
335            .collect();
336        let lon = node_ids
337            .iter()
338            .map(|&i| (((i - 1) as usize % N) * 10_000) as i32)
339            .collect();
340        let mut member_ways_present: Vec<i64> = ways.iter().map(|w| w.id).collect();
341        member_ways_present.sort_unstable();
342        Extract {
343            ref_counts: vec![2; node_ids.len()], // every node is an intersection
344            member_nodes_present: node_ids.clone(),
345            member_ways_present,
346            node_ids,
347            lat,
348            lon,
349            ways,
350            relations,
351            node_tags: Vec::new(),
352            pois: Vec::new(),
353        }
354    }

484        use crate::street::p2p::{Req, Router};
485        use std::collections::HashMap;
486        let mut net = network_from_extract(grid(Vec::new()));
487        let vertex = |r: usize, c: usize| {

489                .find(|&v| {

493        };

497                .unwrap() as u32
498        };
499        // eastbound along row 3: no going straight from column 0 to 4
500        let e: Vec<u32> = (0..4)
501            .map(|c| edge(vertex(C, c), vertex(C, c + 1)))
502            .collect();
503        let r = TurnRestriction {
504            relation: 7,
505            only: false,
506            from_edge: e[0],
507            via_edges: vec![e[1], e[2]],
508            to_edge: e[3],
509        };
510        net.restrictions.push(r.clone());
511        net.reverse_restrictions.push(r);

514        let mut scratch = Vec::new();
515        let on = |c: f64| (C as f64 * 1e-3, c * 1e-3);
516        let split = |p: (f64, f64), scratch: &mut Vec<u32>| {
517            crate::street::link::find_split(&net, p.0, p.1, Mode::Car, scratch).unwrap()
518        };
519        let (o, d) = (split(on(0.5), &mut scratch), split(on(3.5), &mut scratch));
520        let seconds = |fresh: Option<HashMap<u64, bool>>| {
521            let mut router = Router::new(&net, Mode::Car, Req::new(3.6, 12.0, 2), false);
522            router.time_limit = 3600;
523            router.via_fresh = fresh;
524            router.set_origin_split(o);
525            router.set_destination(d);
526            router.route();
527            let last = router.final_state(&d).unwrap();
528            (router.states[last as usize].dur, router.via_used.clone())
529        };
530        let (built, used) = seconds(Some(HashMap::new()));
531        assert_eq!(seconds(None).0, built);
532        // the restriction's array reversed: the straight way is open

538        // a reverse search reverses the array on every check

540        back.time_limit = 3600;
541        back.via_fresh = Some(HashMap::new());
542        assert!(back.set_egress_origin(d));
543        back.route();
544        assert!(back.via_fresh.as_ref().unwrap().contains_key(&7));
545    }

583        assert!(net.restrictions.is_empty());
584        assert!(t.elapsed().as_secs() < 5, "{:?}", t.elapsed());
585    }

587    #[test]
588    fn restrictions_force_a_detour_only_where_they_apply() {
589        let free = car_seconds(vec![]);
590        // forbidding the turn the direct route takes costs a detour
591        let no_turn = car_seconds(vec![restriction("no_right_turn", 1003, node(C, C), 2003)]);
592        assert!(no_turn > free, "{no_turn} > {free}");
593        // "only straight on" forbids it too
594        let only = car_seconds(vec![restriction(
595            "only_straight_on",
596            1003,
597            node(C, C),
598            1103,
599        )]);
600        assert!(only > free, "{only} > {free}");
601        // a restriction on another turn changes nothing
602        let other = car_seconds(vec![restriction("no_left_turn", 1003, node(C, C), 2103)]);
603        assert_eq!(other, free);
604        // an "only" restriction that the direct route obeys changes nothing
605        let obeyed = car_seconds(vec![restriction("only_right_turn", 1003, node(C, C), 2003)]);
606        assert_eq!(obeyed, free);
607    }
