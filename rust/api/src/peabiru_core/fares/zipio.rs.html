1//! The fare structure file: a `.zip` holding five CSV files, in the format
2//! of `data.table::fwrite()`, read back as `data.table::fread()` reads it
3//! (`write_fare_structure()`, `read_fare_structure()`).
4//!
5//! The CSV text is `fwrite`'s (its quoting, its numbers, `\n` line ends);
6//! the archive around it is deflated as `zip::zip()`'s, with the same
7//! entries in the same order, but dated 1980-01-01 and compressed by
8//! miniz_oxide, so the `.zip` bytes differ from those of `zip::zip()` (whose bytes carry
9//! the files' modification times anyway) while every file in it is the
10//! same.
11

16/// A column of a table written as `fwrite` writes it.
17#[derive(Debug, Clone)]
18pub enum FwriteColumn {
19    /// Character; `None` is `NA` (written empty).
20    Text(Vec<Option<String>>),
21    /// Double; NaN is `NA` (written empty).
22    Number(Vec<f64>),
23    /// Integer; `None` is `NA`.
24    Integer(Vec<Option<i32>>),
25    /// Logical; `None` is `NA`.
26    Logical(Vec<Option<bool>>),
27}
28
29impl FwriteColumn {
30    fn len(&self) -> usize {
31        match self {
32            FwriteColumn::Text(v) => v.len(),
33            FwriteColumn::Number(v) => v.len(),
34            FwriteColumn::Integer(v) => v.len(),
35            FwriteColumn::Logical(v) => v.len(),
36        }
37    }
38}

40/// `fwrite`'s text of a character field (`quote = "auto"`): quoted, with
41/// quotes doubled, when it holds a comma, a quote or a line end, or is
42/// empty (an empty string is `""`, `NA` nothing).
43fn fwrite_text(s: &str, out: &mut String) {
44    if s.is_empty() || s.contains([',', '"', '\n', '\r']) {
45        out.push('"');
46        out.push_str(&s.replace('"', "\"\""));
47        out.push('"');
48    } else {
49        out.push_str(s);
50    }
51}

53/// `fwrite`'s text of a double: up to 15 significant digits, trailing
54/// zeros dropped, in fixed or scientific notation, whichever is shorter
55/// (fixed on a tie; `scipen = 0`).
56pub(crate) fn fwrite_number(x: f64, out: &mut String) {
57    if x.is_nan() {
58        return;
59    }
60    if x.is_infinite() {
61        out.push_str(if x > 0.0 { "Inf" } else { "-Inf" });
62        return;
63    }
64    if x == 0.0 {
65        out.push('0');
66        return;
67    }
68    // 15 significant digits, correctly rounded
69    let e = format!("{:.14e}", x);
70    let (mant, exp) = e.split_once('e').expect("exponent");
71    let exp: i32 = exp.parse().expect("exponent");
72    let neg = mant.starts_with('-');
73    let digits: String = mant.chars().filter(char::is_ascii_digit).collect();
74    let digits = digits.trim_end_matches('0');
75    let digits = if digits.is_empty() { "0" } else { digits };
76    let nd = digits.len() as i32;
77    // fixed notation
78    let mut fixed = String::new();
79    if exp >= 0 {
80        if nd <= exp + 1 {
81            fixed.push_str(digits);
82            for _ in nd..exp + 1 {
83                fixed.push('0');
84            }
85        } else {
86            fixed.push_str(&digits[..(exp + 1) as usize]);
87            fixed.push('.');
88            fixed.push_str(&digits[(exp + 1) as usize..]);
89        }
90    } else {
91        fixed.push_str("0.");
92        for _ in 0..(-exp - 1) {
93            fixed.push('0');
94        }
95        fixed.push_str(digits);
96    }
97    // scientific notation, at least two exponent digits
98    let mut sci = String::new();
99    sci.push_str(&digits[..1]);
100    if nd > 1 {
101        sci.push('.');
102        sci.push_str(&digits[1..]);
103    }
104    sci.push('e');
105    sci.push(if exp < 0 { '-' } else { '+' });
106    let a = exp.unsigned_abs();
107    if a < 10 {
108        sci.push('0');
109    }
110    sci.push_str(&a.to_string());
111    if neg {
112        out.push('-');
113    }
114    out.push_str(if fixed.len() > sci.len() {
115        &sci
116    } else {
117        &fixed
118    });
119}

133/// [`fwrite_csv`] with the line end given (`fwrite`'s `eol`, `"\r\n"` by
134/// default on Windows); line breaks inside quoted fields are left alone.
135fn fwrite_csv_eol(columns: &[(String, FwriteColumn)], eol: &str) -> Result<String, RouteError> {
136    let n = columns.first().map_or(0, |c| c.1.len());
137    if columns.iter().any(|c| c.1.len() != n) {
138        return Err(RouteError::Invalid(
139            "the columns of a fare structure table differ in length".into(),
140        ));
141    }
142    let mut s = String::new();
143    for (k, (name, _)) in columns.iter().enumerate() {
144        if k > 0 {
145            s.push(',');
146        }
147        fwrite_text(name, &mut s);
148    }
149    s.push_str(eol);
150    for i in 0..n {
151        for (k, (_, col)) in columns.iter().enumerate() {
152            if k > 0 {
153                s.push(',');
154            }
155            match col {
156                FwriteColumn::Text(v) => {
157                    if let Some(t) = &v[i] {
158                        fwrite_text(t, &mut s);
159                    }
160                }
161                FwriteColumn::Number(v) => fwrite_number(v[i], &mut s),
162                FwriteColumn::Integer(v) => {
163                    if let Some(x) = v[i] {
164                        s.push_str(&x.to_string());
165                    }
166                }
167                FwriteColumn::Logical(v) => {
168                    if let Some(b) = v[i] {
169                        s.push_str(if b { "TRUE" } else { "FALSE" });
170                    }
171                }
172            }
173        }
174        s.push_str(eol);
175    }
176    Ok(s)
177}

179/// CRC-32 (IEEE) of `data`.
180fn crc32(data: &[u8]) -> u32 {
181    let mut table = [0u32; 256];
182    for (i, t) in table.iter_mut().enumerate() {
183        let mut c = i as u32;
184        for _ in 0..8 {
185            c = if c & 1 != 0 {
186                0xEDB8_8320 ^ (c >> 1)
187            } else {
188                c >> 1
189            };
190        }
191        *t = c;
192    }
193    let mut c = !0u32;
194    for &b in data {
195        c = table[((c ^ u32::from(b)) & 0xff) as usize] ^ (c >> 8);
196    }
197    !c
198}

200/// A `.zip` archive of `files` (name, bytes), each entry deflated (level 9,
201/// as `zip::zip()`) and dated 1980-01-01.
202pub fn zip_deflated(files: &[(&str, &[u8])]) -> Vec<u8> {
203    let mut out = Vec::new();
204    let mut central = Vec::new();
205    let le16 = |v: &mut Vec<u8>, x: u16| v.extend_from_slice(&x.to_le_bytes());
206    let le32 = |v: &mut Vec<u8>, x: u32| v.extend_from_slice(&x.to_le_bytes());
207    for (name, data) in files {
208        let offset = out.len() as u32;
209        let crc = crc32(data);
210        let packed = miniz_oxide::deflate::compress_to_vec(data, 9);
211        let (n, len, size) = (name.len() as u16, data.len() as u32, packed.len() as u32);
212        le32(&mut out, 0x0403_4b50);
213        le16(&mut out, 20); // version needed
214        le16(&mut out, 0x0800); // UTF-8 names
215        le16(&mut out, 8); // deflated
216        le16(&mut out, 0); // time
217        le16(&mut out, 0x21); // date 1980-01-01
218        le32(&mut out, crc);
219        le32(&mut out, size);
220        le32(&mut out, len);
221        le16(&mut out, n);
222        le16(&mut out, 0);
223        out.extend_from_slice(name.as_bytes());
224        out.extend_from_slice(&packed);
225        le32(&mut central, 0x0201_4b50);
226        le16(&mut central, 0x031e); // made by: Unix
227        le16(&mut central, 20);
228        le16(&mut central, 0x0800);
229        le16(&mut central, 8);
230        le16(&mut central, 0);
231        le16(&mut central, 0x21);
232        le32(&mut central, crc);
233        le32(&mut central, size);
234        le32(&mut central, len);
235        le16(&mut central, n);
236        le16(&mut central, 0); // extra
237        le16(&mut central, 0); // comment
238        le16(&mut central, 0); // disk
239        le16(&mut central, 0); // internal attributes
240        le32(&mut central, 0o100644 << 16); // -rw-r--r--
241        le32(&mut central, offset);
242        central.extend_from_slice(name.as_bytes());
243    }
244    let cd_offset = out.len() as u32;
245    let cd_len = central.len() as u32;
246    out.extend_from_slice(&central);
247    le32(&mut out, 0x0605_4b50);
248    le16(&mut out, 0);
249    le16(&mut out, 0);
250    le16(&mut out, files.len() as u16);
251    le16(&mut out, files.len() as u16);
252    le32(&mut out, cd_len);
253    le32(&mut out, cd_offset);
254    le16(&mut out, 0);
255    out
256}

380/// The bytes of a fare structure `.zip` file. The five CSV files in it
381/// are byte for byte what `data.table::fwrite()` writes on the same
382/// platform (its line ends: `\r\n` on Windows, `\n` elsewhere, and in
383/// WebAssembly); the archive deflates them and leaves them undated (see
384/// the module notes).
385pub fn write_fare_structure(s: &FareStructure) -> Result<Vec<u8>, RouteError> {
386    let csvs = fare_csvs(s, crate::output::LINE_END)?;
387    let files: Vec<(&str, &[u8])> = FILES
388        .iter()
389        .zip(&csvs)
390        .map(|(n, c)| (*n, c.as_bytes()))
391        .collect();
392    Ok(zip_deflated(&files))
393}

395/// A CSV file as `fread` splits it: the header and the rows of fields,
396/// unquoted fields stripped of surrounding spaces (`strip.white`), and
397/// whether each field was quoted.
398fn fread_rows(text: &str) -> Vec<Vec<(String, bool)>> {
399    let text = text.strip_prefix('\u{feff}').unwrap_or(text);
400    let mut rows = Vec::new();
401    let mut row = Vec::new();
402    let mut field = String::new();
403    let mut quoted = false;
404    let mut in_quotes = false;
405    let mut chars = text.chars().peekable();
406    let mut any = false;
407    let push = |row: &mut Vec<(String, bool)>, field: &mut String, quoted: bool| {
408        let f = if quoted {
409            std::mem::take(field)
410        } else {
411            let t = field.trim_matches([' ', '\t']).to_string();
412            field.clear();
413            t
414        };
415        row.push((f, quoted));
416    };
417    while let Some(c) = chars.next() {
418        any = true;
419        if in_quotes {
420            if c == '"' {
421                if chars.peek() == Some(&'"') {
422                    chars.next();
423                    field.push('"');
424                } else {
425                    in_quotes = false;
426                }
427            } else {
428                field.push(c);
429            }
430            continue;
431        }
432        match c {
433            '"' if field.trim().is_empty() && !quoted => {
434                field.clear();
435                quoted = true;
436                in_quotes = true;
437            }
438            ',' => {
439                push(&mut row, &mut field, quoted);
440                quoted = false;
441            }
442            '\r' | '\n' => {
443                if c == '\r' && chars.peek() == Some(&'\n') {
444                    chars.next();
445                }
446                push(&mut row, &mut field, quoted);
447                quoted = false;
448                if !(row.len() == 1 && row[0].0.is_empty() && !row[0].1) {
449                    rows.push(std::mem::take(&mut row));
450                } else {
451                    row.clear();
452                }
453                any = false;
454            }
455            _ => field.push(c),
456        }
457    }
458    if any {
459        push(&mut row, &mut field, quoted);
460        if !(row.len() == 1 && row[0].0.is_empty() && !row[0].1) {
461            rows.push(row);
462        }
463    }
464    rows
465}

467/// One table read by `fread(file, select = ...)`.
468struct Table {
469    file: &'static str,
470    header: Vec<String>,
471    rows: Vec<Vec<(String, bool)>>,
472}
473
474impl Table {
475    fn read(arch: &Archive, file: &'static str, latin1: bool) -> Result<Table, RouteError> {
476        let e = arch
477            .find(file)
478            .ok_or_else(|| RouteError::Invalid(format!("the fare structure file has no {file}")))?;
479        let bytes = arch
480            .read(e)
481            .map_err(|e| RouteError::Invalid(format!("{file}: {e}")))?;
482        let text = if latin1 {
483            bytes.iter().map(|&b| char::from(b)).collect()
484        } else {
485            String::from_utf8_lossy(&bytes).into_owned()
486        };
487        let mut rows = fread_rows(&text);
488        if rows.is_empty() {
489            return Err(RouteError::Invalid(format!("{file} is empty")));
490        }
491        let header = rows.remove(0).into_iter().map(|f| f.0).collect();
492        Ok(Table { file, header, rows })
493    }
494
495    fn column(&self, name: &str) -> Result<Vec<(&str, bool)>, RouteError> {
496        let k =
497            self.header.iter().position(|h| h == name).ok_or_else(|| {
498                RouteError::Invalid(format!("{} has no column {name}", self.file))
499            })?;
500        Ok(self
501            .rows
502            .iter()
503            .map(|r| r.get(k).map_or(("", false), |(s, q)| (s.as_str(), *q)))
504            .collect())
505    }
506
507    /// A character column: `NA` (unquoted) is missing, an empty field the
508    /// empty string.
509    fn text(&self, name: &str) -> Result<Vec<Option<String>>, RouteError> {
510        Ok(self
511            .column(name)?
512            .into_iter()
513            .map(|(s, q)| (q || s != "NA").then(|| s.to_string()))
514            .collect())
515    }
516
517    fn number(&self, name: &str) -> Result<Vec<f64>, RouteError> {
518        self.column(name)?
519            .into_iter()
520            .map(|(s, _)| {
521                fread_number(s).ok_or_else(|| {
522                    RouteError::Invalid(format!(
523                        "{}: {name} value {s:?} is not a number",
524                        self.file
525                    ))
526                })
527            })
528            .collect()
529    }
530
531    fn logical(&self, name: &str) -> Result<Vec<Option<bool>>, RouteError> {
532        self.column(name)?
533            .into_iter()
534            .map(|(s, _)| match s {
535                "TRUE" | "True" | "true" | "T" => Ok(Some(true)),
536                "FALSE" | "False" | "false" | "F" => Ok(Some(false)),
537                "" | "NA" => Ok(None),
538                _ => Err(RouteError::Invalid(format!(
539                    "{}: {name} value {s:?} is not TRUE or FALSE",
540                    self.file
541                ))),
542            })
543            .collect()
544    }
545
546    /// The `value` of the row whose `setting` is `name`.
547    fn setting(&self, name: &str) -> Result<(String, bool), RouteError> {
548        let settings = self.column("setting")?;
549        let values = self.column("value")?;
550        let hits: Vec<_> = settings
551            .iter()
552            .zip(&values)
553            .filter(|((s, _), _)| *s == name)
554            .map(|(_, (v, q))| (v.to_string(), *q))
555            .collect();
556        match hits.as_slice() {
557            [one] => Ok(one.clone()),
558            _ => Err(RouteError::Invalid(format!(
559                "{} must have one row for {name}, it has {}",
560                self.file,
561                hits.len()
562            ))),
563        }
564    }
565}

567/// A number as `fread` reads it (`NA` and an empty field are NaN).
568fn fread_number(s: &str) -> Option<f64> {
569    match s {
570        "" | "NA" | "NaN" => Some(f64::NAN),
571        "Inf" | "+Inf" | "inf" | "INF" => Some(f64::INFINITY),
572        "-Inf" | "-inf" | "-INF" => Some(f64::NEG_INFINITY),
573        _ => s.parse::<f64>().ok().filter(|x| x.is_finite()),
574    }
575}

577/// R's `as.integer()` of a double: truncated toward zero, NaN for `NA`
578/// or beyond the range of an R integer.
579fn as_integer(x: f64) -> f64 {
580    if x.is_nan() || x.abs() >= 2_147_483_648.0 || x.trunc() == f64::from(i32::MIN) {
581        f64::NAN
582    } else {
583        x.trunc()
584    }
585}

587/// The fare structure in the bytes of a fare structure `.zip`
588/// file: the global settings (the counts made integers as `as.integer()`
589/// does: `Inf` becomes `NA`, NaN here), the three tables with their known
590/// columns, and the debug settings. `encoding` is fread's: `"UTF-8"`
591/// (the default) and `"unknown"` read the text as UTF-8, `"Latin-1"` as
592/// ISO 8859-1.
593pub fn read_fare_structure(bytes: &[u8], encoding: &str) -> Result<FareStructure, RouteError> {
594    let latin1 = match encoding {
595        "UTF-8" | "unknown" => false,
596        "Latin-1" => true,
597        _ => {
598            return Err(RouteError::Invalid(format!(
599                "encoding must be one of \"unknown\", \"UTF-8\" and \"Latin-1\", got {encoding:?}"
600            )))
601        }
602    };
603    let arch = Archive::from_bytes(bytes)
604        .map_err(|e| RouteError::Invalid(format!("fare structure file: {e}")))?;
605    let g = Table::read(&arch, FILES[0], latin1)?;
606    // the value column is numeric when every value reads as a number
607    let num = |name: &str| -> Result<f64, RouteError> {
608        let (v, _) = g.setting(name)?;
609        fread_number(&v)
610            .ok_or_else(|| RouteError::Invalid(format!("{}: {name} {v:?} is not a number", g.file)))
611    };
612    let max_discounted_transfers = as_integer(num("max_discounted_transfers")?);
613    let transfer_time_allowance = as_integer(num("transfer_time_allowance")?);
614    let fare_cap = num("fare_cap")?;
615    let t = Table::read(&arch, FILES[1], latin1)?;
616    let types = t.text("type")?;
617    let (unl, same, use_route) = (
618        t.logical("unlimited_transfers")?,
619        t.logical("allow_same_route_transfer")?,
620        t.logical("use_route_fare")?,
621    );
622    let fares = t.number("fare")?;
623    let mut fares_per_type = Vec::with_capacity(types.len());
624    for k in 0..types.len() {
625        let req = |v: Option<bool>, col: &str| {
626            v.ok_or_else(|| {
627                RouteError::Invalid(format!("{}: {col} is NA in row {}", t.file, k + 1))
628            })
629        };
630        fares_per_type.push(FarePerType {
631            fare_type: types[k].clone(),
632            unlimited_transfers: req(unl[k], "unlimited_transfers")?,
633            allow_same_route_transfer: req(same[k], "allow_same_route_transfer")?,
634            use_route_fare: req(use_route[k], "use_route_fare")?,
635            fare: fares[k],
636        });
637    }
638    let tr = Table::read(&arch, FILES[2], latin1)?;
639    let (a, b, f) = (
640        tr.text("first_leg")?,
641        tr.text("second_leg")?,
642        tr.number("fare")?,
643    );
644    let fares_per_transfer = (0..a.len())
645        .map(|k| FarePerTransfer {
646            first_leg: a[k].clone(),
647            second_leg: b[k].clone(),
648            fare: f[k],
649        })
650        .collect();
651    let r = Table::read(&arch, FILES[3], latin1)?;
652    let cols = [
653        r.text("agency_id")?,
654        r.text("agency_name")?,
655        r.text("route_id")?,
656        r.text("route_short_name")?,
657        r.text("route_long_name")?,
658        r.text("mode")?,
659    ];
660    let route_fare = r.number("route_fare")?;
661    let fare_type = r.text("fare_type")?;
662    let fares_per_route = (0..route_fare.len())
663        .map(|k| FarePerRoute {
664            agency_id: cols[0][k].clone(),
665            agency_name: cols[1][k].clone(),
666            route_id: cols[2][k].clone(),
667            route_short_name: cols[3][k].clone(),
668            route_long_name: cols[4][k].clone(),
669            mode: cols[5][k].clone(),
670            route_fare: route_fare[k],
671            fare_type: fare_type[k].clone(),
672        })
673        .collect();
674    let d = Table::read(&arch, FILES[4], latin1)?;
675    let debug = |name: &str| -> Result<String, RouteError> {
676        let (v, q) = d.setting(name)?;
677        if !q && v == "NA" {
678            return Err(RouteError::Invalid(format!("{}: {name} is NA", d.file)));
679        }
680        Ok(v)
681    };
682    Ok(FareStructure {
683        max_discounted_transfers,
684        transfer_time_allowance,
685        fare_cap,
686        fares_per_type,
687        fares_per_transfer,
688        fares_per_route,
689        debug_output_file: debug("output_file")?,
690        debug_trip_info: debug("trip_info")?,
691    })
