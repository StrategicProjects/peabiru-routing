1//! Transit fares: the fare structure (the object of
2//! `setup_fare_structure()`, `read_fare_structure()` and
3//! `write_fare_structure()`) and the rule-based fare calculator
4//! (`RuleBasedInRoutingFareCalculator`), which R5's multi-criteria router
5//! (McRAPTOR) calls for every state.
6//!
7//! A [`FareStructure`] holds the values as doubles, `Inf` included. The
8//! calculator was written to receive them as JSON (`jsonlite::toJSON`,
9//! four decimal places), from which Jackson reads the
10//! fares into `float`s and the counts into `int`s, and the calculator works
11//! in integer cents (`Math.round(fare * 100.0f)`); [`FareStructure`] goes
12//! through the same conversions before routing, so that the same values
13//! give the same fares.
14//!
15//! The calculator, the structure and its builder are translations to Rust,
16//! by the peabiru authors, of r5r's Java classes `FareStructure`,
17//! `FareStructureBuilder` and `RuleBasedInRoutingFareCalculator`
18//! (Copyright (c) 2020 r5r contributors, MIT licence); see
19//! `LICENSE-THIRD-PARTY`.

81/// A fare structure: how much each ride costs and which transfers are
82/// discounted. Counts and minutes are doubles, `Inf` for no limit. Start
83/// from [`setup_fare_structure`](crate::setup_fare_structure) or
84/// [`read_fare_structure`] and change the fields.
85///
86/// The rules, as the fare calculator applies them:
87/// the first ride costs its route's fare; each later ride costs the
88/// transfer fare between the two types minus the fare of the previous ride
89/// when a discounted transfer applies (a row of `fares_per_transfer` for
90/// the two types, the boardings at most `transfer_time_allowance` minutes
91/// apart, not the same route unless the type allows it, and fewer than
92/// `max_discounted_transfers` discounts used so far), and its full fare
93/// otherwise; a ride of the same type as the previous one is free when the
94/// type has unlimited transfers. The total is capped at `fare_cap`.
95#[derive(Debug, Clone, PartialEq)]
96#[non_exhaustive]
97pub struct FareStructure {
98    /// Most transfers with a discount (default 1; `Inf` is read as -1,
99    /// with which no transfer is ever discounted).
100    pub max_discounted_transfers: f64,
101    /// Minutes between the boardings of a discounted transfer (default
102    /// 120; `Inf` is read as -1, with which no transfer is ever discounted).
103    pub transfer_time_allowance: f64,
104    /// Highest total fare of a trip (`Inf`, the default, or 0: no cap).
105    pub fare_cap: f64,
106    #[allow(missing_docs)]
107    pub fares_per_type: Vec<FarePerType>,
108    #[allow(missing_docs)]
109    pub fares_per_transfer: Vec<FarePerTransfer>,
110    #[allow(missing_docs)]
111    pub fares_per_route: Vec<FarePerRoute>,
112    /// The `output_file` of the debug settings, kept for the file format
113    /// (no debug file is written, whatever it says).
114    pub debug_output_file: String,
115    /// The `trip_info` of the debug settings, kept for the file format.
116    pub debug_trip_info: String,
117}

172/// The value of a Java `float` as R reads it back from Jackson's JSON
173/// (`Float.toString`, then a double): `4.8f` gives `4.8`.
174pub(crate) fn float_to_r(f: f32) -> f64 {
175    // Rust prints the shortest decimal that reads back as the same float,
176    // as Java's `Float.toString` does (JDK 19 and later)
177    format!("{f}").parse().unwrap_or(f64::from(f))
178}

180/// The text `jsonlite::toJSON(x, digits = 4)` writes for a number
181/// (`num_to_char`: `modp_dtoa2` with four decimals for magnitudes between
182/// 1e-5 and 2^31 - 1, `%.*g` otherwise).
183pub(crate) fn jsonlite_number(x: f64) -> String {
184    const PREC: i32 = 4;
185    if x.is_nan() {
186        return "NaN".into();
187    }
188    if x.is_infinite() {
189        return if x > 0.0 { "Inf".into() } else { "-Inf".into() };
190    }
191    if x.abs() < 2_147_483_647.0 && x.abs() > 1e-5 {
192        return modp_dtoa2(x, PREC);
193    }
194    let decimals = (15f64.min(1f64.max(x.log10()) + f64::from(PREC))).ceil() as usize;
195    c_format_g(x, decimals.max(1))
196}

198/// `modp_dtoa2` of stringencoders, as jsonlite uses it.
199fn modp_dtoa2(value: f64, prec: i32) -> String {
200    const POW10: [f64; 10] = [1., 10., 100., 1e3, 1e4, 1e5, 1e6, 1e7, 1e8, 1e9];
201    let neg = value < 0.0;
202    let value = value.abs();
203    let mut whole = value as i32;
204    let tmp = (value - f64::from(whole)) * POW10[prec as usize];
205    let mut frac = tmp as u32;
206    let diff = tmp - f64::from(frac);
207    if diff > 0.5 {
208        frac += 1;
209        if f64::from(frac) >= POW10[prec as usize] {
210            frac = 0;
211            whole += 1;
212        }
213    } else if diff == 0.5 && frac & 1 == 1 {
214        // a tie goes to the even last digit (jsonlite 2.0's modp_dtoa2)
215        frac += 1;
216        if f64::from(frac) >= POW10[prec as usize] {
217            frac = 0;
218            whole += 1;
219        }
220    }
221    let mut s = String::new();
222    if neg {
223        s.push('-');
224    }
225    s.push_str(&whole.to_string());
226    if frac != 0 {
227        let mut count = prec;
228        while frac % 10 == 0 {
229            count -= 1;
230            frac /= 10;
231        }
232        let digits = frac.to_string();
233        s.push('.');
234        for _ in digits.len() as i32..count {
235            s.push('0');
236        }
237        s.push_str(&digits);
238    }
239    s
240}

242/// C's `%.*g` with `p` significant digits (only its value matters here,
243/// which `parse` reads back).
244fn c_format_g(x: f64, p: usize) -> String {
245    format!("{:.*e}", p - 1, x)
246}

248/// Jackson reading a JSON number into a Java `float`: the double the text
249/// stands for, rounded to a float (Jackson 2.10 `getFloatValue`).
250fn jackson_float(text: &str) -> f32 {
251    text.parse::<f64>().map_or(f32::NAN, |d| d as f32)
252}
253
254/// Jackson reading a JSON number into a Java `int`: an integer literal
255/// that must fit, or a decimal truncated toward zero
256/// (`ACCEPT_FLOAT_AS_INT`) when within range.
257fn jackson_int(text: &str, name: &str) -> Result<i32, String> {
258    let out_of_range = || format!("{name} ({text}) is out of the range of a 32-bit integer");
259    if text.contains(['.', 'e', 'E']) {
260        let d: f64 = text.parse().map_err(|_| out_of_range())?;
261        if d < f64::from(i32::MIN) || d > f64::from(i32::MAX) {
262            return Err(out_of_range());
263        }
264        Ok(d as i32)
265    } else {
266        text.parse::<i32>().map_err(|_| out_of_range())
267    }
268}

270/// Java `Math.round(float)`.
271fn java_round(f: f32) -> i32 {
272    // floor(f + 0.5) is exact in f64 for any float; `as` saturates, NaN -> 0
273    (f64::from(f) + 0.5).floor() as i32
274}

276/// Integer cents of a fare: `Math.round(fare * 100.0f)`.
277fn cents(f: f32) -> i32 {
278    java_round(f * 100.0f32)

295/// The work of [`crate::setup_fare_structure`].
296pub(crate) fn setup_fare_structure(
297    layer: &TransitLayer,
298    base_fare: f64,
299    by: &str,
300) -> Result<FareStructure, RouteError> {
301    // rJava::.jfloat(base_fare), and Jackson writing the float back
302    let fare_f = base_fare as f32;
303    if !fare_f.is_finite() {
304        return Err(RouteError::Invalid(format!(
305            "base_fare must be finite, got {base_fare}"
306        )));
307    }
308    let fare = float_to_r(fare_f);
309    let mut types: JavaHashMap<Option<String>> = JavaHashMap::new();
310    let mut routes = Vec::with_capacity(layer.routes.len());
311    for r in &layer.routes {
312        let mode = mode_name(r.route_type).map_err(RouteError::Invalid)?;
313        let t = match by {
314            "MODE" => Some(mode.to_string()),
315            "AGENCY" | "AGENCY_ID" => r.agency_id.clone(),
316            "AGENCY_NAME" => r.agency_name.clone(),
317            _ => Some("GENERIC".to_string()),
318        };
319        let h = string_hash(t.as_deref());
320        types.insert(t.clone(), h).map_err(RouteError::Invalid)?;
321        routes.push(FarePerRoute {
322            agency_id: r.agency_id.clone(),
323            agency_name: r.agency_name.clone(),
324            route_id: Some(r.route_id.clone()),
325            route_short_name: r.short_name.clone(),
326            route_long_name: r.long_name.clone(),
327            mode: Some(mode.to_string()),
328            route_fare: fare,
329            fare_type: t,
330        });
331    }
332    let names: Vec<Option<String>> = types.order().map(|i| types.key(i).clone()).collect();
333    // transfers: a HashSet of "a&b" strings (null is "null"), split on "&"
334    let mut transfers: JavaHashMap<String> = JavaHashMap::new();
335    for a in &names {
336        for b in &names {
337            let s = format!(
338                "{}&{}",
339                a.as_deref().unwrap_or("null"),
340                b.as_deref().unwrap_or("null")
341            );
342            let h = string_hash(Some(&s));
343            transfers.insert(s, h).map_err(RouteError::Invalid)?;
344        }
345    }
346    let mut fares_per_transfer = Vec::new();
347    for i in transfers.order() {
348        let legs = java_split_amp(transfers.key(i));
349        if legs.len() < 2 {
350            return Err(RouteError::Invalid(format!(
351                "this fare structure cannot be built: the transfer {:?} does not split into \
352                 two fare types",
353                transfers.key(i)
354            )));
355        }
356        fares_per_transfer.push(FarePerTransfer {
357            first_leg: Some(legs[0].clone()),
358            second_leg: Some(legs[1].clone()),
359            fare,
360        });
361    }
362    Ok(FareStructure {
363        fares_per_type: names
364            .into_iter()
365            .map(|t| FarePerType {
366                fare_type: t,
367                fare,
368                ..FarePerType::default()
369            })
370            .collect(),
371        fares_per_transfer,
372        fares_per_route: routes,
373        ..FareStructure::default()
374    })
375}

377/// Java's `s.split("&")`: the pieces, trailing empty ones dropped.
378fn java_split_amp(s: &str) -> Vec<String> {
379    let mut v: Vec<String> = s.split('&').map(str::to_string).collect();
380    while v.last().is_some_and(String::is_empty) {
381        v.pop();
382    }
383    v
384}

386impl FareStructure {
387    /// The checks on the values of a fare structure: counts,
388    /// minutes and cap not negative (nor `NA`), fare types present and
389    /// unique, fares finite and not negative, every transfer and route
390    /// naming one of the types, and no missing value in the route table.
391    pub fn validate(&self) -> Result<(), RouteError> {
392        self.check().map_err(RouteError::Invalid)
393    }
394
395    /// [`FareStructure::validate`], then the checks made as soon as a
396    /// structure is given for a network (even for
397    /// a street-only trip): its numbers as they are read (fares into
398    /// 32-bit floats, counts into 32-bit integers), and a row of
399    /// `fares_per_route` for the `agency_id` and `route_id` of every
400    /// route the network's trip patterns run on. For a network without
401    /// public transport, the same as [`FareStructure::validate`].
402    pub fn validate_for(&self, network: &crate::Network) -> Result<(), RouteError> {
403        match network.transit() {
404            Some(layer) => FareCalculator::new(self, layer).map(|_| ()),
405            None => self.validate(),
406        }
407    }
408
409    pub(crate) fn check(&self) -> Result<(), String> {
410        let number = |name: &str, x: f64| {
411            if x.is_nan() || x < 0.0 {
412                Err(format!(
413                    "fare_structure${name} must be a number >= 0, got {x}"
414                ))
415            } else {
416                Ok(())
417            }
418        };
419        number("max_discounted_transfers", self.max_discounted_transfers)?;
420        number("transfer_time_allowance", self.transfer_time_allowance)?;
421        number("fare_cap", self.fare_cap)?;
422        let fare = |table: &str, k: usize, x: f64| {
423            if !x.is_finite() || x < 0.0 {
424                Err(format!(
425                    "fare_structure${table} row {}: fares must be finite and >= 0, got {x}",
426                    k + 1
427                ))
428            } else {
429                Ok(())
430            }
431        };
432        let mut seen = std::collections::HashSet::new();
433        for (k, t) in self.fares_per_type.iter().enumerate() {
434            let Some(name) = &t.fare_type else {
435                return Err(format!(
436                    "fare_structure$fares_per_type row {}: type is missing",
437                    k + 1
438                ));
439            };
440            if !seen.insert(name.as_str()) {
441                return Err(format!(
442                    "fare_structure$fares_per_type: type {name:?} is repeated"
443                ));
444            }
445            fare("fares_per_type", k, t.fare)?;
446        }
447        let known = |table: &str, col: &str, k: usize, v: &Option<String>| match v {
448            None => Err(format!(
449                "fare_structure${table} row {}: {col} is missing",
450                k + 1
451            )),
452            Some(s) if !seen.contains(s.as_str()) => Err(format!(
453                "fare_structure${table} row {}: {col} {s:?} is not a type of fares_per_type",
454                k + 1
455            )),
456            _ => Ok(()),
457        };
458        for (k, t) in self.fares_per_transfer.iter().enumerate() {
459            known("fares_per_transfer", "first_leg", k, &t.first_leg)?;
460            known("fares_per_transfer", "second_leg", k, &t.second_leg)?;
461            fare("fares_per_transfer", k, t.fare)?;
462        }
463        for (k, r) in self.fares_per_route.iter().enumerate() {
464            for (col, v) in [
465                ("agency_id", &r.agency_id),
466                ("agency_name", &r.agency_name),
467                ("route_id", &r.route_id),
468                ("route_short_name", &r.route_short_name),
469                ("route_long_name", &r.route_long_name),
470                ("mode", &r.mode),
471            ] {
472                if v.is_none() {
473                    return Err(format!(
474                        "fare_structure$fares_per_route row {}: {col} is missing",
475                        k + 1
476                    ));
477                }
478            }
479            fare("fares_per_route", k, r.route_fare)?;
480            known("fares_per_route", "fare_type", k, &r.fare_type)?;
481        }
482        Ok(())
483    }
484}

572impl FareCalculator {
573    /// `RuleBasedInRoutingFareCalculator(transitLayer, json)` with
574    /// `loadFareInformation()`: the structure as Java reads it, every
575    /// pattern's route looked up by `agency_id:route_id`.
576    pub(crate) fn new(s: &FareStructure, layer: &TransitLayer) -> Result<Self, RouteError> {
577        s.check().map_err(RouteError::Invalid)?;
578        let inv = RouteError::Invalid;
579        // set_fare_structure(): Inf becomes -1, then JSON and Jackson
580        let java_int = |x: f64, name: &str| -> Result<i32, RouteError> {
581            let x = if x == f64::INFINITY { -1.0 } else { x };
582            jackson_int(&jsonlite_number(x), name).map_err(inv)
583        };
584        let max_discounted = java_int(s.max_discounted_transfers, "max_discounted_transfers")?;
585        let allowance_min = java_int(s.transfer_time_allowance, "transfer_time_allowance")?;
586        let cap_f = as_float(if s.fare_cap == f64::INFINITY {
587            -1.0
588        } else {
589            s.fare_cap
590        });
591        // indexTransportType: HashMap put, a later type with the same name
592        // wins (validation rules repeats out)
593        let mut type_index = std::collections::HashMap::new();
594        let mut types = Vec::with_capacity(s.fares_per_type.len());
595        let mut type_cents = Vec::with_capacity(s.fares_per_type.len());
596        for (i, t) in s.fares_per_type.iter().enumerate() {
597            type_index.insert(t.fare_type.clone(), i as u32);
598            types.push(TypeRule {
599                unlimited: t.unlimited_transfers,
600                same_route: t.allow_same_route_transfer,
601            });
602            type_cents.push(cents(as_float(t.fare)));
603        }
604        let lookup = |t: &Option<String>| -> Result<u32, RouteError> {
605            type_index.get(t).copied().ok_or_else(|| {
606                RouteError::Invalid(format!(
607                    "fare type {:?} is not in fares_per_type",
608                    t.as_deref().unwrap_or("NA")
609                ))
610            })
611        };
612        // indexRouteInfo: agency_id:route_id -> the last FarePerRoute
613        let mut by_route = std::collections::HashMap::new();
614        for (i, r) in s.fares_per_route.iter().enumerate() {
615            by_route.insert(
616                format!(
617                    "{}:{}",
618                    r.agency_id.as_deref().unwrap_or(""),
619                    r.route_id.as_deref().unwrap_or("")
620                ),
621                i as u32,
622            );
623        }
624        let mut route_rule: Vec<Option<(u32, i32)>> = vec![None; s.fares_per_route.len()];
625        let mut patterns = Vec::with_capacity(layer.patterns.len());
626        for pi in 0..layer.patterns.len() {
627            let ri = &layer.routes[layer.pattern_route[pi] as usize];
628            let key = format!(
629                "{}:{}",
630                ri.agency_id.as_deref().unwrap_or("null"),
631                ri.route_id
632            );
633            let Some(&k) = by_route.get(&key) else {
634                return Err(RouteError::Invalid(format!(
635                    "fare_structure$fares_per_route has no row for agency_id {:?} and \
636                     route_id {:?} of the network",
637                    ri.agency_id.as_deref().unwrap_or("NA"),
638                    ri.route_id
639                )));
640            };
641            let rule = match route_rule[k as usize] {
642                Some(r) => r,
643                None => {
644                    let r = &s.fares_per_route[k as usize];
645                    let ti = lookup(&r.fare_type)?;
646                    let t = &s.fares_per_type[ti as usize];
647                    let c = if t.use_route_fare {
648                        cents(as_float(r.route_fare))
649                    } else {
650                        type_cents[ti as usize]
651                    };
652                    route_rule[k as usize] = Some((ti, c));
653                    (ti, c)
654                }
655            };
656            patterns.push((rule.0, rule.1, k));
657        }
658        // equal route_ids (missing ones included), equal numbers
659        let mut classes: std::collections::HashMap<Option<&str>, u32> = Default::default();
660        let route_class = patterns
661            .iter()
662            .map(|&(_, _, k)| {
663                let next = classes.len() as u32;
664                *classes
665                    .entry(s.fares_per_route[k as usize].route_id.as_deref())
666                    .or_insert(next)
667            })
668            .collect();
669        let n = types.len();
670        let mut transfers = vec![vec![None; n]; n];
671        for t in &s.fares_per_transfer {
672            let (a, b) = (lookup(&t.first_leg)?, lookup(&t.second_leg)?);
673            transfers[a as usize][b as usize] = Some(TransferRule {
674                fare: cents(as_float(t.fare)),
675                first_full: type_cents[a as usize],
676                second_full: type_cents[b as usize],
677            });
678        }
679        // the loop of `fare` over the transfers from the last ride's type
680        let max_allowance = transfers
681            .iter()
682            .map(|from: &Vec<Option<TransferRule>>| {
683                let mut max_allowance = 0;
684                for t in from.iter().flatten() {
685                    let allowance = t
686                        .first_full
687                        .wrapping_add(t.second_full)
688                        .wrapping_sub(t.fare);
689                    max_allowance = max_allowance.max(allowance);
690                }
691                max_allowance
692            })
693            .collect();
694        Ok(FareCalculator {
695            max_allowance,
696            max_discounted,
697            allowance_s: allowance_min.wrapping_mul(60),
698            cap_positive: cap_f > 0.0,
699            cap: cents(cap_f),
700            types,
701            transfers,
702            patterns,
703            route_ids: s
704                .fares_per_route
705                .iter()
706                .map(|r| r.route_id.clone())
707                .collect(),
708            route_class,
709        })
710    }

712    fn full(&self, pattern: i32) -> i32 {
713        self.patterns[pattern as usize].1

716    fn type_of(&self, pattern: i32) -> u32 {
717        self.patterns[pattern as usize].0

720    /// `isTransferAllowed`.
721    fn transfer_allowed(&self, a: i32, b: i32) -> bool {
722        let (ta, tb) = (self.type_of(a), self.type_of(b));
723        if ta != tb {
724            return true;
725        }
726        if self.types[ta as usize].same_route {
727            return true;
728        }
729        // `route_ids` of the two patterns differ: their classes do
730        self.route_class[a as usize] != self.route_class[b as usize]
731    }

733    /// Whether the fares of trips of at most `rides` rides are far from the
734    /// ends of an `i32`, so that what is paid less the value of the
735    /// allowance (which R5's `FareDominatingList` compares) is the
736    /// difference of the two numbers, not a wrapped one. Each ride changes
737    /// what is paid by a full fare or by a transfer fare less a full fare,
738    /// and the value of an allowance is at most two full fares, all at most
739    /// `big` here; so `|paid| + |value| <= (rides + 2) * big`.
740    pub(crate) fn exact_within(&self, rides: usize) -> bool {
741        let mut full = 0i64;
742        for &(_, cents, _) in &self.patterns {
743            full = full.max(i64::from(cents).abs());
744        }
745        let mut transfer = 0i64;
746        for t in self.transfers.iter().flatten().flatten() {
747            transfer = transfer.max(i64::from(t.fare).abs());
748            full = full
749                .max(i64::from(t.first_full).abs())
750                .max(i64::from(t.second_full).abs());
751        }
752        let big = 2 * full + transfer;
753        (rides as i64 + 2).saturating_mul(big) < 1 << 30
754    }

756    /// One more ride, `pattern` boarded at `board`, after the rides behind
757    /// `s`: the first ride's setup of [`FareCalculator::fare`] when `s` has
758    /// no ride, one turn of its loop otherwise. The statements are those of
759    /// `fare`, in its order, on the same variables, so folding the rides of
760    /// a trip from `FareProgress::default()` leaves the variables as `fare`
761    /// has them after its loop (`prev == current` there, hence one `last`).
762    pub(crate) fn ride(&self, s: &FareProgress, pattern: i32, board: i32) -> FareProgress {
763        if s.last < 0 {
764            // paid = full(first); prev = current = first; discounts = 0;
765            // prev_board = first_board; current_board = -1
766            return FareProgress {
767                paid: self.full(pattern),
768                last: pattern,
769                prev_board: board,
770                discounts: 0,
771                current_board: -1,
772            };
773        }
774        let mut n = *s;
775        // current = pattern; current_board = board
776        n.last = pattern;
777        n.current_board = board;
778        let (ta, tb) = (self.type_of(s.last), self.type_of(pattern));
779        if ta == tb && self.types[ta as usize].unlimited {
780            // prev = current; continue (prev_board stays)
781            return n;
782        }
783        if s.discounts >= self.max_discounted {
784            n.paid = s.paid.wrapping_add(self.full(pattern));
785        } else {
786            let gap = board.wrapping_sub(s.prev_board);
787            match self.transfers[ta as usize][tb as usize] {
788                Some(t) if self.transfer_allowed(s.last, pattern) && gap <= self.allowance_s => {
789                    n.paid = s.paid.wrapping_add(t.fare.wrapping_sub(self.full(s.last)));
790                    n.discounts = s.discounts + 1;
791                }
792                _ => n.paid = s.paid.wrapping_add(self.full(pattern)),
793            }
794        }
795        n.prev_board = board;
796        n
797    }

799    /// What [`FareCalculator::fare`] returns for the rides behind `s`: its
800    /// statements after the loop.
801    pub(crate) fn bounds(&self, s: &FareProgress) -> FareBounds {
802        let empty = |paid| FareBounds {
803            paid,
804            value: 0,
805            number: 0,
806            expiration: 0,
807        };
808        if s.last < 0 {
809            return empty(0);
810        }
811        let mut paid = s.paid;
812        if self.cap_positive {
813            paid = paid.min(self.cap);
814        }
815        let remaining = self.max_discounted.wrapping_sub(s.discounts);
816        if remaining <= 0 {
817            return empty(paid);
818        }
819        let full = self.full(s.last);
820        let mut max_allowance = self.max_allowance[self.type_of(s.last) as usize];
821        if self.cap_positive && paid > self.cap {
822            max_allowance = max_allowance.max(full);
823        }
824        FareBounds {
825            paid,
826            value: max_allowance,
827            number: remaining,
828            expiration: s.current_board.wrapping_add(self.allowance_s),
829        }
830    }

832    /// `calculateFare` for the rides of a state, `(pattern, board time)`
833    /// in travel order.
834    pub(crate) fn fare(&self, rides: &[(i32, i32)]) -> FareBounds {
835        let empty = |paid| FareBounds {
836            paid,
837            value: 0,
838            number: 0,
839            expiration: 0,
840        };
841        let Some(&(first, first_board)) = rides.first() else {
842            return empty(0);
843        };
844        let mut paid = self.full(first);
845        let mut prev = first;
846        let mut discounts = 0i32;
847        let mut prev_board = first_board;
848        let mut current = first;
849        let mut current_board = -1;
850        let mut last_type = self.type_of(first);
851        for &(pattern, board) in &rides[1..] {
852            current = pattern;
853            current_board = board;
854            let (ta, tb) = (self.type_of(prev), self.type_of(current));
855            last_type = tb;
856            if ta == tb && self.types[ta as usize].unlimited {
857                prev = current;
858                continue;
859            }
860            if discounts >= self.max_discounted {
861                paid = paid.wrapping_add(self.full(current));
862            } else {
863                // getIntegrationFare
864                let gap = current_board.wrapping_sub(prev_board);
865                match self.transfers[ta as usize][tb as usize] {
866                    Some(t) if self.transfer_allowed(prev, current) && gap <= self.allowance_s => {
867                        paid = paid.wrapping_add(t.fare.wrapping_sub(self.full(prev)));
868                        discounts += 1;
869                    }
870                    _ => paid = paid.wrapping_add(self.full(current)),
871                }
872            }
873            prev = current;
874            prev_board = current_board;
875        }
876        if self.cap_positive {
877            paid = paid.min(self.cap);
878        }
879        let remaining = self.max_discounted.wrapping_sub(discounts);
880        if remaining <= 0 {
881            return empty(paid);
882        }
883        let full = self.full(current);
884        let mut max_allowance = 0;
885        for t in self.transfers[last_type as usize].iter().flatten() {
886            let allowance = t
887                .first_full
888                .wrapping_add(t.second_full)
889                .wrapping_sub(t.fare);
890            max_allowance = max_allowance.max(allowance);
891        }
892        if self.cap_positive && paid > self.cap {
893            max_allowance = max_allowance.max(full);
894        }
895        FareBounds {
896            paid,
897            value: max_allowance,
898            number: remaining,
899            expiration: current_board.wrapping_add(self.allowance_s),
900        }
901    }

910        for (x, s) in [
