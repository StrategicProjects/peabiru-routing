1//! How GDAL's GeoJSON driver writes a coordinate, as `sf::st_write()` calls
2//! it (no layer options: no `COORDINATE_PRECISION`, no
3//! `SIGNIFICANT_FIGURES`, not RFC 7946). The speed polygons of a scenario
4//! reach R5 through such a file, so R5 routes on the coordinates read back
5//! from that text; the bindings without GDAL (Python, JavaScript) apply the
6//! same rule.
7//!
8//! The rule, from GDAL 3.8.5 (`OGR_json_double_with_precision_to_string`
9//! in `ogrgeojsonwriter.cpp`, `OGRFormatDouble`, `intelliround` and
10//! `removeTrailingZeros` in `ogrutils.cpp`; the same in GDAL 3.6.0 and
11//! 3.10.0): a magnitude above 1e50 is written with `%.18g` (which reads
12//! back exactly); any other finite value in fixed notation with 15
13//! decimals, then "intelligently rounded" (a tail of five zeros or five
14//! nines before the last digit, or of seven to two in larger numbers, is
15//! taken for binary round-off: dropped, or rounded up), then without
16//! trailing zeros (keeping one after the point).
17
18/// The text GDAL writes for `v` (checked against sf::st_write on 3.5
19/// million doubles); above 1e50 in magnitude GDAL writes `%.18g` and this
20/// the shortest exact form, which reads back as the same double.
21pub fn gdal_geojson_text(v: f64) -> String {
22    if v.is_nan() {
23        return "nan".into();
24    }
25    if v.is_infinite() {
26        return if v > 0.0 { "inf" } else { "-inf" }.into();
27    }
28    if v.abs() > 1e50 {
29        // %.18g: 17 significant digits read back exactly, as does Rust's
30        // shortest form (the text differs, the value does not)
31        return format!("{v:e}");
32    }
33    remove_trailing_zeros(intelliround(format!("{v:.15}")))
34}
35
36/// The coordinate R5 reads for `v` in such a file: GDAL's text for it
37/// ([`gdal_geojson_text`]) read back as the nearest double, as Java's
38/// `Double.parseDouble` and GDAL read it. Infinite and NaN values are
39/// returned as they are.
40pub fn gdal_geojson_coordinate(v: f64) -> f64 {
41    if !v.is_finite() || v.abs() > 1e50 {
42        return v;
43    }
44    gdal_geojson_text(v)
45        .parse()
46        .expect("GDAL's fixed notation is a decimal number")
47}

49/// `intelliround` of GDAL 3.8.5, on the text of `std::fixed` output.
50fn intelliround(mut s: String) -> String {
51    let Some(dot) = s.find('.') else {
52        return s;
53    };
54    if s.contains(['e', 'E']) {
55        return s;
56    }
57    let b = s.as_bytes();
58    let mut before = dot - 1;
59    if b[0] == b'-' {
60        before -= 1;
61    }
62    let i = b.len();
63    if i <= 10 {
64        return s;
65    }
66    let all = |c: u8, from: usize, to: usize| (from..=to).all(|k| b[i - k] == c);
67    // the digits at i-3 .. i-7 must be `c` unless the number has at least
68    // 4 .. 8 digits before the point
69    let tail =
70        |c: u8| (3..=7).all(|k| before > k || b[i - k] == c) && b[i - 8] == c && b[i - 9] == c;
71    if all(b'0', 2, 6) {
72        s.pop();
73    } else if dot < i - 8 && tail(b'0') {
74        s.truncate(i - 8);
75    } else if all(b'9', 2, 6) {
76        s.truncate(i - 6);
77        s = roundup(s);
78    } else if dot < i - 9 && tail(b'9') {
79        s.truncate(i - 9);
80        s = roundup(s);
81    }
82    s
83}

85/// GDAL's `roundup`: add 1 to the last digit, carrying.
86fn roundup(s: String) -> String {
87    let (negative, digits) = match s.strip_prefix('-') {
88        Some(d) => (true, d),
89        None => (false, s.as_str()),
90    };
91    let mut d: Vec<u8> = digits.bytes().collect();
92    let mut pos = d.len();
93    while pos > 0 {
94        pos -= 1;
95        if d[pos] == b'.' {
96            continue;
97        }
98        d[pos] += 1;
99        if d[pos] != b':' {
100            break;
101        }
102        d[pos] = b'0';
103        if pos == 0 {
104            d.insert(0, b'1');
105        }
106    }
107    let mut out = String::with_capacity(d.len() + 1);
108    if negative {
109        out.push('-');
110    }
111    out.push_str(std::str::from_utf8(&d).expect("ASCII"));
112    out
113}

115/// GDAL's `removeTrailingZeros`: no zeros after the last nonzero decimal,
116/// but one digit after the point.
117fn remove_trailing_zeros(mut s: String) -> String {
118    if !s.contains('.') {
119        return s;
120    }
121    let keep = s.trim_end_matches('0').len();
122    s.truncate(keep);
123    if s.ends_with('.') {
124        s.push('0');
125    }
126    s
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132
133    /// Texts GDAL 3.8.5 writes (checked with sf::st_write).
134    #[test]
135    fn as_gdal() {
136        for (v, t) in [
137            (-30.000000000000004, "-30.0"),
138            (-51.2, "-51.2"),
139            (0.1 + 0.2, "0.3"),
140            (-0.0, "-0.0"),
141            (0.0, "0.0"),
142            (1e-20, "0.0"),
143            (-1e-20, "-0.0"),
144            (12345.000000011, "12345.000000010999429"),
145            (12345678.9999999, "12345678.999999899417162"),
146            (45.12300000001, "45.123000000010002"),
147            (1.0000000000001, "1.0000000000001"),
148            (1.0 / 3.0, "0.333333333333333"),
149            (2.0 / 3.0, "0.666666666666667"),
150            (123.4567, "123.4567"),
151            (0.9999999999999, "1.0"),
152            (0.99999999999999, "1.0"),
153        ] {
154            assert_eq!(gdal_geojson_text(v), t, "{v:e}");
155        }
156        assert_eq!(gdal_geojson_coordinate(-30.000000000000004), -30.0);
157        assert_eq!(gdal_geojson_coordinate(1e60), 1e60);
158    }
