15/// A routable network: the street network built from an OpenStreetMap
16/// extract and, when GTFS feeds were given, the public transport layer
17/// built from them with its stops linked to those streets.
18///
19/// Every routing function of the crate takes a `&Network`. It is built once
20/// ([`build_from_dir`], [`build_from_bytes`],
21/// [`build_from_bytes_with_elevation`], or [`build_network`] for streets
22/// only) and does not change afterwards, so it can be shared by threads
23/// (`Network` is `Send` and `Sync`) and used for any number of calls. The
24/// only state it keeps between calls are caches that make later calls
25/// faster and never change their results (the contraction hierarchies of
26/// [`street_travel_time_matrix`] and the grid links of
27/// [`street_travel_time_surface`]); clones share them.
28///
29/// ```no_run
30/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
31/// let network = peabiru_core::build_from_dir("r/inst/extdata/spo", &Default::default())?;
32/// println!("{} street vertices", network.street().n_vertices());
33/// if let Some(transit) = network.transit() {
34///     println!("{} stops from {:?}", transit.n_stops(), transit.feed_ids());
35/// }
36/// for w in network.build_warnings() {
37///     eprintln!("warning: {w}");
38/// }
39/// # Ok(())
40/// # }
41/// ```
42#[derive(Clone)]
43pub struct Network {
44    street: StreetNetwork,
45    /// Empty without feeds.
46    layer: TransitLayer,
47    /// Made once the stops are linked, when there is at least one feed.
48    data: Option<TransitData>,
49}
50
51impl std::fmt::Debug for Network {
52    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
53        f.debug_struct("Network")
54            .field("vertices", &self.street.n_vertices())
55            .field("edges", &self.street.n_edges())
56            .field("elevation", &self.street.elevation())
57            .field("feeds", &self.layer.feed_ids())
58            .field("stops", &self.layer.n_stops())
59            .finish()
60    }
61}

63impl Network {
64    /// A network of `street` and `layer`, whose stops are already linked to
65    /// those streets.
66    pub(crate) fn new(street: StreetNetwork, layer: TransitLayer) -> Self {
67        let data = (!layer.feed_ids().is_empty()).then(|| TransitData::new(&street, &layer));
68        Network {
69            street,
70            layer,
71            data,
72        }
73    }
74
75    /// The street network: its size, box and elevation, as
76    /// [`StreetNetwork`]'s methods report them.
77    pub fn street(&self) -> &StreetNetwork {
78        &self.street
79    }
80
81    /// The public transport layer, `None` when the network was built
82    /// without GTFS feeds (then the functions that route by public
83    /// transport return [`RouteError::Invalid`]).
84    pub fn transit(&self) -> Option<&TransitLayer> {
85        self.data.as_ref().map(|_| &self.layer)
86    }
87
88    /// The warnings of the build, one per line, for the caller to show: one
89    /// per GTFS feed built from in spite of its GTFS errors of priority
90    /// `HIGH` (with [`BuildOptions::ignore_gtfs_errors`]; see
91    /// [`TransitLayer::feed_warnings`]), then those about the elevation raster
92    /// (see [`StreetNetwork::elevation_warnings`]). Empty for a clean build.
93    pub fn build_warnings(&self) -> Vec<String> {
94        let mut w = match self.transit() {
95            Some(t) => t.feed_warnings(),
96            None => Vec::new(),
97        };
98        w.extend(self.street.elevation_warnings().iter().cloned());
99        w
100    }
101
102    /// The layer, empty without feeds.
103    pub(crate) fn layer(&self) -> &TransitLayer {
104        &self.layer
105    }
106
107    /// The layer and its data, `None` without feeds.
108    pub(crate) fn transit_pair(&self) -> Option<(&TransitLayer, &TransitData)> {
109        self.data.as_ref().map(|d| (&self.layer, d))
110    }
111
112    /// The layer and its data, an error without feeds.
113    pub(crate) fn transit_parts(&self) -> Result<(&TransitLayer, &TransitData), RouteError> {
114        self.transit_pair().ok_or_else(|| {
115            RouteError::Invalid("the network has no public transport (no GTFS feed)".into())
116        })
117    }
118}

120/// Read an OpenStreetMap `.pbf` file and build the street network for
121/// walking, cycling and driving, without
122/// public transport ([`Network::transit`] is `None`; see [`build_from_dir`]
123/// for a network with GTFS feeds).
124///
125/// A file that cannot be read or decoded, or that gives no street edge at
126/// all (empty, a header only, or no routable way), is an error naming the
127/// file.
128/// [`BuildOptions::elevation`] needs a raster, so it is not used here
129/// (see [`build_from_bytes_with_elevation`]).
130pub fn build_network(
131    path: impl AsRef<std::path::Path>,
132    opts: &BuildOptions,
133) -> Result<Network, NetworkError> {
134    let street = crate::street::build::build_network(path, opts)?;
135    Ok(Network::new(street, TransitLayer::default()))
136}

138/// Build a network from a directory holding one OpenStreetMap `.pbf` file
139/// and any number of GTFS `.zip` feeds,
140/// and link the stops to the streets; with [`BuildOptions::elevation`],
141/// apply slope costs from the directory's `.tif` raster (see
142/// [`build_from_bytes_with_elevation`] for the rasters supported).
143///
144/// Files are recognised by their extension in lower case only (`.pbf`,
145/// `.zip`); with several `.pbf` files the last one listed is
146/// used, and the feeds are loaded in the order the operating system lists
147/// them (see [`build_from_bytes`]). A directory without GTFS
148/// feeds gives a network without public transport ([`Network::transit`] is
149/// `None`), good for street routing only.
150///
151/// The build fails when a feed has GTFS errors of priority
152/// `HIGH`, or is one gtfs-lib cannot load at all
153/// ([`NetworkError::high_priority_gtfs_errors`],
154/// [`gtfs_errors`](crate::gtfs_errors)), unless
155/// [`BuildOptions::ignore_gtfs_errors`] is set; what the build has to say
156/// about the feeds and the raster is in [`Network::build_warnings`].
157///
158/// This reads the files and builds the network as [`build_from_bytes`]
159/// does from the same bytes, with the feeds in the directory's order;
160/// errors name the files by their path.
161pub fn build_from_dir(dir: impl AsRef<Path>, opts: &BuildOptions) -> Result<Network, NetworkError> {
162    let (street, layer) = crate::transit::build_from_dir(dir, opts)?;
163    Ok(Network::new(street, layer))
164}

166/// Build a network from data held in memory: the bytes of an OpenStreetMap
167/// `.pbf` file and of any number of GTFS `.zip` feeds, each with its file
168/// name. The network is the one [`build_from_dir`] builds from a directory
169/// holding the same files, and the stops are linked to the streets in the
170/// same way.
171///
172/// The feeds are loaded in the order of `feeds`. That order matters: the
173/// stops, patterns and, where two feeds offer equally good trips, the
174/// routes chosen follow it. [`build_from_dir`] loads them in the order the
175/// operating system lists the directory, which differs from
176/// one file system to another; to reproduce a network built from a
177/// directory, pass the feeds in the order of its
178/// [`TransitLayer::feed_reports`].
179///
180/// A feed's name stands for its file name: it names the feed in errors
181/// and in [`FeedReport`](crate::FeedReport)s and, when the feed has no
182/// `feed_info.feed_id`, gives its feed id (the name without a trailing
183/// `.zip`), which prefixes its stop ids. Use the names the files have on
184/// disk (such as `"poa_eptc.zip"`) and keep them distinct. Errors about
185/// the `.pbf` data name it `OpenStreetMap data`. With no feeds, the network
186/// has no public transport, good for street routing only.
187///
188/// The build fails when a feed has GTFS errors of priority
189/// `HIGH`, or is one gtfs-lib cannot load at all
190/// ([`NetworkError::high_priority_gtfs_errors`],
191/// [`gtfs_errors`](crate::gtfs_errors)), unless
192/// [`BuildOptions::ignore_gtfs_errors`] is set.
193///
194/// ```no_run
195/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
196/// let pbf = std::fs::read("r/inst/extdata/spo/spo_osm.pbf")?;
197/// let gtfs = std::fs::read("r/inst/extdata/spo/spo_gtfs.zip")?;
198/// let opts = peabiru_core::BuildOptions::default();
199/// let network = peabiru_core::build_from_bytes(&pbf, &[("spo_gtfs.zip", &gtfs)], &opts)?;
200/// assert_eq!(network.transit().unwrap().feed_ids(), ["spo_gtfs"]);
201/// # Ok(())
202/// # }
203/// ```
204pub fn build_from_bytes(
205    pbf: &[u8],
206    feeds: &[(&str, &[u8])],
207    opts: &BuildOptions,
208) -> Result<Network, NetworkError> {
209    let (street, layer) = crate::transit::build_from_bytes(pbf, feeds, opts)?;
210    Ok(Network::new(street, layer))
211}

213/// [`build_from_bytes`] with an elevation raster: the bytes of a GeoTIFF
214/// file, as [`build_from_dir`] reads the `.tif` file of a directory. The
215/// slope costs of [`BuildOptions::elevation`] are applied to walking and
216/// cycling (with [`Elevation::None`](crate::Elevation::None) the raster is
217/// not read). Errors about the raster name it `elevation raster`.
218///
219/// The raster must be a GeoTIFF of one band of 8, 16 or 32-bit integers
220/// or 32 or 64-bit floats, uncompressed or compressed with LZW,
221/// Deflate or PackBits, north up, georeferenced by a tie point and a pixel
222/// scale (what GDAL writes), in WGS 84 longitude and latitude (EPSG:4326)
223/// or in another system whose coordinate transform peabiru computes
224/// itself: SIRGAS 2000 (4674), SAD69 (4618), Corrego
225/// Alegre 1970-72 (4225), their UTM zones used in Brazil, SIRGAS 2000 /
226/// Brazil Polyconic (5880), WGS 84 / UTM (326xx, 327xx) and WGS 84 /
227/// Pseudo-Mercator (3857). Other rasters are an error that says what to
228/// change. In a system other than EPSG:4326, the pixel read for a sample
229/// within nanometres of the boundary between two pixels of different
230/// values (or of the raster's edge) depends on the last bits of the
231/// projection's sines and cosines; peabiru computes them itself, bit
232/// for bit the same way, so it reads the same pixel on every
233/// machine, and counts such samples in
234/// [`StreetNetwork::elevation_warnings`] (and so in
235/// [`Network::build_warnings`]). A sample that falls on
236/// the outer edge (inside the envelope, past the last pixel) is an error.
237pub fn build_from_bytes_with_elevation(
238    pbf: &[u8],
239    feeds: &[(&str, &[u8])],
240    raster: &[u8],
241    opts: &BuildOptions,
242) -> Result<Network, NetworkError> {
243    let (street, layer) =
244        crate::transit::build_from_bytes_with_elevation(pbf, feeds, raster, opts)?;
245    Ok(Network::new(street, layer))
246}

518/// The trade-off between travel time and fare. For every cutoff of `fare_cutoffs`, in the order given, the
519/// travel times are those of the multi-criteria router (McRAPTOR, see
520/// [`TransitParams::fare_structure`]) with `max_fare` set to the cutoff
521/// (`p.max_fare` is not used); a destination and percentile gets a row at
522/// each cutoff where its travel time differs from that of the previous
523/// cutoff with a row, and is below `max_trip_duration`.
524///
525/// `p.fare_structure` is required. `draws_per_minute` is 1
526/// whatever `p.draws_per_minute` says, so that `time_window` departure
527/// times are sampled. With `p.transit_modes` empty the trip is by
528/// `p.mode` alone: the
529/// street travel time, the same at every cutoff, with
530/// `p.max_trip_duration` as the limit (lower it to the mode's own limit
531/// first); otherwise the network must have public transport
532/// ([`RouteError::Invalid`]). Points are `(lat, lon)` degrees.
533pub fn pareto_frontier(
534    network: &Network,
535    origins: &[(f64, f64)],
536    destinations: &[(f64, f64)],
537    p: &TransitParams,
538    fare_cutoffs: &[f64],
539) -> Result<ParetoFrontier, RouteError> {
540    let transit = if p.transit_modes.is_empty() {
541        None
542    } else {
543        Some(network.transit_parts()?)
544    };
545    crate::transit::pareto::pareto_frontier(
546        &network.street,
547        transit,
548        origins,
549        destinations,
550        p,
551        fare_cutoffs,
552    )
553}
