1//! peabiru's own multi-criteria trip planner, [`trip_options`]: for each
2//! origin-destination pair, every trip worth considering on the chosen
3//! criteria (total time, fare, transfers, time on foot) over a departure
4//! window. Unlike the rest of the crate it is not a port: it
5//! builds on R5's network semantics (street searches, stop linking,
6//! transfers, boarding rules) so that its times agree with the other
7//! functions, and adds its own search ([`search`]) and fare comparisons
8//! (`fares::incremental`).

24/// Seconds to which [`trip_options`] compares total times when it decides
25/// which trips of a minute's exact frontier to return: a tenth of a minute,
26/// the resolution the times are reported in, so that no option of a minute
27/// looks dominated by another in the reported values. (Compared to the
28/// second, two trips whose reported total times are equal and whose other
29/// values make one look better would both be returned; on 200 pairs of
30/// Porto Alegre with all four criteria that was 2% of the minutes.)
31pub(crate) const TIME_RESOLUTION_S: i32 = 6;
32
33/// A criterion [`trip_options`] compares trips on. More may be added, so a
34/// `match` on it needs a wildcard arm.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
36#[non_exhaustive]
37pub enum Criterion {
38    /// Door-to-door time from the departure minute, waits included.
39    Time,
40    /// The fare of the trip by the fare structure (needs one).
41    Fare,
42    /// Transfers: rides minus one (0 for a trip without public transport).
43    Transfers,
44    /// Time on foot: the walking legs (access and egress when they are on
45    /// foot, and the walks between stops).
46    Walk,
47}
48
49impl Criterion {
50    /// Every criterion, in this order.
51    pub const ALL: &'static [Criterion] = &[
52        Criterion::Time,
53        Criterion::Fare,
54        Criterion::Transfers,
55        Criterion::Walk,
56    ];
57
58    /// `"TIME"`, `"FARE"`, `"TRANSFERS"` or `"WALK"`.
59    pub fn name(self) -> &'static str {
60        match self {
61            Criterion::Time => "TIME",
62            Criterion::Fare => "FARE",
63            Criterion::Transfers => "TRANSFERS",
64            Criterion::Walk => "WALK",
65        }
66    }
67
68    /// The criterion [`Criterion::name`] gives `name` (upper case), or
69    /// `None`.
70    pub fn from_name(name: &str) -> Option<Criterion> {
71        Criterion::ALL.iter().copied().find(|c| c.name() == name)
72    }
73}

75/// The chosen criteria as flags.
76#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
77pub(crate) struct Criteria {
78    pub(crate) time: bool,
79    pub(crate) fare: bool,
80    pub(crate) transfers: bool,
81    pub(crate) walk: bool,
82}

84impl Criteria {
85    pub(crate) fn of(list: &[Criterion]) -> Self {
86        Criteria {
87            time: list.contains(&Criterion::Time),
88            fare: list.contains(&Criterion::Fare),
89            transfers: list.contains(&Criterion::Transfers),
90            walk: list.contains(&Criterion::Walk),
91        }
92    }

95/// Parameters of [`trip_options`]. The street and public transport
96/// settings mean what they mean in [`ItineraryParams`](crate::ItineraryParams),
97/// with the same defaults. Fill it in by changing the fields of
98/// [`TripOptionsParams::default()`].
99#[derive(Debug, Clone)]
100#[non_exhaustive]
101pub struct TripOptionsParams {
102    /// The street mode of the trip without public transport and of the
103    /// access to it (default [`Mode::Walk`]).
104    pub mode: Mode,
105    /// The public transport modes; empty (the default) for street trips
106    /// only (then the only option is the direct trip).
107    pub transit_modes: Vec<TransitMode>,
108    /// The street mode from public transport to the destination (default
109    /// [`Mode::Walk`]).
110    pub egress_mode: Mode,
111    /// Service date `yyyymmdd` (needed with public transport).
112    pub date: u32,
113    /// First departure, seconds after midnight.
114    pub departure_time: i32,
115    /// Departure window in minutes (default 10): the trips are searched
116    /// from each minute `departure_time`, `departure_time + 60`, ...
117    pub time_window: i32,
118    /// The criteria, in any order; empty (the default) for all four when
119    /// there is a fare structure and all but [`Criterion::Fare`] otherwise.
120    /// `Fare` needs a fare structure.
121    pub criteria: Vec<Criterion>,
122    /// Longest walk, bicycle and car leg in minutes (default 120 each).
123    pub max_walk_time: i32,
124    #[allow(missing_docs)]
125    pub max_bike_time: i32,
126    #[allow(missing_docs)]
127    pub max_car_time: i32,
128    /// Trip limit in whole minutes (default 120): trips longer than this
129    /// are left out.
130    pub max_trip_duration: i32,
131    /// Walking speed, km/h (default 3.6).
132    pub walk_speed: f64,
133    /// Cycling speed, km/h (default 12).
134    pub bike_speed: f64,
135    /// Most vehicles boarded (default 3).
136    pub max_rides: usize,
137    /// Highest level of traffic stress a cyclist rides on, 1 to 4
138    /// (default 2).
139    pub max_lts: i32,
140    /// The fare structure (default `None`): needed for the fare
141    /// criterion, and gives every option its fare.
142    pub fare_structure: Option<Arc<FareStructure>>,
143    /// Highest fare (default infinity), read as a 32-bit float as the
144    /// other functions do; a finite value needs a `fare_structure`. The
145    /// options are the trips not dominated among those within the limit.
146    pub max_fare: f64,
147    /// Most options per pair (default 0: no limit); the first ones in the
148    /// order of [`TripOption::option`] are kept.
149    pub max_options: usize,
150    /// Seconds to which time on foot is compared (default 60: whole
151    /// minutes, rounded half up; 1: exact). The search is exact; of the
152    /// trips of each minute's exact frontier, those that another trip beats
153    /// once walking is rounded to this step (and total time to a tenth of a
154    /// minute, the resolution it is reported in) are left out, so that two
155    /// trips a few seconds apart are not both returned. Of trips equal in
156    /// those terms the one with the least exact total time, then transfers,
157    /// walking and fare is kept. The values returned are exact.
158    pub walk_resolution: i32,
159    /// Build each option's legs ([`TripOption::itinerary`], default true).
160    pub itineraries: bool,
161    /// Fill in the street edges, OpenStreetMap ids and stops of the legs,
162    /// as [`ItineraryParams::osm_link_ids`](crate::ItineraryParams::osm_link_ids)
163    /// (default false).
164    pub osm_link_ids: bool,
165    /// Car speeds and levels of traffic stress changed for this call, as in
166    /// [`ItineraryParams::scenario`](crate::ItineraryParams::scenario).
167    pub scenario: crate::Scenario,
168    /// Worker threads; 0 means all cores.
169    pub n_threads: usize,
170    /// Stops the computation when cancelled.
171    pub cancel: CancelToken,
172}

205impl TripOptionsParams {
206    /// The criteria a call uses: [`TripOptionsParams::criteria`], or its
207    /// default; an error for none valid (`Fare` without a fare structure).
208    pub fn resolved_criteria(&self) -> Result<Vec<Criterion>, RouteError> {
209        let mut c = if self.criteria.is_empty() {
210            Criterion::ALL
211                .iter()
212                .copied()
213                .filter(|&c| c != Criterion::Fare || self.fare_structure.is_some())
214                .collect()
215        } else {
216            self.criteria.clone()
217        };
218        c.sort_unstable();
219        c.dedup();
220        if c.contains(&Criterion::Fare) && self.fare_structure.is_none() {
221            return Err(RouteError::Invalid(
222                "the FARE criterion needs a fare_structure; create one with \
223                 setup_fare_structure() or read_fare_structure()"
224                    .into(),
225            ));
226        }
227        Ok(c)
228    }
229}

231/// One option of a pair: a way of making the trip that is not dominated
232/// on the chosen criteria at one or more departure minutes of the window.
233///
234/// Two trips found at the same or different minutes are the same option
235/// when they take the same street modes, ride the same routes between the
236/// same stops in the same order (whichever vehicles of those routes), and
237/// have the same fare, transfers and time on foot. Their total time,
238/// which counts the wait from the departure minute, changes from minute to
239/// minute; the option reports the trip of its *representative* minute, the
240/// one with the least total time (the earliest of equal ones).
241#[derive(Debug, Clone, PartialEq)]
242#[non_exhaustive]
243pub struct TripOption {
244    /// The pair (index into the origins and destinations given).
245    pub pair: usize,
246    /// Index within the pair, from 1, in the order of total time, then
247    /// fare, transfers, time on foot, representative minute and
248    /// description.
249    pub option: u32,
250    /// Door to door from the representative minute, seconds, and minutes
251    /// with one decimal place.
252    pub total_time_s: i32,
253    #[allow(missing_docs)]
254    pub total_time: f64,
255    /// The fare, in the currency of the fare structure (`None` without
256    /// one).
257    pub fare: Option<f64>,
258    /// Rides minus one; 0 without public transport.
259    pub transfers: u32,
260    /// Seconds on foot, and minutes with one decimal place.
261    pub walk_time_s: i32,
262    #[allow(missing_docs)]
263    pub walk_time: f64,
264    /// The representative departure minute (seconds after midnight), and
265    /// the arrival from it.
266    pub departure_s: i32,
267    #[allow(missing_docs)]
268    pub arrival_s: i32,
269    /// Every departure minute at which the option is on the frontier,
270    /// ascending (seconds after midnight).
271    pub departures: Vec<i32>,
272    /// The total time at each of [`TripOption::departures`], seconds, and
273    /// minutes with one decimal place.
274    pub totals_s: Vec<i32>,
275    #[allow(missing_docs)]
276    pub totals: Vec<f64>,
277    /// The street modes and the rides in order, as `WALK > BUS 343 > RAIL
278    /// LINHA1 > WALK` (route short names, else their ids).
279    pub description: String,
280    /// The legs of the trip of the representative minute, in the format
281    /// of [`detailed_itineraries`](crate::detailed_itineraries) (with
282    /// [`TripOptionsParams::itineraries`]).
283    pub itinerary: Option<Itinerary>,
284}
285
286impl TripOption {
287    /// How many departure minutes the option is on the frontier at.
288    pub fn n_departures(&self) -> usize {
289        self.departures.len()
290    }
291}

293/// peabiru's multi-criteria trip planner: for each origin-destination pair
294/// (`origins[i]` to `destinations[i]`, `(lat, lon)` degrees), the trips not
295/// dominated on the chosen criteria at each departure minute of the
296/// window, merged over the window into options.
297///
298/// At each minute the result is the exact Pareto set of the trips the
299/// network allows (the direct trip by the street mode, and public
300/// transport trips from the stops of the access search to those of the
301/// egress search, boarding at least 60 s after reaching a stop), one trip
302/// per vector of values: a trip is dominated when another is no worse on
303/// every chosen criterion and better on one; of trips with the same values
304/// on the chosen criteria, the one with the least total time, then
305/// transfers, time on foot and fare is kept. Of that exact set, the trips
306/// are then compared with total time to a tenth of a minute and walking to
307/// [`TripOptionsParams::walk_resolution`] (a minute by default), and those
308/// beaten in those terms are left out; the values returned are exact. There are no random draws:
309/// frequency-based trips are boarded at their expected times (half a
310/// headway of waiting; the scheduled departures with `exact_times = 1`),
311/// so the result does not change from call to call and differs on purpose
312/// from the Monte Carlo draws of the other functions. See the guide of the packages for the
313/// definitions and how this differs from `pareto_frontier()` and
314/// `detailed_itineraries()`.
315///
316/// Errors: invalid parameters, the fare criterion without a fare
317/// structure, cancellation, and a car search that cannot be routed
318/// reproducibly (see [`detailed_itineraries`](crate::detailed_itineraries)).
319pub fn trip_options(
320    network: &crate::Network,
321    origins: &[(f64, f64)],
322    destinations: &[(f64, f64)],
323    p: &TripOptionsParams,
324) -> Result<Vec<Vec<TripOption>>, RouteError> {
325    let (net, transit) = (network.street(), network.transit_pair());
326    crate::itinerary::trip_options_impl(net, transit, origins, destinations, p)
327}

329/// Write [`trip_options`]' result to `dir`, one file per pair with options,
330/// `<dir>/from_<origin id>_to_<destination id>.csv`: one row per leg of each
331/// option (with [`TripOptionsParams::itineraries`]) or per option, with the
332/// pair's ids and points, the option's columns (`option`, `total_time`,
333/// `fare` with a fare structure, `transfers`, `walk_time`,
334/// `departure_time`, `arrival_time`, `n_departures`, `first_departure`,
335/// `last_departure`, `description`) and the leg's (`segment`, `mode`,
336/// `cumulative_fare` with a fare structure, `segment_duration`, `wait`,
337/// `distance`, `route`, the four id columns with `osm_link_ids`, and the
338/// `geometry` as WKT unless `drop_geometry`). Numbers are written as in
339/// the files of [`CsvOutput`](crate::CsvOutput). `ids` and
340/// `points` are those of the pairs given to [`trip_options`].
341///
342/// The file names are built from the ids: an id holding a path separator
343/// (`/`, `\\`) or NUL (on Windows also `< > : " | ? *` or a control
344/// character) is an error of kind `InvalidInput`, before anything is written.
345#[allow(clippy::type_complexity)]
346pub fn write_trip_options_csv(
347    dir: &str,
348    ids: &[(String, String)],
349    points: &[((f64, f64), (f64, f64))],
350    options: &[Vec<TripOption>],
351    drop_geometry: bool,
352    osm_link_ids: bool,
353) -> std::io::Result<()> {
354    use crate::output::{csv_field, java_double, LINE_END};
355    crate::output::check_file_ids(ids.iter().flat_map(|(a, b)| [a.as_str(), b.as_str()]))
356        .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidInput, e.to_string()))?;
357    let clock = |s: i32| format!("{:02}:{:02}:{:02}", s / 3600, (s / 60) % 60, s % 60);
358    for (k, opts) in options.iter().enumerate() {
359        let Some(first) = opts.first() else {
360            continue;
361        };
362        let (from, to) = (&ids[k].0, &ids[k].1);
363        let ((olat, olon), (dlat, dlon)) = points[k];
364        let fares = first.fare.is_some();
365        let legs = first.itinerary.is_some();
366        let mut s = String::from("from_id,from_lat,from_lon,to_id,to_lat,to_lon,option,total_time");
367        if fares {
368            s.push_str(",fare");
369        }
370        s.push_str(
371            ",transfers,walk_time,departure_time,arrival_time,n_departures,first_departure,\
372             last_departure,description",
373        );
374        if legs {
375            s.push_str(",segment,mode");
376            if fares {
377                s.push_str(",cumulative_fare");
378            }
379            s.push_str(",segment_duration,wait,distance,route");
380            if osm_link_ids {
381                s.push_str(",osm_id_list,edge_id_list,board_stop_id,alight_stop_id");
382            }
383            if !drop_geometry {
384                s.push_str(",geometry");
385            }
386        }
387        s.push_str(LINE_END);
388        for x in opts {
389            let mut head = String::new();
390            csv_field(from, &mut head);
391            for v in [olat, olon] {
392                head.push(',');
393                java_double(v, &mut head);
394            }
395            head.push(',');
396            csv_field(to, &mut head);
397            for v in [dlat, dlon] {
398                head.push(',');
399                java_double(v, &mut head);
400            }
401            head.push_str(&format!(",{},", x.option));
402            java_double(x.total_time, &mut head);
403            if let Some(f) = x.fare {
404                head.push(',');
405                java_double(f, &mut head);
406            }
407            head.push_str(&format!(",{},", x.transfers));
408            java_double(x.walk_time, &mut head);
409            head.push_str(&format!(
410                ",{},{},{},{},{},",
411                clock(x.departure_s),
412                clock(x.arrival_s),
413                x.n_departures(),
414                clock(x.departures[0]),
415                clock(*x.departures.last().expect("a departure"))
416            ));
417            csv_field(&x.description, &mut head);
418            match &x.itinerary {
419                Some(it) if legs => {
420                    for (j, l) in it.legs.iter().enumerate() {
421                        s.push_str(&head);
422                        s.push_str(&format!(",{},", j + 1));
423                        csv_field(&l.mode, &mut s);
424                        if let Some(f) = l.cumulative_fare {
425                            s.push(',');
426                            java_double(f, &mut s);
427                        }
428                        for v in [l.segment_duration, l.wait] {
429                            s.push(',');
430                            java_double(v, &mut s);
431                        }
432                        s.push_str(&format!(",{},", l.distance));
433                        csv_field(&l.route, &mut s);
434                        if osm_link_ids {
435                            for v in [
436                                l.osm_id_list(),
437                                l.edge_id_list(),
438                                l.board_stop_id.clone(),
439                                l.alight_stop_id.clone(),
440                            ] {
441                                s.push(',');
442                                csv_field(&v, &mut s);
443                            }
444                        }
445                        if !drop_geometry {
446                            s.push(',');
447                            csv_field(&l.wkt(), &mut s);
448                        }
449                        s.push_str(LINE_END);
450                    }
451                }
452                _ => {
453                    s.push_str(&head);
454                    s.push_str(LINE_END);
455                }
456            }
457        }
458        std::fs::write(format!("{dir}/from_{from}_to_{to}.csv"), s)?;
459    }
460    Ok(())
