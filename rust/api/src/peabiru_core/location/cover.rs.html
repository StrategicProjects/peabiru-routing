9/// A candidate in the greedy queue, with the score it had after `round`
10/// sites were open. Scores only fall as sites open, so a stale score is an
11/// upper bound.
12struct Entry {
13    score: f64,
14    site: usize,
15    round: usize,
16}
17
18impl PartialEq for Entry {
19    fn eq(&self, other: &Self) -> bool {
20        self.cmp(other) == Ordering::Equal
21    }
22}
23impl Eq for Entry {}
24impl PartialOrd for Entry {
25    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
26        Some(self.cmp(other))
27    }
28}
29impl Ord for Entry {
30    /// The largest score first, then the lowest site.
31    fn cmp(&self, other: &Self) -> Ordering {
32        self.score
33            .total_cmp(&other.score)
34            .then(other.site.cmp(&self.site))
35    }
36}

38/// The demand points of a problem and who covers them.
39struct Cover<'a> {
40    p: &'a Problem<'a>,
41    cutoff: f64,
42    /// Covered by an existing facility.
43    fixed: Vec<bool>,
44    /// What a demand point is worth to the choice of sites: its weight
45    /// (maximal covering), or 1 for any point with weight (set covering).
46    worth: Vec<f64>,
47    /// How many of the chosen sites cover each demand point.
48    count: Vec<u32>,
49    /// The exclusive or of `position + 1` of the chosen sites covering each
50    /// demand point: with a count of 1, the one that does.
51    who: Vec<u32>,
52}
53
54impl Cover<'_> {
55    /// The demand points site `j` covers.
56    fn covered_by(&self, j: usize) -> impl Iterator<Item = usize> + '_ {
57        let (rows, times) = self.p.costs.site(j);
58        rows.iter()
59            .zip(times)
60            .filter(|(_, &t)| t <= self.cutoff)
61            .map(|(&i, _)| i as usize)
62    }
63
64    /// What site `j` would add to the chosen sites, in `values`.
65    fn gain(&self, j: usize, values: &[f64]) -> f64 {
66        self.covered_by(j)
67            .filter(|&i| !self.fixed[i] && self.count[i] == 0)
68            .map(|i| values[i])
69            .fold(0.0, |a, b| a + b)
70    }
71
72    fn open(&mut self, j: usize, position: usize) {
73        let (rows, times) = self.p.costs.site(j);
74        for (&i, &t) in rows.iter().zip(times) {
75            if t <= self.cutoff {
76                self.count[i as usize] += 1;
77                self.who[i as usize] ^= position as u32 + 1;
78            }
79        }
80    }
81
82    fn close(&mut self, j: usize, position: usize) {
83        let (rows, times) = self.p.costs.site(j);
84        for (&i, &t) in rows.iter().zip(times) {
85            if t <= self.cutoff {
86                self.count[i as usize] -= 1;
87                self.who[i as usize] ^= position as u32 + 1;
88            }
89        }
90    }
91
92    /// The demand covered, by the existing facilities and the chosen sites.
93    fn covered(&self) -> f64 {
94        (0..self.fixed.len())
95            .filter(|&i| self.fixed[i] || self.count[i] > 0)
96            .map(|i| self.p.demand[i])
97            .fold(0.0, |a, b| a + b)
98    }
99}

101pub(super) fn solve(p: &Problem) -> Result<LocationSolution, RouteError> {
102    let n = p.costs.n_demand();
103    let model = p.params.model;
104    let cancel = &p.params.cancel;
105    let mut c = Cover {
106        p,
107        cutoff: p.params.cutoff,
108        fixed: vec![false; n],
109        worth: match model {
110            LocationModel::SetCover => p
111                .demand
112                .iter()
113                .map(|&w| f64::from(u8::from(w > 0.0)))
114                .collect(),
115            _ => p.demand.to_vec(),
116        },
117        count: vec![0; n],
118        who: vec![0; n],
119    };
120    for &e in p.existing {
121        let rows: Vec<usize> = c.covered_by(e).collect();
122        for i in rows {
123            c.fixed[i] = true;
124        }
125    }
126    let covered_before = c.covered();
127
128    // what each candidate would cover alone, and who nobody covers
129    let mut potential = vec![0f64; p.costs.n_sites()];
130    let mut servable = c.fixed.clone();
131    let mut queue = BinaryHeap::with_capacity(p.candidates.len());
132    for &j in p.candidates {
133        cancel.check()?;
134        potential[j] = c.gain(j, p.demand);
135        for i in c.covered_by(j) {
136            servable[i] = true;
137        }
138        queue.push(Entry {
139            score: c.gain(j, &c.worth),
140            site: j,
141            round: 0,
142        });
143    }
144    let unservable: Vec<usize> = (0..n)
145        .filter(|&i| p.demand[i] > 0.0 && !servable[i])
146        .collect();
147
148    // greedy: the candidate worth most to the points still uncovered
149    let limit = match model {
150        LocationModel::SetCover => usize::MAX,
151        _ => p.params.n_facilities,
152    };
153    let floor = match model {
154        LocationModel::SetCover => 0.0,
155        _ => p.params.min_gain,
156    };
157    let mut chosen: Vec<usize> = Vec::new();
158    while chosen.len() < limit {
159        cancel.check()?;
160        let Some(top) = queue.pop() else { break };
161        if top.round < chosen.len() {
162            queue.push(Entry {
163                score: c.gain(top.site, &c.worth),
164                site: top.site,
165                round: chosen.len(),
166            });
167            continue;
168        }
169        if top.score <= 0.0 || top.score < floor {
170            break;
171        }
172        c.open(top.site, chosen.len());
173        chosen.push(top.site);
174    }
175    let objective_of = |c: &Cover, chosen: &[usize]| match model {
176        LocationModel::SetCover => chosen.len() as f64,
177        _ => c.covered(),
178    };
179    let objective_greedy = objective_of(&c, &chosen);
180
181    let mut n_swaps = 0;
182    if p.params.swap && model == LocationModel::SetCover {
183        // drop the sites the others make redundant, the last chosen first
184        let mut k = chosen.len();
185        while k > 0 {
186            k -= 1;
187            cancel.check()?;
188            let j = chosen[k];
189            let needed = c
190                .covered_by(j)
191                .any(|i| !c.fixed[i] && c.count[i] == 1 && c.worth[i] > 0.0);
192            if !needed {
193                // positions matter to `who` only while swapping: rebuild
194                chosen.remove(k);
195                c.count.iter_mut().for_each(|x| *x = 0);
196                c.who.iter_mut().for_each(|x| *x = 0);
197                for (position, &site) in chosen.iter().enumerate() {
198                    c.open(site, position);
199                }
200                n_swaps += 1;
201            }
202        }
203    } else if p.params.swap && !chosen.is_empty() {
204        n_swaps = substitute(&mut c, &mut chosen)?;
205    }
206    let objective = objective_of(&c, &chosen);
207    let covered = c.covered();
208
209    // the chosen sites by what each adds
210    let mut left = chosen;
211    let (mut sites, mut marginal, mut cumulative) = (Vec::new(), Vec::new(), Vec::new());
212    c.count.iter_mut().for_each(|x| *x = 0);
213    let mut so_far = covered_before;
214    while !left.is_empty() {
215        let mut best = 0;
216        let mut best_score = f64::NEG_INFINITY;
217        for (k, &j) in left.iter().enumerate() {
218            let score = c.gain(j, &c.worth);
219            if score > best_score || (score == best_score && j < left[best]) {
220                (best, best_score) = (k, score);
221            }
222        }
223        let j = left.swap_remove(best);
224        let gain = c.gain(j, p.demand);
225        c.open(j, sites.len());
226        so_far += gain;
227        sites.push(j);
228        marginal.push(gain);
229        cumulative.push(so_far);
230    }
231
232    Ok(LocationSolution {
233        model,
234        sites,
235        marginal,
236        cumulative,
237        potential,
238        assigned_site: Vec::new(),
239        assigned_minutes: Vec::new(),
240        objective_before: match model {
241            LocationModel::SetCover => 0.0,
242            _ => covered_before,
243        },
244        objective_greedy,
245        objective,
246        covered_before,
247        covered,
248        total_demand: p.demand.iter().fold(0.0, |a, b| a + b),
249        unservable_demand: unservable.iter().fold(0.0, |a, &i| a + p.demand[i]),
250        unservable,
251        n_swaps,
252    })
253}

255/// Vertex substitution for maximal covering: while some swap of a chosen
256/// site for a closed candidate covers more demand, make the best of them.
257fn substitute(c: &mut Cover, chosen: &mut [usize]) -> Result<usize, RouteError> {
258    let p = c.p;
259    let mut n_swaps = 0;
260    let mut current = c.covered();
261    loop {
262        // what closing each chosen site would uncover
263        let lost: Vec<f64> = chosen
264            .iter()
265            .enumerate()
266            .map(|(r, &j)| {
267                c.covered_by(j)
268                    .filter(|&i| !c.fixed[i] && c.count[i] == 1 && c.who[i] == r as u32 + 1)
269                    .map(|i| p.demand[i])
270                    .fold(0.0, |a, b| a + b)
271            })
272            .collect();
273        let closed: Vec<usize> = p
274            .candidates
275            .iter()
276            .copied()
277            .filter(|j| !chosen.contains(j))
278            .collect();
279        let view = &*c;
280        let swaps: Vec<Option<(Key, usize)>> = map_of(
281            &closed,
282            p.params.n_threads,
283            &p.params.cancel,
284            || vec![0f64; chosen.len()],
285            |j, shared: &mut Vec<f64>| {
286                // demand `j` covers that nobody covers, and that only the
287                // chosen site `r` covers (kept if `r` is the one to close)
288                let mut alone = 0f64;
289                for i in view.covered_by(j) {
290                    if view.fixed[i] {
291                        continue;
292                    }
293                    match view.count[i] {
294                        0 => alone += p.demand[i],
295                        1 => shared[view.who[i] as usize - 1] += p.demand[i],
296                        _ => {}
297                    }
298                }
299                let mut best: Option<(f64, usize)> = None;
300                for (r, kept) in shared.iter_mut().enumerate() {
301                    let delta = alone + *kept - lost[r];
302                    *kept = 0.0;
303                    if delta > 0.0 && best.map_or(true, |b| delta > b.0) {
304                        best = Some((delta, r));
305                    }
306                }
307                best.map(|(delta, r)| ([-delta, 0.0], r))
308            },
309        )?;
310        let Some(k) = first_best(&swaps) else { break };
311        let (j, r) = (closed[k], swaps[k].as_ref().map_or(0, |s| s.1));
312        let out = chosen[r];
313        c.close(out, r);
314        c.open(j, r);
315        let now = c.covered();
316        if now > current {
317            chosen[r] = j;
318            current = now;
319            n_swaps += 1;
320        } else {
321            // a gain that was only rounding
322            c.close(j, r);
323            c.open(out, r);
324            break;
325        }
326    }
327    Ok(n_swaps)
