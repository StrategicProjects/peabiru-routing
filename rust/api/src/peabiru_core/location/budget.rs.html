1//! Sharing a budget of facilities among groups (municipalities, districts)
2//! from the curve of each: what its first, second, third facility adds.
3
4use crate::RouteError;
5
6/// How [`allocate_budget`] shares the facilities out.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
8#[non_exhaustive]
9pub enum BudgetCriterion {
10    /// The most demand covered in all: each facility goes where it adds
11    /// most.
12    Utilitarian,
13    /// The worst-off group first: each facility goes to the group with
14    /// the lowest share of its demand covered.
15    Maximin,
16}
17
18impl BudgetCriterion {
19    /// The criterion's name in the packages: `utilitarian` or `maximin`.
20    pub fn name(self) -> &'static str {
21        match self {
22            BudgetCriterion::Utilitarian => "utilitarian",
23            BudgetCriterion::Maximin => "maximin",
24        }
25    }
26
27    /// The criterion of that name (`None` for any other text).
28    pub fn from_name(name: &str) -> Option<BudgetCriterion> {
29        Some(match name {
30            "utilitarian" => BudgetCriterion::Utilitarian,
31            "maximin" => BudgetCriterion::Maximin,
32            _ => return None,
33        })
34    }
35}
36
37/// One group of [`allocate_budget`].
38#[derive(Debug, Clone, PartialEq)]
39pub struct BudgetGroup<'a> {
40    /// What each further facility adds in the group, in order: the
41    /// `marginal` of a [`LocationSolution`](crate::LocationSolution) of
42    /// maximal covering solved without swaps. It may not increase.
43    pub marginal: &'a [f64],
44    /// The demand covered in the group before any new facility.
45    pub covered: f64,
46    /// The demand of the group, for its share covered. NaN when unknown:
47    /// the shares are then NaN and [`BudgetCriterion::Maximin`] is
48    /// refused.
49    pub total: f64,
50}
51
52/// What [`allocate_budget_with`] takes besides the criterion, one entry
53/// per group in each list that is given. With none (the default) it is
54/// [`allocate_budget`].
55#[derive(Debug, Clone, Copy, PartialEq, Default)]
56#[non_exhaustive]
57pub struct BudgetRules<'a> {
58    /// How much the demand of each group counts (1 without the list),
59    /// finite and above zero. With [`BudgetCriterion::Utilitarian`] the
60    /// gains of a group are multiplied by its weight; with
61    /// [`BudgetCriterion::Maximin`] its share covered is divided by it.
62    pub weights: Option<&'a [f64]>,
63    /// The priority tier of each group (0 without the list), the lowest
64    /// first: a facility only goes to a group when no group of a lower
65    /// tier has one left to open that adds anything.
66    pub tiers: Option<&'a [u32]>,
67    /// Whether each group may receive facilities (all without the list).
68    /// The others get none and still count in the demand covered and in
69    /// the lowest share.
70    pub eligible: Option<&'a [bool]>,
71}
72
73/// How [`allocate_budget`] shared the facilities out, one entry per
74/// facility in the order they were given: the answer for any smaller
75/// budget is the beginning of the lists.
76#[derive(Debug, Clone, PartialEq, Default)]
77#[non_exhaustive]
78pub struct BudgetAllocation {
79    /// The group that gets each facility.
80    pub group: Vec<usize>,
81    /// What that facility adds.
82    pub gain: Vec<f64>,
83    /// The demand covered in all the groups after it, what was covered
84    /// before included.
85    pub covered: Vec<f64>,
86    /// The lowest share covered among the groups after it (NaN without
87    /// the totals of the groups).
88    pub lowest_share: Vec<f64>,
89    /// How many facilities each group got.
90    pub units: Vec<usize>,
91    /// The order in which the groups got their first facility: 0 for the
92    /// first group to receive one, 1 for the next, `None` for a group
93    /// that got none. One number per group that sums up the criterion,
94    /// the weights and the tiers.
95    pub priority: Vec<Option<usize>>,
96}

98/// Share `budget` facilities among groups, given what each further
99/// facility adds in each group.
100///
101/// With [`BudgetCriterion::Utilitarian`] each facility goes to the group
102/// where the next one adds most. As the gains of a group do not increase,
103/// this covers the most demand possible with every budget up to `budget`
104/// (marginal analysis: Fox, 1966). With [`BudgetCriterion::Maximin`] each
105/// facility goes to the group with the lowest share of its demand covered
106/// that still has a facility to open, which gives the highest possible
107/// lowest share with every budget. Ties go to the larger gain, then to
108/// the group listed first. Fewer than `budget` facilities come back when
109/// no group has a facility left that adds anything.
110///
111/// Comparing the `covered` of the two criteria for the same budget gives
112/// what the fairer allocation costs (Bertsimas, Farias and Trichakis,
113/// 2011).
114///
115/// Errors ([`RouteError::Invalid`]): a gain that is negative, not finite
116/// or larger than the one before it in its group; a `covered` that is
117/// negative or not finite; a `total` that is negative or infinite; and
118/// [`BudgetCriterion::Maximin`] with a total that is NaN.
119pub fn allocate_budget(
120    groups: &[BudgetGroup],
121    budget: usize,
122    criterion: BudgetCriterion,
123) -> Result<BudgetAllocation, RouteError> {
124    allocate_budget_with(groups, budget, criterion, &BudgetRules::default())
125}
126
127/// [`allocate_budget`] with weights, priority tiers and groups left out
128/// ([`BudgetRules`]).
129///
130/// With weights, [`BudgetCriterion::Utilitarian`] gives each facility to
131/// the group where the next one adds most once multiplied by the weight
132/// of the group: the weighted gains of a group still do not increase, so
133/// this is the largest weighted sum for every budget.
134/// [`BudgetCriterion::Maximin`] gives it to the group with the lowest
135/// share covered divided by its weight. With tiers the criterion only
136/// compares the groups of the lowest tier that still has a facility to
137/// open that adds something. A group that is not eligible gets none.
138///
139/// The gains, the demand covered and the lowest share of the result are
140/// never weighted and take in every group, eligible or not, so that runs
141/// with and without the rules can be compared.
142///
143/// Errors ([`RouteError::Invalid`]): those of [`allocate_budget`]; a list
144/// of [`BudgetRules`] whose length is not the number of groups; and a
145/// weight that is not finite or not above zero.
146pub fn allocate_budget_with(
147    groups: &[BudgetGroup],
148    budget: usize,
149    criterion: BudgetCriterion,
150    rules: &BudgetRules,
151) -> Result<BudgetAllocation, RouteError> {
152    let invalid = |m: String| Err(RouteError::Invalid(m));
153    for (what, len) in [
154        ("weights", rules.weights.map(<[f64]>::len)),
155        ("tiers", rules.tiers.map(<[u32]>::len)),
156        ("eligible", rules.eligible.map(<[bool]>::len)),
157    ] {
158        if len.is_some_and(|n| n != groups.len()) {
159            return invalid(format!(
160                "the {what} of a budget allocation must be one per group: got {}, the groups are {}",
161                len.unwrap_or(0),
162                groups.len()
163            ));
164        }
165    }
166    if let Some((g, w)) = rules.weights.and_then(|w| {
167        w.iter()
168            .enumerate()
169            .find(|(_, &w)| !(w.is_finite() && w > 0.0))
170    }) {
171        return invalid(format!(
172            "the weight of a group must be finite and above zero, got {w} for group {g}"
173        ));
174    }
175    let weight = |g: usize| rules.weights.map_or(1.0, |w| w[g]);
176    let tier = |g: usize| rules.tiers.map_or(0, |t| t[g]);
177    let eligible = |g: usize| rules.eligible.map_or(true, |e| e[g]);
178    for (g, group) in groups.iter().enumerate() {
179        for (k, &x) in group.marginal.iter().enumerate() {
180            if !(x.is_finite() && x >= 0.0) {
181                return invalid(format!(
182                    "the gains must be finite and not negative, got {x} for facility {} of group {g}",
183                    k + 1
184                ));
185            }
186            if k > 0 && x > group.marginal[k - 1] * (1.0 + 1e-9) {
187                return invalid(format!(
188                    "the gains of a group may not increase: facility {} of group {g} adds {x}, the one before it {} (curves of max_cover solved without swaps do not increase)",
189                    k + 1,
190                    group.marginal[k - 1]
191                ));
192            }
193        }
194        if !(group.covered.is_finite() && group.covered >= 0.0) {
195            return invalid(format!(
196                "the demand covered in a group must be finite and not negative, got {} for group {g}",
197                group.covered
198            ));
199        }
200        if group.total < 0.0 || group.total.is_infinite() {
201            return invalid(format!(
202                "the demand of a group must be finite and not negative, got {} for group {g}",
203                group.total
204            ));
205        }
206        if criterion == BudgetCriterion::Maximin && group.total.is_nan() {
207            return invalid(format!(
208                "the maximin criterion needs the demand of every group, missing for group {g}"
209            ));
210        }
211    }
212    let mut covered: Vec<f64> = groups.iter().map(|g| g.covered).collect();
213    let share = |g: usize, covered: &[f64]| -> f64 {
214        let total = groups[g].total;
215        if total > 0.0 {
216            covered[g] / total
217        } else if total == 0.0 {
218            1.0
219        } else {
220            f64::NAN
221        }
222    };
223    let mut out = BudgetAllocation {
224        units: vec![0; groups.len()],
225        priority: vec![None; groups.len()],
226        ..BudgetAllocation::default()
227    };
228    let mut entered = 0usize;
229    let mut sum = covered.iter().fold(0.0, |a, b| a + b);
230    while out.group.len() < budget {
231        // the groups with a facility left that adds something; `worth` is
232        // the gain times the weight of the group
233        let mut best: Option<(usize, f64, f64)> = None;
234        for (g, group) in groups.iter().enumerate() {
235            if !eligible(g) {
236                continue;
237            }
238            let Some(&gain) = group.marginal.get(out.units[g]) else {
239                continue;
240            };
241            if gain <= 0.0 {
242                continue;
243            }
244            let worth = gain * weight(g);
245            let wins = match best {
246                None => true,
247                Some((b, _, _)) if tier(g) != tier(b) => tier(g) < tier(b),
248                Some((b, _, b_worth)) => match criterion {
249                    BudgetCriterion::Utilitarian => worth > b_worth,
250                    _ => {
251                        let mine = share(g, &covered) / weight(g);
252                        let theirs = share(b, &covered) / weight(b);
253                        mine < theirs || (mine == theirs && worth > b_worth)
254                    }
255                },
256            };
257            if wins {
258                best = Some((g, gain, worth));
259            }
260        }
261        let Some((g, gain, _)) = best else { break };
262        if out.units[g] == 0 {
263            out.priority[g] = Some(entered);
264            entered += 1;
265        }
266        out.units[g] += 1;
267        covered[g] += gain;
268        sum += gain;
269        out.group.push(g);
270        out.gain.push(gain);
271        out.covered.push(sum);
272        out.lowest_share.push(
273            (0..groups.len())
274                .map(|g| share(g, &covered))
275                .fold(f64::INFINITY, f64::min),
276        );
277    }
278    if groups.is_empty() || groups.iter().any(|g| g.total.is_nan()) {
279        out.lowest_share.iter_mut().for_each(|x| *x = f64::NAN);
280    }
281    Ok(out)
282}
