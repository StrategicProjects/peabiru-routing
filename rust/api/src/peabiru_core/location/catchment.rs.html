1//! Two-step floating catchment area: supply against demand around each
2//! place.

4use super::CostTable;
5use crate::transit::decay::Decay;
6use crate::{CancelToken, DecayFunction, RouteError};

8/// How [`floating_catchment`] weights travel time. Fill it in by changing
9/// the fields of [`FloatingCatchmentParams::default()`].
10#[derive(Debug, Clone)]
11#[non_exhaustive]
12pub struct FloatingCatchmentParams {
13    /// The catchment in minutes. With [`DecayFunction::Step`] a pair
14    /// counts in full up to it (`time <= cutoff`) and not at all beyond;
15    /// the other functions take it as the accessibility functions do (the
16    /// half-life of `Exponential`, the centre of `Linear`, the median of
17    /// `Logistic`), and `FixedExponential` ignores it. No default: it must
18    /// be set, finite and positive, except for `FixedExponential`.
19    pub cutoff: f64,
20    /// The weight of a pair by its travel time (default
21    /// [`DecayFunction::Step`], the two-step floating catchment area in
22    /// its original form). Any other function gives the enhanced form,
23    /// with the weight falling continuously with travel time: every pair
24    /// of the table counts with its weight, so the table should reach as
25    /// far as the weights matter.
26    pub decay: DecayFunction,
27    /// Cancels the computation: checked between facilities.
28    pub cancel: CancelToken,
29}
30
31impl Default for FloatingCatchmentParams {
32    fn default() -> Self {
33        FloatingCatchmentParams {
34            cutoff: f64::NAN,
35            decay: DecayFunction::Step,
36            cancel: CancelToken::default(),
37        }
38    }
39}
40
41/// The result of [`floating_catchment`].
42#[derive(Debug, Clone, PartialEq)]
43#[non_exhaustive]
44pub struct FloatingCatchment {
45    /// Per demand point, the accessibility index: the sum, over the
46    /// facilities it reaches, of the facility's ratio times the weight of
47    /// the pair. Supply per unit of demand (places per resident, say).
48    pub index: Vec<f64>,
49    /// Per facility, its supply divided by the demand in its catchment
50    /// (0 when the catchment holds no demand).
51    pub ratio: Vec<f64>,
52    /// Per facility, the demand in its catchment: the sum of the demand
53    /// weights times the weight of each pair.
54    pub catchment_demand: Vec<f64>,
55    /// The supply of the facilities whose catchment holds no demand. The
56    /// rest of the supply is what the index shares out: the sum of demand
57    /// times index equals the total supply less this.
58    pub unreached_supply: f64,
59}
60
61/// The two-step floating catchment area of Luo and Wang (2003) and, with a
62/// decay function, its enhanced form (Luo and Qi, 2009).
63///
64/// Step 1: each facility's supply is divided by the demand within its
65/// catchment, each demand point weighted by the decay of its travel time.
66/// Step 2: each demand point adds up the ratios of the facilities within
67/// its catchment, weighted the same way. `costs` gives the minutes from
68/// each demand point (row) to each facility (column), `demand` the weight
69/// of each demand point and `supply` the capacity of each facility.
70///
71/// Errors: [`RouteError::Invalid`] for weights or capacities that do not
72/// match the table, are negative or are not finite, a cutoff that is not
73/// set, or a decay parameter out of range; [`RouteError::Cancelled`].
74pub fn floating_catchment(
75    costs: &CostTable,
76    demand: &[f64],
77    supply: &[f64],
78    params: &FloatingCatchmentParams,
79) -> Result<FloatingCatchment, RouteError> {
80    let invalid = |m: String| Err(RouteError::Invalid(m));
81    if demand.len() != costs.n_demand() {
82        return invalid(format!(
83            "demand has {} weights, but the table has {} demand points",
84            demand.len(),
85            costs.n_demand()
86        ));
87    }
88    if supply.len() != costs.n_sites() {
89        return invalid(format!(
90            "supply has {} capacities, but the table has {} facilities",
91            supply.len(),
92            costs.n_sites()
93        ));
94    }
95    if let Some(i) = demand.iter().position(|w| !(w.is_finite() && *w >= 0.0)) {
96        return invalid(format!(
97            "demand weights must be finite and not negative, got {} for demand point {i}",
98            demand[i]
99        ));
100    }
101    if let Some(j) = supply.iter().position(|w| !(w.is_finite() && *w >= 0.0)) {
102        return invalid(format!(
103            "supply must be finite and not negative, got {} for facility {j}",
104            supply[j]
105        ));
106    }
107    let fixed = matches!(params.decay, DecayFunction::FixedExponential { .. });
108    let cutoff_max = f64::from(i32::MAX / 60);
109    if !fixed && !(params.cutoff > 0.0 && params.cutoff <= cutoff_max) {
110        return invalid(format!(
111            "floating_catchment needs a cutoff in minutes, finite and positive, got {}",
112            params.cutoff
113        ));
114    }
115    let decay = Decay::new(params.decay.clone()).map_err(RouteError::Invalid)?;
116    let step = matches!(params.decay, DecayFunction::Step);
117    let cutoff_seconds = if fixed {
118        0
119    } else {
120        (params.cutoff * 60.0).round() as i32
121    };
122    let weight = |minutes: f64| -> f64 {
123        if step {
124            f64::from(u8::from(minutes <= params.cutoff))
125        } else {
126            // `as` saturates: a time beyond i32 seconds weighs nothing
127            decay.weight(cutoff_seconds, (minutes * 60.0).round() as i32)
128        }
129    };
130
131    let n_sites = costs.n_sites();
132    let mut ratio = vec![0f64; n_sites];
133    let mut catchment_demand = vec![0f64; n_sites];
134    let mut index = vec![0f64; costs.n_demand()];
135    let mut unreached_supply = 0f64;
136    for j in 0..n_sites {
137        params.cancel.check()?;
138        let (rows, times) = costs.site(j);
139        let around: f64 = rows
140            .iter()
141            .zip(times)
142            .map(|(&i, &t)| demand[i as usize] * weight(t))
143            .fold(0.0, |a, b| a + b);
144        catchment_demand[j] = around;
145        if around > 0.0 {
146            ratio[j] = supply[j] / around;
147            for (&i, &t) in rows.iter().zip(times) {
148                index[i as usize] += ratio[j] * weight(t);
149            }
150        } else {
151            unreached_supply += supply[j];
152        }
153    }
154    Ok(FloatingCatchment {
155        index,
156        ratio,
157        catchment_demand,
158        unreached_supply,
159    })
160}
