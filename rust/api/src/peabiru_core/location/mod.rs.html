1//! Facility location on a table of travel times: where to open facilities
2//! ([`locate_facilities`]) and how supply compares with demand around each
3//! place ([`floating_catchment`]).
4//!
5//! Everything here works on a [`CostTable`] (demand point x site ->
6//! minutes, a missing pair being unreachable) and never looks at how the
7//! table was computed: by a street or a public transport matrix of this
8//! crate, or by anything else.
9//!
10//! The four siting models are solved by heuristics, with no integer
11//! programming solver:
12//!
13//! - maximal covering (Church and ReVelle, 1974): greedy, each site the
14//!   one that covers the most demand still uncovered, which reaches at
15//!   least `1 - 1/e` of the best coverage possible (Nemhauser, Wolsey and
16//!   Fisher, 1978), then vertex substitution;
17//! - location set covering (Toregas et al., 1971): greedy by the number
18//!   of demand points newly covered, then the sites the others make
19//!   redundant are dropped; with a coverage target below 1 (partial set
20//!   covering), greedy by the demand newly covered until the target is
21//!   reached;
22//! - p-median (Hakimi, 1964): greedy, then the vertex substitution of
23//!   Teitz and Bart (1968);
24//! - p-center (Hakimi, 1964): greedy on the largest travel time (ties by
25//!   the weighted sum), then vertex substitution on the same pair.
26//!
27//! Maximal covering also has a capacitated form (Current and Storbeck,
28//! 1988), in which a facility serves only so much demand and a site may
29//! open more than once: greedy by the demand each further unit would
30//! serve, the nearest demand first.
31//!
32//! Ties are always broken by the lowest site index, the candidates are
33//! evaluated in parallel only where each one's value is computed on its
34//! own, and every sum is taken in a fixed order: the result does not
35//! depend on the number of threads.

58/// The siting model of [`locate_facilities`].
59#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
60#[non_exhaustive]
61pub enum LocationModel {
62    /// Maximal covering: with [`LocationParams::n_facilities`] new
63    /// facilities, cover the most demand within
64    /// [`LocationParams::cutoff`] minutes.
65    MaxCover,
66    /// Location set covering: the fewest new facilities that leave every
67    /// demand point some site can cover within
68    /// [`LocationParams::cutoff`] minutes, or, with
69    /// [`LocationParams::coverage`] below 1, that share of the demand.
70    SetCover,
71    /// p-median: [`LocationParams::n_facilities`] new facilities with the
72    /// smallest demand-weighted sum of travel times to the nearest
73    /// facility.
74    PMedian,
75    /// p-center: [`LocationParams::n_facilities`] new facilities with the
76    /// smallest travel time of the demand point farthest from its nearest
77    /// facility.
78    PCenter,
79}
80
81impl LocationModel {
82    /// The model's name in the packages: `max_cover`, `set_cover`,
83    /// `p_median` or `p_center`.
84    pub fn name(self) -> &'static str {
85        match self {
86            LocationModel::MaxCover => "max_cover",
87            LocationModel::SetCover => "set_cover",
88            LocationModel::PMedian => "p_median",
89            LocationModel::PCenter => "p_center",
90        }
91    }
92
93    /// The model of that name (`None` for any other text).
94    pub fn from_name(name: &str) -> Option<LocationModel> {
95        Some(match name {
96            "max_cover" => LocationModel::MaxCover,
97            "set_cover" => LocationModel::SetCover,
98            "p_median" => LocationModel::PMedian,
99            "p_center" => LocationModel::PCenter,
100            _ => return None,
101        })
102    }
103
104    fn is_cover(self) -> bool {
105        matches!(self, LocationModel::MaxCover | LocationModel::SetCover)
106    }
107}

109/// What [`locate_facilities`] solves. Fill it in by changing the fields of
110/// [`LocationParams::default()`].
111#[derive(Debug, Clone)]
112#[non_exhaustive]
113pub struct LocationParams {
114    /// The model (default: maximal covering).
115    pub model: LocationModel,
116    /// How many new facilities to open (default 1). Fewer are returned
117    /// when no further site improves the objective. Ignored by
118    /// [`LocationModel::SetCover`], which opens as many as it takes.
119    pub n_facilities: usize,
120    /// The covering models' service standard in minutes: a demand point is
121    /// covered by a site at most this far (`time <= cutoff`). No default:
122    /// it must be set for [`LocationModel::MaxCover`] and
123    /// [`LocationModel::SetCover`]; the other models ignore it.
124    pub cutoff: f64,
125    /// Sites of the table (column indices) that are already open. They
126    /// count first: only the demand they leave uncovered (covering
127    /// models), or the travel time they leave (p-median, p-center), drives
128    /// the choice of the new sites. They are never closed or moved. Every
129    /// other site of the table is a candidate.
130    pub existing: Vec<usize>,
131    /// Improve the greedy solution by vertex substitution (default
132    /// `true`): swap an open new site for a closed candidate while that
133    /// improves the objective, taking at each step the best swap of all.
134    /// For [`LocationModel::SetCover`] the improvement is to drop the
135    /// sites the others make redundant. The result is never worse than
136    /// the greedy one ([`LocationSolution::objective_greedy`]).
137    pub swap: bool,
138    /// [`LocationModel::MaxCover`] stops opening sites when the best one
139    /// left would cover less than this much uncovered demand (default 0:
140    /// it stops only when a site would cover nothing).
141    pub min_gain: f64,
142    /// For [`LocationModel::SetCover`], the share of the total demand
143    /// (the sum of the weights) to cover, the demand the existing
144    /// facilities cover included: a number above 0 and at most 1 (default
145    /// 1). With 1 every demand point with weight that some site can cover
146    /// is covered, whatever its weight. Below 1 the model is the partial
147    /// set covering: the sites are chosen by the demand each newly covers
148    /// and no more are opened once the target is reached; when the target
149    /// is out of reach (too much demand that no site covers), all the
150    /// demand that can be covered is.
151    pub coverage: f64,
152    /// For [`LocationModel::MaxCover`], how much demand one facility at
153    /// each site of the table can serve (one value per site, existing
154    /// facilities included; finite and positive). `None` (the default): a
155    /// facility serves all the demand within the cutoff. With it the
156    /// model is the capacitated maximal covering: a candidate may open
157    /// more than once ([`LocationSolution::sites`] then repeats it),
158    /// `n_facilities` counts the facilities opened, not the sites, and may
159    /// exceed the number of candidates; demand is served from the nearest
160    /// points outwards and a point may be served in part, or by more than
161    /// one facility. The existing facilities serve first, each pair of
162    /// facility and demand point in the order of their travel times.
163    /// [`LocationParams::swap`] has no effect.
164    pub capacity: Option<Vec<f64>>,
165    /// For [`LocationModel::PMedian`] and [`LocationModel::PCenter`], the
166    /// minutes charged to a demand point that reaches no open facility
167    /// (while some site of the table could serve it). `None` (the
168    /// default): twice the largest time of the table, so that serving a
169    /// point always beats shortening the trip of another.
170    pub unreachable_time: Option<f64>,
171    /// Worker threads (0 = all cores). The result does not depend on it.
172    pub n_threads: usize,
173    /// Cancels the computation: checked between candidates.
174    pub cancel: CancelToken,
175}

195/// The sites [`locate_facilities`] chose and what they achieve.
196///
197/// The *objective* is, by model: the demand covered (maximal covering),
198/// the number of new sites (set covering), the demand-weighted sum of
199/// minutes to the nearest facility (p-median) and the largest such time
200/// (p-center). Only the first is to be maximised.
201#[derive(Debug, Clone, PartialEq)]
202#[non_exhaustive]
203pub struct LocationSolution {
204    /// The model solved.
205    pub model: LocationModel,
206    /// The new sites (column indices of the table; with
207    /// [`LocationParams::capacity`], one entry per facility opened, so a
208    /// site may repeat), ordered by what each adds: the first is the site of the solution that does best alone
209    /// (after the existing facilities), the second the one that adds most
210    /// to it, and so on. Without [`LocationParams::swap`] this is the
211    /// order in which the greedy heuristic chose them.
212    pub sites: Vec<usize>,
213    /// What each site of `sites` adds to the ones before it: demand newly
214    /// covered (covering models), or by how much the objective falls
215    /// (p-median, p-center).
216    pub marginal: Vec<f64>,
217    /// The state after each site of `sites`: demand covered so far,
218    /// including what the existing facilities cover (covering models), or
219    /// the objective so far (p-median, p-center).
220    pub cumulative: Vec<f64>,
221    /// Per site of the table, what it would add if it were the only new
222    /// facility, in the unit of `marginal`: for the covering models, the
223    /// demand not covered by the existing facilities that it would cover
224    /// alone. 0 for the existing facilities.
225    pub potential: Vec<f64>,
226    /// Per demand point, the nearest open facility, existing or new
227    /// (`None` when it reaches none); ties go to the lowest site index.
228    pub assigned_site: Vec<Option<usize>>,
229    /// Per demand point, the minutes to `assigned_site` (NaN when `None`).
230    pub assigned_minutes: Vec<f64>,
231    /// The objective with the existing facilities only. For set covering,
232    /// 0.
233    pub objective_before: f64,
234    /// The objective of the greedy solution, before the improvement.
235    pub objective_greedy: f64,
236    /// The objective of the solution.
237    pub objective: f64,
238    /// The demand covered by the existing facilities (covering models; NaN
239    /// for the others).
240    pub covered_before: f64,
241    /// The demand covered by the solution (covering models; NaN for the
242    /// others).
243    pub covered: f64,
244    /// For [`LocationModel::MaxCover`], a ceiling on the demand that any
245    /// choice of at most [`LocationParams::n_facilities`] new sites can
246    /// cover, existing facilities included: the optimum is between
247    /// `covered` and this value, so `covered / upper_bound` is a floor on
248    /// the share of the optimum that the solution reaches. It is the
249    /// smaller of two bounds that hold for any covering problem, the
250    /// demand covered by a set of sites plus the `n_facilities` largest
251    /// gains of the sites outside it, taken at the existing facilities
252    /// alone and at the solution, and never more than the demand some
253    /// site can cover. NaN for the other models.
254    pub upper_bound: f64,
255    /// With [`LocationParams::capacity`], how much of the demand of each
256    /// demand point is served, by the existing facilities and the new
257    /// ones: between 0 and its weight. Empty otherwise.
258    pub served: Vec<f64>,
259    /// With [`LocationParams::capacity`], who serves whom: one entry per
260    /// pair of demand point and facility site with demand served between
261    /// them, as `(demand point, site, demand served)`, ordered by demand
262    /// point, then by site. The existing facilities are included, and the
263    /// facilities opened at the same site are added up. Empty otherwise.
264    pub flows: Vec<(usize, usize, f64)>,
265    /// The sum of the demand weights.
266    pub total_demand: f64,
267    /// The demand points with weight that no site of the table, existing
268    /// or candidate, can serve: within the cutoff for the covering models,
269    /// at all for p-median and p-center. They are left out of the
270    /// objectives.
271    pub unservable: Vec<usize>,
272    /// The sum of the weights of `unservable`.
273    pub unservable_demand: f64,
274    /// How many improving swaps were made (set covering: redundant sites
275    /// dropped).
276    pub n_swaps: usize,
277}

279/// Choose where to open facilities.
280///
281/// `costs` gives the minutes from each demand point to each site;
282/// `demand` the weight of each demand point (residents, pupils...), which
283/// must be finite and not negative. The sites in
284/// [`LocationParams::existing`] are already open; all the others are
285/// candidates. See [`LocationModel`] for the four models,
286/// [`LocationParams`] for what can be set and [`LocationSolution`] for
287/// what comes back.
288///
289/// The models are solved by greedy construction followed by vertex
290/// substitution: the solution is good and reproducible, not proven
291/// optimal. For maximal covering the greedy coverage of the demand the
292/// existing facilities leave uncovered is at least `1 - 1/e` (63%) of the
293/// best possible, and [`LocationSolution::upper_bound`] says how close
294/// to the optimum a given solution is. In set covering and p-center the points with weight 0
295/// are not demand; the others count the same whatever their weight.
296///
297/// Errors: [`RouteError::Invalid`] for weights that do not match the
298/// table, a negative or non-finite weight, capacities that do not match
299/// the sites, are not positive or come with another model, an index in `existing` outside
300/// the table or repeated, no candidate, `n_facilities` of 0 or more than
301/// the candidates, a covering model without a cutoff, or a coverage
302/// target outside (0, 1];
303/// [`RouteError::Cancelled`].
304pub fn locate_facilities(
305    costs: &CostTable,
306    demand: &[f64],
307    params: &LocationParams,
308) -> Result<LocationSolution, RouteError> {
309    let invalid = |m: String| Err(RouteError::Invalid(m));
310    if demand.len() != costs.n_demand() {
311        return invalid(format!(
312            "demand has {} weights, but the table has {} demand points",
313            demand.len(),
314            costs.n_demand()
315        ));
316    }
317    if let Some(i) = demand.iter().position(|w| !(w.is_finite() && *w >= 0.0)) {
318        return invalid(format!(
319            "demand weights must be finite and not negative, got {} for demand point {i}",
320            demand[i]
321        ));
322    }
323    let mut is_existing = vec![false; costs.n_sites()];
324    for &e in &params.existing {
325        if e >= costs.n_sites() {
326            return invalid(format!(
327                "existing facility {e} is not a site of the table, which has {}",
328                costs.n_sites()
329            ));
330        }
331        if std::mem::replace(&mut is_existing[e], true) {
332            return invalid(format!("existing facility {e} is listed more than once"));
333        }
334    }
335    let candidates: Vec<usize> = (0..costs.n_sites()).filter(|&j| !is_existing[j]).collect();
336    if candidates.is_empty() {
337        return invalid(
338            "there is no candidate site: every site of the table is an existing facility".into(),
339        );
340    }
341    if let Some(capacity) = &params.capacity {
342        if params.model != LocationModel::MaxCover {
343            return invalid(format!(
344                "capacity is for the max_cover model, not {}",
345                params.model.name()
346            ));
347        }
348        if capacity.len() != costs.n_sites() {
349            return invalid(format!(
350                "capacity has {} values, but the table has {} sites",
351                capacity.len(),
352                costs.n_sites()
353            ));
354        }
355        if let Some(j) = capacity.iter().position(|k| !(k.is_finite() && *k > 0.0)) {
356            return invalid(format!(
357                "capacities must be finite and positive, got {} for site {j}",
358                capacity[j]
359            ));
360        }
361    }
362    if params.model != LocationModel::SetCover {
363        if params.n_facilities == 0 {
364            return invalid("n_facilities must be at least 1".into());
365        }
366        if params.capacity.is_none() && params.n_facilities > candidates.len() {
367            return invalid(format!(
368                "n_facilities is {}, but there are only {} candidate sites",
369                params.n_facilities,
370                candidates.len()
371            ));
372        }
373    }
374    if params.model.is_cover() {
375        if !(params.cutoff.is_finite() && params.cutoff >= 0.0) {
376            return invalid(format!(
377                "the {} model needs a cutoff in minutes, finite and not negative, got {}",
378                params.model.name(),
379                params.cutoff
380            ));
381        }
382        if !(params.min_gain.is_finite() && params.min_gain >= 0.0) {
383            return invalid(format!(
384                "min_gain must be finite and not negative, got {}",
385                params.min_gain
386            ));
387        }
388        if params.model == LocationModel::SetCover
389            && !(params.coverage > 0.0 && params.coverage <= 1.0)
390        {
391            return invalid(format!(
392                "coverage must be above 0 and at most 1, got {}",
393                params.coverage
394            ));
395        }
396    } else if let Some(u) = params.unreachable_time {
397        if !(u.is_finite() && u > 0.0) {
398            return invalid(format!(
399                "unreachable_time must be finite and positive, got {u}"
400            ));
401        }
402    }
403    params.cancel.check()?;
404    let problem = Problem {
405        costs,
406        demand,
407        existing: &params.existing,
408        candidates: &candidates,
409        params,
410    };
411    let mut solution = if let Some(capacity) = &params.capacity {
412        capacity::solve(&problem, capacity)?
413    } else if params.model.is_cover() {
414        cover::solve(&problem)?
415    } else {
416        median::solve(&problem)?
417    };
418    assign(&problem, &mut solution);
419    Ok(solution)
420}

422/// A validated call of [`locate_facilities`].
423struct Problem<'a> {
424    costs: &'a CostTable,
425    demand: &'a [f64],
426    existing: &'a [usize],
427    /// The sites that are not existing facilities, ascending.
428    candidates: &'a [usize],
429    params: &'a LocationParams,
430}
431
432/// The nearest open facility of each demand point.
433fn assign(p: &Problem, s: &mut LocationSolution) {
434    let n = p.costs.n_demand();
435    let mut site: Vec<Option<usize>> = vec![None; n];
436    let mut minutes = vec![f64::NAN; n];
437    let mut open: Vec<usize> = p.existing.iter().chain(&s.sites).copied().collect();
438    open.sort_unstable();
439    for &j in &open {
440        let (rows, times) = p.costs.site(j);
441        for (&i, &t) in rows.iter().zip(times) {
442            let i = i as usize;
443            // ascending site index: a strictly shorter time only
444            if site[i].is_none() || t < minutes[i] {
445                site[i] = Some(j);
446                minutes[i] = t;
447            }
448        }
449    }
450    s.assigned_site = site;
451    s.assigned_minutes = minutes;
452}

454/// A value to minimise, compared field by field.
455type Key = [f64; 2];
456
457fn better(a: &Key, b: &Key) -> bool {
458    a[0] < b[0] || (a[0] == b[0] && a[1] < b[1])
459}

461/// `f(item, scratch)` for every item, in the order of `items`. The items
462/// are shared out in contiguous blocks among the workers, each with a
463/// scratch value of its own, and every value is computed on its own: the
464/// answer is the same for any number of workers.
465fn map_of<S: Send, T: Send>(
466    items: &[usize],
467    n_threads: usize,
468    cancel: &CancelToken,
469    scratch: impl Fn() -> S + Sync,
470    f: impl Fn(usize, &mut S) -> T + Sync,
471) -> Result<Vec<T>, RouteError> {
472    let workers = if items.len() < 64 {
473        1
474    } else {
475        let cores = sys::cores();
476        let asked = if n_threads == 0 {
477            cores
478        } else {
479            n_threads.min(cores)
480        };
481        asked.clamp(1, items.len())
482    };
483    let block = items.len().div_ceil(workers).max(1);
484    let (scratch, f) = (&scratch, &f);
485    let run = move |part: &[usize]| -> Result<Vec<T>, RouteError> {
486        let mut s = scratch();
487        let mut out = Vec::with_capacity(part.len());
488        for (n, &j) in part.iter().enumerate() {
489            if n % 64 == 0 {
490                cancel.check()?;
491            }
492            out.push(f(j, &mut s));
493        }
494        Ok(out)
495    };
496    if workers == 1 {
497        return run(items);
498    }
499    let mut all = Vec::with_capacity(items.len());
500    for part in sys::scoped(items.chunks(block), run) {
501        all.extend(part?);
502    }
503    Ok(all)
504}

506/// The position of the smallest key, the first of them on a tie (`None`
507/// entries are out of the running).
508fn first_best<P>(values: &[Option<(Key, P)>]) -> Option<usize> {
509    let mut best: Option<usize> = None;
510    for (k, v) in values.iter().enumerate() {
511        if let Some((key, _)) = v {
512            let wins = match best.and_then(|b| values[b].as_ref()) {
513                Some((b, _)) => better(key, b),
514                None => true,
515            };
516            if wins {
517                best = Some(k);
518            }
519        }
520    }
521    best
