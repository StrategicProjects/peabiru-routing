1//! Facility location on a table of travel times: where to open facilities
2//! ([`locate_facilities`]) and how supply compares with demand around each
3//! place ([`floating_catchment`]).
4//!
5//! Everything here works on a [`CostTable`] (demand point x site ->
6//! minutes, a missing pair being unreachable) and never looks at how the
7//! table was computed: by a street or a public transport matrix of this
8//! crate, or by anything else.
9//!
10//! The four siting models are solved by heuristics, with no integer
11//! programming solver:
12//!
13//! - maximal covering (Church and ReVelle, 1974): greedy, each site the
14//!   one that covers the most demand still uncovered, which reaches at
15//!   least `1 - 1/e` of the best coverage possible (Nemhauser, Wolsey and
16//!   Fisher, 1978), then vertex substitution;
17//! - location set covering (Toregas et al., 1971): greedy by the number
18//!   of demand points newly covered, then the sites the others make
19//!   redundant are dropped;
20//! - p-median (Hakimi, 1964): greedy, then the vertex substitution of
21//!   Teitz and Bart (1968);
22//! - p-center (Hakimi, 1964): greedy on the largest travel time (ties by
23//!   the weighted sum), then vertex substitution on the same pair.
24//!
25//! Ties are always broken by the lowest site index, the candidates are
26//! evaluated in parallel only where each one's value is computed on its
27//! own, and every sum is taken in a fixed order: the result does not
28//! depend on the number of threads.

43/// The siting model of [`locate_facilities`].
44#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
45#[non_exhaustive]
46pub enum LocationModel {
47    /// Maximal covering: with [`LocationParams::n_facilities`] new
48    /// facilities, cover the most demand within
49    /// [`LocationParams::cutoff`] minutes.
50    MaxCover,
51    /// Location set covering: the fewest new facilities that leave every
52    /// demand point some site can cover within
53    /// [`LocationParams::cutoff`] minutes.
54    SetCover,
55    /// p-median: [`LocationParams::n_facilities`] new facilities with the
56    /// smallest demand-weighted sum of travel times to the nearest
57    /// facility.
58    PMedian,
59    /// p-center: [`LocationParams::n_facilities`] new facilities with the
60    /// smallest travel time of the demand point farthest from its nearest
61    /// facility.
62    PCenter,
63}
64
65impl LocationModel {
66    /// The model's name in the packages: `max_cover`, `set_cover`,
67    /// `p_median` or `p_center`.
68    pub fn name(self) -> &'static str {
69        match self {
70            LocationModel::MaxCover => "max_cover",
71            LocationModel::SetCover => "set_cover",
72            LocationModel::PMedian => "p_median",
73            LocationModel::PCenter => "p_center",
74        }
75    }
76
77    /// The model of that name (`None` for any other text).
78    pub fn from_name(name: &str) -> Option<LocationModel> {
79        Some(match name {
80            "max_cover" => LocationModel::MaxCover,
81            "set_cover" => LocationModel::SetCover,
82            "p_median" => LocationModel::PMedian,
83            "p_center" => LocationModel::PCenter,
84            _ => return None,
85        })
86    }
87
88    fn is_cover(self) -> bool {
89        matches!(self, LocationModel::MaxCover | LocationModel::SetCover)
90    }
91}

93/// What [`locate_facilities`] solves. Fill it in by changing the fields of
94/// [`LocationParams::default()`].
95#[derive(Debug, Clone)]
96#[non_exhaustive]
97pub struct LocationParams {
98    /// The model (default: maximal covering).
99    pub model: LocationModel,
100    /// How many new facilities to open (default 1). Fewer are returned
101    /// when no further site improves the objective. Ignored by
102    /// [`LocationModel::SetCover`], which opens as many as it takes.
103    pub n_facilities: usize,
104    /// The covering models' service standard in minutes: a demand point is
105    /// covered by a site at most this far (`time <= cutoff`). No default:
106    /// it must be set for [`LocationModel::MaxCover`] and
107    /// [`LocationModel::SetCover`]; the other models ignore it.
108    pub cutoff: f64,
109    /// Sites of the table (column indices) that are already open. They
110    /// count first: only the demand they leave uncovered (covering
111    /// models), or the travel time they leave (p-median, p-center), drives
112    /// the choice of the new sites. They are never closed or moved. Every
113    /// other site of the table is a candidate.
114    pub existing: Vec<usize>,
115    /// Improve the greedy solution by vertex substitution (default
116    /// `true`): swap an open new site for a closed candidate while that
117    /// improves the objective, taking at each step the best swap of all.
118    /// For [`LocationModel::SetCover`] the improvement is to drop the
119    /// sites the others make redundant. The result is never worse than
120    /// the greedy one ([`LocationSolution::objective_greedy`]).
121    pub swap: bool,
122    /// [`LocationModel::MaxCover`] stops opening sites when the best one
123    /// left would cover less than this much uncovered demand (default 0:
124    /// it stops only when a site would cover nothing).
125    pub min_gain: f64,
126    /// For [`LocationModel::PMedian`] and [`LocationModel::PCenter`], the
127    /// minutes charged to a demand point that reaches no open facility
128    /// (while some site of the table could serve it). `None` (the
129    /// default): twice the largest time of the table, so that serving a
130    /// point always beats shortening the trip of another.
131    pub unreachable_time: Option<f64>,
132    /// Worker threads (0 = all cores). The result does not depend on it.
133    pub n_threads: usize,
134    /// Cancels the computation: checked between candidates.
135    pub cancel: CancelToken,
136}
137
138impl Default for LocationParams {
139    fn default() -> Self {
140        LocationParams {
141            model: LocationModel::MaxCover,
142            n_facilities: 1,
143            cutoff: f64::NAN,
144            existing: Vec::new(),
145            swap: true,
146            min_gain: 0.0,
147            unreachable_time: None,
148            n_threads: 0,
149            cancel: CancelToken::default(),
150        }
151    }
152}

154/// The sites [`locate_facilities`] chose and what they achieve.
155///
156/// The *objective* is, by model: the demand covered (maximal covering),
157/// the number of new sites (set covering), the demand-weighted sum of
158/// minutes to the nearest facility (p-median) and the largest such time
159/// (p-center). Only the first is to be maximised.
160#[derive(Debug, Clone, PartialEq)]
161#[non_exhaustive]
162pub struct LocationSolution {
163    /// The model solved.
164    pub model: LocationModel,
165    /// The new sites (column indices of the table), ordered by what each
166    /// adds: the first is the site of the solution that does best alone
167    /// (after the existing facilities), the second the one that adds most
168    /// to it, and so on. Without [`LocationParams::swap`] this is the
169    /// order in which the greedy heuristic chose them.
170    pub sites: Vec<usize>,
171    /// What each site of `sites` adds to the ones before it: demand newly
172    /// covered (covering models), or by how much the objective falls
173    /// (p-median, p-center).
174    pub marginal: Vec<f64>,
175    /// The state after each site of `sites`: demand covered so far,
176    /// including what the existing facilities cover (covering models), or
177    /// the objective so far (p-median, p-center).
178    pub cumulative: Vec<f64>,
179    /// Per site of the table, what it would add if it were the only new
180    /// facility, in the unit of `marginal`: for the covering models, the
181    /// demand not covered by the existing facilities that it would cover
182    /// alone. 0 for the existing facilities.
183    pub potential: Vec<f64>,
184    /// Per demand point, the nearest open facility, existing or new
185    /// (`None` when it reaches none); ties go to the lowest site index.
186    pub assigned_site: Vec<Option<usize>>,
187    /// Per demand point, the minutes to `assigned_site` (NaN when `None`).
188    pub assigned_minutes: Vec<f64>,
189    /// The objective with the existing facilities only. For set covering,
190    /// 0.
191    pub objective_before: f64,
192    /// The objective of the greedy solution, before the improvement.
193    pub objective_greedy: f64,
194    /// The objective of the solution.
195    pub objective: f64,
196    /// The demand covered by the existing facilities (covering models; NaN
197    /// for the others).
198    pub covered_before: f64,
199    /// The demand covered by the solution (covering models; NaN for the
200    /// others).
201    pub covered: f64,
202    /// The sum of the demand weights.
203    pub total_demand: f64,
204    /// The demand points with weight that no site of the table, existing
205    /// or candidate, can serve: within the cutoff for the covering models,
206    /// at all for p-median and p-center. They are left out of the
207    /// objectives.
208    pub unservable: Vec<usize>,
209    /// The sum of the weights of `unservable`.
210    pub unservable_demand: f64,
211    /// How many improving swaps were made (set covering: redundant sites
212    /// dropped).
213    pub n_swaps: usize,
214}

216/// Choose where to open facilities.
217///
218/// `costs` gives the minutes from each demand point to each site;
219/// `demand` the weight of each demand point (residents, pupils...), which
220/// must be finite and not negative. The sites in
221/// [`LocationParams::existing`] are already open; all the others are
222/// candidates. See [`LocationModel`] for the four models,
223/// [`LocationParams`] for what can be set and [`LocationSolution`] for
224/// what comes back.
225///
226/// The models are solved by greedy construction followed by vertex
227/// substitution: the solution is good and reproducible, not proven
228/// optimal. For maximal covering the greedy coverage of the demand the
229/// existing facilities leave uncovered is at least `1 - 1/e` (63%) of the
230/// best possible. In set covering and p-center the points with weight 0
231/// are not demand; the others count the same whatever their weight.
232///
233/// Errors: [`RouteError::Invalid`] for weights that do not match the
234/// table, a negative or non-finite weight, an index in `existing` outside
235/// the table or repeated, no candidate, `n_facilities` of 0 or more than
236/// the candidates, or a covering model without a cutoff;
237/// [`RouteError::Cancelled`].
238pub fn locate_facilities(
239    costs: &CostTable,
240    demand: &[f64],
241    params: &LocationParams,
242) -> Result<LocationSolution, RouteError> {
243    let invalid = |m: String| Err(RouteError::Invalid(m));
244    if demand.len() != costs.n_demand() {
245        return invalid(format!(
246            "demand has {} weights, but the table has {} demand points",
247            demand.len(),
248            costs.n_demand()
249        ));
250    }
251    if let Some(i) = demand.iter().position(|w| !(w.is_finite() && *w >= 0.0)) {
252        return invalid(format!(
253            "demand weights must be finite and not negative, got {} for demand point {i}",
254            demand[i]
255        ));
256    }
257    let mut is_existing = vec![false; costs.n_sites()];
258    for &e in &params.existing {
259        if e >= costs.n_sites() {
260            return invalid(format!(
261                "existing facility {e} is not a site of the table, which has {}",
262                costs.n_sites()
263            ));
264        }
265        if std::mem::replace(&mut is_existing[e], true) {
266            return invalid(format!("existing facility {e} is listed more than once"));
267        }
268    }
269    let candidates: Vec<usize> = (0..costs.n_sites()).filter(|&j| !is_existing[j]).collect();
270    if candidates.is_empty() {
271        return invalid(
272            "there is no candidate site: every site of the table is an existing facility".into(),
273        );
274    }
275    if params.model != LocationModel::SetCover {
276        if params.n_facilities == 0 {
277            return invalid("n_facilities must be at least 1".into());
278        }
279        if params.n_facilities > candidates.len() {
280            return invalid(format!(
281                "n_facilities is {}, but there are only {} candidate sites",
282                params.n_facilities,
283                candidates.len()
284            ));
285        }
286    }
287    if params.model.is_cover() {
288        if !(params.cutoff.is_finite() && params.cutoff >= 0.0) {
289            return invalid(format!(
290                "the {} model needs a cutoff in minutes, finite and not negative, got {}",
291                params.model.name(),
292                params.cutoff
293            ));
294        }
295        if !(params.min_gain.is_finite() && params.min_gain >= 0.0) {
296            return invalid(format!(
297                "min_gain must be finite and not negative, got {}",
298                params.min_gain
299            ));
300        }
301    } else if let Some(u) = params.unreachable_time {
302        if !(u.is_finite() && u > 0.0) {
303            return invalid(format!(
304                "unreachable_time must be finite and positive, got {u}"
305            ));
306        }
307    }
308    params.cancel.check()?;
309    let problem = Problem {
310        costs,
311        demand,
312        existing: &params.existing,
313        candidates: &candidates,
314        params,
315    };
316    let mut solution = if params.model.is_cover() {
317        cover::solve(&problem)?
318    } else {
319        median::solve(&problem)?
320    };
321    assign(&problem, &mut solution);
322    Ok(solution)
323}

325/// A validated call of [`locate_facilities`].
326struct Problem<'a> {
327    costs: &'a CostTable,
328    demand: &'a [f64],
329    existing: &'a [usize],
330    /// The sites that are not existing facilities, ascending.
331    candidates: &'a [usize],
332    params: &'a LocationParams,
333}
334
335/// The nearest open facility of each demand point.
336fn assign(p: &Problem, s: &mut LocationSolution) {
337    let n = p.costs.n_demand();
338    let mut site: Vec<Option<usize>> = vec![None; n];
339    let mut minutes = vec![f64::NAN; n];
340    let mut open: Vec<usize> = p.existing.iter().chain(&s.sites).copied().collect();
341    open.sort_unstable();
342    for &j in &open {
343        let (rows, times) = p.costs.site(j);
344        for (&i, &t) in rows.iter().zip(times) {
345            let i = i as usize;
346            // ascending site index: a strictly shorter time only
347            if site[i].is_none() || t < minutes[i] {
348                site[i] = Some(j);
349                minutes[i] = t;
350            }
351        }
352    }
353    s.assigned_site = site;
354    s.assigned_minutes = minutes;
355}

357/// A value to minimise, compared field by field.
358type Key = [f64; 2];
359
360fn better(a: &Key, b: &Key) -> bool {
361    a[0] < b[0] || (a[0] == b[0] && a[1] < b[1])
362}

364/// `f(item, scratch)` for every item, in the order of `items`. The items
365/// are shared out in contiguous blocks among the workers, each with a
366/// scratch value of its own, and every value is computed on its own: the
367/// answer is the same for any number of workers.
368fn map_of<S: Send, T: Send>(
369    items: &[usize],
370    n_threads: usize,
371    cancel: &CancelToken,
372    scratch: impl Fn() -> S + Sync,
373    f: impl Fn(usize, &mut S) -> T + Sync,
374) -> Result<Vec<T>, RouteError> {
375    let workers = if items.len() < 64 {
376        1
377    } else {
378        let cores = sys::cores();
379        let asked = if n_threads == 0 {
380            cores
381        } else {
382            n_threads.min(cores)
383        };
384        asked.clamp(1, items.len())
385    };
386    let block = items.len().div_ceil(workers).max(1);
387    let (scratch, f) = (&scratch, &f);
388    let run = move |part: &[usize]| -> Result<Vec<T>, RouteError> {
389        let mut s = scratch();
390        let mut out = Vec::with_capacity(part.len());
391        for (n, &j) in part.iter().enumerate() {
392            if n % 64 == 0 {
393                cancel.check()?;
394            }
395            out.push(f(j, &mut s));
396        }
397        Ok(out)
398    };
399    if workers == 1 {
400        return run(items);
401    }
402    let mut all = Vec::with_capacity(items.len());
403    for part in sys::scoped(items.chunks(block), run) {
404        all.extend(part?);
405    }
406    Ok(all)
407}

409/// The position of the smallest key, the first of them on a tie (`None`
410/// entries are out of the running).
411fn first_best<P>(values: &[Option<(Key, P)>]) -> Option<usize> {
412    let mut best: Option<usize> = None;
413    for (k, v) in values.iter().enumerate() {
414        if let Some((key, _)) = v {
415            let wins = match best.and_then(|b| values[b].as_ref()) {
416                Some((b, _)) => better(key, b),
417                None => true,
418            };
419            if wins {
420                best = Some(k);
421            }
422        }
423    }
424    best
