1//! commons-math3 3.0 `FastMath.exp`, bit for bit, as bundled in the R5 7.6
2//! jar (`r5-v7.6-all.jar`). R5's decay functions call it instead of
3//! `Math.exp`, and it is not correctly rounded, so `f64::exp` (the platform
4//! libm) differs from it in the last bit for some arguments.

16/// `FastMath.exp(x)` (commons-math3 3.0 `FastMath.exp(double, double,
17/// double[])` with `extra = 0`, `hiPrec = null`; ported from the bytecode).
18pub fn exp(x: f64) -> f64 {
19    exp_extra(x, 0.0)
20}
21
22/// `FastMath.exp(x, extra, null)`: `exp(x + extra)` for a small `extra`
23/// (what `FastMath.pow` calls).
24pub(crate) fn exp_extra(x: f64, extra: f64) -> f64 {
25    const MAX_INDEX: i32 = 750; // EXP_INT_TABLE_MAX_INDEX
26    let int_part_a: f64;
27    let int_part_b: f64;
28    let mut int_val: i32;
29    // `x < 0.0` is `dcmpg; ifge`: NaN takes the non-negative branch.
30    if x < 0.0 {
31        // `(int) -x`: Java's d2i saturates like Rust's `as`.
32        int_val = (-x) as i32;
33        if int_val > 746 {
34            return 0.0;
35        }
36        if int_val > 709 {
37            // subnormal result
38            return exp_extra(x + 40.191_406_25, extra) / 285_040_095_144_011_776.0;
39        }
40        if int_val == 709 {
41            return exp_extra(x + 1.494_140_625, extra) / 4.455_505_956_692_757;
42        }
43        int_val += 1;
44        int_part_a = f64::from_bits(EXP_INT_TABLE_A[(MAX_INDEX - int_val) as usize]);
45        int_part_b = f64::from_bits(EXP_INT_TABLE_B[(MAX_INDEX - int_val) as usize]);
46        int_val = -int_val;
47    } else {
48        int_val = x as i32;
49        if int_val > 709 {
50            return f64::INFINITY;
51        }
52        int_part_a = f64::from_bits(EXP_INT_TABLE_A[(MAX_INDEX + int_val) as usize]);
53        int_part_b = f64::from_bits(EXP_INT_TABLE_B[(MAX_INDEX + int_val) as usize]);
54    }
55    // Greatest multiple of 2^-10 below the fractional part. `x - int_val` is
56    // in [0, 1) here (NaN gives index 0), so the index is in 0..=1024.
57    let int_frac = ((x - f64::from(int_val)) * 1024.0) as i32;
58    let frac_part_a = f64::from_bits(EXP_FRAC_TABLE_A[int_frac as usize]);
59    let frac_part_b = f64::from_bits(EXP_FRAC_TABLE_B[int_frac as usize]);
60    let epsilon = x - (f64::from(int_val) + f64::from(int_frac) / 1024.0);
61    // Remez polynomial for exp(epsilon) - 1, evaluated in the same order.
62    let mut z = 0.041_687_017_387_645_07;
63    z = z * epsilon + 0.166_666_650_502_308_3;
64    z = z * epsilon + 0.500_000_000_004_268_7;
65    z = z * epsilon + 1.0;
66    z = z * epsilon + -3.940_510_424_527_919E-20;
67    let temp_a = int_part_a * frac_part_a;
68    let temp_b = int_part_a * frac_part_b + int_part_b * frac_part_a + int_part_b * frac_part_b;
69    let temp_c = temp_b + temp_a;
70    if extra != 0.0 {
71        temp_c * extra * z + temp_c * extra + temp_c * z + temp_b + temp_a
72    } else {
73        temp_c * z + temp_b + temp_a
74    }
75}
