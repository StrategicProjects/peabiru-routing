1//! The functions of `java.lang.StrictMath` that the coordinate transforms
2//! of GeoTools call through `java.lang.Math` (sine, cosine, tangent,
3//! arc tangent, natural logarithm), ported from fdlibm 5.3, which
4//! `StrictMath` specifies. They are pure IEEE double arithmetic, so they
5//! give the same bits on every platform (the system's `sin` and `cos`
6//! differ between platforms in the last bit).
7//!
8//! fdlibm is Copyright (C) 1993 by Sun Microsystems, Inc.: "Developed at
9//! SunSoft, a Sun Microsystems, Inc. business. Permission to use, copy,
10//! modify, and distribute this software is freely granted, provided that
11//! this notice is preserved."
12//!
13//! Only arguments in the range the transforms use are reduced the fdlibm
14//! way (|x| up to 2^19 pi/2, far beyond the radians of any longitude); a
15//! larger argument gives NaN.
16
17#![allow(
18    clippy::excessive_precision,
19    clippy::many_single_char_names,
20    clippy::approx_constant,
21    clippy::needless_late_init
22)] // fdlibm's constants and layout, as written

24#[inline]
25fn hi(x: f64) -> i32 {
26    (x.to_bits() >> 32) as i32
27}
28
29#[inline]
30fn lo(x: f64) -> u32 {
31    x.to_bits() as u32
32}
33
34#[inline]
35fn with_hi_lo(h: i32, l: u32) -> f64 {
36    f64::from_bits((u64::from(h as u32) << 32) | u64::from(l))
37}

39/// `__kernel_sin(x, y, iy)` on `[-pi/4, pi/4]`.
40fn k_sin(x: f64, y: f64, iy: bool) -> f64 {
41    const S1: f64 = -1.66666666666666324348e-01;
42    const S2: f64 = 8.33333333332248946124e-03;
43    const S3: f64 = -1.98412698298579493134e-04;
44    const S4: f64 = 2.75573137070700676789e-06;
45    const S5: f64 = -2.50507602534068634195e-08;
46    const S6: f64 = 1.58969099521155010221e-10;
47    let ix = hi(x) & 0x7fff_ffff;
48    if ix < 0x3e40_0000 && (x as i32) == 0 {
49        return x;
50    }
51    let z = x * x;
52    let v = z * x;
53    let r = S2 + z * (S3 + z * (S4 + z * (S5 + z * S6)));
54    if !iy {
55        x + v * (S1 + z * r)
56    } else {
57        x - ((z * (0.5 * y - v * r) - y) - v * S1)
58    }
59}

61/// `__kernel_cos(x, y)` on `[-pi/4, pi/4]`.
62fn k_cos(x: f64, y: f64) -> f64 {
63    const C1: f64 = 4.16666666666666019037e-02;
64    const C2: f64 = -1.38888888888741095749e-03;
65    const C3: f64 = 2.48015872894767294178e-05;
66    const C4: f64 = -2.75573143513906633035e-07;
67    const C5: f64 = 2.08757232129817482790e-09;
68    const C6: f64 = -1.13596475577881948265e-11;
69    let ix = hi(x) & 0x7fff_ffff;
70    if ix < 0x3e40_0000 && (x as i32) == 0 {
71        return 1.0;
72    }
73    let z = x * x;
74    let r = z * (C1 + z * (C2 + z * (C3 + z * (C4 + z * (C5 + z * C6)))));
75    if ix < 0x3fd3_3333 {
76        1.0 - (0.5 * z - (z * r - x * y))
77    } else {
78        let qx = if ix > 0x3fe9_0000 {
79            0.28125
80        } else {
81            with_hi_lo(ix - 0x0020_0000, 0)
82        };
83        let hz = 0.5 * z - qx;
84        let a = 1.0 - qx;
85        a - (hz - (z * r - x * y))
86    }
87}

89/// `__ieee754_rem_pio2` for |x| up to about 2^19 pi/2: `x - n pi/2` as
90/// `(y0, y1)` and `n`; `None` beyond.
91fn rem_pio2(x: f64) -> Option<(i32, f64, f64)> {
92    const NPIO2_HW: [i32; 32] = [
93        0x3FF921FB, 0x400921FB, 0x4012D97C, 0x401921FB, 0x401F6A7A, 0x4022D97C, 0x4025FDBB,
94        0x402921FB, 0x402C463A, 0x402F6A7A, 0x4031475C, 0x4032D97C, 0x40346B9C, 0x4035FDBB,
95        0x40378FDB, 0x403921FB, 0x403AB41B, 0x403C463A, 0x403DD85A, 0x403F6A7A, 0x40407E4C,
96        0x4041475C, 0x4042106C, 0x4042D97C, 0x4043A28C, 0x40446B9C, 0x404534AC, 0x4045FDBB,
97        0x4046C6CB, 0x40478FDB, 0x404858EB, 0x404921FB,
98    ];
99    const INVPIO2: f64 = 6.36619772367581382433e-01;
100    const PIO2_1: f64 = 1.57079632673412561417e+00;
101    const PIO2_1T: f64 = 6.07710050650619224932e-11;
102    const PIO2_2: f64 = 6.07710050630396597660e-11;
103    const PIO2_2T: f64 = 2.02226624879595063154e-21;
104    const PIO2_3: f64 = 2.02226624871116645580e-21;
105    const PIO2_3T: f64 = 8.47842766036889956997e-32;
106    let hx = hi(x);
107    let ix = hx & 0x7fff_ffff;
108    if ix <= 0x3fe9_21fb {
109        return Some((0, x, 0.0));
110    }
111    if ix < 0x4002_d97c {
112        // |x| < 3pi/4: one subtraction of pi/2
113        return Some(if hx > 0 {
114            let mut z = x - PIO2_1;
115            if ix != 0x3ff9_21fb {
116                let y0 = z - PIO2_1T;
117                (1, y0, (z - y0) - PIO2_1T)
118            } else {
119                z -= PIO2_2;
120                let y0 = z - PIO2_2T;
121                (1, y0, (z - y0) - PIO2_2T)
122            }
123        } else {
124            let mut z = x + PIO2_1;
125            if ix != 0x3ff9_21fb {
126                let y0 = z + PIO2_1T;
127                (-1, y0, (z - y0) + PIO2_1T)
128            } else {
129                z += PIO2_2;
130                let y0 = z + PIO2_2T;
131                (-1, y0, (z - y0) + PIO2_2T)
132            }
133        });
134    }
135    if ix > 0x4139_21fb {
136        return None;
137    }
138    let t = x.abs();
139    let n = (t * INVPIO2 + 0.5) as i32;
140    let fnum = f64::from(n);
141    let mut r = t - fnum * PIO2_1;
142    let mut w = fnum * PIO2_1T;
143    let y0;
144    if n < 32 && ix != NPIO2_HW[(n - 1) as usize] {
145        y0 = r - w;
146    } else {
147        let j = ix >> 20;
148        let mut y = r - w;
149        let mut i = j - ((hi(y) >> 20) & 0x7ff);
150        if i > 16 {
151            let t = r;
152            w = fnum * PIO2_2;
153            r = t - w;
154            w = fnum * PIO2_2T - ((t - r) - w);
155            y = r - w;
156            i = j - ((hi(y) >> 20) & 0x7ff);
157            if i > 49 {
158                let t = r;
159                w = fnum * PIO2_3;
160                r = t - w;
161                w = fnum * PIO2_3T - ((t - r) - w);
162                y = r - w;
163            }
164        }
165        y0 = y;
166    }
167    let y1 = (r - y0) - w;
168    Some(if hx < 0 { (-n, -y0, -y1) } else { (n, y0, y1) })
169}

171/// `StrictMath.sin`.
172pub(crate) fn sin(x: f64) -> f64 {
173    let ix = hi(x) & 0x7fff_ffff;
174    if ix <= 0x3fe9_21fb {
175        return k_sin(x, 0.0, false);
176    }
177    if ix >= 0x7ff0_0000 {
178        return f64::NAN;
179    }
180    match rem_pio2(x) {
181        Some((n, y0, y1)) => match n & 3 {
182            0 => k_sin(y0, y1, true),
183            1 => k_cos(y0, y1),
184            2 => -k_sin(y0, y1, true),
185            _ => -k_cos(y0, y1),
186        },
187        None => f64::NAN,
188    }
189}

191/// `StrictMath.cos`.
192pub(crate) fn cos(x: f64) -> f64 {
193    let ix = hi(x) & 0x7fff_ffff;
194    if ix <= 0x3fe9_21fb {
195        return k_cos(x, 0.0);
196    }
197    if ix >= 0x7ff0_0000 {
198        return f64::NAN;
199    }
200    match rem_pio2(x) {
201        Some((n, y0, y1)) => match n & 3 {
202            0 => k_cos(y0, y1),
203            1 => -k_sin(y0, y1, true),
204            2 => -k_cos(y0, y1),
205            _ => k_sin(y0, y1, true),
206        },
207        None => f64::NAN,
208    }
209}

211/// `__kernel_tan(x, y, k)` on `[-pi/4, pi/4]`: tan if `k == 1`, -1/tan
212/// if `k == -1` (fdlibm 5.3, with the fix for arguments near pi/4).
213fn k_tan(mut x: f64, mut y: f64, iy: i32) -> f64 {
214    const T: [f64; 13] = [
215        3.33333333333334091986e-01,
216        1.33333333333201242699e-01,
217        5.39682539762260521377e-02,
218        2.18694882948595424599e-02,
219        8.86323982359930005737e-03,
220        3.59207910759131235356e-03,
221        1.45620945432529025516e-03,
222        5.88041240820264096874e-04,
223        2.46463134818469906812e-04,
224        7.81794442939557092300e-05,
225        7.14072491382608190305e-05,
226        -1.85586374855275456654e-05,
227        2.59073051863633712884e-05,
228    ];
229    const PIO4: f64 = 7.85398163397448278999e-01;
230    const PIO4LO: f64 = 3.06161699786838301793e-17;
231    let hx = hi(x);
232    let ix = hx & 0x7fff_ffff;
233    if ix < 0x3e30_0000 && (x as i32) == 0 {
234        if ((ix as u32) | lo(x)) | ((iy + 1) as u32) == 0 {
235            return 1.0 / x.abs();
236        } else if iy == 1 {
237            return x;
238        } else {
239            // compute -1/(x+y) carefully
240            let w = x + y;
241            let z = with_hi_lo(hi(w), 0);
242            let v = y - (z - x);
243            let a = -1.0 / w;
244            let t = with_hi_lo(hi(a), 0);
245            let s = 1.0 + t * z;
246            return t + a * (s + t * v);
247        }
248    }
249    let big = ix >= 0x3fe5_9428;
250    if big {
251        if hx < 0 {
252            x = -x;
253            y = -y;
254        }
255        let z = PIO4 - x;
256        let w = PIO4LO - y;
257        x = z + w;
258        y = 0.0;
259    }
260    let z = x * x;
261    let w = z * z;
262    let r = T[1] + w * (T[3] + w * (T[5] + w * (T[7] + w * (T[9] + w * T[11]))));
263    let v = z * (T[2] + w * (T[4] + w * (T[6] + w * (T[8] + w * (T[10] + w * T[12])))));
264    let s = z * x;
265    let r = y + z * (s * (r + v) + y);
266    let r = r + T[0] * s;
267    let w = x + r;
268    if big {
269        let v = f64::from(iy);
270        let sign = f64::from(1 - ((hx >> 30) & 2));
271        return sign * (v - 2.0 * (x - (w * w / (w + v) - r)));
272    }
273    if iy == 1 {
274        w
275    } else {
276        // -1/(x+r) with care
277        let z = with_hi_lo(hi(w), 0);
278        let v = r - (z - x);
279        let a = -1.0 / w;
280        let t = with_hi_lo(hi(a), 0);
281        let s = 1.0 + t * z;
282        t + a * (s + t * v)
283    }
284}

286/// `StrictMath.tan`.
287pub(crate) fn tan(x: f64) -> f64 {
288    let ix = hi(x) & 0x7fff_ffff;
289    if ix <= 0x3fe9_21fb {
290        return k_tan(x, 0.0, 1);
291    }
292    if ix >= 0x7ff0_0000 {
293        return f64::NAN;
294    }
295    match rem_pio2(x) {
296        Some((n, y0, y1)) => k_tan(y0, y1, 1 - ((n & 1) << 1)),
297        None => f64::NAN,
298    }
299}

301/// `StrictMath.atan`.
302pub(crate) fn atan(x: f64) -> f64 {
303    const ATANHI: [f64; 4] = [
304        4.63647609000806093515e-01,
305        7.85398163397448278999e-01,
306        9.82793723247329054082e-01,
307        1.57079632679489655800e+00,
308    ];
309    const ATANLO: [f64; 4] = [
310        2.26987774529616870924e-17,
311        3.06161699786838301793e-17,
312        1.39033110312309984516e-17,
313        6.12323399573676603587e-17,
314    ];
315    const AT: [f64; 11] = [
316        3.33333333333329318027e-01,
317        -1.99999999998764832476e-01,
318        1.42857142725034663711e-01,
319        -1.11111104054623557880e-01,
320        9.09088713343650656196e-02,
321        -7.69187620504482999495e-02,
322        6.66107313738753120669e-02,
323        -5.83357013379057348645e-02,
324        4.97687799461593236017e-02,
325        -3.65315727442169155270e-02,
326        1.62858201153657823623e-02,
327    ];
328    let hx = hi(x);
329    let ix = hx & 0x7fff_ffff;
330    if ix >= 0x4410_0000 {
331        // |x| >= 2^66
332        if ix > 0x7ff0_0000 || (ix == 0x7ff0_0000 && lo(x) != 0) {
333            return x + x; // NaN
334        }
335        return if hx > 0 {
336            ATANHI[3] + ATANLO[3]
337        } else {
338            -ATANHI[3] - ATANLO[3]
339        };
340    }
341    let id: i32;
342    let mut x = x;
343    if ix < 0x3fdc_0000 {
344        // |x| < 0.4375
345        if ix < 0x3e20_0000 {
346            // |x| < 2^-29 (the "raise inexact" test always passes)
347            return x;
348        }
349        id = -1;
350    } else {
351        x = x.abs();
352        if ix < 0x3ff3_0000 {
353            if ix < 0x3fe6_0000 {
354                id = 0;
355                x = (2.0 * x - 1.0) / (2.0 + x);
356            } else {
357                id = 1;
358                x = (x - 1.0) / (x + 1.0);
359            }
360        } else if ix < 0x4003_8000 {
361            id = 2;
362            x = (x - 1.5) / (1.0 + 1.5 * x);
363        } else {
364            id = 3;
365            x = -1.0 / x;
366        }
367    }
368    let z = x * x;
369    let w = z * z;
370    let s1 = z * (AT[0] + w * (AT[2] + w * (AT[4] + w * (AT[6] + w * (AT[8] + w * AT[10])))));
371    let s2 = w * (AT[1] + w * (AT[3] + w * (AT[5] + w * (AT[7] + w * AT[9]))));
372    if id < 0 {
373        return x - x * (s1 + s2);
374    }
375    let i = id as usize;
376    let z = ATANHI[i] - ((x * (s1 + s2) - ATANLO[i]) - x);
377    if hx < 0 {
378        -z
379    } else {
380        z
381    }
382}

384/// `StrictMath.atan2(y, x)`.
385pub(crate) fn atan2(y: f64, x: f64) -> f64 {
386    const PI_O_4: f64 = 7.8539816339744827900e-01;
387    const PI_O_2: f64 = 1.5707963267948965580e+00;
388    const PI: f64 = 3.1415926535897931160e+00;
389    const PI_LO: f64 = 1.2246467991473531772e-16;
390    let hx = hi(x);
391    let ix = hx & 0x7fff_ffff;
392    let lx = lo(x);
393    let hy = hi(y);
394    let iy = hy & 0x7fff_ffff;
395    let ly = lo(y);
396    if ((ix as u32) | ((lx | lx.wrapping_neg()) >> 31)) > 0x7ff0_0000
397        || ((iy as u32) | ((ly | ly.wrapping_neg()) >> 31)) > 0x7ff0_0000
398    {
399        return x + y; // NaN
400    }
401    if ((hx.wrapping_sub(0x3ff0_0000)) as u32 | lx) == 0 {
402        return atan(y); // x = 1
403    }
404    let m = ((hy >> 31) & 1) | ((hx >> 30) & 2);
405    if ((iy as u32) | ly) == 0 {
406        return match m {
407            0 | 1 => y,
408            2 => PI,
409            _ => -PI,
410        };
411    }
412    if ((ix as u32) | lx) == 0 {
413        return if hy < 0 { -PI_O_2 } else { PI_O_2 };
414    }
415    if ix == 0x7ff0_0000 {
416        return if iy == 0x7ff0_0000 {
417            match m {
418                0 => PI_O_4,
419                1 => -PI_O_4,
420                2 => 3.0 * PI_O_4,
421                _ => -3.0 * PI_O_4,
422            }
423        } else {
424            match m {
425                0 => 0.0,
426                1 => -0.0,
427                2 => PI,
428                _ => -PI,
429            }
430        };
431    }
432    if iy == 0x7ff0_0000 {
433        return if hy < 0 { -PI_O_2 } else { PI_O_2 };
434    }
435    let k = (iy - ix) >> 20;
436    let z = if k > 60 {
437        PI_O_2 + 0.5 * PI_LO
438    } else if hx < 0 && k < -60 {
439        0.0
440    } else {
441        atan((y / x).abs())
442    };
443    match m {
444        0 => z,
445        1 => -z,
446        2 => PI - (z - PI_LO),
447        _ => (z - PI_LO) - PI,
448    }
449}

451/// `StrictMath.log` (`__ieee754_log`).
452pub(crate) fn log(x: f64) -> f64 {
453    const LN2_HI: f64 = 6.93147180369123816490e-01;
454    const LN2_LO: f64 = 1.90821492927058770002e-10;
455    const TWO54: f64 = 1.80143985094819840000e+16;
456    const LG1: f64 = 6.666666666666735130e-01;
457    const LG2: f64 = 3.999999999940941908e-01;
458    const LG3: f64 = 2.857142874366239149e-01;
459    const LG4: f64 = 2.222219843214978396e-01;
460    const LG5: f64 = 1.818357216161805012e-01;
461    const LG6: f64 = 1.531383769920937332e-01;
462    const LG7: f64 = 1.479819860511658591e-01;
463    let mut hx = hi(x);
464    let lx = lo(x);
465    let mut k = 0i32;
466    let mut x = x;
467    if hx < 0x0010_0000 {
468        if ((hx & 0x7fff_ffff) as u32 | lx) == 0 {
469            return f64::NEG_INFINITY;
470        }
471        if hx < 0 {
472            return f64::NAN;
473        }
474        k -= 54;
475        x *= TWO54;
476        hx = hi(x);
477    }
478    if hx >= 0x7ff0_0000 {
479        return x + x;
480    }
481    k += (hx >> 20) - 1023;
482    hx &= 0x000f_ffff;
483    let i = (hx + 0x95f64) & 0x0010_0000;
484    x = with_hi_lo(hx | (i ^ 0x3ff0_0000), lo(x));
485    k += i >> 20;
486    let f = x - 1.0;
487    if (0x000f_ffff & (2 + hx)) < 3 {
488        // |f| < 2^-20
489        if f == 0.0 {
490            if k == 0 {
491                return 0.0;
492            }
493            let dk = f64::from(k);
494            return dk * LN2_HI + dk * LN2_LO;
495        }
496        let r = f * f * (0.5 - 0.33333333333333333 * f);
497        if k == 0 {
498            return f - r;
499        }
500        let dk = f64::from(k);
501        return dk * LN2_HI - ((r - dk * LN2_LO) - f);
502    }
503    let s = f / (2.0 + f);
504    let dk = f64::from(k);
505    let z = s * s;
506    let mut i = hx - 0x6147a;
507    let w = z * z;
508    let j = 0x6b851 - hx;
509    let t1 = w * (LG2 + w * (LG4 + w * LG6));
510    let t2 = z * (LG1 + w * (LG3 + w * (LG5 + w * LG7)));
511    i |= j;
512    let r = t2 + t1;
513    if i > 0 {
514        let hfsq = 0.5 * f * f;
515        if k == 0 {
516            f - (hfsq - s * (hfsq + r))
517        } else {
518            dk * LN2_HI - ((hfsq - (s * (hfsq + r) + dk * LN2_LO)) - f)
519        }
520    } else if k == 0 {
521        f - s * (f - r)
522    } else {
523        dk * LN2_HI - ((s * (f - r) - dk * LN2_LO) - f)
524    }
525}
