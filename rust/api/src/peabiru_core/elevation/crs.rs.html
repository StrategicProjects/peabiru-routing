1//! The coordinate reference systems of elevation rasters other than WGS 84
2//! longitude and latitude, and the transform R5 applies to every sampled
3//! point (`CRS.findMathTransform(DefaultGeographicCRS.WGS84, crs)` in
4//! GeoTools 25.2 with its EPSG database, `gt-epsg-hsql`).
5//!
6//! The transforms are written from the published formulas, not from
7//! GeoTools' code (which is LGPL): the geocentric conversion of EPSG
8//! Guidance Note 7-2 (method 9602), its inverse by Toms' non-iterative
9//! method (R. Toms, "An improved algorithm for geocentric to geodetic
10//! coordinate conversion", 1996, as in NIMA's public-domain `geocent.c`),
11//! the geocentric translation (method 9603), the Transverse Mercator series
12//! of Snyder (Map Projections: A Working Manual, 1987, equations 8-9 and
13//! 8-10, as in PROJ.4's `tmerc`), the American Polyconic (Snyder 18-1 to
14//! 18-3, PROJ.4's `poly`) and the spherical Popular Visualisation Pseudo
15//! Mercator (method 1024). The order of the floating-point operations
16//! (and the way each step derives the eccentricity from the axes) was then
17//! matched, by trying the algebraically equal forms, to what GeoTools
18//! returns, double for double, on hundreds of thousands of points per
19//! system (`bench/compare/crs/crs_parity.sh`, whose Java harness calls
20//! GeoTools from the R5 jar). Sines, cosines, tangents, arc tangents and
21//! logarithms are fdlibm's ([`crate::fdlibm`]), the values of
22//! `java.lang.StrictMath`.
23//!
24//! **Platforms.** GeoTools calls `java.lang.Math`, whose sine and cosine
25//! (and on x86_64 tangent and logarithm) the JVM computes with its own
26//! code, which differs from `StrictMath` in the last bit for a few per cent
27//! of arguments (measured on arm64: Java 21 gives another sine for 6% of
28//! them). R5's own result then depends on the machine it runs on, by a few
29//! nanometres. The sampled pixel is `Math.round` of the grid coordinate, so
30//! such a difference only matters when a point falls within a few
31//! nanometres of the boundary between two pixels or of the raster's edge.
32//! [`Crs::margin`] bounds how far GeoTools' point may be from
33//! [`Crs::apply`]'s; the sampler reads the pixel `StrictMath` gives (the
34//! same on every machine) and counts the samples that close to a boundary
35//! between different values, for a warning.
36//!
37//! The table of systems ([`lookup`]) is the subset of the EPSG registry
38//! that GeoTools' database gives for Brazilian and world-wide data, with
39//! the parameters (ellipsoids and datum shifts) exactly as GeoTools has
40//! them (read from the jar with `CRS.decode` and `findMathTransform`).
41

58/// An ellipsoid as GeoTools keeps it: semi-major and semi-minor axes (the
59/// minor one derived from the inverse flattening, `a - a / invf`).
60#[derive(Debug, Clone, Copy, PartialEq)]
61pub(crate) struct Ellipsoid {
62    a: f64,
63    b: f64,
64}

87/// The geodetic datum of a system, as the step from WGS 84 longitude and
88/// latitude to the system's own.
89#[derive(Debug, Clone, Copy, PartialEq)]
90pub(crate) enum Datum {
91    /// WGS 84 itself: no step.
92    Wgs84,
93    /// Another ellipsoid with a zero datum shift (SIRGAS 2000): to
94    /// geocentric coordinates on WGS 84 and back on the other ellipsoid.
95    Ellipsoid(Ellipsoid),
96    /// A geocentric translation (the inverse of the datum's `TOWGS84`)
97    /// between the two conversions.
98    Shift(Ellipsoid, [f64; 3]),
99}
100
101impl Datum {
102    fn ellipsoid(&self) -> Ellipsoid {
103        match *self {
104            Datum::Wgs84 => WGS84,
105            Datum::Ellipsoid(e) | Datum::Shift(e, _) => e,
106        }
107    }
108}

110const SIRGAS2000: Datum = Datum::Ellipsoid(GRS80);
111/// SAD69 (EPSG:6618), `TOWGS84[-59, -11, -52]` in GeoTools' database.
112const SAD69: Datum = Datum::Shift(GRS67M, [59.0, 11.0, 52.0]);
113/// Corrego Alegre 1970-72 (EPSG:6225), `TOWGS84[-206.05, 168.28, -3.82]`,
114/// inverted by GeoTools' matrix inversion (hence the last bits).
115const CORREGO_ALEGRE: Datum =
116    Datum::Shift(INTL24, [206.05000000000004, -168.28, 3.8200000000000003]);

118/// A map projection.

120pub(crate) enum Projection {
121    /// Transverse Mercator (EPSG method 9807) with the latitude of origin

123    /// easting and northing (metres).

127    Polyconic { cm: f64, fe: f64, fn_: f64 },
128    /// Popular Visualisation Pseudo Mercator (EPSG method 1024), on the
129    /// sphere of the ellipsoid's semi-major axis.

135pub(crate) struct Crs {

152    Crs {

154        datum,

158            fe: 500_000.0,

161        residual,
162        name,
163    }

251/// `e^2 = (a^2 - b^2) / a^2`.
252fn e2(e: Ellipsoid) -> f64 {
253    (e.a * e.a - e.b * e.b) / (e.a * e.a)
254}
255
256/// Longitude and latitude in degrees on ellipsoid `e` (height 0) to
257/// geocentric coordinates.
258pub(crate) fn to_geocentric(e: Ellipsoid, lon: f64, lat: f64) -> [f64; 3] {
259    let l = to_radians(lon);
260    let p = to_radians(lat);
261    let e2 = e2(e);
262    let sin_p = fdlibm::sin(p);
263    let cos_p = fdlibm::cos(p);
264    let rn = e.a / (1.0 - e2 * (sin_p * sin_p)).sqrt();
265    let x = rn * cos_p * fdlibm::cos(l);
266    let y = rn * cos_p * fdlibm::sin(l);
267    let z = rn * (1.0 - e2) * sin_p;
268    [x, y, z]
269}

271/// Geocentric coordinates to longitude and latitude in degrees on
272/// ellipsoid `e` (Toms' method).
273pub(crate) fn from_geocentric(e: Ellipsoid, c: [f64; 3]) -> (f64, f64) {
274    const AD_C: f64 = 1.0026;
275    let [x, y, z] = c;
276    let e2 = e2(e);
277    let ep2 = (e.a * e.a - e.b * e.b) / (e.b * e.b);
278    let lon = fdlibm::atan2(y, x);
279    let w2 = x * x + y * y;
280    let w = w2.sqrt();
281    let t0 = z * AD_C;
282    let s0 = (t0 * t0 + w2).sqrt();
283    let sin_b0 = t0 / s0;
284    let cos_b0 = w / s0;
285    let t1 = z + (e.b * ep2) * (sin_b0 * sin_b0 * sin_b0);
286    let sum = w - (e.a * e2) * (cos_b0 * cos_b0 * cos_b0);
287    let s1 = (t1 * t1 + sum * sum).sqrt();
288    let sin_p1 = t1 / s1;
289    let cos_p1 = sum / s1;
290    let lat = fdlibm::atan(sin_p1 / cos_p1);
291    (to_degrees(lon), to_degrees(lat))
292}

294/// Transverse Mercator on ellipsoid `e`, from degrees.
295pub(crate) fn transverse_mercator(
296    e: Ellipsoid,
297    cm: f64,
298    k0: f64,
299    fe: f64,
300    fn_: f64,
301    lon: f64,
302    lat: f64,
303) -> (f64, f64) {
304    const FC1: f64 = 1.0;
305    const FC2: f64 = 0.5;
306    const FC3: f64 = 0.16666666666666666666;
307    const FC4: f64 = 0.08333333333333333333;
308    const FC5: f64 = 0.05;
309    const FC6: f64 = 0.03333333333333333333;
310    const FC7: f64 = 0.02380952380952380952;
311    const FC8: f64 = 0.01785714285714285714;
312    // (the map projections derive it from the axis ratio)
313    let es = 1.0 - (e.b * e.b) / (e.a * e.a);
314    let esp = es / (1.0 - es);
315    let x = roll(to_radians(lon) - to_radians(cm));
316    let y = to_radians(lat);
317    let sinphi = fdlibm::sin(y);
318    let cosphi = fdlibm::cos(y);
319    let mut t = if cosphi.abs() > 1e-10 {
320        sinphi / cosphi
321    } else {
322        0.0
323    };
324    t *= t;
325    let mut al = cosphi * x;
326    let als = al * al;
327    al /= (1.0 - es * sinphi * sinphi).sqrt();
328    let n = esp * cosphi * cosphi;
329    let my = mlfn(es, y, sinphi, cosphi)
330        + sinphi
331            * al
332            * x
333            * FC2
334            * (1.0
335                + FC4
336                    * als
337                    * (5.0 - t
338                        + n * (9.0 + 4.0 * n)
339                        + FC6
340                            * als
341                            * (61.0
342                                + t * (t - 58.0)
343                                + n * (270.0 - 330.0 * t)
344                                + FC8 * als * (1385.0 + t * (t * (543.0 - t) - 3111.0)))));
345    let mx = al
346        * (FC1
347            + FC3
348                * als
349                * (1.0 - t
350                    + n
351                    + FC5
352                        * als
353                        * (5.0
354                            + t * (t - 18.0)
355                            + n * (14.0 - 58.0 * t)
356                            + FC7 * als * (61.0 + t * (t * (179.0 - t) - 479.0)))));
357    let scale = k0 * e.a;
358    (scale * mx + fe, scale * my + fn_)
359}

361/// `MapProjection.rollLongitude`: an angle in radians brought into
362/// [-pi, pi].
363fn roll(x: f64) -> f64 {
364    const TWO_PI: f64 = 2.0 * std::f64::consts::PI;
365    x - TWO_PI * (x / TWO_PI + 0.5).floor()
366}

368/// The meridian distance on the unit ellipsoid of eccentricity squared
369/// `es` (Snyder's series, PROJ.4's `pj_mlfn`).
370fn mlfn(es: f64, phi: f64, sphi: f64, cphi: f64) -> f64 {
371    const C00: f64 = 1.0;
372    const C02: f64 = 0.25;
373    const C04: f64 = 0.046875;
374    const C06: f64 = 0.01953125;
375    const C08: f64 = 0.01068115234375;
376    const C22: f64 = 0.75;
377    const C44: f64 = 0.46875;
378    const C46: f64 = 0.01302083333333333333;
379    const C48: f64 = 0.00712076822916666666;
380    const C66: f64 = 0.36458333333333333333;
381    const C68: f64 = 0.00569661458333333333;
382    const C88: f64 = 0.3076171875;
383    let en0 = C00 - es * (C02 + es * (C04 + es * (C06 + es * C08)));
384    let en1 = es * (C22 - es * (C04 + es * (C06 + es * C08)));
385    let mut t = es * es;
386    let en2 = t * (C44 - es * (C46 + es * C48));
387    t *= es;
388    let en3 = t * (C66 - es * C68);
389    let en4 = t * es * C88;
390    let c = cphi * sphi;
391    let s = sphi * sphi;
392    en0 * phi - c * (en1 + s * (en2 + s * (en3 + s * en4)))
393}

395/// The American Polyconic on ellipsoid `e` (Snyder, equations 18-1 to
396/// 18-3, as PROJ.4's `poly`), from degrees, the latitude of origin at the
397/// equator.
398fn polyconic(e: Ellipsoid, cm: f64, fe: f64, fn_: f64, lon: f64, lat: f64) -> (f64, f64) {
399    const TOL: f64 = 1e-10;
400    let es = 1.0 - (e.b * e.b) / (e.a * e.a);
401    let lam = roll(to_radians(lon) - to_radians(cm));
402    let phi = to_radians(lat);
403    let (x, y) = if phi.abs() <= TOL {
404        (lam, 0.0)
405    } else {
406        let sp = fdlibm::sin(phi);
407        let cp = fdlibm::cos(phi);
408        let ms = if cp.abs() > TOL {
409            cp / (1.0 - es * (sp * sp)).sqrt() / sp
410        } else {
411            0.0
412        };
413        let l = lam * sp;
414        (
415            ms * fdlibm::sin(l),
416            mlfn(es, phi, sp, cp) + ms * (1.0 - fdlibm::cos(l)),
417        )
418    };
419    (e.a * x + fe, e.a * y + fn_)
420}

422/// The pseudo-Mercator, from degrees.
423fn pseudo_mercator(a: f64, lon: f64, lat: f64) -> (f64, f64) {
424    let x = to_radians(lon);
425    let y = fdlibm::log(fdlibm::tan(
426        std::f64::consts::FRAC_PI_4 + 0.5 * to_radians(lat),
427    ));
428    (a * x + 0.0, a * y + 0.0)
429}

438/// How far, in metres, a point projected by GeoTools may be from where
439/// [`Crs::apply`] puts it because of the JVM's own sines and cosines:
440/// 5e-8 m, more than six times the largest difference measured between
441/// GeoTools on Java 21 (arm64, with the JVM's intrinsics) and these
442/// transforms, 7.5e-9 m over 1.5 million points per system (those within 8
443/// degrees of the central meridian).
444const MARGIN_M: f64 = 5e-8;
445/// The same for a geographic system, in degrees: seven times the largest
446/// difference measured (2.8e-14 degrees, 3 nm).
447const MARGIN_DEG: f64 = 2e-13;
448
449impl Crs {
450    /// `"EPSG:31982 (SIRGAS 2000 / UTM zone 22S)"`.
451    pub(crate) fn label(&self) -> String {
452        match self.projection {
453            Some(Projection::TransverseMercator { cm, fn_, .. }) if self.name.ends_with("UTM") => {
454                format!(
455                    "EPSG:{} ({} zone {}{})",
456                    self.code,
457                    self.name,
458                    (cm as i32 + 183) / 6,
459                    if fn_ > 0.0 { 'S' } else { 'N' }
460                )
461            }
462            _ => format!("EPSG:{} ({})", self.code, self.name),
463        }
464    }
465
466    /// How far GeoTools' result may be from [`Crs::apply`]'s on some JVM
467    /// (see the module notes), in the system's units; `None` for WGS 84
468    /// longitude and latitude, where nothing is computed. A sample this
469    /// close to a boundary between pixels of different values is read as
470    /// `StrictMath` reads it and counted for the build's warning.
471    pub(crate) fn margin(&self) -> Option<f64> {
472        match (self.datum, self.projection) {
473            (Datum::Wgs84, None) => None,
474            (_, None) => Some(MARGIN_DEG),
475            (_, Some(_)) => Some(MARGIN_M),
476        }
477    }
478
479    /// WGS 84 longitude and latitude (degrees) in this system.
480    pub(crate) fn apply(&self, lon: f64, lat: f64) -> (f64, f64) {
481        let (mut lon, mut lat) = (lon, lat);
482        match self.datum {
483            Datum::Wgs84 => {}
484            Datum::Ellipsoid(e) => {
485                (lon, lat) = from_geocentric(e, to_geocentric(WGS84, lon, lat));
486            }
487            Datum::Shift(e, d) => {
488                let c = to_geocentric(WGS84, lon, lat);
489                (lon, lat) = from_geocentric(e, [c[0] + d[0], c[1] + d[1], c[2] + d[2]]);
490            }
491        }
492        let e = self.datum.ellipsoid();
493        let (mut x, mut y) = match self.projection {
494            None => (lon, lat),
495            Some(Projection::TransverseMercator { cm, k0, fe, fn_ }) => {
496                transverse_mercator(e, cm, k0, fe, fn_, lon, lat)
497            }
498            Some(Projection::Polyconic { cm, fe, fn_ }) => polyconic(e, cm, fe, fn_, lon, lat),
499            Some(Projection::PseudoMercator) => pseudo_mercator(e.a, lon, lat),
500        };
501        if self.residual {
502            let w = 1.0 + f64::EPSILON;
503            x = (x + f64::from_bits(0x3de0000000000000)) / w;
504            y = (y + f64::from_bits(0x3e20000000000000)) / w;
505        }
506        (x, y)
507    }
508}

583                (gx.to_bits(), gy.to_bits()),

606    #[test]
607    fn the_table() {
608        // the zones as GeoTools' database has them
609        let cm = |code| match lookup(code).unwrap().projection {
610            Some(Projection::TransverseMercator { cm, fn_, .. }) => (cm, fn_),
611            p => panic!("{p:?}"),
612        };
613        assert_eq!(cm(32601), (-177.0, 0.0));
614        assert_eq!(cm(32760), (177.0, 10_000_000.0));
615        assert_eq!(cm(31965), (-117.0, 0.0));
616        assert_eq!(cm(31977), (-81.0, 10_000_000.0));
617        assert_eq!(cm(31985), (-33.0, 10_000_000.0));
618        assert_eq!(cm(6210), (-45.0, 0.0));
619        assert_eq!(cm(6211), (-39.0, 0.0));
620        assert_eq!(cm(5396), (-27.0, 10_000_000.0));
621        assert_eq!(cm(29168), (-75.0, 0.0));
622        assert_eq!(cm(29195), (-33.0, 10_000_000.0));
623        assert_eq!(cm(22521), (-57.0, 10_000_000.0));
624        assert!(lookup(32761).is_none() && lookup(29173).is_none() && lookup(2193).is_none());
625        assert!(lookup(32722).unwrap().residual && !lookup(32622).unwrap().residual);
626        assert_eq!(
627            lookup(31982).unwrap().label(),
628            "EPSG:31982 (SIRGAS 2000 / UTM zone 22S)"
629        );
630        assert_eq!(lookup(4674).unwrap().label(), "EPSG:4674 (SIRGAS 2000)");
631        assert_eq!(lookup(4326).unwrap().margin(), None);
632        assert_eq!(lookup(4674).unwrap().margin(), Some(MARGIN_DEG));
633        assert_eq!(lookup(3857).unwrap().margin(), Some(MARGIN_M));
634        // WGS 84 is the identity
635        let w = lookup(4326).unwrap();
636        assert_eq!(w.apply(-51.25, -30.01), (-51.25, -30.01));
637    }
