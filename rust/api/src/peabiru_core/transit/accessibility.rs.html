1//! Accessibility to opportunities (`accessibility()`): for each
2//! origin, the sum over destinations of `opportunities * weight(t)`, where
3//! `t` is a travel-time percentile in seconds and `weight` a decay function
4//! of `t` and a cutoff (R5 `TravelTimeReducer.recordTravelTimePercentilesForTarget`,
5//! `AccessibilityResult`, and the ported `AccessibilityEstimator`).
6//!
7//! Travel times come from the same computation as the travel-time
8//! matrices: [`super::matrix`] for walk + transit and
9//! [`crate::street::router`] for street-only modes. Unlike the matrices,
10//! nothing is rounded to minutes: the decay is applied to the percentile in
11//! whole seconds (`TravelTimeReducer.java:217-226`), and the sum is the
12//! raw double, unrounded.
13

24/// `AnalysisWorkerTask.MAX_REGIONAL_CUTOFFS`.
25const MAX_CUTOFFS: usize = 12;
26
27/// How to weight the opportunities; the defaults are `cutoffs = 30` and
28/// `decay_function = "step"`). Fill it in by changing the fields of
29/// [`AccessibilityParams::default()`].
30#[derive(Debug, Clone)]
31#[non_exhaustive]
32pub struct AccessibilityParams {
33    /// Cutoffs in whole minutes: 1 to 12 values in 0..=120, in any order
34    /// (they are sorted ascending). Pass the single placeholder `0` with
35    /// [`DecayFunction::FixedExponential`], which ignores it.
36    pub cutoffs: Vec<i32>,
37    /// The decay function and its parameter.
38    pub decay: DecayFunction,
39}
40
41impl Default for AccessibilityParams {
42    fn default() -> Self {
43        AccessibilityParams {
44            cutoffs: vec![30],
45            decay: DecayFunction::Step,
46        }
47    }
48}

50/// Accessibility per origin, opportunity column, percentile and cutoff:
51/// the raw double, unrounded.
52///
53/// The values are held in one buffer, origin by origin, then opportunity
54/// column, percentile and cutoff (the cutoff varies fastest), which
55/// [`into_values`](Accessibility::into_values) hands over without copying.
56#[derive(Debug, Clone, PartialEq)]
57pub struct Accessibility {
58    /// Percentiles in output order (ascending).
59    percentiles: Vec<i32>,
60    /// Cutoffs in minutes, in output order (ascending).
61    cutoffs: Vec<i32>,
62    n_opportunities: usize,
63    /// `[origin][opportunity][percentile][cutoff]`, flattened.
64    values: Vec<f64>,
65}
66
67impl Accessibility {
68    /// Number of origins.
69    pub fn n_origins(&self) -> usize {
70        match self.per_origin() {
71            0 => 0,
72            per => self.values.len() / per,
73        }
74    }
75
76    /// Number of opportunity columns, in the order of the destinations'
77    /// columns (see [`Destinations::new`]).
78    pub fn n_opportunities(&self) -> usize {
79        self.n_opportunities
80    }
81
82    /// The percentiles, in output order (ascending).
83    pub fn percentiles(&self) -> &[i32] {
84        &self.percentiles
85    }
86
87    /// The cutoffs in minutes, in output order (ascending). With
88    /// [`DecayFunction::FixedExponential`], which ignores cutoffs, whatever
89    /// [`AccessibilityParams::cutoffs`] held (usually the placeholder
90    /// `0`).
91    pub fn cutoffs(&self) -> &[i32] {
92        &self.cutoffs
93    }
94
95    /// Values per origin: opportunity columns x percentiles x cutoffs.
96    pub fn per_origin(&self) -> usize {
97        self.n_opportunities * self.percentiles.len() * self.cutoffs.len()
98    }
99
100    /// Accessibility of `origin` to opportunity column `opportunity` at the
101    /// `percentile`-th percentile and `cutoff`-th cutoff (indices into
102    /// [`percentiles`](Accessibility::percentiles) and
103    /// [`cutoffs`](Accessibility::cutoffs)). Panics when an index is out of
104    /// range.
105    pub fn get(&self, origin: usize, opportunity: usize, percentile: usize, cutoff: usize) -> f64 {
106        let (np, nc) = (self.percentiles.len(), self.cutoffs.len());
107        assert!(
108            origin < self.n_origins()
109                && opportunity < self.n_opportunities
110                && percentile < np
111                && cutoff < nc,
112            "({origin}, {opportunity}, {percentile}, {cutoff}) is out of range"
113        );
114        self.values[origin * self.per_origin() + (opportunity * np + percentile) * nc + cutoff]
115    }
116
117    /// Every value, origin by origin, then opportunity column, percentile
118    /// and cutoff (the cutoff varies fastest).
119    pub fn values(&self) -> &[f64] {
120        &self.values
121    }
122
123    /// Every value, in the order of [`values`](Accessibility::values),
124    /// without copying.
125    pub fn into_values(self) -> Vec<f64> {
126        self.values
127    }
128}

130/// Destination points with their opportunity counts: `opportunities[k][j]`
131/// is the count of opportunity kind `k` at `points[j]` (one column per
132/// `opportunities_colnames` entry of the packages). Made with
133/// [`Destinations::new`].
134#[derive(Debug, Clone, Copy)]
135#[non_exhaustive]
136pub struct Destinations<'a> {
137    /// `(lat, lon)` degrees.
138    pub points: &'a [(f64, f64)],
139    /// One column per kind of opportunity, one count per point.
140    pub opportunities: &'a [Vec<f64>],
141}
142
143impl<'a> Destinations<'a> {
144    /// Destination `points` (`(lat, lon)` degrees) and their opportunity
145    /// counts, one column per kind of opportunity with one count per
146    /// point. The accessibility functions check that every column has one
147    /// count per point.
148    pub fn new(points: &'a [(f64, f64)], opportunities: &'a [Vec<f64>]) -> Self {
149        Destinations {
150            points,
151            opportunities,
152        }
153    }
154}

156/// The validated request: decay, cutoffs in seconds with their zero points,
157/// percentiles and opportunity counts (R5 `TravelTimeReducer` constructor).
158struct Reducer {
159    decay: Decay,
160    percentiles: Vec<i32>,
161    cutoffs_minutes: Vec<i32>,
162    cutoffs_seconds: Vec<i32>,
163    zero_points: Vec<i32>,
164    /// `[opportunity][destination]`, NaN replaced by 0.
165    opportunities: Vec<Vec<f64>>,
166    /// Destinations where some opportunity count is positive; the others
167    /// cannot change any sum.
168    useful: Vec<bool>,
169}
170
171impl Reducer {
172    fn new(
173        percentiles: &[i32],
174        a: &AccessibilityParams,
175        opportunities: &[Vec<f64>],
176        n_destinations: usize,
177    ) -> Result<Self, String> {
178        // AnalysisWorkerTask.validatePercentiles (sorted first)
179        check_percentiles(percentiles)?;
180        let mut percentiles = percentiles.to_vec();
181        percentiles.sort_unstable();
182        // AnalysisWorkerTask.validateCutoffsMinutes (sorted first)
183        let mut cutoffs = a.cutoffs.clone();
184        cutoffs.sort_unstable();
185        if cutoffs.is_empty() || cutoffs.len() > MAX_CUTOFFS {
186            return Err(format!(
187                "between 1 and {MAX_CUTOFFS} cutoffs are allowed, got {}",
188                cutoffs.len()
189            ));
190        }
191        if let Some(c) = cutoffs.iter().find(|c| !(0..=120).contains(*c)) {
192            return Err(format!("cutoffs must be 0 to 120 minutes, got {c}"));
193        }
194        if opportunities.is_empty() {
195            return Err("at least one opportunity column is required".into());
196        }
197        let mut cols = Vec::with_capacity(opportunities.len());
198        for (k, col) in opportunities.iter().enumerate() {
199            if col.len() != n_destinations {
200                return Err(format!(
201                    "opportunity column {k} has {} values for {n_destinations} destinations",
202                    col.len()
203                ));
204            }
205            if let Some(v) = col.iter().find(|v| v.is_infinite()) {
206                return Err(format!(
207                    "opportunity column {k} has a non-finite value ({v})"
208                ));
209            }
210            // a missing count (NaN) is 0; R5 would skip a NaN anyway
211            // (`!(count > 0)`).
212            cols.push(
213                col.iter()
214                    .map(|&v| if v.is_nan() { 0.0 } else { v })
215                    .collect::<Vec<f64>>(),
216            );
217        }
218        let decay = Decay::new(a.decay.clone())?;
219        let cutoffs_seconds: Vec<i32> = cutoffs.iter().map(|c| c * 60).collect();
220        let zero_points = cutoffs_seconds
221            .iter()
222            .map(|&c| decay.reaches_zero_at(c))
223            .collect();
224        let useful = (0..n_destinations)
225            .map(|j| cols.iter().any(|c| c[j] > 0.0))
226            .collect();
227        Ok(Reducer {
228            decay,
229            percentiles,
230            cutoffs_minutes: cutoffs,
231            cutoffs_seconds,
232            zero_points,
233            opportunities: cols,
234            useful,
235        })
236    }
237
238    fn per_origin(&self) -> usize {
239        self.opportunities.len() * self.percentiles.len() * self.cutoffs_seconds.len()
240    }
241
242    /// `TravelTimeReducer.extractTravelTimePercentilesAndRecord` for one
243    /// target: sort the per-iteration seconds, take the percentiles, add the
244    /// weighted opportunities into `acc` (`[opportunity][percentile][cutoff]`).
245    fn record(&self, target: usize, times: &mut [i32], pct: &mut Vec<i32>, acc: &mut [f64]) {
246        if !self.useful[target] {
247            return;
248        }
249        times.sort_unstable();
250        let n = times.len();
251        pct.clear();
252        pct.extend(
253            self.percentiles
254                .iter()
255                .map(|&p| times[percentile_index(p, n).min(n - 1)]),
256        );
257        let (np, nc) = (self.percentiles.len(), self.cutoffs_seconds.len());
258        // TravelTimeReducer.java:250-279
259        for k in 0..self.opportunities.len() {
260            let count = self.opportunities[k][target];
261            // `if (!(opportunityCountAtTarget > 0)) continue;`
262            if count.is_nan() || count <= 0.0 {
263                continue;
264            }
265            for (p, &t) in pct.iter().enumerate() {
266                if t == UNREACHED {
267                    // percentiles are ascending: the rest are UNREACHED too
268                    break;
269                }
270                // backward through the cutoffs, stopping at the first whose
271                // zero point the time reaches
272                for c in (0..nc).rev() {
273                    if t >= self.zero_points[c] {
274                        break;
275                    }
276                    let w = self.decay.weight(self.cutoffs_seconds[c], t);
277                    if w > 0.0 {
278                        acc[(k * np + p) * nc + c] += count * w;
279                    }
280                }
281            }
282        }
283    }
284
285    fn finish(self, n_origins: usize, values: Vec<f64>) -> Accessibility {
286        debug_assert_eq!(values.len(), n_origins * self.per_origin());
287        Accessibility {
288            n_opportunities: self.opportunities.len(),
289            percentiles: self.percentiles,
290            cutoffs: self.cutoffs_minutes,
291            values,
292        }
293    }
294}

296/// The work of [`crate::transit_accessibility`].
297pub(crate) fn transit_accessibility(
298    net: &StreetNetwork,
299    layer: &TransitLayer,
300    data: &TransitData,
301    origins: &[(f64, f64)],
302    destinations: &Destinations,
303    p: &TransitParams,
304    a: &AccessibilityParams,
305) -> Result<Accessibility, RouteError> {
306    let r = Reducer::new(
307        &p.percentiles,
308        a,
309        destinations.opportunities,
310        destinations.points.len(),
311    )
312    .map_err(RouteError::Invalid)?;
313    let q = Query::new(net, layer, data, destinations.points, p)?;
314    let per = r.per_origin();
315    let rows = par_origins(p.n_threads, origins.len(), &p.cancel, |i, scratch| {
316        let mut acc = vec![0.0; per];
317        let mut pct = Vec::with_capacity(r.percentiles.len());
318        // An unlinked origin records nothing: all zeros
319        // (TravelTimeComputer.java:313-317).
320        q.route_origin(origins[i], scratch, |j, times| {
321            r.record(j, times, &mut pct, &mut acc)
322        })?;
323        Ok(acc)
324    })?;
325    let rows = rows.into_iter().collect::<Result<Vec<_>, RouteError>>()?;
326    Ok(r.finish(origins.len(), rows.concat()))
327}

329/// The work of [`crate::street_accessibility`].
330pub(crate) fn street_accessibility(
331    net: &StreetNetwork,
332    origins: &[(f64, f64)],
333    destinations: &Destinations,
334    mode: Mode,
335    params: &RouteParams,
336    percentiles: &[i32],
337    a: &AccessibilityParams,
338) -> Result<Accessibility, RouteError> {
339    let r = Reducer::new(
340        percentiles,
341        a,
342        destinations.opportunities,
343        destinations.points.len(),
344    )
345    .map_err(RouteError::Invalid)?;
346    let per = r.per_origin();
347    let rows = travel_time_seconds(
348        net,
349        origins,
350        destinations.points,
351        mode,
352        params,
353        |_, times| {
354            let mut acc = vec![0.0; per];
355            if let Some(times) = times {
356                let mut pct = Vec::with_capacity(r.percentiles.len());
357                let mut one = [0i32];
358                for (j, &t) in times.iter().enumerate() {
359                    one[0] = t;
360                    r.record(j, &mut one, &mut pct, &mut acc);
361                }
362            }
363            acc
364        },
365    )?;
366    Ok(r.finish(origins.len(), rows.concat()))
367}

369/// The trip limits of `accessibility()`, derived from what the user asked.
370/// Given the user's `max_trip_duration` and `max_walk_time` (`None` = `Inf`)
371/// in minutes, returns the `(max_trip_duration, max_walk_time)` to route
372/// with:
373/// - `max_trip_duration` may not be below the largest cutoff (an error);
374/// - with the `step` function it becomes the largest cutoff;
375/// - `max_walk_time = Inf` then becomes that `max_trip_duration`;
376/// - for walking alone (`walk_only`, no transit) `max_trip_duration` is
377///   reduced to `max_walk_time`. (Bicycle and car alone have their own
378///   limits, which also default to `max_trip_duration`.)
379pub fn accessibility_trip_limits(
380    decay: &DecayFunction,
381    cutoffs: Option<&[i32]>,
382    max_trip_duration: i32,
383    max_walk_time: Option<i32>,
384    walk_only: bool,
385) -> Result<(i32, i32), RouteError> {
386    if max_trip_duration < 1 {
387        return Err(RouteError::Invalid(format!(
388            "max_trip_duration must be at least 1, got {max_trip_duration}"
389        )));
390    }
391    let mut m = max_trip_duration;
392    if let Some(cutoffs) = cutoffs {
393        let top = cutoffs.iter().copied().max().unwrap_or(0);
394        if m < top {
395            return Err(RouteError::Invalid(
396                "'max_trip_duration' cannot be shorter than 'max(cutoffs)'".into(),
397            ));
398        }
399        if *decay == DecayFunction::Step {
400            m = top;
401        }
402    }
403    let w = match max_walk_time {
404        Some(w) if w >= 1 => w,
405        Some(w) => {
406            return Err(RouteError::Invalid(format!(
407                "max_walk_time must be at least 1, got {w}"
408            )))
409        }
410        None => m,
411    };
412    if walk_only && w < m {
413        m = w;
414    }
415    Ok((m, w))
416}

429    #[test]
430    fn step_reduction_counts_strictly_below_each_cutoff() {
431        let opp = vec![vec![1.0, 2.0, 4.0, 8.0, 0.0, -5.0]];
432        let a = params(DecayFunction::Step, &[30, 10]);
433        let r = Reducer::new(&[50], &a, &opp, 6).unwrap();
434        assert_eq!(r.cutoffs_minutes, vec![10, 30]);
435        let mut acc = vec![0.0; r.per_origin()];
436        let mut pct = Vec::new();
437        for (j, t) in [0, 599, 600, 1799, 100, 100].into_iter().enumerate() {
438            r.record(j, &mut [t], &mut pct, &mut acc);
439        }
440        // cutoff 10 min: t < 600 -> 1 + 2; 30 min: + 4 + 8; zero and
441        // negative counts never count
442        assert_eq!(acc, vec![3.0, 15.0]);
443    }
