1//! Per-departure travel times with the routes ridden:
2//! `expanded_travel_time_matrix()` and `arrival_travel_time_matrix()` with
3//! public transport in `mode` (the modes, access and egress of
4//! [`TransitParams`]). A record without public transport has the access
5//! mode as its routes (`[WALK]`, `[BICYCLE]`, `[CAR]`: Java's
6//! `directModes.toString()`).
7//!
8//! Both are one R5 regional task per origin with `includePathResults`:
9//! the RAPTOR search keeps a path to every stop at every iteration
10//! (`FastRaptorWorker.retainPaths`), propagation keeps the path and egress
11//! behind each destination's best time per iteration
12//! (`PerTargetPropagater.propagateTransit`), and `TravelTimeReducer
13//! .recordPathsForTarget` turns them into one record per iteration. The
14//! ported `TravelTimeMatrixComputer` (see LICENSE-THIRD-PARTY; the offsets
15//! below are those of its bytecode)
16//! then breaks each record down into minutes (`extractPathResults`), sorts
17//! the records of each departure minute (`PATH_ORDER`) and numbers them as
18//! draws (`populateTravelTimesBreakdown`), or for arrival keeps the latest
19//! departure that arrives in time (`filterLatestBeforeArrivalTime`). The R
20//! wrappers finally blank out rows whose `total_time` exceeds
21//! `max_trip_duration`.
22

49/// One trip, every time in minutes with one decimal.
50#[derive(Debug, Clone, PartialEq)]
51#[non_exhaustive]
52pub struct Breakdown {
53    /// Walking from the origin to the first stop (or the whole walk of a
54    /// walk-only trip).
55    pub access_time: f64,
56    /// Waiting at stops, the first boarding included.
57    pub wait_time: f64,
58    /// In vehicles.
59    pub ride_time: f64,
60    /// Walking between stops.
61    pub transfer_time: f64,
62    /// Walking from the last stop to the destination.
63    pub egress_time: f64,
64    /// Route ids joined by `|`, or `[WALK]` for a walk-only trip.
65    pub routes: String,
66    /// Vehicles boarded.
67    pub n_rides: u32,
68    /// Total travel time: the sum of the five times
69    /// above rounded to one decimal (when that sum is 0, the trip's own
70    /// total, rounded).
71    pub total_time: f64,
72}

74/// One row of an expanded or arrival matrix (the origin is the row's place

76#[derive(Debug, Clone, PartialEq)]
77#[non_exhaustive]
78pub struct ExpandedRow {

124/// `Math.round(double)` for the values met here: the nearest integer,
125/// ties towards positive infinity.
126fn java_round(x: f64) -> f64 {
127    let f = x.floor();
128    if x - f >= 0.5 {
129        f + 1.0
130    } else {
131        f
132    }
133}

135/// `PathBreakdown.roundTo1Place`: `Math.round(x * 10) / 10.0`.
136fn round1(x: f64) -> f64 {
137    java_round(x * 10.0) / 10.0
138}

146/// The ported `TravelTimeMatrixComputer$PathBreakdown` with its raw
147/// (unrounded) minute fields.
148#[derive(Debug, Clone)]
149struct PathBreakdown {
150    access: f64,
151    wait: f64,
152    ride: f64,
153    transfer: f64,
154    egress: f64,
155    total: f64,
156    /// Shared between the records of one path.
157    routes: Arc<str>,
158    n_rides: u32,
159}
160
161impl PathBreakdown {
162    /// A walk-only record: every component 0, `routes = directModes`
163    /// (`walk_routes` holds `[WALK]`).
164    fn walk(walk_routes: &Arc<str>, total: f64) -> Self {
165        PathBreakdown {
166            access: 0.0,
167            wait: 0.0,
168            ride: 0.0,
169            transfer: 0.0,
170            egress: 0.0,
171            total,
172            routes: walk_routes.clone(),
173            n_rides: 0,
174        }
175    }
176
177    /// `PathBreakdown.unreachable()`: total `Integer.MAX_VALUE` minutes.
178    fn unreachable(walk_routes: &Arc<str>) -> Self {
179        Self::walk(walk_routes, f64::from(i32::MAX))
180    }
181
182    /// The row of the result (`addPathToDataframe`): `total_time` is the
183    /// rounded sum of the rounded components when positive
184    /// (`getCombinedTravelTime`), else the rounded total; the caller then
185    /// sets the row to `None` when `total_time > max_trip_duration`.
186    fn report(&self, max_trip_min: i32) -> Option<Breakdown> {
187        let (a, w, r, t, e) = (
188            round1(self.access),
189            round1(self.wait),
190            round1(self.ride),
191            round1(self.transfer),
192            round1(self.egress),
193        );
194        let combined = round1(a + w + r + t + e);
195        let total = if combined > 0.0 {
196            combined
197        } else {
198            round1(self.total)
199        };
200        (total <= f64::from(max_trip_min)).then(|| Breakdown {
201            access_time: a,
202            wait_time: w,
203            ride_time: r,
204            transfer_time: t,
205            egress_time: e,
206            routes: self.routes.to_string(),
207            n_rides: self.n_rides,
208            total_time: total,
209        })
210    }
211}

213/// `PATH_ORDER`: total time, routes (Java `String.compareTo`, UTF-16 code
214/// units), then access, wait, ride, transfer and egress (`Double.compare`).
215fn path_order(a: &PathBreakdown, b: &PathBreakdown) -> Ordering {
216    a.total
217        .total_cmp(&b.total)
218        .then_with(|| a.routes.encode_utf16().cmp(b.routes.encode_utf16()))
219        .then_with(|| a.access.total_cmp(&b.access))
220        .then_with(|| a.wait.total_cmp(&b.wait))
221        .then_with(|| a.ride.total_cmp(&b.ride))
222        .then_with(|| a.transfer.total_cmp(&b.transfer))
223        .then_with(|| a.egress.total_cmp(&b.egress))
224}

226/// The records of one destination: transit records per departure minute
227/// (sorted by `PATH_ORDER`) and walk-only records (`DIRECT_KEY = -1`).
228#[derive(Default)]
229struct DestinationPaths {
230    /// Indexed by minute of the window (0 = first departure); a minute's
231    /// list is filled when [`Records::minute`] first asks for it.
232    by_minute: Vec<Vec<PathBreakdown>>,
233    /// Whether `by_minute[k]` has been filled.
234    built: Vec<bool>,
235    walk: Vec<PathBreakdown>,
236}

238/// What the transit records of a destination are built from: the best
239/// path and egress behind each iteration's time after propagation.
240struct TransitSource<'a> {
241    layer: &'a TransitLayer,
242    paths: &'a PathArena,
243    best: &'a [i32],
244    best_path: &'a [u32],
245    best_egress: &'a [i32],
246    access_time: &'a [i32],
247    /// The route list of each path, built once per path and origin.
248    routes: &'a mut [Option<Arc<str>>],
249    per_minute: usize,
250    to_time: i32,
251}
252
253/// One destination's records as the callers of [`Context::origin_paths`]
254/// read them: the walk-only records, sorted, and each minute's transit
255/// records, built and sorted on first request. The arrival search usually
256/// reads only the last minutes of its window, so the other minutes are
257/// never built; the expanded matrix reads them all.
258struct Records<'a> {
259    dp: &'a mut DestinationPaths,
260    src: Option<TransitSource<'a>>,
261    /// Whether some iteration reached the destination by transit.
262    has_transit: bool,
263}
264
265impl Records<'_> {
266    fn is_empty(&self) -> bool {
267        self.dp.walk.is_empty() && !self.has_transit
268    }
269
270    fn walk(&self) -> &[PathBreakdown] {
271        &self.dp.walk
272    }
273
274    /// The transit records of departure minute `k` (0 = first departure)
275    /// in `PATH_ORDER`: one per iteration of that minute with a transit
276    /// path, in iteration order before the (stable) sort, as
277    /// `extractPathResults` adds them.
278    fn minute(&mut self, k: usize) -> Result<&[PathBreakdown], String> {
279        if !self.dp.built[k] {
280            let list = &mut self.dp.by_minute[k];
281            list.clear();
282            if let Some(s) = &mut self.src {
283                // iterations run from the latest minute down
284                let block = self.dp.built.len() - 1 - k;
285                let dep = s.to_time - 60 * (1 + block as i32);
286                let end = ((block + 1) * s.per_minute).min(s.best_path.len());
287                for i in block * s.per_minute..end {
288                    let path_k = s.best_path[i];
289                    if path_k == NO_PATH {
290                        continue;
291                    }
292                    let path = s.paths.legs(path_k);
293                    let routes = s.routes[path_k as usize]
294                        .get_or_insert_with(|| route_list(s.layer, path))
295                        .clone();
296                    list.push(transit_breakdown(
297                        path,
298                        routes,
299                        dep,
300                        s.access_time,
301                        s.best_egress[i],
302                        s.best[i],
303                    )?);
304                }
305                list.sort_by(path_order); // stable, as `List.sort`
306            }
307            self.dp.built[k] = true;
308        }
309        Ok(&self.dp.by_minute[k])
310    }
311}

313/// What every origin shares.
314struct Context<'a> {
315    net: &'a StreetNetwork,
316    /// The scenario's network, which the access and direct searches run on
317    /// (`None`: they run on `net`).
318    scen: Option<StreetNetwork>,
319    layer: &'a TransitLayer,
320    data: &'a TransitData,
321    filtered: Filtered,
322    p: &'a TransitParams,
323    dest_splits: Vec<Option<Split>>,
324    /// Per destination, `(stop, egress seconds)` in the iteration order of
325    /// R5's per-point `TIntIntHashMap` (`EgressCostTable
326    /// .destructivelyTransposeForPropagationAsNeeded`): ties between stops
327    /// are won by the first one met. Stops at or beyond the egress limit
328    /// are left out.
329    egress: Vec<Vec<(u32, i32)>>,
330    /// Stops within egress reach of some destination.
331    retain: Vec<bool>,
332    walk_mms: i32,
333    /// The access and direct street search.
334    access: Access,
335    /// `[WALK]` (`[BICYCLE]`, `[CAR]`), the routes of a record without
336    /// public transport: Java's `directModes.toString()`.
337    walk_routes: Arc<str>,
338    /// The draws per minute (`populateExpandedResults` offs 13-46):
339    /// `numberOfMonteCarloDraws / timeWindowSize` with frequencies, else 1.
340    draws_per_minute: usize,
341}
342
343impl<'a> Context<'a> {
344    fn new(
345        net: &'a StreetNetwork,
346        layer: &'a TransitLayer,
347        data: &'a TransitData,
348        destinations: &[(f64, f64)],
349        p: &'a TransitParams,
350    ) -> Result<Self, RouteError> {
351        validate(layer, destinations.len(), p).map_err(RouteError::Invalid)?;
352        let active = layer.active_services(p.date, weekday(p.date));
353        let filtered =
354            Filtered::new(layer, &active, &p.transit_modes).map_err(RouteError::Invalid)?;
355        let walk_speed = (p.walk_speed * 5.0 / 18.0) as f32;
356        let walk_mms = (walk_speed * 1000.0f32) as i32;
357        let link = |mode| {
358            link_points(
359                net,
360                destinations.len(),
361                |j| destinations[j],
362                mode,
363                p.n_threads,
364                &p.cancel,
365            )
366            .ok_or(RouteError::Cancelled)
367        };
368        // the destinations by the egress mode (walking) and, for the
369        // direct times, by the access mode
370        let egress_splits = link(p.egress_mode)?;
371        let egress = if p.egress_mode == Mode::Walk {
372            walk::egress_tables(
373                layer,
374                &data.trees,
375                net,
376                destinations,
377                &egress_splits,
378                p.n_threads,
379                &p.cancel,
380            )
381        } else {
382            mode_egress_tables(
383                net,
384                layer,
385                p.egress_mode,
386                Targets::Points(destinations),
387                &egress_splits,
388                p.n_threads,
389                &p.cancel,
390            )
391        }
392        .ok_or(RouteError::Cancelled)?;
393        let dest_splits = if p.mode == p.egress_mode {
394            egress_splits
395        } else {
396            link(p.mode)?
397        };
398        let scen =
399            crate::street::scenario::scenario_network(net, &p.scenario, p.mode == Mode::Car)?;
400        let access = Access::new(scen.as_ref().unwrap_or(net), p);
401        let seconds = egress_seconds(p);
402        let mut retain = vec![false; layer.stop_ids.len()];
403        let egress = egress
404            .into_iter()
405            .map(|table| {
406                // `table` is in increasing stop order, the order R5 puts the
407                // stops into the map; the stops beyond the egress limit are
408                // skipped afterwards, in propagation.
409                let mut keys = TroveIntKeys::new();
410                for &(s, _) in &table {
411                    keys.insert(s as i32);
412                    retain[s as usize] = true;
413                }
414                keys.iteration_order()
415                    .filter_map(|s| {
416                        let i = table
417                            .binary_search_by_key(&(s as u32), |&(t, _)| t)
418                            .expect("stop in egress table");
419                        let (stop, cost) = table[i];
420                        Some((stop, seconds(cost)?))
421                    })
422                    .collect()
423            })
424            .collect();
425        let draws_per_minute = if layer.has_frequencies {
426            ((p.time_window * p.draws_per_minute) / p.time_window) as usize
427        } else {
428            1
429        };
430        Ok(Context {
431            net,
432            scen,
433            layer,
434            data,
435            filtered,
436            p,
437            dest_splits,
438            egress,
439            retain,
440            walk_mms,
441            walk_routes: format!("[{}]", access.mode().r5_name()).into(),
442            access,
443            draws_per_minute,
444        })
445    }
446
447    /// The records of every destination from origin `(lat, lon)`, handed to
448    /// `on_destination(j, records)` in destination order; `false` when the
449    /// origin is not linked (R5 then reports nothing).
450    fn origin_paths(
451        &self,
452        lat: f64,
453        lon: f64,
454        sc: &mut OriginScratch,
455        mut on_destination: impl FnMut(usize, &mut Records) -> Result<(), String>,
456    ) -> Result<bool, String> {
457        let p = self.p;
458        let m = p.max_trip_duration;
459        let max_t = m * 60;
460        let tw = p.time_window as usize;
461        let mut access = Vec::new();
462        if !self.access.search(
463            self.scen.as_ref().unwrap_or(self.net),
464            self.layer,
465            (lat, lon),
466            &self.dest_splits,
467            &mut sc.split,
468            &mut sc.search,
469            &mut sc.direct,
470            &mut access,
471        ) {
472            return Ok(false);
473        }
474        let direct = &sc.direct;
475        let dp = &mut sc.records;
476        dp.by_minute.resize_with(tw, Vec::new);
477        dp.built.resize(tw, false);
478        let clear = |dp: &mut DestinationPaths| {
479            dp.built.iter_mut().for_each(|b| *b = false);
480            dp.walk.clear();
481        };
482        if access.is_empty() {
483            // No stop reached: R5 records the direct time as every percentile
484            // and no path result (`TravelTimeComputer.java:304-315`); the result then
485            // has a walk record of `percentile minutes` when <= M
486            // (`extractPathResults` offs 528-604; totalTime is `(double)
487            // minutes`).
488            for (j, &d) in direct.iter().enumerate() {
489                clear(dp);
490                let minutes = if d == UNREACHED { UNREACHED } else { d / 60 };
491                if minutes <= m {
492                    dp.walk
493                        .push(PathBreakdown::walk(&self.walk_routes, f64::from(minutes)));
494                }
495                let mut rec = Records {
496                    dp: &mut *dp,
497                    src: None,
498                    has_transit: false,
499                };
500                on_destination(j, &mut rec)?;
501            }
502            return Ok(true);
503        }
504        let req = RaptorRequest {
505            from_time: p.departure_time,
506            time_window: p.time_window,
507            max_rides: p.max_rides,
508            max_duration: max_t,
509            walk_mms: self.walk_mms,
510            max_walk_min: p.max_walk_time,
511            monte_carlo_draws: p.time_window * p.draws_per_minute,
512            seed: p.seed,
513            origin: (lat, lon),
514        };
515        raptor::route_paths_into(
516            self.layer,
517            &self.data.transfers,
518            &self.filtered,
519            &access,
520            &req,
521            &self.retain,
522            &p.cancel,
523            &mut sc.times,
524            &mut sc.arena,
525        );
526        if p.cancel.is_cancelled() {
527            // the search stopped early; the caller discards the origin
528            return Ok(true);
529        }
530        let (times, paths) = (&sc.times, &sc.arena);
531        let n_stops = self.layer.stop_ids.len();
532        // Per stop, the least duration over the iterations: a lower bound
533        // on every candidate through that stop.
534        let stop_min = &mut sc.stop_min;
535        stop_min.clear();
536        stop_min.resize(n_stops, UNREACHED);
537        for row in times.chunks_exact(n_stops.max(1)) {
538            for (m, &t) in stop_min.iter_mut().zip(row) {
539                *m = (*m).min(t);
540            }
541        }
542        let stop_min = &*stop_min;
543        let per_minute = req.iterations_per_minute(self.layer.has_frequencies)?;
544        let n_iter = times.len() / n_stops.max(1);
545        let to_time = p.departure_time + p.time_window * 60;
546        let mut access_time = vec![-1; self.layer.stop_ids.len()];
547        for &(s, t) in &access {
548            access_time[s as usize] = t;
549        }
550        // The route list of each path, built once per path.
551        let mut routes: Vec<Option<Arc<str>>> = vec![None; paths.len()];
552        let mut best = vec![0; n_iter];
553        let mut best_path: Vec<u32> = vec![NO_PATH; n_iter];
554        let mut best_egress = vec![0; n_iter];
555        for (j, &d) in direct.iter().enumerate() {
556            if p.cancel.is_cancelled() {
557                return Ok(true);
558            }
559            best.iter_mut().for_each(|t| *t = d);
560            best_path.iter_mut().for_each(|x| *x = NO_PATH);
561            propagate(
562                &self.egress[j],
563                times,
564                stop_min,
565                n_stops,
566                paths,
567                max_t,
568                &mut best,
569                &mut best_path,
570                &mut best_egress,
571            );
572            // `recordPathsForTarget` + `extractPathResults`: the walk-only
573            // records now, the transit ones per minute on request.
574            clear(dp);
575            for i in 0..n_iter {
576                if best_path[i] == NO_PATH && best[i] < UNREACHED {
577                    dp.walk
578                        .push(PathBreakdown::walk(&self.walk_routes, minutes_f32(best[i])));
579                }
580            }
581            dp.walk.sort_by(path_order);
582            let mut rec = Records {
583                dp: &mut *dp,
584                has_transit: best_path.iter().any(|&k| k != NO_PATH),
585                src: Some(TransitSource {
586                    layer: self.layer,
587                    paths,
588                    best: &best,
589                    best_path: &best_path,
590                    best_egress: &best_egress,
591                    access_time: &access_time,
592                    routes: &mut routes,
593                    per_minute,
594                    to_time,
595                }),
596            };
597            on_destination(j, &mut rec)?;
598        }
599        Ok(true)
600    }
601}

617/// `PerTargetPropagater.propagateTransit` for one destination, keeping the
618/// path (an index into `paths.paths`) and egress time behind each
619/// iteration's best time. Only a strict improvement replaces them, so equal
620/// times keep the stop met first. `stop_min[s]` is the least of stop `s`'s
621/// durations over the iterations.
622#[allow(clippy::too_many_arguments)]
623fn propagate(
624    egress: &[(u32, i32)],
625    times: &[i32],
626    stop_min: &[i32],
627    n_stops: usize,
628    paths: &PathArena,
629    max_t: i32,
630    best: &mut [i32],
631    best_path: &mut [u32],
632    best_egress: &mut [i32],
633) {
634    // No candidate at or above `bound` replaces anything: the smaller of
635    // max_t and the worst time so far (only strict improvements count).
636    let mut bound = best.iter().copied().max().map_or(max_t, |w| max_t.min(w));
637    for &(stop, e) in egress {
638        // every iteration's candidate through this stop is at least its
639        // least duration plus the egress walk
640        if stop_min[stop as usize].saturating_add(e) >= bound {
641            continue;
642        }
643        let mut worst = 0;
644        for (i, t) in best.iter_mut().enumerate() {
645            let ts = times[i * n_stops + stop as usize];
646            if ts < max_t && ts < *t {
647                // saturating: past i32::MAX is past max_t
648                let c = ts.saturating_add(e);
649                if c < max_t && c < *t {
650                    *t = c;
651                    if let Some(k) = paths.index(i, stop as usize) {
652                        best_path[i] = k;
653                        best_egress[i] = e;
654                    }
655                }
656            }
657            worst = worst.max(*t);
658        }
659        bound = max_t.min(worst);
660    }
661}

672/// The breakdown of one transit iteration (`extractPathResults` offs
673/// 162-393): access to the first boarding stop, waits derived from the
674/// departure (`Path.computeWaitTimes`), rides, transfers and egress.
675fn transit_breakdown(
676    path: &[Leg],
677    routes: Arc<str>,
678    departure: i32,
679    access_time: &[i32],
680    egress: i32,
681    total: i32,
682) -> Result<PathBreakdown, String> {
683    let access = path
684        .first()
685        .map(|l| access_time[l.board_stop as usize])
686        .filter(|&t| t >= 0)
687        .ok_or("path without access to its first boarding stop")?;
688    // `TransitPath::wait_times`, summed
689    let mut ready = departure + access;
690    let mut wait: i32 = 0;
691    for l in path {
692        ready += l.transfer_time;
693        let w = l.board_time - ready;
694        if w < 0 {
695            return Err("path with a negative wait".into());
696        }
697        wait += w;
698        ready = l.board_time + l.ride_time;
699    }
700    Ok(PathBreakdown {
701        access: minutes_f32(access),
702        wait: minutes_f32(wait),
703        ride: minutes_f32(path.iter().map(|l| l.ride_time).sum()),
704        transfer: minutes_f32(path.iter().map(|l| l.transfer_time).sum()),
705        egress: minutes_f32(egress),
706        total: minutes_f32(total),
707        routes,
708        n_rides: path.len() as u32,
709    })
710}

712fn validate(layer: &TransitLayer, n_destinations: usize, p: &TransitParams) -> Result<(), String> {
713    if n_destinations > MAX_PATH_DESTINATIONS {
714        return Err(format!(
715            "at most {MAX_PATH_DESTINATIONS} destinations are supported with paths"
716        ));
717    }
718    // the bounds shared with the plain matrix (they also keep the walk
719    // speed a non-zero divisor); percentiles are not used here
720    p.check()?;
721    let per_minute = if layer.has_frequencies {
722        p.draws_per_minute as u64
723    } else {
724        1
725    };
726    let cells = p.time_window as u64 * per_minute * (layer.stop_ids.len() as u64).max(1);
727    if cells > MAX_ITERATION_STOPS {
728        return Err("time_window x draws_per_minute is too large for this network".into());
729    }
730    Ok(())
731}

733/// Run `f` for every origin on `n_threads` threads, results in origin
734/// order; `Cancelled` when `cancel` is, and then the workers stop before
735/// their next origin.
736fn per_origin<T: Send>(
737    origins: &[(f64, f64)],
738    n_threads: usize,
739    cancel: &CancelToken,
740    f: impl Fn(f64, f64, &mut OriginScratch) -> Result<T, String> + Sync,
741) -> Result<Vec<T>, RouteError> {
742    let n_threads = worker_threads(n_threads, origins.len());
743    let next = AtomicUsize::new(0);
744    let done: Vec<Vec<(usize, Result<T, String>)>> = crate::sys::scoped(0..n_threads, |_| {
745        let mut done = Vec::new();
746        let mut scratch = OriginScratch::default();
747        while !cancel.is_cancelled() {
748            let i = next.fetch_add(1, AtomicOrdering::Relaxed);
749            if i >= origins.len() {
750                break;
751            }
752            let (lat, lon) = origins[i];
753            done.push((i, f(lat, lon, &mut scratch)));
754        }
755        done
756    });
757    cancel.check()?;
758    let mut out: Vec<Option<T>> = (0..origins.len()).map(|_| None).collect();
759    for (i, r) in done.into_iter().flatten() {
760        out[i] = Some(r.map_err(RouteError::Invalid)?);
761    }
762    Ok(out
763        .into_iter()
764        .map(|r| r.expect("every origin routed"))
765        .collect())
766}

768/// The work of [`crate::expanded_travel_time_matrix`].
769pub(crate) fn expanded_travel_time_matrix(
770    net: &StreetNetwork,
771    layer: &TransitLayer,
772    data: &TransitData,
773    origins: &[(f64, f64)],
774    destinations: &[(f64, f64)],
775    p: &TransitParams,
776) -> Result<ExpandedMatrix, RouteError> {
777    let ctx = Context::new(net, layer, data, destinations, p)?;
778    let dpm = ctx.draws_per_minute;
779    let m = p.max_trip_duration;
780    per_origin(origins, p.n_threads, &p.cancel, |lat, lon, scratch| {
781        let mut rows = Vec::new();
782        ctx.origin_paths(lat, lon, scratch, |j, dp| {
783            if dp.is_empty() {
784                return Ok(());
785            }
786            let walk_first = dp.walk().first().cloned();
787            for k in 0..p.time_window as usize {
788                let departure = p.departure_time + 60 * k as i32;
789                let mut draw = 0;
790                for pb in dp.minute(k)? {
791                    draw += 1;
792                    rows.push(ExpandedRow {
793                        to: j as u32,
794                        departure,
795                        draw,
796                        trip: pb.report(m),
797                    });
798                }
799                if (draw as usize) < dpm {
800                    let filler = walk_first
801                        .clone()
802                        .unwrap_or_else(|| PathBreakdown::unreachable(&ctx.walk_routes));
803                    let trip = filler.report(m);
804                    for d in draw as usize + 1..=dpm {
805                        rows.push(ExpandedRow {
806                            to: j as u32,
807                            departure,
808                            draw: d as u32,
809                            trip: trip.clone(),
810                        });
811                    }
812                }
813            }
814            Ok(())
815        })?;
816        Ok(rows)
817    })
818    .map(ExpandedMatrix)
819}

821/// The service date and departure of an arrival search: the departure is
822/// `max_trip_duration` before the arrival time, and the search covers the
823/// window `[departure, arrival)` minute by minute.
824/// Returns `(yyyymmdd, seconds after midnight)`; an earlier day when the
825/// window starts before midnight. Daylight-saving shifts are not modelled.
826/// An error when `date` is not a calendar date (see
827/// [`validate_date`]), `max_trip_duration` is under 1 minute or the
828/// departure falls before year 1.
829pub fn arrival_departure(
830    date: u32,
831    arrival_time: i32,
832    max_trip_duration: i32,
833) -> Result<(u32, i32), RouteError> {
834    validate_date(date)?;
835    if max_trip_duration < 1 {
836        return Err(RouteError::Invalid(format!(
837            "max_trip_duration must be at least 1, got {max_trip_duration}"
838        )));
839    }
840    let t = i64::from(arrival_time) - i64::from(max_trip_duration) * 60;
841    if t >= 0 {
842        return Ok((date, t as i32));
843    }
844    let days = t.div_euclid(86_400);
845    let date = civil_from_days(days_from_civil(date) + days)
846        .ok_or_else(|| format!("max_trip_duration {max_trip_duration} min reaches before year 1"))
847        .map_err(RouteError::Invalid)?;
848    Ok((date, t.rem_euclid(86_400) as i32))
849}

851/// The work of [`crate::arrival_travel_time_matrix`].
852pub(crate) fn arrival_travel_time_matrix(
853    net: &StreetNetwork,
854    layer: &TransitLayer,
855    data: &TransitData,
856    origins: &[(f64, f64)],
857    destinations: &[(f64, f64)],
858    p: &TransitParams,
859) -> Result<ExpandedMatrix, RouteError> {
860    if p.time_window != p.max_trip_duration {
861        return Err(RouteError::Invalid(
862            "an arrival search needs time_window = max_trip_duration".into(),
863        ));
864    }
865    let ctx = Context::new(net, layer, data, destinations, p)?;
866    let dpm = ctx.draws_per_minute;
867    let m = p.max_trip_duration;
868    let arrival = p.departure_time + m * 60;
869    // Arrival of a record leaving at `dep`: `dep + Math.round(totalTime * 60)`
870    // (in i64: a long walk-only total plus a late departure can pass i32).
871    let arrival = i64::from(arrival);
872    let arrives =
873        |dep: i32, pb: &PathBreakdown| i64::from(dep) + java_round(pb.total * 60.0) as i64;
874    per_origin(origins, p.n_threads, &p.cancel, |lat, lon, scratch| {
875        let mut rows = Vec::new();
876        ctx.origin_paths(lat, lon, scratch, |j, dp| {
877            if dp.is_empty() {
878                return Ok(());
879            }
880            for k in (0..p.time_window as usize).rev() {
881                let departure = p.departure_time + 60 * k as i32;
882                let mut count = 0;
883                for pb in dp.minute(k)? {
884                    count += 1;
885                    if arrives(departure, pb) <= arrival {
886                        rows.push(ExpandedRow {
887                            to: j as u32,
888                            departure,
889                            draw: count,
890                            trip: pb.report(m),
891                        });
892                        return Ok(());
893                    }
894                }
895                if (count as usize) < dpm {
896                    if let Some(pb) = dp.walk().first() {
897                        if arrives(departure, pb) <= arrival {
898                            rows.push(ExpandedRow {
899                                to: j as u32,
900                                departure,
901                                draw: 1,
902                                trip: pb.report(m),
903                            });
904                            return Ok(());
905                        }
906                    }
907                }
908            }
909            Ok(())
910        })?;
911        Ok(rows)
912    })
913    .map(ExpandedMatrix)
914}
