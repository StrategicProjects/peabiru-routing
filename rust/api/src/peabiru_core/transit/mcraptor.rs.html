1//! R5 7.6's multi-criteria router, `McRaptorSuboptimalPathProfileRouter`,
2//! ported literally (tie-breaking and iteration orders included), with the
3//! dominating lists used with it: R5's `FareDominatingList` (time and
4//! fare, with the rule-based fare calculator) and the ported `TripPlanner`
5//! wrapper of `SuboptimalDominatingList` (time, with `suboptimal_minutes`).
6//!
7//! It is not range RAPTOR: it samples `monte_carlo_draws` departure times
8//! in the window by a constrained random walk (R5's
9//! `generateDepartureTimesToSample`, with the generator seeded as below),
10//! and runs a full search from each. Every state is kept as an object with
11//! a back pointer, as R5 does; the states of a draw live in an arena that
12//! the next draw reuses.
13//!
14//! What differs from R5 is only how the same decisions are computed (each
15//! difference is argued where it is made):
16//!
17//! - the dominating lists hold, beside each state, the values the
18//!   comparisons read ([`Entry`]), and a removal compacts the list in place;
19//! - a state's fare comes from the fare variables of the state behind it
20//!   ([`FareCalculator::ride`]) instead of from all its rides, and is
21//!   worked out once per boarding rather than once per stop alighted at;
22//! - a state that no list accepts is not kept (R5 leaves it to the garbage
23//!   collector);
24//! - `bestStatesBeforeRound` keeps only the states a round can board with.
25

173/// `DominatingList.add` of R5's two lists, which differ in `dominates`
174/// alone: `new` is refused, and the list left as it is at that point, at
175/// the first state of the list that dominates it; the states before that
176/// one which `new` dominates are removed; `new` goes to the end when no
177/// state refuses it.
178///
179/// R5 (and the first version of this port) removes each state at once,
180/// shifting the rest of the list down, then goes on from the same index.
181/// Here the states that stay are moved down as the scan goes (`w` is where
182/// the next one goes, `r` the next to examine), and a refusal moves the
183/// states not yet examined down as well. Both examine every state once, in
184/// the list's order, with the same two tests on the same values, and leave
185/// the survivors in their order followed by the unexamined ones: the same
186/// list, and the same answer.
187#[inline(always)]
188fn list_add(list: &mut Vec<Entry>, new: Entry, dominates: impl Fn(&Entry, &Entry) -> bool) -> bool {
189    let n = list.len();
190    let mut w = 0;
191    for r in 0..n {
192        let old = list[r];
193        if dominates(&old, &new) {
194            if w != r {
195                list.copy_within(r..n, w);
196                list.truncate(n - (r - w));
197            }
198            return false;
199        }
200        if !dominates(&new, &old) {
201            if w != r {
202                list[w] = old;
203            }
204            w += 1;
205        }
206    }
207    list.truncate(w);
208    list.push(new);
209    true
210}

212/// `add` of the `TripPlanner` wrapper of `SuboptimalDominatingList`: a
213/// state more than `max_duration` after the draw's `departure` is refused;
214/// one that the list accepts shrinks `max_duration` to its own duration
215/// when `shrink` (the shortest path is asked for and the state is at
216/// the destination).
217fn suboptimal_add(
218    list: &mut Vec<Entry>,
219    new: Entry,
220    suboptimal_minutes: i32,
221    departure: i32,
222    max_duration: &mut i32,
223    shrink: bool,
224) -> bool {
225    if new.time.wrapping_sub(departure) > *max_duration {
226        return false;
227    }
228    // SuboptimalDominatingList.dominates(a, b): one access mode
229    let threshold = suboptimal_minutes.wrapping_mul(60);
230    let dominates = |a: &Entry, b: &Entry| {
231        (a.round < b.round && a.time <= b.time) || a.time.wrapping_add(threshold) < b.time
232    };
233    if !list_add(list, new, dominates) {
234        return false;
235    }
236    if shrink {
237        *max_duration = (*max_duration).min(new.time.wrapping_sub(departure));
238    }
239    true
240}

242/// [`fare_better_or_equal`] of two states of the same time, from their
243/// fares.
244#[inline(always)]
245fn fare_no_worse(a: &FareBounds, b: &FareBounds) -> bool {
246    a.paid <= b.paid.wrapping_sub(b.value)
247        || (a.paid <= b.paid
248            && a.value >= b.value
249            && a.expiration >= b.expiration
250            && a.number >= b.number)
251}

253/// `FareDominatingList.betterOrEqual(a, b)`.
254#[inline(always)]
255fn fare_better_or_equal(a: &Entry, b: &Entry) -> bool {
256    a.time <= b.time
257        && (a.paid <= b.consumed
258            || (a.paid <= b.paid
259                && a.value >= b.value
260                && a.expiration >= b.expiration
261                && a.number >= b.number))
262}

264/// A bit set over stops or patterns.
265#[derive(Clone)]
266struct Bits(Vec<u64>);
267
268impl Bits {
269    fn new(n: usize) -> Self {
270        Bits(vec![0; n.div_ceil(64)])
271    }
272    fn set(&mut self, i: usize) {
273        self.0[i / 64] |= 1 << (i % 64);
274    }
275    fn clear(&mut self) {
276        self.0.iter_mut().for_each(|w| *w = 0);
277    }
278    fn is_empty(&self) -> bool {
279        self.0.iter().all(|&w| w == 0)
280    }
281    fn and(&mut self, o: &Bits) {
282        for (a, b) in self.0.iter_mut().zip(&o.0) {
283            *a &= b;
284        }
285    }
286    fn or(&mut self, o: &Bits) {
287        for (a, b) in self.0.iter_mut().zip(&o.0) {
288            *a |= b;
289        }
290    }
291    fn ones(&self) -> Vec<usize> {
292        let mut v = Vec::new();
293        for (wi, &w) in self.0.iter().enumerate() {
294            let mut w = w;
295            while w != 0 {
296                v.push(wi * 64 + w.trailing_zeros() as usize);
297                w &= w - 1;
298            }
299        }
300        v
301    }
302}

387/// The times of a pattern's trips, laid out for the search.
388struct Timetable {
389    /// The trips running on the date, as indices into `pattern.trips`, in
390    /// its order.
391    trips: Vec<u32>,
392    /// `departures[pos * trips.len() + k]`: when the `k`-th running trip
393    /// leaves the pattern's stop at `pos`; `arrivals` likewise.
394    departures: Vec<i32>,
395    arrivals: Vec<i32>,
396    /// Per running trip: the first position from which its arrivals never
397    /// decrease to the end of the pattern (0 for a trip whose times are in
398    /// order).
399    settled: Vec<u32>,
400    /// Per position: the pattern does not come back to this stop later.
401    last_visit: Vec<bool>,
402    /// Per position: the pattern has been at this stop before.
403    revisit: Vec<bool>,
404}

406/// The timetables of the patterns a search may ride that run by schedule
407/// only: the same numbers as `pattern.trips[..].departures` and
408/// `.arrivals`, copied stop by stop so that the trips at one stop are side
409/// by side (R5, and the search without a timetable, follow a pointer per
410/// trip). Made once per request and shared by its searches.
411pub(crate) struct Schedules(Vec<Option<Timetable>>);
412
413impl Schedules {
414    /// `allowed` and `active` as [`McRaptor::new`] takes them.
415    pub(crate) fn new(layer: &TransitLayer, allowed: &[bool], active: &[bool]) -> Self {
416        Schedules(
417            layer
418                .patterns
419                .iter()
420                .enumerate()
421                .map(|(pi, p)| {
422                    let n = p.stops.len();
423                    // only where the search reads a time of every trip at
424                    // every stop: a pattern it may ride, by schedule alone
425                    // (so that no trip is a frequency trip), each trip with
426                    // a time per stop
427                    if !allowed[pi]
428                        || !p.has_schedules
429                        || p.has_frequencies
430                        || p.trips.iter().any(|t| {
431                            t.is_frequency() || t.departures.len() != n || t.arrivals.len() != n
432                        })
433                    {
434                        return None;
435                    }
436                    let trips: Vec<u32> = (0..p.trips.len() as u32)
437                        .filter(|&t| active[p.trips[t as usize].service as usize])
438                        .collect();
439                    let mut departures = Vec::with_capacity(n * trips.len());
440                    let mut arrivals = Vec::with_capacity(n * trips.len());
441                    for pos in 0..n {
442                        for &t in &trips {
443                            departures.push(p.trips[t as usize].departures[pos]);
444                            arrivals.push(p.trips[t as usize].arrivals[pos]);
445                        }
446                    }
447                    let settled = trips
448                        .iter()
449                        .map(|&t| {
450                            let a = &p.trips[t as usize].arrivals;
451                            let mut from = n.saturating_sub(1);
452                            while from > 0 && a[from - 1] <= a[from] {
453                                from -= 1;
454                            }
455                            from as u32
456                        })
457                        .collect();
458                    let last_visit = (0..n)
459                        .map(|pos| !p.stops[pos + 1..].contains(&p.stops[pos]))
460                        .collect();
461                    let revisit = (0..n)
462                        .map(|pos| p.stops[..pos].contains(&p.stops[pos]))
463                        .collect();
464                    Some(Timetable {
465                        trips,
466                        departures,
467                        arrivals,
468                        settled,
469                        last_visit,
470                        revisit,
471                    })
472                })
473                .collect(),
474        )
475    }
476}

503            Some(seed) => MersenneTwister::from_long(i64::from(objects_hash(&[
504                seed,

525        let exact_fares = match dom {

527            Dominance::Suboptimal { .. } => false,
528        };
529        McRaptor {
530            layer,
531            transfers,
532            allowed,
533            active,
534            schedules,
535            req,
536            dom,
537            egress,
538            egress_time,

540            offsets: Offsets::new(layer, offsets_mt),

565    /// The seed of the departure-time sample
566    /// (`R5TravelTimeComputer.seedOffsets`): `Objects.hash(fromLat,
567    /// fromLon)` without a seed, `Objects.hash(seed, fromLat, fromLon, 1)`
568    /// with one.
569    pub(crate) fn departure_seed(seed: Option<i32>, (lat, lon): (f64, f64)) -> i64 {
570        let h = match seed {
571            None => objects_hash(&[double_hash(lat), double_hash(lon)]),
572            Some(s) => objects_hash(&[s, double_hash(lat), double_hash(lon), 1]),
573        };
574        i64::from(h)

577    /// `generateDepartureTimesToSample`, repeated until it gives exactly
578    /// `monte_carlo_draws` times, as R5 does (an error rather than R5's
579    /// endless loop when that cannot happen).
580    fn departure_times(&mut self) -> Result<Vec<i32>, String> {
581        let r = self.req;
582        if r.monte_carlo_draws == 200 {
583            return Err(
584                "the multi-criteria router does not accept exactly 200 Monte Carlo draws; \
585                        use a slightly different time window or number of draws per minute"
586                    .into(),
587            );
588        }
589        if r.monte_carlo_draws <= 0 {
590            return Err("the multi-criteria router needs at least one Monte Carlo draw".into());
591        }
592        let mean = (r.to_time - r.from_time) / r.monte_carlo_draws;
593        if mean < 2 {
594            return Err(format!(
595                "the multi-criteria router cannot sample {} departure times in a window of {} s; \
596                 use fewer draws per minute",
597                r.monte_carlo_draws,
598                r.to_time - r.from_time
599            ));
600        }
601        let half = mean / 2;
602        for _ in 0..1_000_000 {
603            let mut v = Vec::new();
604            let mut t = r.from_time + self.mt.next_int(mean);
605            while t < r.to_time {
606                v.push(t);
607                t += self.mt.next_int(mean) + half;
608            }
609            if v.len() == r.monte_carlo_draws as usize {
610                return Ok(v);
611            }
612        }
613        Err(format!(
614            "the multi-criteria router cannot sample exactly {} departure times in this \
615             window; use fewer draws per minute",
616            r.monte_carlo_draws
617        ))
618    }

620    /// `route()`: every draw in turn, handing `sink(departure, out)` the
621    /// draw's result. Errors are R5's exceptions.
622    pub(crate) fn route(
623        &mut self,
624        access: &[(u32, i32)],
625        cancel: &CancelToken,
626        sink: impl FnMut(i32, DrawOut<'_>),
627    ) -> Result<(), String> {
628        let deps = self.departure_times()?;
629        self.route_departures(&deps, access, cancel, sink)
630    }

632    /// [`McRaptor::route`]'s draws from the departure times `deps` (for
633    /// comparing with it at chosen times).
634    pub(crate) fn route_departures(
635        &mut self,
636        deps: &[i32],
637        access: &[(u32, i32)],
638        cancel: &CancelToken,
639        mut sink: impl FnMut(i32, DrawOut<'_>),
640    ) -> Result<(), String> {
641        let n_stops = self.layer.stop_ids.len();
642        let mut times = vec![UNREACHED; n_stops];
643        let mut final_states = Vec::new();
644        for &dep in deps {
645            if cancel.is_cancelled() {
646                return Ok(());
647            }
648            self.departure = dep;
649            self.offsets.randomize(self.layer);
650            for &s in &self.with_bag {
651                let bag = &mut self.bags[s as usize];
652                bag.best.clear();
653                bag.nt.clear();
654                self.has_bag[s as usize] = false;
655                self.before_range[s as usize] = (0, 0);
656            }
657            self.with_bag.clear();
658            self.arena.clear();
659            self.progress.clear();
660            self.touched_patterns.clear();
661            self.touched_stops.clear();
662            // TripPlanner.AccessTimes: target pruning per draw
663            self.best_at_target = i32::MAX;
664            self.round = 0;
665            for &(stop, t) in access {
666                let from = FareFrom::Access;
667                if self.add_state(stop, -1, -1, dep.wrapping_add(t), -1, -1, -1, NONE, 0, from)? {
668                    self.touched_stops.set(stop as usize);
669                }
670            }
671            self.mark_patterns();
672            self.round += 1;
673            while self.do_one_round()? && self.round < self.req.max_rides as i32 + 1 {}
674            if self.egress.is_some() {
675                self.propagate_to_destination(&mut final_states);
676                sink(dep, DrawOut::Final(&self.arena, &final_states));
677            } else {
678                self.collate(&mut times);
679                sink(dep, DrawOut::Times(&times));
680            }
681        }
682        Ok(())
683    }

695    /// The entry of a new state for the fare list, with its fare, or `None`
696    /// when `FareDominatingList.add` refuses it before looking at the list:
697    /// too late for the draw, or over `max_fare`. R5 makes these two tests
698    /// in each list the state is offered to (`best`, then `nonTransfer`);
699    /// they read only the state and the request, so both lists answer alike
700    /// and neither changes.
701    ///
702    /// The fare is `calculateFare` of the state's rides. R5 hands the
703    /// calculator the whole chain of the state; the fare is a function of
704    /// the rides alone (their patterns and board times, in order), so:
705    ///
706    /// - a state with no state behind it has no rides: `fare(&[])`;
707    /// - a state that is not off a vehicle has the rides of the state
708    ///   behind it, hence its fare (which the caller reads from that
709    ///   state's list entry, a copy of it);
710    /// - a state off a vehicle has the rides of the state behind it and one
711    ///   more, `(pattern, board_time)`: [`FareCalculator::ride`] from the
712    ///   variables kept for that state, and [`FareCalculator::bounds`].
713    ///   The states alighting along a pattern from the same boarding have
714    ///   the same state behind, pattern and board time, hence the same
715    ///   fare: it is worked out for the first that needs it and kept in
716    ///   the boarding's memo (emptied whenever the boarding changes).
717    ///
718    /// Every state behind another was accepted by a list, so its variables
719    /// are in `progress` (`add_state` keeps only accepted states, each
720    /// with its variables).
721    #[allow(clippy::too_many_arguments)]
722    #[inline(always)]
723    fn fare_entry(
724        &self,
725        calc: &FareCalculator,
726        id: u32,
727        time: i32,
728        round: i32,
729        pattern: i32,
730        board_time: i32,
731        back: u32,
732        from: &mut FareFrom<'_>,
733    ) -> Option<(Entry, FareBounds)> {
734        let max_clock = self.departure.wrapping_add(self.req.max_trip_s);
735        if time > max_clock {
736            return None;
737        }
738        let f = match from {
739            FareFrom::Access => calc.bounds(&FareProgress::default()),
740            FareFrom::Walk(f) => *f,
741            FareFrom::Ride(memo) => match memo {
742                Some((f, _)) => *f,
743                None => {
744                    let p = calc.ride(&self.progress[back as usize], pattern, board_time);
745                    let f = calc.bounds(&p);
746                    **memo = Some((f, p));
747                    f
748                }
749            },
750        };
751        let consumed = f.paid.wrapping_sub(f.value);
752        if consumed > self.req.max_fare {
753            return None;
754        }
755        let e = Entry {
756            id,
757            time,
758            round,
759            paid: f.paid,
760            consumed,
761            value: f.value,
762            expiration: f.expiration,
763            number: f.number,
764        };
765        Some((e, f))
766    }

768    /// The fare variables of a state whose entry [`McRaptor::fare_entry`]
769    /// made from `from`.
770    fn progress_from(&self, back: u32, from: &FareFrom<'_>) -> FareProgress {
771        match from {
772            FareFrom::Access => FareProgress::default(),
773            FareFrom::Walk(_) => self.progress[back as usize],
774            FareFrom::Ride(memo) => memo.expect("fare_entry").1,
775        }
776    }

778    /// `addState`. `back_time` is the time of the state `back` (not read
779    /// without one); `from` gives the state's fare with the fare list.
780    ///
781    /// R5 makes the state, then offers it to the lists. Here the lists are
782    /// offered its entry first and the state is kept (at the index the
783    /// entry already names, the next of the arena) only when one accepts
784    /// it: a refused state is in no list, nothing points back to it and no
785    /// result reads it.
786    #[allow(clippy::too_many_arguments)]
787    fn add_state(
788        &mut self,
789        stop: u32,
790        board_pos: i32,
791        alight_pos: i32,
792        time: i32,
793        board_time: i32,
794        pattern: i32,
795        trip: i32,
796        back: u32,
797        back_time: i32,
798        mut from: FareFrom<'_>,
799    ) -> Result<bool, String> {
800        let r = self.req;
801        if time > r.to_time.wrapping_add(r.max_trip_s) {
802            return Ok(false);
803        }
804        if r.max_fare < 0
805            && time.wrapping_sub(r.suboptimal_minutes.wrapping_mul(60)) > self.best_at_target
806        {
807            return Ok(false);
808        }
809        if back != NONE && back_time > time {
810            return Err(
811                "internal error in the multi-criteria router: a state is earlier than the one it \
812                        comes from"
813                    .into(),
814            );
815        }
816        let id = self.arena.len() as u32;
817        let si = stop as usize;
818        if !self.has_bag[si] {
819            self.has_bag[si] = true;
820            self.with_bag.push(stop);
821        }
822        let round = self.round;
823        let mut fare = None;
824        let mut progress = FareProgress::default();
825        let optimal = match self.dom {
826            Dominance::Fare(calc) => {
827                match self.fare_entry(calc, id, time, round, pattern, board_time, back, &mut from) {
828                    None => false,
829                    Some((e, f)) => {
830                        let bag = &mut self.bags[si];
831                        let mut optimal = list_add(&mut bag.best, e, fare_better_or_equal);
832                        if pattern != -1 && list_add(&mut bag.nt, e, fare_better_or_equal) {
833                            optimal = true;
834                        }
835                        if optimal {
836                            fare = Some(f);
837                            progress = self.progress_from(back, &from);
838                        }
839                        optimal
840                    }
841                }
842            }
843            Dominance::Suboptimal { .. } => {
844                let e = Entry {
845                    id,
846                    time,
847                    round,
848                    paid: 0,
849                    consumed: 0,
850                    value: 0,
851                    expiration: 0,
852                    number: 0,
853                };
854                // never at the destination (which has no stop): the limit
855                // does not shrink here
856                let (sub, dep) = (r.suboptimal_minutes, self.departure);
857                let (bag, limit) = (&mut self.bags[si], &mut self.max_duration);
858                let mut optimal = suboptimal_add(&mut bag.best, e, sub, dep, limit, false);
859                if pattern != -1 && suboptimal_add(&mut bag.nt, e, sub, dep, limit, false) {
860                    optimal = true;
861                }
862                optimal
863            }
864        };
865        if optimal {
866            self.arena.push(McState {
867                back,
868                time,
869                board_time,
870                pattern,
871                trip,
872                round,
873                stop: stop as i32,
874                board_pos,
875                alight_pos,
876                fare,
877            });
878            self.progress.push(progress);
879        }
880        if self.egress.is_some() && optimal && pattern != -1 {
881            let e = self.egress_time[si];
882            if e != -1 {
883                let at = time.wrapping_add(e);
884                if at < self.best_at_target {
885                    self.best_at_target = at;
886                }
887            }
888        }
889        Ok(optimal)
890    }

892    /// `doOneRound`: `true` when a pattern is to be explored next round.
893    fn do_one_round(&mut self) -> Result<bool, String> {
894        // bestStatesBeforeRound: a copy of every bag's best states. R5
895        // copies each list whole and, when boarding, skips the states
896        // whose round is not the one before this (before anything else is
897        // done with them); the copy here leaves those out, and takes the
898        // time, the one other value boarding reads of a state. A stop
899        // whose bag is made during the round has no copy in R5 and an
900        // empty one here (`before_range` is emptied for every stop at the
901        // start of the draw and written only for the stops with a bag):
902        // nothing boards there either way.
903        let mut before = std::mem::take(&mut self.before);
904        before.clear();
905        let previous = self.round - 1;
906        for &s in &self.with_bag {
907            let start = before.len() as u32;
908            for e in &self.bags[s as usize].best {
909                if e.round == previous {
910                    before.push((e.id, e.time));
911                }
912            }
913            self.before_range[s as usize] = (start, before.len() as u32);
914        }
915        let mut states = std::mem::take(&mut self.on_board);
916        let done = self.ride_patterns(&before, &mut states);
917        self.before = before;
918        self.on_board = states;
919        done?;
920        self.do_transfers()?;
921        self.mark_patterns();
922        self.round += 1;
923        Ok(!self.touched_patterns.is_empty())
924    }

926    /// The patterns of `doOneRound`.
927    fn ride_patterns(
928        &mut self,
929        before: &[(u32, i32)],
930        states: &mut Vec<u32>,
931    ) -> Result<(), String> {
932        // on the last round of a point-to-point search, only the patterns
933        // near the destination
934        if self.round == self.req.max_rides as i32 && self.egress.is_some() {
935            self.touched_patterns.and(&self.near_destination);
936        }
937        let layer = self.layer;
938        for pat in self.touched_patterns.ones() {
939            if !self.allowed[pat] {
940                continue;
941            }
942            let pattern = &layer.patterns[pat];
943            states.clear();
944            if let Some(table) = &self.schedules.0[pat] {
945                let mut riders = std::mem::take(&mut self.riders);
946                let done = self.ride_timetable(pat, table, before, &mut riders);
947                self.riders = riders;
948                done?;
949                continue;
950            }
951            for (pos, &stop) in pattern.stops.iter().enumerate() {
952                // get off the bus
953                for &s in states.iter() {
954                    let Slot {
955                        trip,
956                        board_pos,
957                        freq_time,
958                        ..
959                    } = self.slot[s as usize];
960                    let sched = &pattern.trips[trip as usize];
961                    let (arrival, board_time) = if sched.is_frequency() {
962                        let travel =
963                            sched.arrivals[pos].wrapping_sub(sched.departures[board_pos as usize]);
964                        (freq_time.wrapping_add(travel), freq_time)
965                    } else {
966                        (sched.arrivals[pos], sched.departures[board_pos as usize])
967                    };
968                    // the memo of this boarding: `put` emptied it, and the
969                    // pattern and the board time are the same until the
970                    // next `put` for the state (the board time comes from
971                    // the slot)
972                    let mut memo = self.slot[s as usize].memo;
973                    let back_time = self.arena[s as usize].time;
974                    let from = FareFrom::Ride(&mut memo);
975                    let added = self.add_state(
976                        stop, board_pos, pos as i32, arrival, board_time, pat as i32, trip, s,
977                        back_time, from,
978                    );
979                    self.slot[s as usize].memo = memo;
980                    if added? {
981                        self.touched_stops.set(stop as usize);
982                    }
983                }
984                // get on the bus
985                let (from, to) = self.before_range[stop as usize];
986                for &(s, s_time) in &before[from as usize..to as usize] {
987                    if pattern.has_frequencies && pattern.has_schedules {
988                        return Err(
989                            "the multi-criteria router does not support frequencies and \
990                                    schedules in the same trip pattern"
991                                .into(),
992                        );
993                    }
994                    if pattern.has_schedules {
995                        for (ti, t) in pattern.trips.iter().enumerate() {
996                            if !self.active[t.service as usize] {
997                                continue;
998                            }
999                            if t.departures[pos] > s_time.wrapping_add(BOARD_SLACK) {
1000                                states.push(s);
1001                                self.put(s, (ti as i32, pos as i32, 0));
1002                                break;
1003                            }
1004                        }
1005                    } else if pattern.has_frequencies {
1006                        for (ti, t) in pattern.trips.iter().enumerate() {
1007                            if !self.active[t.service as usize] {
1008                                continue;
1009                            }
1010                            let earliest = s_time.wrapping_add(BOARD_SLACK);
1011                            for (e, f) in t.frequencies.iter().enumerate() {
1012                                let mut departure = f
1013                                    .start
1014                                    .wrapping_add(self.offsets.offsets[pat][ti][e])
1015                                    .wrapping_add(t.departures[pos]);
1016                                let latest = f.end.wrapping_add(t.departures[pos]);
1017                                if earliest > latest {
1018                                    continue;
1019                                }
1020                                if f.headway <= 0 {
1021                                    return Err(format!(
1022                                        "frequency of trip {} has headway {}",
1023                                        t.trip_id, f.headway
1024                                    ));
1025                                }
1026                                while departure < earliest {
1027                                    departure = departure.wrapping_add(f.headway);
1028                                }
1029                                if departure > latest {
1030                                    continue;
1031                                }
1032                                states.push(s);
1033                                self.put(s, (ti as i32, pos as i32, departure));
1034                            }
1035                        }
1036                    }
1037                }
1038            }
1039        }
1040        Ok(())
1041    }

1043    /// One pattern of [`McRaptor::ride_patterns`] from its timetable: what
1044    /// the loop there does for a pattern that runs by schedule only (the
1045    /// frequency branches and the mixed-pattern error cannot be reached
1046    /// for it), reading the same times from the timetable.
1047    ///
1048    /// - Getting off: every trip is scheduled, so the state arrives at
1049    ///   `trip.arrivals[pos]` having boarded at
1050    ///   `trip.departures[board_pos]`; these are `arrivals[pos * n + k]`
1051    ///   and `departures[board_pos * n + k]` for the `k`-th running trip.
1052    /// - Getting on: R5 takes the first trip of the pattern, in its order,
1053    ///   that runs on the date and leaves the stop after the state's time
1054    ///   plus the slack. The timetable has the running trips in that
1055    ///   order: the first `k` with `departures[pos * n + k]` after it.
1056    ///
1057    /// R5 keeps the trip and the board position of the states on board in
1058    /// maps keyed by the state, where a later boarding of the same state
1059    /// replaces an earlier one, for every copy of the state on board. A
1060    /// state boards only at its own stop (the candidates at a position are
1061    /// the best states of that position's stop, each once), so that
1062    /// happens only where the pattern comes back to a stop. Here each
1063    /// state on board carries its own trip and board position
1064    /// ([`Rider`]), and a boarding at such a position rewrites those of
1065    /// the state's earlier copies, with their fare memo emptied.
1066    ///
1067    /// A state on board is left behind once it can only be refused for
1068    /// the rest of the pattern. R5 keeps it on board to the last stop and
1069    /// offers a new state at every stop. The state on board `s` is dropped
1070    /// at `pos` when all of these hold:
1071    ///
1072    /// 1. it boarded at the pattern's last visit to its stop, so the state
1073    ///    does not board again further on and its trip `k` (and the fare
1074    ///    of what it offers, its memo) is the same at every later
1075    ///    position. (A state that boarded at an earlier visit of a stop
1076    ///    the pattern comes back to stays on board whatever its time: a
1077    ///    second boarding there would change its trip.)
1078    /// 2. `pos >= settled[k]`: the arrivals of trip `k` do not decrease
1079    ///    from `pos` to the end of the pattern, so the state offered at
1080    ///    `pos` and at every later position `q` has a time
1081    ///    `arrivals[q * n + k] >= arrivals[pos * n + k]`.
1082    /// 3. one of:
1083    ///    - (a) its arrival at `pos` is later than `limit` (the end of the
1084    ///      window plus the longest trip, the same for the whole search):
1085    ///      `add_state` refuses such a state in its first statement;
1086    ///    - (b) with the fare list, its arrival at `pos` is not before the
1087    ///      time of `s` and either is later than the draw's `max_clock`
1088    ///      or the fare of what it offers, already worked out, is over
1089    ///      `max_fare`. `add_state` then, for the state offered at `pos`
1090    ///      or later: may refuse it in its first two tests (too late, or
1091    ///      target pruning); does not fail the test on going back in time;
1092    ///      makes the stop's bag if it has none; and gets no entry from
1093    ///      [`McRaptor::fare_entry`], which refuses on these same two
1094    ///      grounds (the time, or the same fare against the same limit)
1095    ///      before any list is looked at;
1096    ///    - (c) with the fare list and exact fares, its arrival at `pos` is
1097    ///      not before the time of `s` and it is shadowed: a state `a`
1098    ///      boarded the same trip `k` before it (it comes earlier in the
1099    ///      order of the states on board), at the last visit to its own
1100    ///      stop, and the fare `a` offers is no worse than the fare this
1101    ///      one offers ([`fare_no_worse`]). At `pos` and at every later
1102    ///      position both offer a state of the same time (the trip's
1103    ///      arrival), `a` first. Call the two states offered at a stop
1104    ///      `A` and `B`: `betterOrEqual(A, B)`, at every such stop. Then
1105    ///      offering `B` changes nothing:
1106    ///
1107    ///      - `betterOrEqual` is transitive when fares are exact (what is
1108    ///        paid less the value of the allowance is not a wrapped
1109    ///        number, [`FareCalculator::exact_within`]) since the value of
1110    ///        an allowance is never negative: with `X >= A` and `A >= B`,
1111    ///        by the first rule both times `X.paid <= A.paid - A.value <=
1112    ///        A.paid <= B.paid - B.value`; by the first then the second,
1113    ///        `X.paid <= A.paid - A.value <= B.paid - B.value` (`A` pays no
1114    ///        more and holds no less); by the second then the first,
1115    ///        `X.paid <= A.paid <= B.paid - B.value`; by the second both
1116    ///        times, each of its comparisons is transitive. The same shows
1117    ///        `A.paid - A.value <= B.paid - B.value`.
1118    ///      - No state of a list is better than or equal to another of the
1119    ///        same list: a state enters only after a scan of the whole
1120    ///        list that removed those it is better than or equal to and
1121    ///        found none better than or equal to it.
1122    ///      - If `A` was offered to a list, the list has from then on a
1123    ///        state `X >= A`: `A` itself or the state that refused it; and
1124    ///        `X` leaves a list only in the scan of a state `C >= X`,
1125    ///        which then enters (`C >= A`) or is refused by a state
1126    ///        `F >= C >= A` that stays.
1127    ///      - So when `B` is offered to that list, with `X >= A >= B`: its
1128    ///        scan stops at `X` at the latest, refused; and it removes
1129    ///        nothing on the way, for a state `O` with `B >= O` would have
1130    ///        `X >= O`, two states of one list.
1131    ///      - `A` and `B`, both off a vehicle, are offered to the same two
1132    ///        lists of the stop. When `A` is not offered to them, `B` is
1133    ///        not either or is refused before them: `add_state`'s first
1134    ///        two tests read the time alone (and `best_at_target`, which
1135    ///        only decreases); the test on going back in time passes for
1136    ///        `B` (its arrival is not before the time of its state from
1137    ///        `pos` on); and `fare_entry` refuses `B` when it refuses `A`,
1138    ///        on the time or because `A.paid - A.value`, hence
1139    ///        `B.paid - B.value`, is over `max_fare`.
1140    ///      - When `a` has itself been left behind, by (a) or (b) its
1141    ///        offers would be refused on grounds that refuse `B` too, as
1142    ///        just shown, and by (c) it is shadowed by a state whose fare
1143    ///        is no worse than its own, hence than this one's.
1144    ///
1145    /// In every case the state offered is refused and no list changes. The
1146    /// one trace (b) and (c) can skip is a bag made for a stop that has
1147    /// none, and
1148    /// an empty bag is the same as none for all that reads the bags:
1149    /// `do_one_round`'s copy (no state to board with), `do_transfers` (only
1150    /// stops where a state was accepted), `collate` (no time for a stop
1151    /// without non-transfer states) and `propagate_to_destination` (no
1152    /// state to walk from). The states that stay keep their order, so the
1153    /// others are offered in the order R5 offers them.
1154    fn ride_timetable(
1155        &mut self,
1156        pat: usize,
1157        table: &Timetable,
1158        before: &[(u32, i32)],
1159        states: &mut Vec<Rider>,
1160    ) -> Result<(), String> {
1161        let layer = self.layer;
1162        let n = table.trips.len();
1163        // `add_state`'s first test
1164        let limit = self.req.to_time.wrapping_add(self.req.max_trip_s);
1165        // `fare_entry`'s two tests
1166        let fare_list = matches!(self.dom, Dominance::Fare(_));
1167        let max_clock = self.departure.wrapping_add(self.req.max_trip_s);
1168        let max_fare = self.req.max_fare;
1169        let over =
1170            |memo: &Memo| matches!(memo, Some((f, _)) if f.paid.wrapping_sub(f.value) > max_fare);
1171        // (c): the fares that can shadow, per trip
1172        let shadowing = match self.dom {
1173            Dominance::Fare(calc) if self.exact_fares => Some(calc),
1174            _ => None,
1175        };
1176        let mut shadow = std::mem::take(&mut self.shadow);
1177        let mut shadow_trips = std::mem::take(&mut self.shadow_trips);
1178        for k in shadow_trips.drain(..) {
1179            shadow[k as usize].clear();
1180        }
1181        if shadow.len() < n {
1182            shadow.resize(n, Vec::new());
1183        }
1184        states.clear();
1185        for (pos, &stop) in layer.patterns[pat].stops.iter().enumerate() {
1186            let at = pos * n;
1187            // get off the bus
1188            let mut kept = 0;
1189            for r in 0..states.len() {
1190                let mut rider = states[r];
1191                let k = rider.k as usize;
1192                let arrival = table.arrivals[at + k];
1193                if rider.at_last_visit
1194                    && pos as u32 >= table.settled[k]
1195                    && (arrival > limit
1196                        || (fare_list
1197                            && arrival >= rider.s_time
1198                            && (arrival > max_clock || over(&rider.memo) || rider.shadowed)))
1199                {
1200                    continue;
1201                }
1202                let added = self.add_state(
1203                    stop,
1204                    rider.board_pos as i32,
1205                    pos as i32,
1206                    arrival,
1207                    rider.board_time,
1208                    pat as i32,
1209                    table.trips[k] as i32,
1210                    rider.s,
1211                    rider.s_time,
1212                    FareFrom::Ride(&mut rider.memo),
1213                );
1214                states[kept] = rider;
1215                kept += 1;
1216                if added? {
1217                    self.touched_stops.set(stop as usize);
1218                }
1219            }
1220            states.truncate(kept);
1221            // get on the bus
1222            let (from, to) = self.before_range[stop as usize];
1223            for &(s, s_time) in &before[from as usize..to as usize] {
1224                let earliest = s_time.wrapping_add(BOARD_SLACK);
1225                if let Some(k) = table.departures[at..at + n]
1226                    .iter()
1227                    .position(|&d| d > earliest)
1228                {
1229                    // `trip.departures[board_pos]` of the generic loop
1230                    let board_time = table.departures[at + k];
1231                    if table.revisit[pos] {
1232                        for earlier in states.iter_mut().filter(|r| r.s == s) {
1233                            earlier.k = k as u32;
1234                            earlier.board_pos = pos as u32;
1235                            earlier.board_time = board_time;
1236                            earlier.memo = None;
1237                        }
1238                    }
1239                    let at_last_visit = table.last_visit[pos];
1240                    let mut memo = None;
1241                    let mut shadowed = false;
1242                    if let Some(calc) = shadowing {
1243                        // the fare `fare_entry` would work out at the
1244                        // first stop that needs it
1245                        let p = calc.ride(&self.progress[s as usize], pat as i32, board_time);
1246                        let f = calc.bounds(&p);
1247                        memo = Some((f, p));
1248                        if at_last_visit {
1249                            let seen = &mut shadow[k];
1250                            if seen.iter().any(|a| fare_no_worse(a, &f)) {
1251                                shadowed = true;
1252                            } else {
1253                                // this one shadows whatever those it is
1254                                // no worse than would (transitivity)
1255                                if seen.is_empty() {
1256                                    shadow_trips.push(k as u32);
1257                                }
1258                                seen.retain(|o| !fare_no_worse(&f, o));
1259                                seen.push(f);
1260                            }
1261                        }
1262                    }
1263                    states.push(Rider {
1264                        s,
1265                        s_time,
1266                        k: k as u32,
1267                        board_pos: pos as u32,
1268                        board_time,
1269                        at_last_visit,
1270                        shadowed,
1271                        memo,
1272                    });
1273                }
1274            }
1275        }
1276        self.shadow = shadow;
1277        self.shadow_trips = shadow_trips;
1278        Ok(())
1279    }

1304    /// `doTransfers`.
1305    fn do_transfers(&mut self) -> Result<(), String> {
1306        let mut by_transfer = std::mem::replace(&mut self.by_transfer, Bits(Vec::new()));
1307        by_transfer.clear();
1308        let mut done = Ok(());
1309        'stops: for stop in self.touched_stops.ones() {
1310            // R5 walks from a copy of the stop's non-transfer states. A
1311            // state that walks is added with pattern -1, which goes to the
1312            // `best` list of its stop only: no non-transfer list changes
1313            // during the loop, and this one can be lent instead of copied.
1314            let nt = std::mem::take(&mut self.bags[stop].nt);
1315            for e in &nt {
1316                for &(to, mm) in &self.transfers[stop] {
1317                    let walk = (f64::from(mm) / self.req.walk_mms) as i32;
1318                    let time = e.time.wrapping_add(walk);
1319                    let from = FareFrom::Walk(e.fare());
1320                    match self.add_state(to, -1, -1, time, -1, -1, -1, e.id, e.time, from) {
1321                        Ok(true) => by_transfer.set(to as usize),
1322                        Ok(false) => {}
1323                        Err(err) => {
1324                            done = Err(err);
1325                            self.bags[stop].nt = nt;
1326                            break 'stops;
1327                        }
1328                    }
1329                }
1330            }
1331            self.bags[stop].nt = nt;
1332        }
1333        self.touched_stops.or(&by_transfer);
1334        self.by_transfer = by_transfer;
1335        done
1336    }

1363    /// `doPropagationToDestination`: a state at the destination for every
1364    /// state off a vehicle at an egress stop, in a fresh bag.
1365    fn propagate_to_destination(&mut self, out: &mut Vec<u32>) {
1366        let egress = self.egress.expect("egress");
1367        let mut dest: Vec<Entry> = Vec::new();
1368        for &(stop, e) in egress {
1369            if !self.has_bag[stop as usize] {
1370                continue;
1371            }
1372            // lent, not copied: only `dest` changes in the loop
1373            let nt = std::mem::take(&mut self.bags[stop as usize].nt);
1374            for en in &nt {
1375                let s = en.id;
1376                let id = self.arena.len() as u32;
1377                let time = en.time.wrapping_add(e);
1378                // McRaptorStateBag.add of a walk (pattern -1): best only.
1379                // As in `add_state`, the state is kept when accepted.
1380                let mut fare = None;
1381                let mut progress = FareProgress::default();
1382                let accepted = match self.dom {
1383                    Dominance::Fare(calc) => {
1384                        let mut from = FareFrom::Walk(en.fare());
1385                        match self.fare_entry(calc, id, time, 0, -1, 0, s, &mut from) {
1386                            None => false,
1387                            Some((entry, f)) => {
1388                                fare = Some(f);
1389                                progress = self.progress[s as usize];
1390                                list_add(&mut dest, entry, fare_better_or_equal)
1391                            }
1392                        }
1393                    }
1394                    Dominance::Suboptimal { shortest, .. } => {
1395                        let entry = Entry {
1396                            id,
1397                            time,
1398                            round: 0,
1399                            paid: 0,
1400                            consumed: 0,
1401                            value: 0,
1402                            expiration: 0,
1403                            number: 0,
1404                        };
1405                        let (sub, dep) = (self.req.suboptimal_minutes, self.departure);
1406                        suboptimal_add(&mut dest, entry, sub, dep, &mut self.max_duration, shortest)
1407                    }
1408                };
1409                if accepted {
1410                    self.arena.push(McState {
1411                        back: s,
1412                        time,
1413                        board_time: 0,
1414                        pattern: -1,
1415                        trip: -1,
1416                        round: 0,
1417                        stop: -1,
1418                        board_pos: 0,
1419                        alight_pos: 0,
1420                        fare,
1421                    });
1422                    self.progress.push(progress);
1423                }
1424            }
1425            self.bags[stop as usize].nt = nt;
1426        }
1427        out.clear();
1428        out.extend(dest.iter().map(|e| e.id));
1429    }

1732                            r5::DrawOut::Final(arena, states) => chains.push((
1733                                dep,
1734                                states.iter().map(|&s| chain!(arena, s, r5::NONE)).collect(),
1735                            )),
1736                        })
1737                        .err();
1738                }
1739                for schedules in [&timetables, &no_timetables] {
1740                    let mut got = Found::default();
1741                    let mut router = McRaptor::new(
1742                        &net.layer,
1743                        &net.transfers,
1744                        &net.allowed,
1745                        &net.active,
1746                        schedules,
1747                        &req,
1748                        dom(),
1749                        egress.as_deref(),
1750                        7,
1751                    );
1752                    let (times, chains) = (&mut got.times, &mut got.chains);
1753                    got.error = router
1754                        .route_departures(&deps, &access, &cancel, |dep, out| match out {
1755                            DrawOut::Times(t) => times.push((dep, t.to_vec())),
1756                            DrawOut::Final(arena, states) => chains.push((
1757                                dep,
1758                                states.iter().map(|&s| chain!(arena, s, NONE)).collect(),
1759                            )),
1760                        })
1761                        .err();
1762                    if want.error.is_some() {
1763                        // the draws before the failing one, and the failure
1764                        assert_eq!(got.error, want.error, "seed {seed}");
1765                    }
1766                    assert_eq!(got, want, "seed {seed}");
1767                }
1768                searches += 1;
1769                errors += u64::from(want.error.is_some());
1770                with_states += want
1771                    .chains
1772                    .iter()
1773                    .map(|(_, c)| c.iter().filter(|c| c.len() > 3).count() as u64)
1774                    .sum::<u64>();
1775                with_times += want
1776                    .times
1777                    .iter()
1778                    .map(|(_, t)| t.iter().filter(|&&t| t != UNREACHED).count() as u64)
1779                    .sum::<u64>();
1780            }
1781        }
1782        // the searches find trips (with a transfer or more behind them),
1783        // and some fail as R5 fails
1784        if n_cases >= 8000 {
1785            assert!(
1786                searches > 25_000 && with_states > 1_500 && with_times > 30_000 && errors > 150,
1787                "{searches} {with_states} {with_times} {errors}"
1788            );
1789            assert!(errors < searches / 2, "{errors} of {searches}");
1790        }
1791    }
