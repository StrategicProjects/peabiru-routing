24/// Public transport routing parameters (modes, access, egress and transfers),
25/// with a default for everything but the date and departure time, which
26/// have none. Fill it in by changing the fields of [`TransitParams::default()`]:
27///
28/// ```
29/// let mut p = peabiru_core::TransitParams::default();
30/// p.date = 20190513; // a Monday
31/// p.departure_time = 14 * 3600; // 14:00
32/// p.percentiles = vec![25, 50, 75];
33/// assert!(p.validate().is_ok());
34/// ```
35#[derive(Debug, Clone)]
36#[non_exhaustive]
37pub struct TransitParams {
38    /// Service date `yyyymmdd`; it must be a real calendar date. The
39    /// default, 0, is a placeholder that [`TransitParams::validate`]
40    /// rejects: there is no default date.
41    pub date: u32,
42    /// Departure, seconds after midnight of `date` (up to 7 days; the
43    /// default 0 is midnight).
44    pub departure_time: i32,
45    /// Departure window in minutes: a search departs at every minute of
46    /// `[departure_time, departure_time + time_window)` (default 10).
47    pub time_window: i32,
48    /// Travel-time percentiles: 1 to 5 values in 1..=99 (see
49    /// [`validate_percentiles`]). Any order and repeats are accepted; the
50    /// results list them in ascending order, a repeated value once per
51    /// occurrence (they are sorted before the
52    /// search). The expanded and arrival matrices do not
53    /// use them but check them all the same, so keep the default there.
54    pub percentiles: Vec<i32>,
55    /// Longest walk in minutes at each end of a trip and in each transfer
56    /// (default 120). The packages' `max_walk_time`, whose `Inf` is
57    /// `max_trip_duration` here.
58    pub max_walk_time: i32,
59    /// Trip limit in whole minutes (default 120).
60    pub max_trip_duration: i32,
61    /// Walking speed, km/h (default 3.6).
62    pub walk_speed: f64,
63    /// The street mode that reaches public transport, which is also the
64    /// mode of the trip without it: the street mode next to the transit
65    /// modes in the packages' `mode` (default [`Mode::Walk`]). By bicycle or car the
66    /// search goes on on foot from wherever the vehicle got, within the
67    /// same time limit, so that stops in
68    /// pedestrian-only areas are reached too; the trip without public
69    /// transport keeps those walking ends as well.
70    pub mode: Mode,
71    /// The street mode from public transport to the destinations, the
72    /// packages' `mode_egress` (default [`Mode::Walk`]). By bicycle or car
73    /// every stop is linked again by that mode and searched up to 5 km by bicycle,
74    /// or below 30 minutes by car, at default speeds and any level of
75    /// traffic stress; the egress then must stay below `max_bike_time` or
76    /// `max_car_time`, at `bike_speed` by bicycle.
77    pub egress_mode: Mode,
78    /// Cycling speed, km/h (default 12), for bicycle access and egress.
79    pub bike_speed: f64,
80    /// Highest level of traffic stress a cyclist rides on, 1 to 4 (default
81    /// 2), for bicycle access (bicycle egress ignores it, see
82    /// `egress_mode`).
83    pub max_lts: i32,
84    /// Longest bicycle access in minutes, the walk after it included, and
85    /// longest bicycle egress (default 120). The packages' `max_bike_time`,
86    /// whose `Inf` is `max_trip_duration` here.
87    pub max_bike_time: i32,
88    /// Longest car access in minutes, the walk after it included, and
89    /// longest car egress (default 120). The packages' `max_car_time`, whose
90    /// `Inf` is `max_trip_duration` here.
91    pub max_car_time: i32,
92    /// Most vehicles boarded in one trip (default 3).
93    pub max_rides: usize,
94    /// The public transport modes to ride, the transit modes in the
95    /// packages' `mode` (default: all of them, `"TRANSIT"`). Routes of other modes
96    /// are left out of the search; it must not be empty.
97    pub transit_modes: Vec<TransitMode>,
98    /// Monte Carlo draws per departure minute for frequency-based routes
99    /// (default 5).
100    pub draws_per_minute: i32,
101    /// Seed of the Monte Carlo draws: with the same seed the results are
102    /// the same. `None` (the default) seeds
103    /// from the clock, so frequency-based results vary from
104    /// call to call (on WebAssembly, which has no clock, from a counter:
105    /// they vary from call to call but repeat from one run of the program
106    /// to the next).
107    pub seed: Option<i32>,
108    /// Worker threads; 0 means all cores, and more than the cores
109    /// available run as many as there are cores.
110    pub n_threads: usize,
111    /// Stops the computation when cancelled, from any thread, through a
112    /// clone of it (see [`CancelToken`]); the default is a fresh token,
113    /// which nothing cancels unless a clone of it is kept.
114    pub cancel: CancelToken,
115    /// The fare structure (default `None`): with one, public transport
116    /// is routed with the multi-criteria router
117    /// (McRAPTOR) over `time_window x draws_per_minute` sampled departure
118    /// times instead of RAPTOR at every minute, keeping per stop the
119    /// fastest trip within `max_fare`. The structure must cover every
120    /// route of the network ([`FareStructure::validate_for`]).
121    pub fare_structure: Option<Arc<FareStructure>>,
122    /// The highest fare of a trip (default infinity: no limit), in the currency of
123    /// the fare structure, read as a 32-bit float; a finite value needs a
124    /// `fare_structure`.
125    pub max_fare: f64,
126    /// Car speeds and levels of traffic stress changed for this call
127    /// (the packages' `new_carspeeds`, `carspeed_scale`, `new_lts`); the default
128    /// changes nothing. It changes the street search from the
129    /// origins (access and direct trips) but not the egress from the stops.
130    pub scenario: crate::Scenario,
131}

133impl TransitParams {
134    /// Check the parameters before routing: the percentiles (see
135    /// [`validate_percentiles`]); counts that are positive and fit 32-bit
136    /// integers; a walk speed of at least 1 mm/s; and what keeps the
137    /// arithmetic in 32-bit seconds: minutes times 60 must fit, and so must the
138    /// departure plus the window plus `max_trip_duration`, the latest clock
139    /// time a search can hold. A departure is at most 7 days after
140    /// midnight of the service day. Every public transport function calls
141    /// it first; [`RouteError::Invalid`] says what is wrong.
142    pub fn validate(&self) -> Result<(), RouteError> {
143        self.check().map_err(RouteError::Invalid)
144    }
145
146    /// [`TransitParams::validate`] with the message alone.
147    pub(crate) fn check(&self) -> Result<(), String> {
148        check_date(self.date)?;
149        check_percentiles(&self.percentiles)?;
150        let check_in = |name: &str, v: i64, lo: i64, hi: i64, unit: &str| {
151            if (lo..=hi).contains(&v) {
152                Ok(())
153            } else {
154                Err(format!("{name} must be from {lo} to {hi}{unit}, got {v}"))
155            }
156        };
157        let check = |name: &str, v: i64, lo: i64, hi: i64| check_in(name, v, lo, hi, "");
158        let int_max = i64::from(i32::MAX);
159        let max_minutes = int_max / 60;
160        check_in(
161            "departure_time",
162            self.departure_time.into(),
163            0,
164            7 * 86_400,
165            " seconds",
166        )?;
167        check_in(
168            "time_window",
169            self.time_window.into(),
170            1,
171            int_max,
172            " minutes",
173        )?;
174        check("draws_per_minute", self.draws_per_minute.into(), 1, int_max)?;
175        // R5 receives `time_window * draws_per_minute` as an `int`
176        let draws = i64::from(self.time_window) * i64::from(self.draws_per_minute);
177        if draws > int_max {
178            return Err(format!(
179                "time_window x draws_per_minute ({} x {} = {draws} Monte Carlo draws) must be \
180                 at most {int_max}",
181                self.time_window, self.draws_per_minute
182            ));
183        }
184        // max_trip_duration first: the packages pass max_walk_time = Inf
185        // as max_trip_duration, so a value too large is reported under the
186        // name the user set
187        let mins = " minutes";
188        check_in(
189            "max_trip_duration",
190            self.max_trip_duration.into(),
191            1,
192            max_minutes,
193            mins,
194        )?;
195        check_in(
196            "max_walk_time",
197            self.max_walk_time.into(),
198            1,
199            max_minutes,
200            mins,
201        )?;
202        check_in(
203            "max_bike_time",
204            self.max_bike_time.into(),
205            1,
206            max_minutes,
207            mins,
208        )?;
209        check_in(
210            "max_car_time",
211            self.max_car_time.into(),
212            1,
213            max_minutes,
214            mins,
215        )?;
216        let last_clock = i64::from(self.departure_time)
217            + 60 * (i64::from(self.time_window) + i64::from(self.max_trip_duration));
218        if last_clock > int_max {
219            return Err(format!(
220                "time_window + max_trip_duration ({} + {} min) after a departure at {} s must \
221                 end within {int_max} s (about 68 years); reduce time_window or \
222                 max_trip_duration",
223                self.time_window, self.max_trip_duration, self.departure_time
224            ));
225        }
226        check(
227            "max_rides",
228            i64::try_from(self.max_rides).unwrap_or(i64::MAX),
229            1,
230            int_max,
231        )?;
232        check_speed("walk_speed", self.walk_speed)?;
233        check_speed("bike_speed", self.bike_speed)?;
234        // 1 to 4 (R5 treats anything else as 4)
235        if !(1..=4).contains(&self.max_lts) {
236            return Err(format!(
237                "max_lts must be from 1 to 4 (the highest level of traffic stress), got {}",
238                self.max_lts
239            ));
240        }
241        if self.transit_modes.is_empty() {
242            return Err("transit_modes must hold at least one public transport mode".into());
243        }
244        check_max_fare(self.max_fare, self.fare_structure.is_some())?;
245        self.scenario.check()
246    }
247}

249/// `max_fare` must be a number >= 0, finite only with a fare structure.
250pub(crate) fn check_max_fare(max_fare: f64, fares: bool) -> Result<(), String> {
251    if max_fare.is_nan() || max_fare < 0.0 {
252        return Err(format!("max_fare must be a number >= 0, got {max_fare}"));
253    }
254    if max_fare.is_finite() && !fares {
255        return Err(format!(
256            "max_fare requires a fare_structure: max_fare is {max_fare} but there is no \
257             fare_structure, so no fares are calculated and the limit would be ignored"
258        ));
259    }
260    Ok(())
261}

263/// R5's `request.maxFare`: `Math.round(maxFare * 100.0f)` of `max_fare` as
264/// a Java `float`, `Integer.MAX_VALUE` for an infinite `max_fare`.
265pub(crate) fn max_fare_cents(max_fare: f64) -> i32 {
266    let f = if max_fare.is_finite() {
267        max_fare as f32
268    } else {
269        i32::MAX as f32
270    };
271    (f64::from(f * 100.0f32) + 0.5).floor() as i32
272}

332/// `AnalysisWorkerTask.MAX_PERCENTILES`.
333pub const MAX_PERCENTILES: usize = 5;
334
335/// Check travel-time percentiles: 1 to 5 of them, each in
336/// 1..=99, in any
337/// order and with repeats (the results sort them). Street
338/// modes have one travel time, reported for every percentile; they take
339/// the same check, so that an empty list is an error there too.
340pub fn validate_percentiles(percentiles: &[i32]) -> Result<(), RouteError> {
341    check_percentiles(percentiles).map_err(RouteError::Invalid)
342}
343
344/// [`validate_percentiles`] with the message alone.
345pub(crate) fn check_percentiles(percentiles: &[i32]) -> Result<(), String> {
346    if percentiles.is_empty() || percentiles.len() > MAX_PERCENTILES {
347        return Err(format!(
348            "between 1 and {MAX_PERCENTILES} percentiles are allowed, got {}",
349            percentiles.len()
350        ));
351    }
352    if let Some(p) = percentiles.iter().find(|p| !(1..=99).contains(*p)) {
353        return Err(format!("percentiles must be in 1..=99, got {p}"));
354    }
355    Ok(())
356}

358/// Days in month `m` (1-12) of year `y`, proleptic Gregorian calendar.
359fn days_in_month(y: i64, m: i64) -> i64 {
360    let leap = (y % 4 == 0 && y % 100 != 0) || y % 400 == 0;
361    match m {
362        2 if leap => 29,
363        2 => 28,
364        4 | 6 | 9 | 11 => 30,
365        _ => 31,
366    }
367}

369/// Check that `yyyymmdd` is a real calendar date (Gregorian, years 1 to
370/// 9999): `20190513` passes, `20190230`, `20191301` and `0` do not. The
371/// service calendar is looked up by this date, so a date that does not
372/// exist would silently match no service.
373pub fn validate_date(date: u32) -> Result<(), RouteError> {
374    check_date(date).map_err(RouteError::Invalid)
375}
376
377/// [`validate_date`] with the message alone.
378pub(crate) fn check_date(date: u32) -> Result<(), String> {
379    let (y, m, d) = (
380        i64::from(date / 10000),
381        i64::from((date / 100) % 100),
382        i64::from(date % 100),
383    );
384    if (1..=9999).contains(&y) && (1..=12).contains(&m) && d >= 1 && d <= days_in_month(y, m) {
385        Ok(())
386    } else {
387        Err(format!(
388            "date must be a calendar date written yyyymmdd, got {date}"
389        ))
390    }
391}

393/// Days since 1970-01-01 of a valid `yyyymmdd` (Howard Hinnant's
394/// `days_from_civil`).
395pub(crate) fn days_from_civil(date: u32) -> i64 {
396    let (y, m, d) = (
397        i64::from(date / 10000),
398        i64::from((date / 100) % 100),
399        i64::from(date % 100),
400    );
401    let y = if m <= 2 { y - 1 } else { y };
402    let era = y.div_euclid(400);
403    let yoe = y - era * 400;
404    let doy = (153 * (if m > 2 { m - 3 } else { m + 9 }) + 2) / 5 + d - 1;
405    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
406    era * 146_097 + doe - 719_468
407}
408
409/// The `yyyymmdd` of a day count since 1970-01-01 (`civil_from_days`), or
410/// `None` outside years 1 to 9999.
411pub(crate) fn civil_from_days(days: i64) -> Option<u32> {
412    let z = days + 719_468;
413    let era = z.div_euclid(146_097);
414    let doe = z - era * 146_097;
415    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
416    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
417    let mp = (5 * doy + 2) / 153;
418    let d = doy - (153 * mp + 2) / 5 + 1;
419    let m = if mp < 10 { mp + 3 } else { mp - 9 };
420    let y = yoe + era * 400 + i64::from(m <= 2);
421    if !(1..=9999).contains(&y) {
422        return None;
423    }
424    u32::try_from(y * 10000 + m * 100 + d).ok()
425}

427/// Day of week of `yyyymmdd`, 0 = Monday.
428pub fn weekday(date: u32) -> usize {
429    // 1970-01-01 is a Thursday
430    ((days_from_civil(date) + 3).rem_euclid(7)) as usize
431}

459/// The egress time in seconds of an egress cost, or `None` at or beyond
460/// the egress mode's limit (`PerTargetPropagater.propagateTransit`): the
461/// cost is millimetres walked or cycled, at `(int)(speed * 1000)` mm/s, or
462/// seconds driven; the limit is `max_walk_time`, `max_bike_time` or
463/// `max_car_time`.
464pub(crate) fn egress_seconds(p: &TransitParams) -> impl Fn(i32) -> Option<i32> + Sync {
465    let mms = |kmh: f64| (((kmh * 5.0 / 18.0) as f32) * 1000.0f32) as i32;
466    let (speed, limit) = match p.egress_mode {
467        Mode::Walk => (mms(p.walk_speed), p.max_walk_time),
468        Mode::Bicycle => (mms(p.bike_speed), p.max_bike_time),
469        Mode::Car => (1, p.max_car_time),
470    };
471    let limit = limit * 60;
472    move |cost: i32| Some(cost / speed).filter(|&e| e < limit)
473}

475/// Percentile index into `n` sorted values (R5: `ceil(p / 100 * n) - 1`).
476pub(crate) fn percentile_index(p: i32, n: usize) -> usize {
477    ((f64::from(p) / 100.0 * n as f64).ceil() as i64 - 1).max(0) as usize
478}

513/// The street search from an origin of a public transport query (R5
514/// `TravelTimeComputer`, part I): by the access mode within its time
515/// limit, carried on on foot after a bicycle or car (`keepRoutingOnFoot`).
516/// It gives the access times to the stops and the direct times to the
517/// destinations.
518pub(crate) struct Access {
519    /// The graph of the access mode.
520    graph: ModeGraph,
521    /// The walking graph that carries a bicycle or car search on.
522    foot: Option<ModeGraph>,
523    /// `min(max_trip_duration, the access mode's own limit)` in seconds
524    /// (`ProfileRequest.getMaxTimeSeconds`).
525    pub(crate) limit: i32,
526}
527
528impl Access {
529    pub(crate) fn new(net: &StreetNetwork, p: &TransitParams) -> Self {
530        let graph = |mode| ModeGraph::with_speeds(net, mode, p.walk_speed, p.bike_speed, p.max_lts);
531        let mode_limit = match p.mode {
532            Mode::Walk => p.max_walk_time,
533            Mode::Bicycle => p.max_bike_time,
534            Mode::Car => p.max_car_time,
535        };
536        Access {
537            graph: graph(p.mode),
538            foot: (p.mode != Mode::Walk).then(|| graph(Mode::Walk)),
539            limit: p.max_trip_duration.min(mode_limit) * 60,
540        }
541    }
542
543    /// The street mode of the access and of the direct trip.
544    pub(crate) fn mode(&self) -> Mode {
545        self.graph.mode()
546    }
547
548    /// Search from `(lat, lon)`: afterwards `access` holds `(stop, seconds)`
549    /// of the stops reached, in stop order, and `direct` the direct time to
550    /// each destination of `dest_splits` (`UNREACHED` where unreachable).
551    /// `false`, with neither filled in, when the origin cannot be linked to
552    /// the streets by the access mode.
553    #[allow(clippy::too_many_arguments)]
554    pub(crate) fn search(
555        &self,
556        net: &StreetNetwork,
557        layer: &TransitLayer,
558        (lat, lon): (f64, f64),
559        dest_splits: &[Option<crate::street::link::Split>],
560        split: &mut Vec<u32>,
561        search: &mut Option<Search>,
562        direct: &mut Vec<i32>,
563        access: &mut Vec<(u32, i32)>,
564    ) -> bool {
565        let Some(origin) = find_split(net, lat, lon, self.mode(), split) else {
566            return false;
567        };
568        let search = search.get_or_insert_with(|| Search::new(net));
569        access_search_from(
570            net,
571            &self.graph,
572            self.foot.as_ref(),
573            &origin,
574            dest_splits,
575            self.limit,
576            search,
577            direct,
578        );
579        let vertex_time = search.vertex_time();
580        access.clear();
581        access.extend(
582            layer
583                .stop_vertex
584                .iter()
585                .enumerate()
586                .filter(|(_, &v)| v >= 0 && vertex_time[v as usize] < self.limit)
587                .map(|(s, &v)| (s as u32, vertex_time[v as usize])),
588        );
589        true
590    }
591}

593/// The destinations of a [`Query`].
594#[derive(Clone, Copy)]
595enum Dests<'p> {
596    Points(&'p [(f64, f64)]),
597    /// The cells of a grid, whose egress tables R5 builds from the cells
598    /// each stop's envelope overlaps.
599    Grid(&'p crate::WebMercatorGrid),

602/// Per-thread buffers reused across origins.
603#[derive(Default)]
604pub(crate) struct Scratch {
605    split: Vec<u32>,
606    times: Vec<i32>,
607    search: Option<Search>,
608    direct: Vec<i32>,
609    access: Vec<(u32, i32)>,
610    stop_times: StopTimes,
611}

613impl<'a> Query<'a> {
614    pub(crate) fn new(
615        net: &'a StreetNetwork,
616        layer: &'a TransitLayer,
617        data: &'a TransitData,
618        destinations: &[(f64, f64)],
619        p: &'a TransitParams,
620    ) -> Result<Self, RouteError> {
621        Self::build(net, layer, data, Dests::Points(destinations), p)
622    }

624    /// [`Query::new`] to the cells of a grid, whose egress tables R5 builds
625    /// from the cells each stop's envelope overlaps.
626    pub(crate) fn new_grid(
627        net: &'a StreetNetwork,
628        layer: &'a TransitLayer,
629        data: &'a TransitData,
630        grid: &crate::WebMercatorGrid,
631        p: &'a TransitParams,
632    ) -> Result<Self, RouteError> {
633        Self::build(net, layer, data, Dests::Grid(grid), p)
634    }
635
636    fn build(
637        net: &'a StreetNetwork,
638        layer: &'a TransitLayer,
639        data: &'a TransitData,
640        destinations: Dests<'_>,
641        p: &'a TransitParams,
642    ) -> Result<Self, RouteError> {
643        p.check().map_err(RouteError::Invalid)?;
644        // RAPTOR keeps iterations x stops durations per origin; their byte
645        // count must be a valid allocation size
646        let per_minute = if layer.has_frequencies {
647            p.draws_per_minute as u64
648        } else {
649            1
650        };
651        let cells = p.time_window as u64 * per_minute * layer.stop_ids.len() as u64;
652        if cells > isize::MAX as u64 / 4 {
653            return Err(RouteError::Invalid(format!(
654                "time_window x draws_per_minute x stops ({cells}) is too large to allocate"
655            )));
656        }
657        let active = layer.active_services(p.date, weekday(p.date));
658        let filtered =
659            Filtered::new(layer, &active, &p.transit_modes).map_err(RouteError::Invalid)?;
660        let walk_speed = (p.walk_speed * 5.0 / 18.0) as f32;
661        let link = |mode| -> Result<Links, RouteError> {
662            Ok(match destinations {
663                Dests::Points(points) => Arc::new(
664                    link_points(
665                        net,
666                        points.len(),
667                        |j| points[j],
668                        mode,
669                        p.n_threads,
670                        &p.cancel,
671                    )
672                    .ok_or(RouteError::Cancelled)?,
673                ),
674                Dests::Grid(g) => grid_links(net, g, mode, p.n_threads, &p.cancel)?,
675            })
676        };
677        // the destinations by the egress mode (walking) and, for the
678        // direct times, by the access mode
679        let egress_splits = link(p.egress_mode)?;
680        let dest_splits = if p.mode == p.egress_mode {
681            egress_splits.clone()
682        } else {
683            link(p.mode)?
684        };
685        let walk_mms = (walk_speed * 1000.0f32) as i32;
686        // PerTargetPropagater: egress seconds from the cost (mm at the
687        // mode's speed, or seconds by car), stops at or beyond the mode's
688        // egress limit skipped.
689        let seconds = egress_seconds(p);
690        let egress = match (p.egress_mode, destinations) {
691            (Mode::Walk, Dests::Points(points)) => walk::egress_tables(
692                layer,
693                &data.trees,
694                net,
695                points,
696                &egress_splits,
697                p.n_threads,
698                &p.cancel,
699            )
700            .ok_or(RouteError::Cancelled)?
701            .into_iter()
702            .map(|table| {
703                table
704                    .into_iter()
705                    .filter_map(|(stop, mm)| Some((stop, seconds(mm)?)))
706                    .collect()
707            })
708            .collect(),
709            // converted as they are found
710            (Mode::Walk, Dests::Grid(g)) => walk::egress_tables_grid(
711                layer,
712                &data.trees,
713                net,
714                g,
715                &egress_splits,
716                seconds,
717                p.n_threads,
718                &p.cancel,
719            )
720            .ok_or(RouteError::Cancelled)?,
721            (mode, dests) => {
722                let targets = match dests {
723                    Dests::Points(points) => Targets::Points(points),
724                    Dests::Grid(g) => Targets::Grid(g),
725                };
726                mode_egress_tables(
727                    net,
728                    layer,
729                    mode,
730                    targets,
731                    &egress_splits,
732                    p.n_threads,
733                    &p.cancel,
734                )
735                .ok_or(RouteError::Cancelled)?
736                .into_iter()
737                .map(|table| {
738                    table
739                        .into_iter()
740                        .filter_map(|(stop, c)| Some((stop, seconds(c)?)))
741                        .collect()
742                })
743                .collect()
744            }
745        };
746        let fares = match &p.fare_structure {
747            Some(fs) => Some(FareSetup::new(fs, layer, p.date, &p.transit_modes)?),
748            None => None,
749        };
750        let scen =
751            crate::street::scenario::scenario_network(net, &p.scenario, p.mode == Mode::Car)?;
752        let access = Access::new(scen.as_ref().unwrap_or(net), p);
753        Ok(Query {
754            net,
755            scen,
756            layer,
757            data,
758            p,
759            filtered,
760            access,
761            dest_splits,
762            egress,
763            max_t: p.max_trip_duration * 60,
764            walk_mms,
765            fares,
766            max_fare: max_fare_cents(p.max_fare),
767            walk_mms_f: f64::from(walk_speed * 1000.0f32),
768        })
769    }
770
771    /// Route from one origin and hand each destination's travel times in
772    /// seconds to `per_destination(j, times)`, destinations in order: one
773    /// value per RAPTOR iteration, or a single (unvarying) value when no
774    /// stop was reached on foot, `UNREACHED` where unreachable. The slice
775    /// may be reordered by the callee. Returns `false`, without calling it,
776    /// when the origin cannot be linked to the street network (R5 then
777    /// records nothing for the origin).
778    pub(crate) fn route_origin(
779        &self,
780        origin: (f64, f64),
781        scratch: &mut Scratch,
782        per_destination: impl FnMut(usize, &mut [i32]),
783    ) -> Result<bool, RouteError> {
784        self.route_origin_fare(origin, scratch, self.max_fare, per_destination)
785    }
786
787    /// [`Query::route_origin`] with `request.maxFare` set to `max_fare`
788    /// cents (`pareto_frontier()` routes once per fare cutoff).
789    /// Errors are McRAPTOR's exceptions.
790    pub(crate) fn route_origin_fare(
791        &self,
792        (lat, lon): (f64, f64),
793        scratch: &mut Scratch,
794        max_fare: i32,
795        mut per_destination: impl FnMut(usize, &mut [i32]),
796    ) -> Result<bool, RouteError> {
797        let (net, layer, p) = (self.scen.as_ref().unwrap_or(self.net), self.layer, self.p);
798        let Scratch {
799            split,
800            times: tt,
801            search,
802            direct,
803            access,
804            stop_times,
805        } = scratch;
806        // Street search: direct times and access to stops.
807        if !self.access.search(
808            net,
809            layer,
810            (lat, lon),
811            &self.dest_splits,
812            split,
813            search,
814            direct,
815            access,
816        ) {
817            return Ok(false);
818        }
819        let transit = !access.is_empty();
820        if let (true, Some(f)) = (transit, &self.fares) {
821            // R5TravelTimeComputer with a fare calculator: McRAPTOR and
822            // FareDominatingList, times collated per draw
823            let req = McRequest {
824                from_time: p.departure_time,
825                to_time: p.departure_time + p.time_window * 60,
826                max_trip_s: self.max_t,
827                max_rides: p.max_rides,
828                suboptimal_minutes: 0,
829                max_fare,
830                monte_carlo_draws: p.time_window * p.draws_per_minute,
831                walk_mms: self.walk_mms_f,
832                seed: p.seed,
833                origin: (lat, lon),
834            };
835            let mut router = McRaptor::new(
836                layer,
837                &self.data.transfers,
838                &f.allowed,
839                &f.active,
840                &f.schedules,
841                &req,
842                Dominance::Fare(&f.calc),
843                None,
844                McRaptor::departure_seed(p.seed, (lat, lon)),
845            );
846            stop_times.reset(layer.stop_ids.len(), req.monte_carlo_draws.max(0) as usize);
847            let mut k = 0;
848            router
849                .route(access, &p.cancel, |_, out| {
850                    if let DrawOut::Times(t) = out {
851                        stop_times.set_iteration(k, t);
852                        k += 1;
853                    }
854                })
855                .map_err(RouteError::Invalid)?;
856        } else if transit {
857            let req = RaptorRequest {
858                from_time: p.departure_time,
859                time_window: p.time_window,
860                max_rides: p.max_rides,
861                max_duration: self.max_t,
862                walk_mms: self.walk_mms,
863                max_walk_min: p.max_walk_time,
864                monte_carlo_draws: p.time_window * p.draws_per_minute,
865                seed: p.seed,
866                origin: (lat, lon),
867            };
868            raptor::route_stop_major(
869                layer,
870                &self.data.transfers,
871                &self.filtered,
872                access,
873                &req,
874                &p.cancel,
875                stop_times,
876            );
877        }
878        let max_t = self.max_t;
879        for (j, &d) in direct.iter().enumerate() {
880            if p.cancel.is_cancelled() {
881                // the caller discards the origin
882                return Ok(true);
883            }
884            // PerTargetPropagater.propagate: start from the non-transit time,
885            // improve it by vehicle arrival + egress walk. Only the least
886            // candidate per iteration matters, so the order of the stops
887            // does not.
888            tt.clear();
889            if !transit {
890                tt.push(d);
891            } else {
892                tt.resize(stop_times.n_iterations(), d);
893                // No candidate at or above `bound` can improve any
894                // iteration: the smaller of max_trip_duration and the worst
895                // time so far.
896                let mut bound = max_t.min(d);
897                for &(stop, e) in &self.egress[j] {
898                    let stop = stop as usize;
899                    // every iteration's candidate is at least the stop's
900                    // least duration plus the egress walk
901                    if stop_times.min(stop).saturating_add(e) >= bound {
902                        continue;
903                    }
904                    let mut worst = 0;
905                    for (t, &ts) in tt.iter_mut().zip(stop_times.get(stop)) {
906                        // ts + e < max_t also means ts < max_t
907                        let c = ts.saturating_add(e);
908                        let c = if c < max_t { c } else { UNREACHED };
909                        *t = (*t).min(c);
910                        worst = worst.max(*t);
911                    }
912                    bound = max_t.min(worst);
913                }
914            }
915            per_destination(j, tt);
916        }
917        Ok(true)
918    }
919}

921/// Run `f(origin_index, scratch)` for every origin on `n_threads` threads
922/// (0 = all cores; capped at the cores and the origins), results in origin
923/// order; `Cancelled` when `cancel` is, and then the workers stop before
924/// their next origin.
925pub(crate) fn par_origins<T: Send>(
926    n_threads: usize,
927    n_origins: usize,
928    cancel: &CancelToken,
929    f: impl Fn(usize, &mut Scratch) -> T + Sync,
930) -> Result<Vec<T>, RouteError> {
931    let n_threads = worker_threads(n_threads, n_origins);
932    let next = AtomicUsize::new(0);
933    let chunks: Vec<Vec<(usize, T)>> = crate::sys::scoped(0..n_threads, |_| {
934        let mut done = Vec::new();
935        let mut scratch = Scratch::default();
936        while !cancel.is_cancelled() {
937            let i = next.fetch_add(1, Ordering::Relaxed);
938            if i >= n_origins {
939                break;
940            }
941            done.push((i, f(i, &mut scratch)));
942        }
943        done
944    });
945    cancel.check()?;
946    let mut slots: Vec<Option<T>> = (0..n_origins).map(|_| None).collect();
947    for (i, r) in chunks.into_iter().flatten() {
948        slots[i] = Some(r);
949    }
950    Ok(slots
951        .into_iter()
952        .map(|r| r.expect("every origin is routed"))
953        .collect())
954}

956/// A public transport travel-time matrix: for each origin and destination,
957/// whole minutes at each requested percentile, `None` where there is none.
958/// A pair is reported (its lowest percentile is `Some`) when that
959/// percentile is within `max_trip_duration`.
960///
961/// The percentiles are in ascending order, whatever the order of
962/// [`TransitParams::percentiles`]; a repeated percentile
963/// gets one value per occurrence (see [`validate_percentiles`]).
964/// The values are held in one buffer, origin by origin, destination by
965/// destination, percentile by percentile, which
966/// [`into_vec`](TransitMatrix::into_vec) hands over without copying.
967#[derive(Debug, Clone, PartialEq, Eq)]
968pub struct TransitMatrix {
969    n_origins: usize,
970    n_destinations: usize,
971    percentiles: Vec<i32>,
972    /// `minutes[(origin * n_destinations + destination) * n_percentiles + k]`.
973    minutes: Vec<Option<u32>>,
974}
975
976impl TransitMatrix {
977    /// Number of origins.
978    pub fn n_origins(&self) -> usize {
979        self.n_origins
980    }
981
982    /// Number of destinations.
983    pub fn n_destinations(&self) -> usize {
984        self.n_destinations
985    }
986
987    /// The percentiles, ascending: the order of the values of each pair.
988    pub fn percentiles(&self) -> &[i32] {
989        &self.percentiles
990    }
991
992    /// Minutes from origin `origin` to destination `destination` at each
993    /// percentile, in the order of [`percentiles`](TransitMatrix::percentiles).
994    /// Panics when either index is out of range.
995    pub fn get(&self, origin: usize, destination: usize) -> &[Option<u32>] {
996        assert!(
997            origin < self.n_origins && destination < self.n_destinations,
998            "({origin}, {destination}) is outside a {} x {} matrix",
999            self.n_origins,
1000            self.n_destinations
1001        );
1002        let np = self.percentiles.len();
1003        let k = (origin * self.n_destinations + destination) * np;
1004        &self.minutes[k..k + np]
1005    }
1006
1007    /// The values from origin `origin`: destination by destination, one
1008    /// value per percentile each (`n_destinations * percentiles().len()`
1009    /// values). Panics when `origin` is out of range.
1010    pub fn row(&self, origin: usize) -> &[Option<u32>] {
1011        assert!(origin < self.n_origins, "origin {origin} out of range");
1012        let per = self.n_destinations * self.percentiles.len();
1013        &self.minutes[origin * per..(origin + 1) * per]
1014    }
1015
1016    /// The rows in origin order, as [`row`](TransitMatrix::row) gives them.
1017    pub fn rows(&self) -> impl ExactSizeIterator<Item = &[Option<u32>]> + '_ {
1018        (0..self.n_origins).map(move |i| self.row(i))
1019    }
1020
1021    /// The whole matrix as one slice, in the order of
1022    /// [`into_vec`](TransitMatrix::into_vec).
1023    pub fn as_slice(&self) -> &[Option<u32>] {
1024        &self.minutes
1025    }
1026
1027    /// The whole matrix as one vector, without copying: the value for
1028    /// `(origin, destination)` at the `k`-th percentile is at
1029    /// `(origin * n_destinations + destination) * percentiles().len() + k`.
1030    pub fn into_vec(self) -> Vec<Option<u32>> {
1031        self.minutes
1032    }
1033}

1035/// The work of [`crate::transit_travel_time_matrix`].
1036pub(crate) fn transit_travel_time_matrix(
1037    net: &StreetNetwork,
1038    layer: &TransitLayer,
1039    data: &TransitData,
1040    origins: &[(f64, f64)],
1041    destinations: &[(f64, f64)],
1042    p: &TransitParams,
1043) -> Result<TransitMatrix, RouteError> {
1044    let q = Query::new(net, layer, data, destinations, p)?;
1045    let m = p.max_trip_duration;
1046    let mut percentiles = p.percentiles.clone();
1047    percentiles.sort_unstable();
1048    let nd = destinations.len();
1049    let np = percentiles.len();
1050    let rows = par_origins(p.n_threads, origins.len(), &p.cancel, |i, scratch| {
1051        let mut row = vec![None; nd * np];
1052        q.route_origin(origins[i], scratch, |j, tt| {
1053            // TravelTimeReducer.extractTravelTimePercentilesAndRecord
1054            tt.sort_unstable();
1055            let n = tt.len();
1056            let vals: Vec<i32> = percentiles
1057                .iter()
1058                .map(|&pc| {
1059                    let v = tt[percentile_index(pc, n).min(n - 1)];
1060                    if v == UNREACHED {
1061                        UNREACHED
1062                    } else {
1063                        v / 60
1064                    }
1065                })
1066                .collect();
1067            if vals[0] <= m {
1068                for (c, &v) in row[j * np..(j + 1) * np].iter_mut().zip(&vals) {
1069                    if v <= m {
1070                        *c = Some(v as u32);
1071                    }
1072                }
1073            }
1074        })?;
1075        Ok::<_, RouteError>(row)
1076    })?;
1077    let mut minutes = Vec::with_capacity(origins.len() * nd * np);
1078    for row in rows {
1079        let row: Vec<Option<u32>> = row?;
1080        minutes.extend_from_slice(&row);
1081    }
1082    Ok(TransitMatrix {
1083        n_origins: origins.len(),
1084        n_destinations: nd,
1085        percentiles,
1086        minutes,
1087    })
1088}
