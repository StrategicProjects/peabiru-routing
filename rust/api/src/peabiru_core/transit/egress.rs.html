1//! Egress from public transport by bicycle or car (R5 `EgressCostTable`
2//! for `BICYCLE` and `CAR`, then `destructivelyTransposeForPropagationAsNeeded`).
3//!
4//! Walking egress reads the walking trees kept with the network
5//! ([`super::walk::egress_tables`]). For the other modes R5 keeps nothing:
6//! for each destination point set it runs a street search from every
7//! linked stop and extends it to the points around the stop. Those searches
8//! carry no request: they run with the defaults of R5's `ProfileRequest`
9//! (walking 1.3 m/s, any level of traffic stress), whatever the speeds and
10//! limits of the query, which only apply when the costs are turned into
11//! egress times.
12//!
13//! - Bicycle: a distance-minimising search, up to 5 km
14//!   (`BICYCLE_DISTANCE_LINKING_LIMIT_METERS`), to the points within 5 km
15//!   of the stop; the cost is millimetres.
16//! - Car: a time-minimising search (turn costs and restrictions), below
17//!   30 minutes (`CAR_TIME_LINKING_LIMIT_SECONDS`), to the points within
18//!   30 min x 22 m/s of the stop; the cost is seconds, the destination
19//!   edge driven at its car speed and the last stretch off the street at
20//!   1.3 m/s (`OFF_STREET_SPEED_MILLIMETERS_PER_SECOND`).
21//!
22//! Each search starts from the coordinates of the stop's vertex, linked to
23//! the streets by the mode (R5 links the stop again rather than starting
24//! from its vertex, whose connector cars may not use).
25

36/// `EgressCostTable.BICYCLE_DISTANCE_LINKING_LIMIT_METERS`.
37const BIKE_LIMIT_M: i32 = 5000;
38/// `EgressCostTable.CAR_TIME_LINKING_LIMIT_SECONDS`.
39const CAR_LIMIT_S: i32 = 30 * 60;
40/// `EgressCostTable.MAX_CAR_SPEED_METERS_PER_SECOND`.
41const MAX_CAR_SPEED_MS: i32 = 22;
42/// `LinkedPointSet.OFF_STREET_SPEED_MILLIMETERS_PER_SECOND`.
43const OFF_STREET_MMS: i32 = 1300;
44
45/// A stop and its `(target, cost)` pairs.
46type StopTable = (usize, Vec<(u32, i32)>);
47
48/// The destinations of an egress table.
49#[derive(Clone, Copy)]
50pub(crate) enum Targets<'p> {
51    /// Points `(lat, lon)`: those strictly inside a stop's envelope
52    /// (`FreeFormPointSet.getPointsInEnvelope`).
53    Points(&'p [(f64, f64)]),
54    /// The cells of a grid: every cell a stop's envelope overlaps
55    /// (`WebMercatorGridPointSet.getPointsInEnvelope`).
56    Grid(&'p crate::WebMercatorGrid),
57}
58
59impl Targets<'_> {
60    fn len(&self) -> usize {
61        match self {
62            Targets::Points(p) => p.len(),
63            Targets::Grid(g) => g.n_cells(),
64        }
65    }
66}

68/// The egress costs by `mode` (bicycle or car) from every linked stop to
69/// the targets, per target `(stop, cost)` in increasing stop order: the
70/// order R5 fills each target's map in. The cost is millimetres by bicycle
71/// and seconds by car (see the module). `dest_splits` are the targets
72/// linked to the streets by `mode`. The stops are shared out among up to
73/// `n_threads` threads (0 = all cores); the result does not depend on it.
74/// `None` when `cancel` is cancelled, which the workers check between
75/// stops.
76pub(crate) fn mode_egress_tables(
77    net: &StreetNetwork,
78    layer: &TransitLayer,
79    mode: Mode,
80    targets: Targets<'_>,
81    dest_splits: &[Option<Split>],
82    n_threads: usize,
83    cancel: &CancelToken,
84) -> Option<Vec<Vec<(u32, i32)>>> {
85    debug_assert_ne!(mode, Mode::Walk);
86    let radius_m = match mode {
87        Mode::Bicycle => f64::from(BIKE_LIMIT_M),
88        _ => f64::from(CAR_LIMIT_S * MAX_CAR_SPEED_MS),
89    };
90    // points sorted by latitude, to find those of an envelope by binary
91    // search; a NaN latitude is inside no envelope
92    let by_lat: Vec<(f64, u32)> = match targets {
93        Targets::Points(points) => {
94            let mut v: Vec<(f64, u32)> = points
95                .iter()
96                .enumerate()
97                .filter(|(_, p)| !p.0.is_nan())
98                .map(|(i, p)| (p.0, i as u32))
99                .collect();
100            v.sort_unstable_by(|a, b| a.0.total_cmp(&b.0).then(a.1.cmp(&b.1)));
101            v
102        }
103        Targets::Grid(_) => Vec::new(),
104    };
105    let car = (mode == Mode::Car).then(|| ModeGraph::r5_defaults(net, Mode::Car));
106    let n_stops = layer.stop_ids.len();
107    let next = AtomicUsize::new(0);
108    let n_threads = worker_threads(n_threads, n_stops);
109    let per_thread: Vec<Vec<StopTable>> = crate::sys::scoped(0..n_threads, |_| {
110        let mut done = Vec::new();
111        let mut split_scratch = Vec::new();
112        let mut tree_scratch = (mode == Mode::Bicycle).then(|| TreeScratch::new(net));
113        let mut dist = if mode == Mode::Bicycle {
114            vec![i32::MAX; net.n_vertices()]
115        } else {
116            Vec::new()
117        };
118        let mut search = car.as_ref().map(|_| Search::new(net));
119        while !cancel.is_cancelled() {
120            let s = next.fetch_add(1, Ordering::Relaxed);
121            if s >= n_stops {
122                break;
123            }
124            let v = layer.stop_vertex[s];
125            if v < 0 {
126                continue;
127            }
128            let v = v as usize;
129            let (lat, lon) = (
130                f64::from(net.v_lat[v]) / FIXED_FACTOR,
131                f64::from(net.v_lon[v]) / FIXED_FACTOR,
132            );
133            let Some(origin) = find_split(net, lat, lon, mode, &mut split_scratch) else {
134                continue;
135            };
136            let env = stop_envelope_m(layer, net, s, radius_m);
137            let mut table = Vec::new();
138            match mode {
139                Mode::Bicycle => {
140                    let sc = tree_scratch.as_mut().expect("bicycle scratch");
141                    let tree = bike_distance_tree(net, &origin, BIKE_LIMIT_M * 1000, sc);
142                    for &(u, d) in &tree {
143                        dist[u as usize] = d;
144                    }
145                    for_targets(targets, &by_lat, env, |p| {
146                        if let Some(sp) = &dest_splits[p] {
147                            let (d0, d1) = (dist[sp.vertex0 as usize], dist[sp.vertex1 as usize]);
148                            if let Some(c) = egress_cost(d0, d1, sp) {
149                                table.push((p as u32, c));
150                            }
151                        }
152                    });
153                    for &(u, _) in &tree {
154                        dist[u as usize] = i32::MAX;
155                    }
156                }
157                _ => {
158                    let g = car.as_ref().expect("car graph");
159                    let search = search.as_mut().expect("car search");
160                    route(net, g, &origin, CAR_LIMIT_S, search);
161                    let vt = search.vertex_time();
162                    for_targets(targets, &by_lat, env, |p| {
163                        if let Some(sp) = &dest_splits[p] {
164                            if let Some(c) = car_cost(net, vt, sp) {
165                                table.push((p as u32, c));
166                            }
167                        }
168                    });
169                }
170            }
171            if !table.is_empty() {
172                done.push((s, table));
173            }
174        }
175        done
176    });
177    if cancel.is_cancelled() {
178        return None;
179    }
180    let mut by_stop: Vec<StopTable> = per_thread.into_iter().flatten().collect();
181    by_stop.sort_unstable_by_key(|&(s, _)| s);
182    let mut out = vec![Vec::new(); targets.len()];
183    for (s, table) in by_stop {
184        for (p, c) in table {
185            out[p as usize].push((s as u32, c));
186        }
187    }
188    Some(out)
189}

191/// Call `f(target)` for each target of a stop's envelope, in increasing
192/// target order for a grid, by latitude for points (the order does not
193/// matter: each target gets at most one cost per stop).
194fn for_targets(
195    targets: Targets<'_>,
196    by_lat: &[(f64, u32)],
197    env: Envelope,
198    mut f: impl FnMut(usize),
199) {
200    let (south, north, west, east) = env;
201    match targets {
202        Targets::Points(points) => {
203            let lo = by_lat.partition_point(|e| e.0 <= south);
204            let hi = by_lat.partition_point(|e| e.0 < north);
205            for &(_, p) in by_lat.get(lo..hi).unwrap_or(&[]) {
206                let lon = points[p as usize].1;
207                if lon < east && lon > west {
208                    f(p as usize);
209                }
210            }
211        }
212        Targets::Grid(grid) => {
213            let Some(((y0, y1), (x0, x1))) = grid.cells_in_envelope(env) else {
214                return;
215            };
216            let width = grid.width() as usize;
217            for y in y0..=y1 {
218                for x in x0..=x1 {
219                    f(y * width + x);
220                }
221            }
222        }
223    }
224}

226/// Seconds by car from a stop to a linked point, given the stop's search
227/// times at the vertices (`LinkedPointSet.timeToPoint` with the car speed
228/// of the point's edge and 1.3 m/s off the street); `None` when neither
229/// end of the edge was reached or the sum overflows R5's `int`.
230fn car_cost(net: &StreetNetwork, vt: &[i32], sp: &Split) -> Option<i32> {
231    let (t0, t1) = (vt[sp.vertex0 as usize], vt[sp.vertex1 as usize]);
232    if t0 == i32::MAX && t1 == i32::MAX {
233        return None;
234    }
235    let on = (cms_to_ms(net.edge_speed_cms[sp.edge as usize]) * 1000.0f32) as i32;
236    let off = sp.distance_to_edge_mm / OFF_STREET_MMS;
237    let via = |t: i32, d: i32| {
238        let x = t.wrapping_add(d / on).wrapping_add(off);
239        if x < 0 {
240            i32::MAX
241        } else {
242            x
243        }
244    };
245    let c = via(t0, sp.distance0_mm).min(via(t1, sp.distance1_mm));
246    (c != i32::MAX).then_some(c)
247}
