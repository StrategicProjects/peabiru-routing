1//! The Pareto frontier of travel time and fare (a port of
2//! `ParetoFrontierCalculator`, see LICENSE-THIRD-PARTY): for every fare
3//! cutoff, the travel-time matrix of R5's fare router with `maxFare` set to
4//! the cutoff, then per destination and percentile the cutoffs at which
5//! the travel time changes.
6
7use super::matrix::{max_fare_cents, par_origins, percentile_index, Query};
8use super::raptor::UNREACHED;
9use super::TransitLayer;
10use crate::street::router::Mode;
11use crate::street::StreetNetwork;
12use crate::transit::matrix::TransitData;
13use crate::{RouteError, RouteParams, TransitParams};
14
15/// One row of a [`ParetoFrontier`].
16#[derive(Debug, Clone, Copy, PartialEq)]
17#[non_exhaustive]
18pub struct ParetoRow {
19    /// Index into the origins.
20    pub origin: usize,
21    /// Index into the destinations.
22    pub destination: usize,
23    /// The travel-time percentile.
24    pub percentile: i32,
25    /// Whole minutes, below `max_trip_duration`.
26    pub travel_time: i32,
27    /// The fare cutoff at which this travel time is first reached, as the
28    /// double of the 32-bit float it is read into (`4.8` is
29    /// `4.800000190734863`).
30    pub monetary_cost: f64,
31}
32
33/// The result of `pareto_frontier()`: rows origin by origin, destination by
34/// destination, percentile by percentile (ascending), then in the order of
35/// the fare cutoffs.
36#[derive(Debug, Clone, PartialEq)]
37#[non_exhaustive]
38pub struct ParetoFrontier {
39    /// The rows.
40    pub rows: Vec<ParetoRow>,
41}
42
43/// The work of [`crate::pareto_frontier`].
44pub(crate) fn pareto_frontier(
45    net: &StreetNetwork,
46    transit: Option<(&TransitLayer, &TransitData)>,
47    origins: &[(f64, f64)],
48    destinations: &[(f64, f64)],
49    p: &TransitParams,
50    fare_cutoffs: &[f64],
51) -> Result<ParetoFrontier, RouteError> {
52    let Some(fs) = &p.fare_structure else {
53        return Err(RouteError::Invalid(
54            "fare_structure is required; create one with setup_fare_structure() or \
55             read_fare_structure()"
56                .into(),
57        ));
58    };
59    if fare_cutoffs.is_empty() || fare_cutoffs.iter().any(|c| c.is_nan() || *c < 0.0) {
60        return Err(RouteError::Invalid(
61            "fare_cutoffs must hold at least one number, each >= 0".into(),
62        ));
63    }
64    for (k, c) in fare_cutoffs.iter().enumerate() {
65        if fare_cutoffs[..k].contains(c) {
66            return Err(RouteError::Invalid(format!(
67                "fare_cutoffs must be unique, {c} is repeated"
68            )));
69        }
70    }
71    // rJava::.jfloat(fare_cutoffs); the row value is (double) of the float
72    let cutoffs: Vec<f32> = fare_cutoffs.iter().map(|&c| c as f32).collect();
73    let cents: Vec<i32> = cutoffs
74        .iter()
75        .map(|&c| max_fare_cents(f64::from(c)))
76        .collect();
77    let mut p = p.clone();
78    p.draws_per_minute = 1;
79    p.max_fare = f64::INFINITY;
80    let mut percentiles = p.percentiles.clone();
81    percentiles.sort_unstable();
82    let np = percentiles.len();
83    let nd = destinations.len();
84    let max_min = p.max_trip_duration;
85    // per origin: minutes[cutoff][destination * np + k]
86    let per_origin: Vec<Vec<Vec<i32>>> = match transit {
87        Some((layer, data)) => {
88            p.check().map_err(RouteError::Invalid)?;
89            let q = Query::new(net, layer, data, destinations, &p)?;
90            let rows = par_origins(p.n_threads, origins.len(), &p.cancel, |i, scratch| {
91                let mut out = Vec::with_capacity(cents.len());
92                for &c in &cents {
93                    let mut row = vec![UNREACHED; nd * np];
94                    q.route_origin_fare(origins[i], scratch, c, |j, tt| {
95                        tt.sort_unstable();
96                        let n = tt.len();
97                        for (k, &pc) in percentiles.iter().enumerate() {
98                            let v = tt[percentile_index(pc, n).min(n - 1)];
99                            row[j * np + k] = if v == UNREACHED { UNREACHED } else { v / 60 };
100                        }
101                    })?;
102                    out.push(row);
103                }
104                Ok::<_, RouteError>(out)
105            })?;
106            rows.into_iter().collect::<Result<_, _>>()?
107        }
108        None => {
109            // the fare structure is still read in and checked
110            crate::fares::FareStructure::check(fs).map_err(RouteError::Invalid)?;
111            crate::transit::matrix::check_percentiles(&p.percentiles)
112                .map_err(RouteError::Invalid)?;
113            let rp = RouteParams {
114                walk_speed: p.walk_speed,
115                bike_speed: p.bike_speed,
116                max_lts: p.max_lts,
117                max_trip_duration: p.max_trip_duration,
118                n_threads: p.n_threads,
119                cancel: p.cancel.clone(),
120                ..RouteParams::default()
121            };
122            let mode: Mode = p.mode;
123            let m = crate::street::router::street_travel_time_matrix(
124                net,
125                origins,
126                destinations,
127                mode,
128                &rp,
129            )?;
130            (0..origins.len())
131                .map(|i| {
132                    let row: Vec<i32> = (0..nd)
133                        .flat_map(|j| {
134                            let v = m.get(i, j).map_or(UNREACHED, |v| v as i32);
135                            std::iter::repeat(v).take(np)
136                        })
137                        .collect();
138                    vec![row; cents.len()]
139                })
140                .collect()
141        }
142    };
143    let mut rows = Vec::new();
144    for (i, by_cutoff) in per_origin.iter().enumerate() {
145        for j in 0..nd {
146            for (k, &pc) in percentiles.iter().enumerate() {
147                let mut previous = -1;
148                for (c, &cutoff) in cutoffs.iter().enumerate() {
149                    let tt = by_cutoff[c][j * np + k];
150                    if tt != previous && tt < max_min {
151                        rows.push(ParetoRow {
152                            origin: i,
153                            destination: j,
154                            percentile: pc,
155                            travel_time: tt,
156                            monetary_cost: f64::from(cutoff),
157                        });
158                        previous = tt;
159                    }
160                }
161            }
162        }
163    }
164    Ok(ParetoFrontier { rows })
165}
