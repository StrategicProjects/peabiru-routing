14/// Where a point meets the street network: the result of [`find_snap`](crate::find_snap).
15#[derive(Debug, Clone, Copy, PartialEq)]
16#[non_exhaustive]
17pub struct Snap {
18    /// Latitude of the snapped point, degrees.
19    pub lat: f64,
20    /// Longitude of the snapped point, degrees.
21    pub lon: f64,
22    /// Distance from the point to the snapped point, metres (haversine on
23    /// a sphere of radius 6,367,450 m).
24    pub distance_m: f64,
25}
26
27/// The work of [`crate::find_snap`].
28pub(crate) fn find_snap(
29    net: &StreetNetwork,
30    points: &[(f64, f64)],
31    radius_m: f64,
32    mode: Mode,
33) -> Result<Vec<Option<Snap>>, RouteError> {
34    if !(radius_m.is_finite() && radius_m >= 0.0) {
35        return Err(RouteError::Invalid(format!(
36            "the radius must be a finite number of metres >= 0, got {radius_m}"
37        )));
38    }
39    const INITIAL_RADIUS_M: f64 = crate::street::link::INITIAL_LINK_RADIUS_M;
40    let mut scratch = Vec::new();
41    Ok(points
42        .iter()
43        .map(|&(lat, lon)| {
44            let first = if radius_m > INITIAL_RADIUS_M {
45                find_split_radius(net, lat, lon, INITIAL_RADIUS_M, mode, &mut scratch)
46            } else {
47                None
48            };
49            let split =
50                first.or_else(|| find_split_radius(net, lat, lon, radius_m, mode, &mut scratch))?;
51            let snap_lat = f64::from(split.fixed_lat) / FIXED_FACTOR;
52            let snap_lon = f64::from(split.fixed_lon) / FIXED_FACTOR;
53            Some(Snap {
54                lat: snap_lat,
55                lon: snap_lon,
56                distance_m: crate::fastmath_trig::haversine_m(lat, lon, snap_lat, snap_lon),
57            })
58        })
59        .collect())
60}

62impl StreetNetwork {
63    /// The bounding box of the street network as
64    /// `[xmin, ymin, xmax, ymax]` (longitude, latitude degrees).
65    /// It covers every OpenStreetMap node of the ways street edges were
66    /// made from, even edges then dropped, so it can be a little
67    /// larger than the box of the edges. A network without streets gives
68    /// JTS's empty envelope, `[0, 0, -1, -1]`.
69    pub fn bbox(&self) -> [f64; 4] {
70        match self.envelope {
71            Some((min_lat, max_lat, min_lon, max_lon)) => {
72                let d = |f: i32| f64::from(f) / FIXED_FACTOR;
73                [d(min_lon), d(min_lat), d(max_lon), d(max_lat)]
74            }
75            None => [0.0, 0.0, -1.0, -1.0],
76        }
77    }
78}

80/// The vertices of the street network: those at an end
81/// of a street edge (connectors of transit stops and park-and-rides left
82/// out), in a fixed order. Made by [`street_vertices`](crate::street_vertices).
83#[derive(Debug, Clone, Default, PartialEq)]
84#[non_exhaustive]
85pub struct StreetVertices {
86    /// `index`: the vertex index in the network.
87    pub index: Vec<u32>,
88    /// Latitude, degrees.
89    pub lat: Vec<f64>,
90    /// Longitude, degrees.
91    pub lon: Vec<f64>,
92    /// `park_and_ride`: the centre of a park-and-ride.
93    pub park_and_ride: Vec<bool>,
94    /// `bike_sharing`: a bike-sharing station (always `false`: bike sharing
95    /// is not loaded).
96    pub bike_sharing: Vec<bool>,
97}

99/// The work of [`crate::street_vertices`].
100pub(crate) fn street_vertices(net: &StreetNetwork) -> StreetVertices {
101    let mut seen = vec![false; net.n_vertices()];
102    let mut order: Vec<u32> = Vec::new();
103    for e in 0..net.n_edges() {
104        if net.edge_flags[e] & LINK != 0 {
105            continue;
106        }
107        for v in [net.edge_from(e), net.edge_to(e)] {
108            if !seen[v] {
109                seen[v] = true;
110                order.push(v as u32);
111            }
112        }
113    }
114    let order = java_hash_set_order(order);
115    let deg = |f: i32| f64::from(f) / FIXED_FACTOR;
116    StreetVertices {
117        lat: order.iter().map(|&v| deg(net.v_lat[v as usize])).collect(),
118        lon: order.iter().map(|&v| deg(net.v_lon[v as usize])).collect(),
119        park_and_ride: order
120            .iter()
121            .map(|&v| net.v_flags[v as usize] & PARK_AND_RIDE != 0)
122            .collect(),
123        bike_sharing: vec![false; order.len()],
124        index: order,
125    }
126}

128/// Iteration order of a `java.util.HashSet<Integer>` (default capacity and
129/// load factor) after adding `keys`, distinct, in this order. A bucket is
130/// `(h ^ (h >>> 16)) & (capacity - 1)`; resizing keeps the order within a
131/// bucket, so the final order is by bucket, then by insertion. (Java turns
132/// a bucket of more than 8 keys into a tree, which orders it differently;
133/// that needs 9 keys whose spread hashes agree in their low bits, which
134/// vertex indices, dense from 0, do not give.)
135fn java_hash_set_order(keys: Vec<u32>) -> Vec<u32> {
136    let mut capacity = 16usize;
137    while keys.len() > capacity / 4 * 3 {
138        capacity *= 2;
139    }
140    let mask = (capacity - 1) as u32;
141    let bucket = |k: u32| (k ^ (k >> 16)) & mask;
142    let mut keys = keys;
143    keys.sort_by_key(|&k| bucket(k)); // stable
144    keys
145}

147/// The stops of the public transport network, one per stop in
148/// stop index order. Made by [`transit_stops`](crate::transit_stops).
149#[derive(Debug, Clone, Default, PartialEq)]
150#[non_exhaustive]
151pub struct TransitStops {
152    /// `stop_index`.
153    pub stop_index: Vec<u32>,
154    /// `stop_id`, prefixed by the feed id (`feed:stop`).
155    pub stop_id: Vec<String>,
156    /// `stop_name`; `None` when blank.
157    pub stop_name: Vec<Option<String>>,
158    /// Latitude of the stop's street vertex, degrees; `None` for a stop not
159    /// linked to the streets.
160    pub lat: Vec<Option<f64>>,
161    /// Longitude, as `lat`.
162    pub lon: Vec<Option<f64>>,
163    /// `linked_to_street`.
164    pub linked_to_street: Vec<bool>,
165}

167/// The work of [`crate::transit_stops`].
168pub(crate) fn transit_stops(net: &StreetNetwork, layer: &TransitLayer) -> TransitStops {
169    let n = layer.n_stops();
170    let mut t = TransitStops::default();
171    for s in 0..n {
172        let v = layer.stop_vertex[s];
173        let at = (v >= 0).then_some(v as usize);
174        t.stop_index.push(s as u32);
175        t.stop_id.push(layer.stop_ids[s].clone());
176        t.stop_name.push(layer.stop_names.get(s).cloned().flatten());
177        t.lat
178            .push(at.map(|v| f64::from(net.v_lat[v]) / FIXED_FACTOR));
179        t.lon
180            .push(at.map(|v| f64::from(net.v_lon[v]) / FIXED_FACTOR));
181        t.linked_to_street.push(at.is_some());
182    }
183    t
184}

186/// The routes of the public transport network: one row per trip
187/// pattern (the trips of a route with the same stops), in pattern order.
188/// Made by [`transit_routes`](crate::transit_routes).
189#[derive(Debug, Clone, Default, PartialEq)]
190#[non_exhaustive]
191pub struct TransitRoutes {
192    /// `agency_id` of the route's agency.
193    pub agency_id: Vec<Option<String>>,
194    /// `agency_name`.
195    pub agency_name: Vec<Option<String>>,
196    /// `route_id`, as in the feed (not prefixed).
197    pub route_id: Vec<String>,
198    /// `route_long_name`.
199    pub long_name: Vec<Option<String>>,
200    /// `route_short_name`.
201    pub short_name: Vec<Option<String>>,
202    /// `mode`: the transit mode of the GTFS `route_type` (`"BUS"`,
203    /// `"RAIL"`, `"SUBWAY"`, `"TRAM"`, `"FERRY"`, `"CABLE_CAR"`,
204    /// `"GONDOLA"`, `"FUNICULAR"` or `"AIR"`).
205    pub mode: Vec<&'static str>,
206    /// Per row: `geometry[offsets[i]..offsets[i + 1]]`, `[lon, lat]`.
207    offsets: Vec<u32>,
208    geometry: Vec<[f64; 2]>,
209}
210
211impl TransitRoutes {
212    /// Number of rows.
213    pub fn len(&self) -> usize {
214        self.route_id.len()
215    }
216
217    /// Whether there are no rows.
218    pub fn is_empty(&self) -> bool {
219        self.route_id.is_empty()
220    }
221
222    /// The line of row `i` as `[lon, lat]` degrees: the GTFS shape of the
223    /// trip that made the pattern, or else the positions of its linked
224    /// stops (empty when none is linked).
225    pub fn geometry(&self, i: usize) -> &[[f64; 2]] {
226        &self.geometry[self.offsets[i] as usize..self.offsets[i + 1] as usize]
227    }
228}

230/// R5's `TransitLayer.getTransitModes`, by name; `None` for a route type R5
231/// does not know (it fails on it).
232fn transit_mode(route_type: i32) -> Option<&'static str> {
233    Some(match route_type {
234        100..=199 | 300..=499 => "RAIL",
235        200..=299 | 700..=899 => "BUS",
236        500..=699 => "SUBWAY",
237        900..=999 => "TRAM",
238        1000..=1099 | 1200..=1299 => "FERRY",
239        1100..=1199 => "AIR",
240        1300..=1399 => "GONDOLA",
241        1400..=1499 => "FUNICULAR",
242        0 | 12 => "TRAM",
243        1 => "SUBWAY",
244        2 => "RAIL",
245        3 | 11 => "BUS",
246        4 => "FERRY",
247        5 => "CABLE_CAR",
248        6 => "GONDOLA",
249        7 => "FUNICULAR",
250        _ => return None,
251    })
252}

254/// The work of [`crate::transit_routes`].
255pub(crate) fn transit_routes(
256    net: &StreetNetwork,
257    layer: &TransitLayer,
258) -> Result<TransitRoutes, RouteError> {
259    let mut t = TransitRoutes {
260        offsets: vec![0],
261        ..TransitRoutes::default()
262    };
263    for (p, pattern) in layer.patterns.iter().enumerate() {
264        let route = layer
265            .pattern_route
266            .get(p)
267            .and_then(|&r| layer.routes.get(r as usize));
268        let route_type = route.map_or(pattern.route_type, |r| r.route_type);
269        let mode = transit_mode(route_type).ok_or_else(|| {
270            RouteError::Invalid(format!("Unknown GTFS route_type code: {route_type}"))
271        })?;
272        t.agency_id.push(route.and_then(|r| r.agency_id.clone()));
273        t.agency_name
274            .push(route.and_then(|r| r.agency_name.clone()));
275        t.route_id
276            .push(route.map_or_else(|| pattern.route_id.clone(), |r| r.route_id.clone()));
277        t.long_name.push(route.and_then(|r| r.long_name.clone()));
278        t.short_name.push(route.and_then(|r| r.short_name.clone()));
279        t.mode.push(mode);
280        match layer.pattern_shape.get(p).copied().flatten() {
281            Some(s) => t.geometry.extend_from_slice(&layer.shapes[s as usize]),
282            None => {
283                for &s in &pattern.stops {
284                    let v = layer.stop_vertex[s as usize];
285                    if v >= 0 {
286                        let v = v as usize;
287                        t.geometry.push([
288                            f64::from(net.v_lon[v]) / FIXED_FACTOR,
289                            f64::from(net.v_lat[v]) / FIXED_FACTOR,
290                        ]);
291                    }
292                }
293            }
294        }
295        t.offsets.push(t.geometry.len() as u32);
296    }
297    Ok(t)
298}

300/// The work of [`crate::transit_service_dates`].
301pub(crate) fn transit_service_dates(layer: &TransitLayer) -> Result<Vec<u32>, RouteError> {
302    let mut used = vec![false; layer.services.len()];
303    for p in &layer.patterns {
304        for t in &p.trips {
305            if let Some(u) = used.get_mut(t.service as usize) {
306                *u = true;
307            }
308        }
309    }
310    let mut dates = std::collections::BTreeSet::new();
311    for (code, s) in layer.services.iter().enumerate() {
312        if !used[code] {
313            continue;
314        }
315        if let Some((_, start_int, end_int)) = s.calendar {
316            let (Some(start), Some(end)) = (basic_iso_date(start_int), basic_iso_date(end_int))
317            else {
318                let (start, end) = (start_int, end_int);
319                let id = layer.service_ids.get(code).map_or("", String::as_str);
320                return Err(RouteError::Invalid(format!(
321                    "Cannot discover operating dates for service '{id}': invalid calendar \
322                     bounds {start} to {end}. Specify dates explicitly or correct the GTFS \
323                     calendar."
324                )));
325            };
326            let mut d = start;
327            while d <= end {
328                if s.active_on(d, weekday(d)) {
329                    dates.insert(d);
330                }
331                d = next_day(d);
332            }
333        }
334        for &d in s.exceptions.keys() {
335            if check_date(d).is_ok() && s.active_on(d, weekday(d)) {
336                dates.insert(d);
337            }
338        }
339    }
340    Ok(dates.into_iter().collect())
341}

343/// Whether Java's `LocalDate.parse(Integer.toString(d), BASIC_ISO_DATE)`
344/// reads `d`: eight digits making a calendar date; the date.
345fn basic_iso_date(d: i32) -> Option<u32> {
346    u32::try_from(d)
347        .ok()
348        .filter(|&d| (10_000_101..=99_991_231).contains(&d) && check_date(d).is_ok())
349}
350
351/// The day after a valid `yyyymmdd`.
352fn next_day(d: u32) -> u32 {
353    if check_date(d + 1).is_ok() {
354        d + 1
355    } else if (d / 100) % 100 < 12 {
356        (d / 100 + 1) * 100 + 1
357    } else {
358        (d / 10000 + 1) * 10000 + 101
359    }

375    #[test]
376    fn hash_set_order_of_small_and_large_keys() {
377        // below the capacity the order is ascending
378        assert_eq!(java_hash_set_order(vec![5, 3, 9, 0]), vec![0, 3, 5, 9]);
379        // 16 and 0 share bucket 0 of a 16-slot table: insertion order
380        assert_eq!(java_hash_set_order(vec![16, 1, 0]), vec![16, 0, 1]);
381        // a key above 2^16 is spread by its high half
382        let k = (1 << 16) | 2; // (k ^ k >> 16) & 15 = 3
383        assert_eq!(java_hash_set_order(vec![k, 3, 2]), vec![2, k, 3]);
384    }

398    fn network_tables_of_the_sample() {
399        let Some(spo) = crate::sample_data::spo() else {

401        };
402        let (net, layer) =
403            crate::transit::build_from_dir(spo, &crate::BuildOptions::default()).unwrap();

405        // every point of the grid snaps within 300 m, on the streets' grid

408        for s in snaps[..2].iter().map(|s| s.unwrap()) {

455        let dates = transit_service_dates(&layer).unwrap();
