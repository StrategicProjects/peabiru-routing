1//! commons-math3 3.0 `FastMath.toRadians`, `sin`, `cos`, `asin` and `pow`,
2//! bit for bit, as bundled in the R5 7.6 jar (`r5-v7.6-all.jar`), for
3//! R5's haversine distance (`GeometryUtils.distance`, which
4//! `find_snap()` reports). They are not correctly rounded, so the platform
5//! libm differs from them in the last bit for some arguments, and the
6//! haversine formula, which subtracts nearby angles, turns that into
7//! differences of about 1e-9 metres.
8//!
9//! The lookup tables below are `FastMath.SINE_TABLE_A/B`,
10//! `COSINE_TABLE_A/B`, `TANGENT_TABLE_A/B`, `EIGHTHS`, `LN_HI_PREC_COEF`
11//! and `FastMath$lnMant.LN_MANT`, dumped from that jar
12//! by reflection as raw IEEE-754 bits. Arguments beyond 3,294,198 radians,
13//! which FastMath reduces with the Payne-Hanek method, are not ported (no
14//! longitude or latitude in radians comes near them); `sin` and `cos` fall
15//! back to the platform's there.

28/// 2^30, for splitting a double in a high and a low part.
29const HEX_40000000: f64 = 1_073_741_824.0;
30const TWO_POWER_52: f64 = 4.503_599_627_370_496E15;
31const LN_2_A: f64 = 0.693_147_063_255_310_1;
32const LN_2_B: f64 = 1.173_046_352_508_234_8E-7;
33const PI_2_A: f64 = 1.570_796_326_794_896_6;
34const PI_2_B: f64 = 6.123_233_995_736_766E-17;

41/// `FastMath.doubleHighPart`: the value with its 30 low bits cleared
42/// (subnormals unchanged).
43fn double_high_part(d: f64) -> f64 {
44    const SAFE_MIN: f64 = 2.225_073_858_507_201_4E-308;
45    if d > -SAFE_MIN && d < SAFE_MIN {
46        return d;
47    }
48    f64::from_bits(d.to_bits() & 0xFFFF_FFFF_C000_0000)
49}

51/// `FastMath.toRadians(x)`.
52pub fn to_radians(x: f64) -> f64 {
53    if x.is_infinite() || x == 0.0 {
54        return x;
55    }
56    const FACTA: f64 = 0.017_453_290_522_098_54;
57    const FACTB: f64 = 1.997_844_754_509_471E-9;
58    let xa = double_high_part(x);
59    let xb = x - xa;
60    let result = xb * FACTB + xb * FACTA + xa * FACTB + xa * FACTA;
61    if result == 0.0 {
62        return result * x;
63    }
64    result
65}

67fn poly_sine(x: f64) -> f64 {
68    let x2 = x * x;
69    let mut p = 2.755_381_745_227_221_7E-6;
70    p = p * x2 + -1.984_126_965_958_650_5E-4;
71    p = p * x2 + 0.008_333_333_333_329_196;
72    p = p * x2 + -0.166_666_666_666_666_66;
73    p * x2 * x
74}

76fn poly_cosine(x: f64) -> f64 {
77    let x2 = x * x;
78    let mut p = 2.479_773_539_153_719E-5;
79    p = p * x2 + -0.001_388_888_868_903_988_3;
80    p = p * x2 + 0.041_666_666_666_621_166;
81    p = p * x2 + -0.499_999_999_999_999_94;
82    p * x2

85/// `FastMath.sinQ`: sine of `xa + xb` in `[0, pi/2]`.
86fn sin_q(xa: f64, xb: f64) -> f64 {
87    let idx = (xa * 8.0 + 0.5) as i32 as usize;
88    let epsilon = xa - t(&EIGHTHS, idx);
89    let sint_a = t(&SINE_TABLE_A, idx);
90    let sint_b = t(&SINE_TABLE_B, idx);
91    let cost_a = t(&COSINE_TABLE_A, idx);
92    let cost_b = t(&COSINE_TABLE_B, idx);
93    let mut sin_eps_a = epsilon;
94    let mut sin_eps_b = poly_sine(epsilon);
95    let cos_eps_b = poly_cosine(epsilon);
96    let temp = sin_eps_a * HEX_40000000;
97    let temp2 = (sin_eps_a + temp) - temp;
98    sin_eps_b += sin_eps_a - temp2;
99    sin_eps_a = temp2;
100
101    let mut a = 0.0f64;
102    let mut b = 0.0f64;
103    let mut tt = sint_a;
104    let mut c = a + tt;
105    let mut d = -(c - a - tt);
106    a = c;
107    b += d;
108    tt = cost_a * sin_eps_a;
109    c = a + tt;
110    d = -(c - a - tt);
111    a = c;
112    b += d;
113    b = b + sint_a * cos_eps_b + cost_a * sin_eps_b;
114    b = b + sint_b + cost_b * sin_eps_a + sint_b * cos_eps_b + cost_b * sin_eps_b;
115    if xb != 0.0 {
116        tt = ((cost_a + cost_b) * (1.0 + cos_eps_b) - (sint_a + sint_b) * (sin_eps_a + sin_eps_b))
117            * xb;
118        c = a + tt;
119        d = -(c - a - tt);
120        a = c;
121        b += d;
122    }
123    a + b
124}

126/// `FastMath.cosQ`: cosine of `xa + xb` in `[0, pi/2]`.
127fn cos_q(xa: f64, xb: f64) -> f64 {
128    let a = PI_2_A - xa;
129    let mut b = -(a - PI_2_A + xa);
130    b += PI_2_B - xb;
131    sin_q(a, b)
132}

134/// The Cody-Waite reduction of `sin` and `cos` for `xa` in
135/// `(pi/2, 3294198]`: quadrant, and the remainder as `(a, b)`.
136fn cody_waite(xa: f64) -> (i32, f64, f64) {
137    let mut k = (xa * 0.636_619_772_367_581_4) as i32;
138    loop {
139        let mut a = f64::from(-k) * 1.570_796_251_296_997;
140        let mut rem_a = xa + a;
141        let mut rem_b = -(rem_a - xa - a);
142        a = f64::from(-k) * 7.549_789_948_768_648E-8;
143        let mut b = rem_a;
144        rem_a = a + b;
145        rem_b += -(rem_a - b - a);
146        a = f64::from(-k) * 6.123_233_995_736_766E-17;
147        b = rem_a;
148        rem_a = a + b;
149        rem_b += -(rem_a - b - a);
150        if rem_a > 0.0 {
151            return (k & 3, rem_a, rem_b);
152        }
153        k -= 1;
154    }
155}

157/// `FastMath.sin(x)`.
158pub fn sin(x: f64) -> f64 {
159    let mut negative = false;
160    let mut quadrant = 0;
161    let mut xa = x;
162    let mut xb = 0.0;
163    if x < 0.0 {
164        negative = true;
165        xa = -xa;
166    }
167    if xa == 0.0 {
168        return if x.to_bits() >> 63 == 1 { -0.0 } else { 0.0 };
169    }
170    if xa.is_nan() || xa == f64::INFINITY {
171        return f64::NAN;
172    }
173    if xa > 3_294_198.0 {
174        return x.sin();
175    } else if xa > PI_2_A {
176        (quadrant, xa, xb) = cody_waite(xa);
177    }
178    if negative {
179        quadrant ^= 2;
180    }
181    match quadrant {
182        0 => sin_q(xa, xb),
183        1 => cos_q(xa, xb),
184        2 => -sin_q(xa, xb),
185        _ => -cos_q(xa, xb),
186    }
187}

189/// `FastMath.cos(x)`.
190pub fn cos(x: f64) -> f64 {
191    let mut quadrant = 0;
192    let mut xa = x;
193    if x < 0.0 {
194        xa = -xa;
195    }
196    if xa.is_nan() || xa == f64::INFINITY {
197        return f64::NAN;
198    }
199    let mut xb = 0.0;
200    if xa > 3_294_198.0 {
201        return x.cos();
202    } else if xa > PI_2_A {
203        (quadrant, xa, xb) = cody_waite(xa);
204    }
205    match quadrant {
206        0 => cos_q(xa, xb),
207        1 => -sin_q(xa, xb),
208        2 => -cos_q(xa, xb),
209        _ => sin_q(xa, xb),
210    }
211}

213/// `FastMath.atan(xa, xb, leftPlane)`: the arctangent of `xa + xb`.
214fn atan(mut xa: f64, mut xb: f64, left_plane: bool) -> f64 {
215    let mut negate = false;
216    if xa == 0.0 {
217        return if left_plane {
218            std::f64::consts::PI.copysign(xa)
219        } else {
220            xa
221        };
222    }
223    if xa < 0.0 {
224        xa = -xa;
225        xb = -xb;
226        negate = true;
227    }
228    if xa > 1.633_123_935_319_537E16 {
229        return if negate ^ left_plane { -PI_2_A } else { PI_2_A };
230    }
231    let idx = if xa < 1.0 {
232        ((-1.716_814_692_820_413_5 * xa * xa + 8.0) * xa + 0.5) as i32
233    } else {
234        let one_over_xa = 1.0 / xa;
235        (-((-1.716_814_692_820_413_5 * one_over_xa * one_over_xa + 8.0) * one_over_xa) + 13.07)
236            as i32
237    } as usize;
238    let mut eps_a = xa - t(&TANGENT_TABLE_A, idx);
239    let mut eps_b = -(eps_a - xa + t(&TANGENT_TABLE_A, idx));
240    eps_b += xb - t(&TANGENT_TABLE_B, idx);
241    let mut temp = eps_a + eps_b;
242    eps_b = -(temp - eps_a - eps_b);
243    eps_a = temp;
244
245    temp = xa * HEX_40000000;
246    let mut ya = xa + temp - temp;
247    let mut yb = xb + xa - ya;
248    xa = ya;
249    xb += yb;
250
251    if idx == 0 {
252        let denom = 1.0 / (1.0 + (xa + xb) * (t(&TANGENT_TABLE_A, idx) + t(&TANGENT_TABLE_B, idx)));
253        ya = eps_a * denom;
254        yb = eps_b * denom;
255    } else {
256        let mut temp2 = xa * t(&TANGENT_TABLE_A, idx);
257        let mut za = 1.0 + temp2;
258        let mut zb = -(za - 1.0 - temp2);
259        temp2 = xb * t(&TANGENT_TABLE_A, idx) + xa * t(&TANGENT_TABLE_B, idx);
260        temp = za + temp2;
261        zb += -(temp - za - temp2);
262        za = temp;
263        zb += xb * t(&TANGENT_TABLE_B, idx);
264        ya = eps_a / za;
265        temp = ya * HEX_40000000;
266        let yaa = (ya + temp) - temp;
267        let yab = ya - yaa;
268        temp = za * HEX_40000000;
269        let zaa = (za + temp) - temp;
270        let zab = za - zaa;
271        yb = (eps_a - yaa * zaa - yaa * zab - yab * zaa - yab * zab) / za;
272        yb += -eps_a * zb / za / za;
273        yb += eps_b / za;
274    }
275
276    eps_a = ya;
277    eps_b = yb;
278    let eps_a2 = eps_a * eps_a;
279    yb = 0.074_908_222_888_644_72;
280    yb = yb * eps_a2 + -0.090_884_508_661_851_92;
281    yb = yb * eps_a2 + 0.111_110_959_423_133_05;
282    yb = yb * eps_a2 + -0.142_857_142_367_918_2;
283    yb = yb * eps_a2 + 0.199_999_999_999_235_82;
284    yb = yb * eps_a2 + -0.333_333_333_333_332_87;
285    yb = yb * eps_a2 * eps_a;
286    ya = eps_a;
287    temp = ya + yb;
288    yb = -(temp - ya - yb);
289    ya = temp;
290    yb += eps_b / (1.0 + eps_a * eps_a);
291
292    let eighths = t(&EIGHTHS, idx);
293    let mut za = eighths + ya;
294    let mut zb = -(za - eighths - ya);
295    temp = za + yb;
296    zb += -(temp - za - yb);
297    za = temp;
298    let mut result = za + zb;
299    let resultb = -(result - za - zb);
300    if left_plane {
301        const PIA: f64 = std::f64::consts::PI;
302        const PIB: f64 = 1.224_646_799_147_353_2E-16;
303        za = PIA - result;
304        zb = -(za - PIA + result);
305        zb += PIB - resultb;
306        result = za + zb;
307    }
308    if negate ^ left_plane {
309        result = -result;
310    }
311    result
312}

314/// `FastMath.asin(x)`.
315pub fn asin(x: f64) -> f64 {
316    if x.is_nan() {
317        return f64::NAN;
318    }
319    if x > 1.0 || x < -1.0 {
320        return f64::NAN;
321    }
322    if x == 1.0 {
323        return PI_2_A;
324    }
325    if x == -1.0 {
326        return -PI_2_A;
327    }
328    if x == 0.0 {
329        return x;
330    }
331    let mut temp = x * HEX_40000000;
332    let xa = x + temp - temp;
333    let xb = x - xa;
334    let mut ya = xa * xa;
335    let mut yb = xa * xb * 2.0 + xb * xb;
336    ya = -ya;
337    yb = -yb;
338    let mut za = 1.0 + ya;
339    let mut zb = -(za - 1.0 - ya);
340    temp = za + yb;
341    zb += -(temp - za - yb);
342    za = temp;
343    let y = za.sqrt();
344    temp = y * HEX_40000000;
345    ya = y + temp - temp;
346    yb = y - ya;
347    yb += (za - ya * ya - 2.0 * ya * yb - yb * yb) / (2.0 * y);
348    let dx = zb / (2.0 * y);
349    let r = x / y;
350    temp = r * HEX_40000000;
351    let mut ra = r + temp - temp;
352    let mut rb = r - ra;
353    rb += (x - ra * ya - ra * yb - rb * ya - rb * yb) / y;
354    rb += -x * dx / y / y;
355    temp = ra + rb;
356    rb = -(temp - ra - rb);
357    ra = temp;
358    atan(ra, rb, false)
359}

361/// `FastMath.log(x, hiPrec)` with `hiPrec` given: `ln(x)` as a high and a
362/// low part, for a positive normal or subnormal `x` (the caller handles
363/// the other cases).
364fn log_hi_prec(x: f64) -> (f64, f64) {
365    let mut bits = x.to_bits() as i64;
366    let mut exp = (bits >> 52) as i32 - 1023;
367    if bits & 0x7ff0_0000_0000_0000 == 0 {
368        bits <<= 1;
369        while bits & 0x0010_0000_0000_0000 == 0 {
370            exp -= 1;
371            bits <<= 1;
372        }
373    }
374    let lnm = LN_MANT[((bits & 0x000f_fc00_0000_0000) >> 42) as usize];
375    let (lnm0, lnm1) = (f64::from_bits(lnm[0]), f64::from_bits(lnm[1]));
376    let epsilon =
377        (bits & 0x3ff_ffff_ffff) as f64 / (TWO_POWER_52 + (bits & 0x000f_fc00_0000_0000) as f64);
378
379    let mut tmp = epsilon * HEX_40000000;
380    let mut aa = epsilon + tmp - tmp;
381    let mut ab = epsilon - aa;
382    let xa = aa;
383    let mut xb = ab;
384    let numer = (bits & 0x3ff_ffff_ffff) as f64;
385    let denom = TWO_POWER_52 + (bits & 0x000f_fc00_0000_0000) as f64;
386    aa = numer - xa * denom - xb * denom;
387    xb += aa / denom;
388
389    let last = LN_HI_PREC_COEF[LN_HI_PREC_COEF.len() - 1];
390    let mut ya = f64::from_bits(last[0]);
391    let mut yb = f64::from_bits(last[1]);
392    for c in LN_HI_PREC_COEF[..LN_HI_PREC_COEF.len() - 1].iter().rev() {
393        aa = ya * xa;
394        ab = ya * xb + yb * xa + yb * xb;
395        tmp = aa * HEX_40000000;
396        ya = aa + tmp - tmp;
397        yb = aa - ya + ab;
398        aa = ya + f64::from_bits(c[0]);
399        ab = yb + f64::from_bits(c[1]);
400        tmp = aa * HEX_40000000;
401        ya = aa + tmp - tmp;
402        yb = aa - ya + ab;
403    }
404    aa = ya * xa;
405    ab = ya * xb + yb * xa + yb * xb;
406    let lnza = aa + ab;
407    let lnzb = -(lnza - aa - ab);
408
409    let e = f64::from(exp);
410    let mut a = LN_2_A * e;
411    let mut b = 0.0;
412    let mut c = a + lnm0;
413    let mut d = -(c - a - lnm0);
414    a = c;
415    b += d;
416    c = a + lnza;
417    d = -(c - a - lnza);
418    a = c;
419    b += d;
420    c = a + LN_2_B * e;
421    d = -(c - a - LN_2_B * e);
422    a = c;
423    b += d;
424    c = a + lnm1;
425    d = -(c - a - lnm1);
426    a = c;
427    b += d;
428    c = a + lnzb;
429    d = -(c - a - lnzb);
430    a = c;
431    b += d;
432    (a, b)
433}

435/// `FastMath.pow(x, y)` for the finite `x >= 0` and `y` the haversine
436/// formula passes (`y = 2`); other arguments follow FastMath's special
437/// cases where cheap and otherwise the platform's `powf`.
438pub fn pow(x: f64, y: f64) -> f64 {
439    if y == 0.0 {
440        return 1.0;
441    }
442    if x.is_nan() {
443        return x;
444    }
445    if x == 0.0 {
446        let negative_zero = x.to_bits() >> 63 == 1;
447        let odd_integer = y == (y as i64) as f64 && (y as i64) & 1 == 1;
448        if negative_zero && odd_integer {
449            return if y < 0.0 { f64::NEG_INFINITY } else { -0.0 };
450        }
451        return if y < 0.0 {
452            f64::INFINITY
453        } else if y > 0.0 {
454            0.0
455        } else {
456            f64::NAN
457        };
458    }
459    if !x.is_finite() || !y.is_finite() {
460        return x.powf(y);
461    }
462    if x < 0.0 {
463        if y >= TWO_POWER_52 || y <= -TWO_POWER_52 {
464            return pow(-x, y);
465        }
466        if y == (y as i64) as f64 {
467            return if (y as i64) & 1 == 0 {
468                pow(-x, y)
469            } else {
470                -pow(-x, y)
471            };
472        }
473        return f64::NAN;
474    }
475    let (ya, yb) = if y < 8e298 && y > -8e298 {
476        let tmp1 = y * HEX_40000000;
477        let ya = y + tmp1 - tmp1;
478        (ya, y - ya)
479    } else {
480        let tmp1 = y * 9.313_225_746_154_785E-10;
481        let tmp2 = tmp1 * 9.313_225_746_154_785E-10;
482        let ya = (tmp1 + tmp2 - tmp1) * HEX_40000000 * HEX_40000000;
483        (ya, y - ya)
484    };
485    let (mut lna, mut lnb) = log_hi_prec(x);
486    if (lna + lnb).is_infinite() {
487        return lna + lnb;
488    }
489    let tmp1 = lna * HEX_40000000;
490    let tmp2 = lna + tmp1 - tmp1;
491    lnb += lna - tmp2;
492    lna = tmp2;
493    let aa = lna * ya;
494    let ab = lna * yb + lnb * ya + lnb * yb;
495    lna = aa + ab;
496    lnb = -(lna - aa - ab);
497    let mut z = 1.0 / 120.0;
498    z = z * lnb + (1.0 / 24.0);
499    z = z * lnb + (1.0 / 6.0);
500    z = z * lnb + 0.5;
501    z = z * lnb + 1.0;
502    z *= lnb;
503    exp_extra(lna, z)
504}

506/// R5's `GeometryUtils.distance(lat0, lon0, lat1, lon1)`: haversine metres
507/// on a sphere of radius 6,367,450 m, with commons-math's `FastMath`.
508pub fn haversine_m(lat0: f64, lon0: f64, lat1: f64, lon1: f64) -> f64 {
509    let theta0 = to_radians(lat0);
510    let theta1 = to_radians(lat1);
511    let lambda0 = to_radians(lon0);
512    let lambda1 = to_radians(lon1);
513    let theta_comb = (theta1 - theta0) / 2.0;
514    let lambda_comb = (lambda1 - lambda0) / 2.0;
515    let sin2theta = pow(sin(theta_comb), 2.0);
516    let sin2lambda = pow(sin(lambda_comb), 2.0);
517    let under = sin2theta + cos(theta0) * cos(theta1) * sin2lambda;
518    2.0 * 6_367_450.0 * asin(under.sqrt())
519}
