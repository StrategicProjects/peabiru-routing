10/// The street edges of a network: every directed edge but the connectors of transit stops and
11/// park-and-rides, in edge order, with their attributes, geometry and
12/// centroid. Made by [`street_edges`](crate::street_edges).
13#[derive(Debug, Clone, PartialEq)]
14pub struct StreetEdges {
15    /// Directed edge index of each row.
16    edges: Vec<u32>,
17    /// Per row: `(lat, lon)` of the centroid, `None` for a zero-length
18    /// geometry (s2 gives an empty point).
19    centroids: Vec<Option<(f64, f64)>>,
20    osm_id: Vec<i64>,
21    from_vertex: Vec<u32>,
22    to_vertex: Vec<u32>,
23    class: Vec<u8>,
24    length_mm: Vec<i32>,
25    flags: Vec<u32>,
26    speed_cms: Vec<i16>,
27    /// Per row: `geometry[geom_offsets[i]..geom_offsets[i + 1]]`, `(lon, lat)`.
28    geom_offsets: Vec<u32>,
29    geometry: Vec<[f64; 2]>,
30}
31
32/// The work of [`crate::street_edges`].
33pub(crate) fn street_edges(net: &StreetNetwork) -> StreetEdges {
34    let mut t = StreetEdges {
35        edges: Vec::new(),
36        centroids: Vec::new(),
37        osm_id: Vec::new(),
38        from_vertex: Vec::new(),
39        to_vertex: Vec::new(),
40        class: Vec::new(),
41        length_mm: Vec::new(),
42        flags: Vec::new(),
43        speed_cms: Vec::new(),
44        geom_offsets: vec![0],
45        geometry: Vec::new(),
46    };
47    let deg = |f: i32| f64::from(f) / FIXED_FACTOR;
48    let mut pts: Vec<[f64; 2]> = Vec::new();
49    for e in 0..net.n_edges() {
50        let f = net.edge_flags[e];
51        if f & LINK != 0 {
52            continue;
53        }
54        let k = e / 2;
55        // EdgeStore.Edge.getGeometry(): the pair's points, reversed for a
56        // backward edge
57        pts.clear();
58        pts.extend(net.pair_points(k).map(|(lat, lon)| [deg(lon), deg(lat)]));
59        if e % 2 == 1 {
60            pts.reverse();
61        }
62        t.edges.push(e as u32);
63        t.centroids.push(s2_centroid(&pts));
64        t.osm_id.push(net.pair_way[k]);
65        t.from_vertex.push(net.edge_from(e) as u32);
66        t.to_vertex.push(net.edge_to(e) as u32);
67        t.class.push(net.pair_class.get(k).copied().unwrap_or(4));
68        t.length_mm.push(net.pair_length_mm[k]);
69        t.flags.push(f);
70        t.speed_cms.push(net.edge_speed_cms[e]);
71        t.geometry.extend_from_slice(&pts);
72        t.geom_offsets.push(t.geometry.len() as u32);
73    }
74    t
75}

77impl StreetEdges {
78    /// Number of edges (rows).
79    pub fn len(&self) -> usize {

83    /// Whether there are no edges.
84    pub fn is_empty(&self) -> bool {
85        self.edges.is_empty()

91    }

93    /// `osm_id`: the OpenStreetMap way the edge was made from.
94    pub fn osm_id(&self, i: usize) -> i64 {
95        self.osm_id[i]
96    }

98    /// `from_vertex`.
99    pub fn from_vertex(&self, i: usize) -> u32 {
100        self.from_vertex[i]
101    }
102
103    /// `to_vertex`.
104    pub fn to_vertex(&self, i: usize) -> u32 {
105        self.to_vertex[i]
106    }

108    /// `street_class`: `"MOTORWAY"`, `"PRIMARY"`, `"SECONDARY"`,

110    pub fn street_class(&self, i: usize) -> &'static str {
111        match self.class[i] {
112            0 => "MOTORWAY",
113            1 => "PRIMARY",
114            2 => "SECONDARY",
115            3 => "TERTIARY",
116            _ => "OTHER",

120    /// `length` in metres.
121    pub fn length_m(&self, i: usize) -> f64 {
122        f64::from(self.length_mm[i]) / 1000.0
123    }
124
125    /// `walk`: whether pedestrians may use the edge.
126    pub fn walk(&self, i: usize) -> bool {
127        self.flags[i] & ALLOWS_PEDESTRIAN != 0
128    }
129
130    /// `car`: whether cars may use the edge.
131    pub fn car(&self, i: usize) -> bool {
132        self.flags[i] & ALLOWS_CAR != 0
133    }
134
135    /// `car_speed` in km/h (the double of a 32-bit float).
136    pub fn car_speed_kph(&self, i: usize) -> f64 {
137        f64::from(((f64::from(self.speed_cms[i]) / 100.0) * 3.6) as f32)
138    }
139
140    /// `bicycle`: whether bicycles may use the edge.
141    pub fn bicycle(&self, i: usize) -> bool {
142        self.flags[i] & ALLOWS_BIKE != 0
143    }
144
145    /// `bicycle_lts`: level of traffic stress, 1 to 4 (1 when none is set).
146    pub fn bicycle_lts(&self, i: usize) -> i32 {
147        let f = self.flags[i];
148        let mut lts = 1;
149        if f & BIKE_LTS_2 != 0 {
150            lts = 2;
151        }
152        if f & BIKE_LTS_3 != 0 {
153            lts = 3;
154        }
155        if f & BIKE_LTS_4 != 0 {
156            lts = 4;
157        }
158        lts
159    }
160
161    /// The edge's line, `[lon, lat]` degrees from its start to its end.
162    pub fn geometry(&self, i: usize) -> &[[f64; 2]] {
163        &self.geometry[self.geom_offsets[i] as usize..self.geom_offsets[i + 1] as usize]
164    }
165
166    /// `(lat, lon)` degrees of the point to route to for the edge: the
167    /// centroid of its line on the sphere (`sf::st_centroid()` with s2),
168    /// `None` when the line has no length.
169    pub fn centroid(&self, i: usize) -> Option<(f64, f64)> {
170        self.centroids[i]
171    }
172
173    /// Every centroid as `(lat, lon)`, NaN for an edge without one.
174    pub fn centroid_points(&self) -> Vec<(f64, f64)> {
175        self.centroids
176            .iter()
177            .map(|c| c.unwrap_or((f64::NAN, f64::NAN)))
178            .collect()
179    }
180}

182type Vec3 = [f64; 3];
183
184/// s2 `S2LatLng::ToPoint` of a longitude/latitude, as s2's plate carrée
185/// projection unprojects it.
186fn to_point([lon, lat]: [f64; 2]) -> Vec3 {
187    let to_radians = std::f64::consts::PI / 180.0;
188    let phi = to_radians * lat;
189    let theta = to_radians * libm_remainder(lon, 360.0);
190    let cosphi = phi.cos();
191    [theta.cos() * cosphi, theta.sin() * cosphi, phi.sin()]
192}
193
194/// C `remainder()` for the longitudes s2 sees: the identity within
195/// `[-180, 180]`.
196fn libm_remainder(x: f64, y: f64) -> f64 {
197    let n = (x / y).round_ties_even();
198    if n == 0.0 {
199        x
200    } else {
201        x - n * y
202    }
203}

205/// s2's `Vector3::Norm2`: `0 + x*x + y*y + z*z`, one term at a time.
206fn norm2(v: Vec3) -> f64 {
207    let mut s = 0.0;
208    for c in v {
209        s += c * c;
210    }
211    s
212}

214/// `S2::TrueCentroid(a, b)` of a segment: towards its midpoint, scaled by
215/// twice the sine of half its angle.
216fn true_centroid(a: Vec3, b: Vec3) -> Vec3 {
217    let vdiff = [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
218    let vsum = [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
219    let sin2 = norm2(vdiff);
220    let cos2 = norm2(vsum);
221    if cos2 == 0.0 {
222        return [0.0; 3];
223    }
224    let s = (sin2 / cos2).sqrt();
225    [s * vsum[0], s * vsum[1], s * vsum[2]]
226}
227
228/// `s2_centroid()` of a line (s2geography: the sum of the segments' true
229/// centroids, normalised), as `(lat, lon)` degrees; `None` for a zero
230/// vector.
231fn s2_centroid(line: &[[f64; 2]]) -> Option<(f64, f64)> {
232    let mut c = [0.0f64; 3];
233    let mut prev = None;
234    for &p in line {
235        let q = to_point(p);
236        if let Some(a) = prev {
237            let t = true_centroid(a, q);
238            for k in 0..3 {
239                c[k] += t[k];
240            }
241        }
242        prev = Some(q);
243    }
244    let n = norm2(c).sqrt();
245    if n == 0.0 {
246        return None;
247    }
248    let inv = 1.0 / n;
249    let c = [c[0] * inv, c[1] * inv, c[2] * inv];
250    // S2LatLng(point) and S1Angle::degrees()
251    let to_degrees = 180.0 / std::f64::consts::PI;
252    let lat = (c[2] + 0.0).atan2((c[0] * c[0] + c[1] * c[1]).sqrt());
253    let lon = (c[1] + 0.0).atan2(c[0] + 0.0);
254    Some((to_degrees * lat, to_degrees * lon))
255}

257/// One edge of a line-based isochrone: from origin `origin`, the edge in
258/// row `edge` of the [`StreetEdges`] is reached in `travel_time` minutes,
259/// within the band of cutoff `isochrone` (the smallest cutoff at or above
260/// the travel time).
261#[derive(Debug, Clone, Copy, PartialEq)]
262#[non_exhaustive]
263pub struct IsochroneLine {
264    /// Index of the origin.
265    pub origin: usize,
266    /// Row in the [`StreetEdges`] routed to.
267    pub edge: usize,
268    /// `travel_time_p50`, whole minutes, above 0.
269    pub travel_time: u32,
270    /// `isochrone`: the cutoff of the edge's band, minutes.
271    pub isochrone: f64,
272}
273
274/// Line-based isochrones from a travel-time matrix to the edges'
275/// centroids: `minutes` holds one row per origin and one value
276/// per row of the [`StreetEdges`] (the median travel time, `None` where
277/// unreached or beyond the trip limit), and `cutoffs` are checked by
278/// [`isochrone_cutoffs`](crate::isochrone_cutoffs).
279///
280/// Times of 0 are left out; each other edge gets the band `(a, b]` of
281/// consecutive cutoffs (starting from 0) holding its time, and edges beyond
282/// the last cutoff are left out. The result is ordered by origin, then by
283/// band and travel time, both descending, ties in edge order.
284/// [`RouteError::Invalid`](crate::RouteError::Invalid) for bad
285/// cutoffs or a matrix of the wrong size.
286pub fn isochrone_lines(
287    n_origins: usize,
288    n_edges: usize,
289    minutes: &[Option<u32>],
290    cutoffs: &[f64],
291) -> Result<Vec<IsochroneLine>, crate::RouteError> {
292    let cutoffs = super::isochrone_cutoffs(cutoffs)?;
293    if minutes.len() != n_origins * n_edges {
294        return Err(crate::RouteError::Invalid(format!(
295            "a matrix of {n_origins} x {n_edges} needs {} values, got {}",
296            n_origins * n_edges,
297            minutes.len()
298        )));
299    }
300    let mut out = Vec::new();
301    for o in 0..n_origins {
302        let start = out.len();
303        let row = &minutes[o * n_edges..(o + 1) * n_edges];
304        for (edge, t) in row.iter().enumerate() {
305            let Some(t) = *t else { continue };
306            if t == 0 {
307                continue;
308            }
309            // cut(x, breaks = c(0, cutoffs), labels = FALSE): right-closed
310            let x = f64::from(t);
311            let Some(&band) = cutoffs.iter().find(|&&c| x <= c) else {
312                continue;
313            };
314            out.push(IsochroneLine {
315                origin: o,
316                edge,
317                travel_time: t,
318                isochrone: band,
319            });
320        }
321        // order(-isochrone, -travel_time_p50), stable
322        out[start..].sort_by(|a, b| {
323            b.isochrone
324                .total_cmp(&a.isochrone)
325                .then(b.travel_time.cmp(&a.travel_time))
326        });
327    }
328    Ok(out)
329}

335    #[test]
336    fn centroid_of_a_straight_segment_is_its_midpoint() {
337        let (lat, lon) = s2_centroid(&[[-51.0, -30.0], [-51.0, -29.0]]).unwrap();
338        assert!((lat + 29.5).abs() < 1e-12, "{lat}");
339        assert!((lon + 51.0).abs() < 1e-12, "{lon}");
340        assert_eq!(s2_centroid(&[[-51.0, -30.0], [-51.0, -30.0]]), None);
341        assert_eq!(libm_remainder(-51.5, 360.0), -51.5);
342        assert_eq!(libm_remainder(190.0, 360.0), -170.0);
343    }

373    }
