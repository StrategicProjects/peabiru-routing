1//! Isobands by marching squares, a port of the `isobander` of the R package
2//! isoband 0.3.0 (Claus O. Wilke, MIT; `src/isoband.cpp`), the code behind
3//! `isoband::isobands()`.
4//!
5//! Only the bands of isochrones are supported: from 0 up to a cutoff, over
6//! values that are never negative, so a grid value is either inside the
7//! band (`0 <= z < high`) or above it, and the 16 cell configurations of
8//! those two states are the only ones that occur.
9//!
10//! isoband collects the rings by walking a `std::unordered_map` keyed by
11//! grid points, so the order of the rings and the vertex each ring starts
12//! at are those of the hash table, which depend on the C++ standard
13//! library isoband was compiled with. [`HashTable`] reproduces both that R
14//! packages are built with: LLVM's libc++ (macOS) and GCC's libstdc++
15//! (Linux, and Windows through Rtools), chosen by [`CxxLibrary`]. With the
16//! library of the platform the rings come out exactly as isoband gives
17//! them there.
18
19/// The C++ standard library whose `std::unordered_map` decides the order of
20/// isoband's rings.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum CxxLibrary {
23    /// LLVM's libc++, which R packages are compiled with on macOS.
24    Libcxx,
25    /// GCC's libstdc++ (GCC 11 to 15 lay the table out alike), which R
26    /// packages are compiled with on Linux and, through Rtools, on Windows.
27    Libstdcxx,
28}
29
30impl CxxLibrary {
31    /// The library of the platform this crate is compiled for: libc++ on
32    /// Apple targets, libstdc++ on every other native target, and libc++
33    /// on WebAssembly (no R there to follow: fixed, so that the npm package
34    /// gives the same polygons wherever it runs).
35    pub const NATIVE: CxxLibrary = if cfg!(any(target_vendor = "apple", target_family = "wasm")) {
36        CxxLibrary::Libcxx
37    } else {
38        CxxLibrary::Libstdcxx
39    };
40}

42/// Kind of a polygon vertex in grid space (isoband `point_type`).
43const GRID: u8 = 0;

45const H_HI: u8 = 2;

49/// A vertex in grid space: row, column and kind (isoband `grid_point`).
50#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
51struct GridPoint {
52    r: i32,
53    c: i32,
54    kind: u8,
55}
56
57impl GridPoint {
58    /// The default `grid_point()`: off the grid.
59    const NONE: GridPoint = GridPoint {
60        r: -1,
61        c: -1,
62        kind: GRID,
63    };
64
65    /// isoband's `grid_point_hasher` through `std::hash<long long>`, which
66    /// is the identity in libc++ and in libstdc++.
67    fn hash(self) -> u64 {
68        ((i64::from(self.r) << 30) ^ (i64::from(self.c) << 3) ^ i64::from(self.kind)) as u64
69    }
70}

111/// `std::unordered_map<grid_point, point_connect>` as libc++'s
112/// `__hash_table` and libstdc++'s `_Hashtable` lay it out. Both keep one
113/// singly linked list of nodes, buckets pointing at the node before their
114/// first one, put a new key first in its bucket (or first in the list when
115/// its bucket is empty), take the bucket of a hash modulo the bucket count,
116/// unlink erased nodes without moving the others and keep the bucket count
117/// on `clear()`. Iteration follows the list. They differ in when and how
118/// they rehash:
119///
120/// - libc++ starts with no buckets and, past a load factor of 1, grows to
121///   `next_prime(2n + 1)` (powers of 2 aside), moving runs of nodes that
122///   share a bucket together (`__do_rehash`);
123/// - libstdc++ starts with 1 bucket and, past a load factor of 1, grows to
124///   the next prime of its table at or above `max(size + 1, 2n)` (13 for
125///   the first key; [`GNU_BUCKET_COUNTS`] holds the ones reached), relinking the nodes one by one at the front of their
126///   new bucket or of the list (`_M_rehash` for unique keys, with
127///   `_Prime_rehash_policy`).
128#[derive(Debug, Clone)]
129struct HashTable {
130    lib: CxxLibrary,
131    nodes: Vec<Node>,
132    free: Vec<u32>,
133    /// `anchor.next`.
134    first: u32,
135    buckets: Vec<u32>,
136    size: usize,
137    /// libstdc++ `_Prime_rehash_policy::_M_next_resize`: the size past
138    /// which an insertion rehashes (0 before the first one).
139    next_resize: usize,
140}
141
142/// libc++ `__constrain_hash`; equal to `h % bc`, which is libstdc++'s
143/// `_Mod_range_hashing`.
144fn constrain(h: u64, bc: usize) -> usize {
145    let bc = bc as u64;
146    (if bc & (bc - 1) == 0 {
147        h & (bc - 1)
148    } else if h < bc {
149        h
150    } else {
151        h % bc
152    }) as usize
153}
154
155/// libstdc++ `_Prime_rehash_policy::_M_need_rehash` (max load factor 1)
156/// before the `n`-th key goes into a table of `bc` buckets: the new bucket
157/// count, if it rehashes. `next_resize` is the policy's state, the size
158/// past which it rehashes (0 before the first insertion).
159///
160/// The tables [`Isobander::new`] accepts never grow past
161/// [`GNU_MAX_SIZE`] keys, so the next bucket count is always listed.
162fn gnu_need_rehash(next_resize: &mut usize, bc: usize, n: usize) -> Option<usize> {
163    if n <= *next_resize {
164        return None;
165    }
166    let min_bkts = n.max(if *next_resize == 0 { 11 } else { 0 });
167    if min_bkts < bc {
168        *next_resize = bc;
169        return None;
170    }
171    let nbc = gnu_next_bkt((min_bkts + 1).max(bc * 2))
172        .expect("isoband: no more keys than Isobander::new checks for");
173    *next_resize = nbc;
174    Some(nbc)
175}
176
177/// libstdc++ `_Prime_rehash_policy::_M_next_bkt` for the arguments
178/// insertions pass it (see [`GNU_BUCKET_COUNTS`]): the first listed count
179/// at or above `n`, or `None` past the list.
180fn gnu_next_bkt(n: usize) -> Option<usize> {
181    let i = GNU_BUCKET_COUNTS.partition_point(|&p| p < n);
182    GNU_BUCKET_COUNTS.get(i).copied()
183}
184
185/// libc++ `__is_hash_power2`.
186fn is_hash_power2(bc: usize) -> bool {
187    bc > 2 && bc & (bc - 1) == 0
188}
189
190/// libc++ `__next_prime`: the smallest prime at or above `n`.
191fn next_prime(n: usize) -> usize {
192    fn is_prime(n: usize) -> bool {
193        if n < 2 {
194            return false;
195        }
196        let mut d = 2;
197        while d * d <= n {
198            if n % d == 0 {
199                return false;
200            }
201            d += 1;
202        }
203        true
204    }
205    let mut n = n;
206    while !is_prime(n) {
207        n += 1;
208    }
209    n
210}

212/// The bucket counts a libstdc++ `std::unordered_map` that only grows by
213/// insertion goes through, after its initial single bucket: 13 on the first
214/// key, then, each time the size passes the bucket count, the policy's
215/// next bucket count for twice the current one.
216///
217/// These are the values `_Prime_rehash_policy::_M_next_bkt` returns for
218/// the only arguments isoband's map passes it (12 on the first insertion,
219/// twice the bucket count afterwards; `clear()` and `erase()` keep the
220/// bucket count, and isoband never calls `reserve()` or `rehash()`), so
221/// they are the reachable entries of libstdc++'s list of primes, not the
222/// list. Recorded from the real policy by `tests/data/isoband-hash/record.cpp`
223/// (the `policy` line of `libstdcxx.txt`, GCC 16.2; the real containers of
224/// that file, recorded with GCC 11 to 14 and 16, reach the counts up to
225/// 712,697), which the tests check this list against. The list stops two
226/// steps past what the largest grid needs (see [`gnu_max_keys`]).
227const GNU_BUCKET_COUNTS: [usize; 23] = [
228    13, 29, 59, 127, 257, 541, 1109, 2357, 5087, 10273, 20753, 42043, 85229, 172933, 351061,
229    712697, 1447153, 2938679, 5967347, 12117689, 24607243, 49969847, 101473717,
230];
231
232/// The most keys the emulated libstdc++ table can take: past the last
233/// listed bucket count, an insertion would rehash to a count the list does
234/// not have.
235const GNU_MAX_SIZE: usize = GNU_BUCKET_COUNTS[GNU_BUCKET_COUNTS.len() - 1];
236
237/// An upper bound on the keys isoband's map holds for a grid of `cells`
238/// points: each grid point and the crossings of the band's top on the edge
239/// to its right and on the edge below it, so fewer than 3 per point. For
240/// R5's largest grid (5,000,000 cells) that is 15,000,000 keys, which
241/// needs the bucket count 24,607,243.
242fn gnu_max_keys(cells: usize) -> usize {
243    cells.saturating_mul(3)
244}
245
246/// Whether the hash table of `lib` can order the rings of a grid of
247/// `cells` points as the real one does.
248fn check_grid_size(cells: usize, lib: CxxLibrary) -> Result<(), String> {
249    if lib == CxxLibrary::Libstdcxx && gnu_max_keys(cells) > GNU_MAX_SIZE {
250        return Err(format!(
251            "isochrone grid of {cells} cells is too large to order its polygons as \
252             libstdc++ does (at most {} cells)",
253            GNU_MAX_SIZE / 3
254        ));
255    }
256    Ok(())
257}

259impl HashTable {
260    fn new(lib: CxxLibrary) -> Self {
261        HashTable {
262            lib,
263            nodes: Vec::new(),
264            free: Vec::new(),
265            first: NIL,
266            // libstdc++ starts on its single bucket, libc++ on none
267            buckets: match lib {
268                CxxLibrary::Libcxx => Vec::new(),
269                CxxLibrary::Libstdcxx => vec![NIL],
270            },
271            size: 0,
272            next_resize: 0,
273        }
274    }
275
276    fn next_of(&self, p: u32) -> u32 {
277        if p == ANCHOR {
278            self.first
279        } else {
280            self.nodes[p as usize].next
281        }
282    }
283
284    fn set_next(&mut self, p: u32, v: u32) {
285        if p == ANCHOR {
286            self.first = v;
287        } else {
288            self.nodes[p as usize].next = v;
289        }
290    }
291
292    fn find(&self, key: GridPoint) -> Option<u32> {
293        let bc = self.buckets.len();
294        if bc == 0 || self.size == 0 {
295            return None;
296        }
297        let h = key.hash();
298        let ch = constrain(h, bc);
299        let pn = self.buckets[ch];
300        if pn == NIL {
301            return None;
302        }
303        let mut nd = self.next_of(pn);
304        while nd != NIL {
305            let n = &self.nodes[nd as usize];
306            if !(n.hash == h || constrain(n.hash, bc) == ch) {
307                break;
308            }
309            if n.hash == h && n.key == key {
310                return Some(nd);
311            }
312            nd = n.next;
313        }
314        None
315    }
316
317    fn get(&self, key: GridPoint) -> &Connect {
318        let i = self.find(key).expect("isoband: point in the polygon grid");
319        &self.nodes[i as usize].value
320    }
321
322    fn get_mut(&mut self, key: GridPoint) -> &mut Connect {
323        let i = self.find(key).expect("isoband: point in the polygon grid");
324        &mut self.nodes[i as usize].value
325    }
326
327    /// `map[key] = value`.
328    fn set(&mut self, key: GridPoint, value: Connect) {
329        if let Some(i) = self.find(key) {
330            self.nodes[i as usize].value = value;
331            return;
332        }
333        let h = key.hash();
334        match self.lib {
335            CxxLibrary::Libcxx => self.grow_libcxx(),
336            CxxLibrary::Libstdcxx => self.grow_libstdcxx(),
337        }
338        let bc = self.buckets.len();
339        let ch = constrain(h, bc);
340        let node = Node {
341            key,
342            hash: h,
343            value,
344            next: NIL,
345        };
346        let nd = match self.free.pop() {
347            Some(i) => {
348                self.nodes[i as usize] = node;
349                i
350            }
351            None => {
352                self.nodes.push(node);
353                (self.nodes.len() - 1) as u32
354            }
355        };
356        let pn = self.buckets[ch];
357        if pn == NIL {
358            let after = self.first;
359            self.nodes[nd as usize].next = after;
360            self.first = nd;
361            self.buckets[ch] = ANCHOR;
362            if after != NIL {
363                let nh = constrain(self.nodes[after as usize].hash, bc);
364                self.buckets[nh] = nd;
365            }
366        } else {
367            let after = self.next_of(pn);
368            self.nodes[nd as usize].next = after;
369            self.set_next(pn, nd);
370        }
371        self.size += 1;
372    }
373
374    /// Before inserting a new key, libc++ (`__emplace_unique_key_args`).
375    fn grow_libcxx(&mut self) {
376        let bc = self.buckets.len();
377        if (self.size + 1) as f32 > bc as f32 {
378            let grow = 2 * bc + usize::from(!is_hash_power2(bc));
379            let need = ((self.size + 1) as f32).ceil() as usize;
380            self.rehash(grow.max(need));
381        }
382    }
383
384    /// Before inserting a new key, libstdc++ (`_M_insert_unique_node` with
385    /// `_Prime_rehash_policy::_M_need_rehash`, max load factor 1).
386    fn grow_libstdcxx(&mut self) {
387        let bc = self.buckets.len();
388        if let Some(nbc) = gnu_need_rehash(&mut self.next_resize, bc, self.size + 1) {
389            self.rehash_libstdcxx(nbc);
390        }
391    }
392
393    /// libstdc++ `_M_rehash` for unique keys: every node, in list order,
394    /// goes first in its new bucket, or first in the list when the bucket
395    /// is new.
396    fn rehash_libstdcxx(&mut self, nbc: usize) {
397        let mut buckets = vec![NIL; nbc];
398        let mut p = self.first;
399        self.first = NIL;
400        let mut bbegin = 0;
401        while p != NIL {
402            let next = self.nodes[p as usize].next;
403            let b = constrain(self.nodes[p as usize].hash, nbc);
404            if buckets[b] == NIL {
405                self.nodes[p as usize].next = self.first;
406                self.first = p;
407                buckets[b] = ANCHOR;
408                if self.nodes[p as usize].next != NIL {
409                    buckets[bbegin] = p;
410                }
411                bbegin = b;
412            } else {
413                let head = buckets[b];
414                let after = self.next_of(head);
415                self.nodes[p as usize].next = after;
416                self.set_next(head, p);
417            }
418            p = next;
419        }
420        self.buckets = buckets;
421    }
422
423    /// libc++ `__rehash`, growing only (as insertions call it).
424    fn rehash(&mut self, n: usize) {
425        let n = if n == 1 {
426            2
427        } else if n & (n - 1) != 0 {
428            next_prime(n)
429        } else {
430            n
431        };
432        if n > self.buckets.len() {
433            self.do_rehash(n);
434        }
435    }
436
437    /// libc++ `__do_rehash` for unique keys.
438    fn do_rehash(&mut self, nbc: usize) {
439        self.buckets = vec![NIL; nbc];
440        let mut pp = ANCHOR;
441        let mut cp = self.first;
442        if cp == NIL {
443            return;
444        }
445        let mut phash = constrain(self.nodes[cp as usize].hash, nbc);
446        self.buckets[phash] = pp;
447        pp = cp;
448        cp = self.next_of(pp);
449        while cp != NIL {
450            let ch = constrain(self.nodes[cp as usize].hash, nbc);
451            if ch == phash {
452                pp = cp;
453            } else if self.buckets[ch] == NIL {
454                self.buckets[ch] = pp;
455                pp = cp;
456                phash = ch;
457            } else {
458                let np = cp;
459                let after_np = self.nodes[np as usize].next;
460                self.set_next(pp, after_np);
461                let head = self.buckets[ch];
462                let after_head = self.next_of(head);
463                self.nodes[np as usize].next = after_head;
464                self.set_next(head, cp);
465            }
466            cp = self.next_of(pp);
467        }
468    }
469
470    /// `map.erase(key)` (libc++ `remove`).
471    fn erase(&mut self, key: GridPoint) {
472        let Some(cn) = self.find(key) else { return };
473        let bc = self.buckets.len();
474        let ch = constrain(self.nodes[cn as usize].hash, bc);
475        let mut pn = self.buckets[ch];
476        while self.next_of(pn) != cn {
477            pn = self.next_of(pn);
478        }
479        let cn_next = self.nodes[cn as usize].next;
480        let same_bucket =
481            |t: &Self, p: u32| p != NIL && constrain(t.nodes[p as usize].hash, bc) == ch;
482        // the bucket empties when neither neighbour is in it
483        if (pn == ANCHOR || constrain(self.nodes[pn as usize].hash, bc) != ch)
484            && !same_bucket(self, cn_next)
485        {
486            self.buckets[ch] = NIL;
487        }
488        if cn_next != NIL {
489            let nh = constrain(self.nodes[cn_next as usize].hash, bc);
490            if nh != ch {
491                self.buckets[nh] = pn;
492            }
493        }
494        self.set_next(pn, cn_next);
495        self.nodes[cn as usize].next = NIL;
496        self.free.push(cn);
497        self.size -= 1;
498    }
499
500    /// `map.clear()`: the bucket count stays.
501    fn clear(&mut self) {
502        if self.size > 0 {
503            self.nodes.clear();
504            self.free.clear();
505            self.first = NIL;
506            self.buckets.iter_mut().for_each(|b| *b = NIL);
507            self.size = 0;
508        }
509    }
510
511    /// Keys in iteration order.
512    fn keys(&self) -> Vec<GridPoint> {
513        let mut out = Vec::with_capacity(self.size);
514        let mut p = self.first;
515        while p != NIL {
516            out.push(self.nodes[p as usize].key);
517            p = self.nodes[p as usize].next;
518        }
519        out
520    }
521}

546/// The isobander of one grid, reused band after band as isoband does (the
547/// hash table keeps its bucket count from one band to the next, which
548/// decides the order of the rings).
549pub(crate) struct Isobander<'a> {
550    grid: Grid<'a>,
551    vhi: f64,
552    map: HashTable,
553    tmp: Vec<GridPoint>,
554}
555
556/// A ring in grid coordinates (x, y), not closed: the last point is not
557/// repeated.
558pub(crate) type Ring = Vec<[f64; 2]>;
559
560impl<'a> Isobander<'a> {
561    /// An isobander whose rings come in the order isoband gives them when
562    /// compiled with `lib`. Fails for a grid too large for the bucket
563    /// counts of libstdc++ that are reproduced (more than 33 million
564    /// points, far above R5's 5 million), rather than give another order.
565    pub(crate) fn new(grid: Grid<'a>, lib: CxxLibrary) -> Result<Self, String> {
566        debug_assert_eq!(grid.z.len(), grid.nrow() * grid.ncol());
567        check_grid_size(grid.nrow().saturating_mul(grid.ncol()), lib)?;
568        Ok(Isobander {
569            grid,
570            vhi: 0.0,
571            map: HashTable::new(lib),
572            tmp: Vec::with_capacity(8),
573        })
574    }
575
576    /// The rings of the band `[0, high)`, in isoband's order. Every value
577    /// must be finite and at least 0.
578    pub(crate) fn band(&mut self, high: f64) -> Vec<Ring> {
579        self.vhi = high;
580        self.calculate_contour();
581        self.collect()
582    }
583
584    fn poly(&mut self, pts: &[(i32, i32, u8)]) {
585        self.tmp.clear();
586        self.tmp
587            .extend(pts.iter().map(|&(r, c, kind)| GridPoint { r, c, kind }));
588        self.poly_merge();
589    }
590
591    /// isoband `poly_merge`.
592    fn poly_merge(&mut self) {
593        let n = self.tmp.len();
594        let mut to_delete = [false; 8];
595        let mut conn = [Connect::default(); 8];
596        for i in 0..n {
597            conn[i].altpoint = false;
598            conn[i].next = self.tmp[if i + 1 < n { i + 1 } else { 0 }];
599            conn[i].prev = self.tmp[if i >= 1 { i - 1 } else { n - 1 }];
600            let p = self.tmp[i];
601            if let Some(idx) = self.map.find(p) {
602                let g = self.map.nodes[idx as usize].value;
603                let t = &mut conn[i];
604                if !g.altpoint {
605                    let score = 2 * u8::from(t.next == g.prev) + u8::from(t.prev == g.next);
606                    match score {
607                        3 => to_delete[i] = true,
608                        2 => t.next = g.next,
609                        1 => t.prev = g.prev,
610                        _ => {
611                            t.prev2 = g.prev;
612                            t.next2 = g.next;
613                            t.altpoint = true;
614                        }
615                    }
616                } else {
617                    let score = 8 * u8::from(t.next == g.prev2)
618                        + 4 * u8::from(t.prev == g.next2)
619                        + 2 * u8::from(t.next == g.prev)
620                        + u8::from(t.prev == g.next);
621                    match score {
622                        9 => {
623                            t.next = g.next2;
624                            t.prev = g.prev;
625                        }
626                        6 => {
627                            t.next = g.next;
628                            t.prev = g.prev2;
629                        }
630                        8 => {
631                            t.next2 = g.next2;
632                            t.prev2 = t.prev;
633                            t.prev = g.prev;
634                            t.next = g.next;
635                            t.altpoint = true;
636                        }
637                        4 => {
638                            t.prev2 = g.prev2;
639                            t.next2 = t.next;
640                            t.prev = g.prev;
641                            t.next = g.next;
642                            t.altpoint = true;
643                        }
644                        2 => {
645                            t.next = g.next;
646                            t.prev2 = g.prev2;
647                            t.next2 = g.next2;
648                            t.altpoint = true;
649                        }
650                        1 => {
651                            t.prev = g.prev;
652                            t.prev2 = g.prev2;
653                            t.next2 = g.next2;
654                            t.altpoint = true;
655                        }
656                        // isoband stops with "undefined merging configuration"
657                        _ => panic!("isoband: undefined merging configuration {score}"),
658                    }
659                }
660            }
661        }
662        for i in 0..n {
663            let p = self.tmp[i];
664            if to_delete[i] {
665                self.map.erase(p);
666            } else {
667                self.map.set(p, conn[i]);
668            }
669        }
670    }

672    /// isoband `calculate_contour` for a band from 0 (values never below).
673    fn calculate_contour(&mut self) {
674        self.map.clear();
675        let (nrow, ncol) = (self.grid.nrow() as i32, self.grid.ncol() as i32);
676        let vhi = self.vhi;
677        // ternarized: 1 inside the band, 2 at or above its top
678        let t = |z: f64| -> u8 {
679            debug_assert!(z.is_finite() && z >= 0.0);
680            if z >= vhi {
681                2
682            } else {
683                1
684            }
685        };
686        for r in 0..nrow - 1 {
687            for c in 0..ncol - 1 {
688                let g = &self.grid;
689                let (z_tl, z_tr, z_br, z_bl) =
690                    (g.z(r, c), g.z(r, c + 1), g.z(r + 1, c + 1), g.z(r + 1, c));
691                let index = 27 * t(z_tl) + 9 * t(z_tr) + 3 * t(z_br) + t(z_bl);
692                // all polygons are drawn clockwise for proper merging
693                match index {
694                    80 => {}
695                    // single triangle
696                    79 => self.poly(&[(r, c, V_HI), (r + 1, c, H_HI), (r + 1, c, GRID)]),
697                    77 => self.poly(&[(r, c + 1, V_HI), (r + 1, c + 1, GRID), (r + 1, c, H_HI)]),
698                    71 => self.poly(&[(r, c, H_HI), (r, c + 1, GRID), (r, c + 1, V_HI)]),
699                    53 => self.poly(&[(r, c, V_HI), (r, c, GRID), (r, c, H_HI)]),
700                    // single rectangle
701                    76 => self.poly(&[
702                        (r, c, V_HI),
703                        (r, c + 1, V_HI),
704                        (r + 1, c + 1, GRID),
705                        (r + 1, c, GRID),
706                    ]),
707                    68 => self.poly(&[
708                        (r, c, H_HI),
709                        (r, c + 1, GRID),
710                        (r + 1, c + 1, GRID),
711                        (r + 1, c, H_HI),
712                    ]),
713                    44 => self.poly(&[
714                        (r, c, GRID),
715                        (r, c + 1, GRID),
716                        (r, c + 1, V_HI),
717                        (r, c, V_HI),
718                    ]),
719                    52 => self.poly(&[
720                        (r, c, H_HI),
721                        (r + 1, c, H_HI),
722                        (r + 1, c, GRID),
723                        (r, c, GRID),
724                    ]),
725                    // single square
726                    40 => self.poly(&[
727                        (r, c, GRID),
728                        (r, c + 1, GRID),
729                        (r + 1, c + 1, GRID),
730                        (r + 1, c, GRID),
731                    ]),
732                    // single pentagon
733                    49 => self.poly(&[
734                        (r, c, GRID),
735                        (r, c, H_HI),
736                        (r, c + 1, V_HI),
737                        (r + 1, c + 1, GRID),
738                        (r + 1, c, GRID),
739                    ]),
740                    67 => self.poly(&[
741                        (r + 1, c, GRID),
742                        (r, c, V_HI),
743                        (r, c, H_HI),
744                        (r, c + 1, GRID),
745                        (r + 1, c + 1, GRID),
746                    ]),
747                    41 => self.poly(&[
748                        (r, c, GRID),
749                        (r, c + 1, GRID),
750                        (r + 1, c + 1, GRID),
751                        (r + 1, c, H_HI),
752                        (r, c, V_HI),
753                    ]),
754                    43 => self.poly(&[
755                        (r, c, GRID),
756                        (r, c + 1, GRID),
757                        (r, c + 1, V_HI),
758                        (r + 1, c, H_HI),
759                        (r + 1, c, GRID),
760                    ]),
761                    // 6-sided saddles
762                    70 => {
763                        if central_value(z_tl, z_tr, z_bl, z_br) >= vhi {
764                            self.poly(&[(r + 1, c, GRID), (r, c, V_HI), (r + 1, c, H_HI)]);
765                            self.poly(&[(r, c + 1, GRID), (r, c + 1, V_HI), (r, c, H_HI)]);
766                        } else {
767                            self.poly(&[
768                                (r + 1, c, GRID),
769                                (r, c, V_HI),
770                                (r, c, H_HI),
771                                (r, c + 1, GRID),
772                                (r, c + 1, V_HI),
773                                (r + 1, c, H_HI),
774                            ]);
775                        }
776                    }
777                    50 => {
778                        if central_value(z_tl, z_tr, z_bl, z_br) >= vhi {
779                            self.poly(&[(r, c, GRID), (r, c, H_HI), (r, c, V_HI)]);
780                            self.poly(&[(r + 1, c + 1, GRID), (r + 1, c, H_HI), (r, c + 1, V_HI)]);
781                        } else {
782                            self.poly(&[
783                                (r, c, GRID),
784                                (r, c, H_HI),
785                                (r, c + 1, V_HI),
786                                (r + 1, c + 1, GRID),
787                                (r + 1, c, H_HI),
788                                (r, c, V_HI),
789                            ]);
790                        }
791                    }
792                    _ => unreachable!("cell index {index} with values never below 0"),
793                }
794            }
795        }
796    }

798    /// isoband `calc_point_coords` (upper level only).
799    fn coords(&self, p: GridPoint) -> [f64; 2] {
800        let g = &self.grid;
801        let (r, c) = (p.r as usize, p.c as usize);
802        match p.kind {
803            GRID => [g.x[c], g.y[r]],
804            H_HI => [
805                interpolate(
806                    g.x[c],
807                    g.x[c + 1],
808                    g.z(p.r, p.c),
809                    g.z(p.r, p.c + 1),
810                    self.vhi,
811                ),
812                g.y[r],
813            ],
814            V_HI => [
815                g.x[c],
816                interpolate(
817                    g.y[r],
818                    g.y[r + 1],
819                    g.z(p.r, p.c),
820                    g.z(p.r + 1, p.c),
821                    self.vhi,
822                ),
823            ],
824            _ => unreachable!("only the upper level is interpolated"),
825        }
826    }

828    /// isoband `collect`: walk the hash table, following each ring.
829    fn collect(&mut self) -> Vec<Ring> {
830        let mut rings = Vec::new();
831        for start in self.map.keys() {
832            let s = *self.map.get(start);
833            if (s.collected && !s.altpoint) || (s.collected && s.collected2 && s.altpoint) {
834                continue;
835            }
836            let mut ring = Vec::new();
837            let mut cur = start;
838            let mut prev = if s.altpoint && !s.collected2 {
839                s.prev2
840            } else {
841                s.prev
842            };
843            loop {
844                ring.push(self.coords(cur));
845                let pc = self.map.get_mut(cur);
846                let newcur = if pc.altpoint && pc.prev2 == prev {
847                    pc.collected2 = true;
848                    pc.next2
849                } else {
850                    pc.collected = true;
851                    pc.next
852                };
853                prev = cur;
854                cur = newcur;
855                if cur == start {
856                    break;
857                }
858            }
859            rings.push(ring);
860        }
861        rings
862    }
863}

865/// isoband `central_value`: `(z[r,c] + z[r,c+1] + z[r+1,c] + z[r+1,c+1]) / 4`.
866fn central_value(z_tl: f64, z_tr: f64, z_bl: f64, z_br: f64) -> f64 {
867    (z_tl + z_tr + z_bl + z_br) / 4.0
868}

870/// isoband `interpolate`: `x0 + d * (x1 - x0)`, `d = (value - z0) / (z1 - z0)`.
871fn interpolate(x0: f64, x1: f64, z0: f64, z1: f64, value: f64) -> f64 {
872    let d = (value - z0) / (z1 - z0);
873    x0 + d * (x1 - x0)
874}

895            x ^= x >> 7;

901            };
902            if i % 3 == 0 {

904                reference.remove(&(k.r, k.c, k.kind));
905            } else {
906                let mut v = Connect::default();
907                v.prev.r = i;

913                reference.clear();
914            }

921        }
922        // every bucket's chain holds only its own keys, contiguously
923        let bc = t.buckets.len();
924        let order: Vec<usize> = t.keys().iter().map(|k| constrain(k.hash(), bc)).collect();
925        let mut seen = std::collections::HashSet::new();

967    /// Replays the operations of `record.cpp` on the emulated table and
968    /// checks every state the real `std::unordered_map` recorded.
969    fn replay(lib: CxxLibrary, fixture: &str) {
970        let mut lines = fixture.lines().filter(|l| !l.is_empty());
971        let mut sequences = 0;
972        let mut policies = 0;
973        while let Some(head) = lines.next() {
974            let f: Vec<&str> = head.split(' ').collect();
975            if f[0] == "grow" {
976                let want: Vec<usize> = f[1..].iter().map(|v| v.parse().unwrap()).collect();
977                let mut t = HashTable::new(lib);
978                let mut got = vec![t.buckets.len()];
979                for i in 0..400_000 {
980                    let k = GridPoint {
981                        r: i / 700,
982                        c: i % 700,
983                        kind: GRID,
984                    };
985                    t.set(k, Connect::default());
986                    if got.last() != Some(&t.buckets.len()) {
987                        got.push(t.buckets.len());
988                    }
989                }
990                assert_eq!(got, want, "{lib:?}: bucket counts");
991                continue;
992            }
993            if f[0] == "policy" {
994                // libstdc++'s own policy, past the keys any grid holds
995                let want: Vec<usize> = f[1..].iter().map(|v| v.parse().unwrap()).collect();
996                assert_eq!(gnu_policy_counts(60_000_000), want, "{lib:?}: policy");
997                policies += 1;
998                continue;
999            }
1000            assert_eq!(f[0], "seq", "{lib:?}: fixture line {head}");
1001            let n: Vec<u64> = f[1..].iter().map(|v| v.parse().unwrap()).collect();
1002            let (seed, phases, ops, rows, cols) = (n[0], n[1], n[2], n[3], n[4]);
1003            let mut rng = SplitMix(seed);
1004            let mut t = HashTable::new(lib);
1005            for _ in 0..phases {
1006                t.clear();
1007                for i in 0..ops {
1008                    let op = rng.next() % 100;
1009                    let k = GridPoint {
1010                        r: (rng.next() % rows) as i32,
1011                        c: (rng.next() % cols) as i32,
1012                        kind: (rng.next() % 5) as u8,
1013                    };
1014                    if op < 65 {
1015                        let mut v = Connect::default();
1016                        v.prev.r = i as i32;
1017                        t.set(k, v);
1018                    } else {
1019                        t.erase(k);
1020                    }
1021                    if i % 256 == 255 {
1022                        let want = lines.next().unwrap();
1023                        let got = format!("cp {} {} {}", t.buckets.len(), t.size, fnv(&t));
1024                        assert_eq!(got, want, "{lib:?}: seq {seed}, phase op {i}");
1025                    }
1026                }
1027            }
1028            let want = lines.next().unwrap();
1029            let got = if t.size <= 1500 {
1030                format!("end {} {}{}", t.buckets.len(), t.size, key_list(&t))
1031            } else {
1032                format!("end {} {} fnv {}", t.buckets.len(), t.size, fnv(&t))
1033            };
1034            assert_eq!(got, want, "{lib:?}: seq {seed}, end");
1035            sequences += 1;
1036        }
1037        assert_eq!(sequences, 8);
1038        assert_eq!(policies, usize::from(lib == CxxLibrary::Libstdcxx));
1039    }

1061    /// an empty table up to `keys` keys, as `record.cpp` asks the real one.
1062    fn gnu_policy_counts(keys: usize) -> Vec<usize> {
1063        let (mut bc, mut next_resize) = (1, 0);
1064        let mut got = vec![bc];
1065        for n in 0..keys {

1070                bc = nbc;
1071                got.push(bc);
1072            }
1073        }
1074        got
1075    }
