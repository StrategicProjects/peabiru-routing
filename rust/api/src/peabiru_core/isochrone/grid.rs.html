1//! The Web Mercator grid of travel-time surfaces (R5 `WebMercatorExtents`,
2//! `WebMercatorGridPointSet` and the pixel functions of `Grid`).
3
4use crate::street::geom::FIXED_FACTOR;
5use crate::street::StreetNetwork;
6use crate::RouteError;
7
8/// Lowest and highest zoom level R5 accepts (`WebMercatorExtents.MIN_ZOOM`,
9/// `MAX_ZOOM`).
10pub(crate) const MIN_ZOOM: i32 = 9;
11/// See [`MIN_ZOOM`].
12pub(crate) const MAX_ZOOM: i32 = 12;
13/// `WebMercatorExtents.MAX_GRID_CELLS`.
14const MAX_GRID_CELLS: i64 = 5_000_000;
15
16/// `Grid.lonToFractionalPixel`.
17pub(crate) fn lon_to_fractional_pixel(lon: f64, zoom: i32) -> f64 {
18    (lon + 180.0) / 360.0 * 2f64.powi(zoom) * 256.0
19}
20
21/// `Grid.latToFractionalPixel`.
22pub(crate) fn lat_to_fractional_pixel(lat: f64, zoom: i32) -> f64 {
23    let lat_rad = lat.to_radians();
24    (1.0 - (lat_rad.tan() + 1.0 / lat_rad.cos()).ln() / std::f64::consts::PI)
25        * 2f64.powi(zoom - 1)
26        * 256.0
27}
28
29/// `Grid.lonToPixel`: Java `(int)`.
30pub(crate) fn lon_to_pixel(lon: f64, zoom: i32) -> i32 {
31    lon_to_fractional_pixel(lon, zoom) as i32
32}
33
34/// `Grid.latToPixel`: Java `(int)`.
35pub(crate) fn lat_to_pixel(lat: f64, zoom: i32) -> i32 {
36    lat_to_fractional_pixel(lat, zoom) as i32
37}
38
39/// `Grid.pixelToLon` of a pixel's centre (`pixelToCenterLon`).
40fn pixel_center_lon(x: i32, zoom: i32) -> f64 {
41    (f64::from(x) + 0.5) / (2f64.powi(zoom) * 256.0) * 360.0 - 180.0
42}
43
44/// `Grid.pixelToLat` of a pixel's centre (`pixelToCenterLat`).
45fn pixel_center_lat(y: i32, zoom: i32) -> f64 {
46    use std::f64::consts::PI;
47    let tile = (f64::from(y) + 0.5) / 256.0;
48    (PI - tile * PI * 2.0 / 2f64.powi(zoom))
49        .sinh()
50        .atan()
51        .to_degrees()
52}
53
54/// A block of Web Mercator pixels at one zoom level: the destinations of a
55/// travel-time surface. Pixel numbers are
56/// absolute (world) numbers at the zoom level, `x` growing eastward and
57/// `y` southward; cell `i` is pixel `(west + i % width, north + i / width)`.
58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59pub struct WebMercatorGrid {
60    zoom: i32,
61    west: i32,
62    north: i32,
63    width: i32,
64    height: i32,
65}
66
67impl WebMercatorGrid {
68    /// The grid polygon isochrones are interpolated from: the pixels
69    /// covering the street network's envelope (the
70    /// envelope holds every OpenStreetMap node of the ways streets were
71    /// made from). `zoom` must be 9 to 12, and the grid at most 5,000,000
72    /// cells; [`RouteError::Invalid`] otherwise, and for a
73    /// network without streets.
74    pub fn for_network(network: &crate::Network, zoom: i32) -> Result<Self, RouteError> {
75        Self::for_street(network.street(), zoom)
76    }
77
78    /// [`WebMercatorGrid::for_network`] of a street network.
79    pub(crate) fn for_street(net: &StreetNetwork, zoom: i32) -> Result<Self, RouteError> {
80        check_zoom(zoom)?;
81        let Some((min_lat, max_lat, min_lon, max_lon)) = net.envelope else {
82            return Err(RouteError::Invalid(
83                "the network has no streets to make a grid of".into(),
84            ));
85        };
86        let deg = |f: i32| f64::from(f) / FIXED_FACTOR;
87        let north = lat_to_pixel(deg(max_lat), zoom);
88        let west = lon_to_pixel(deg(min_lon), zoom);
89        let height = (lat_to_fractional_pixel(deg(min_lat), zoom) - f64::from(north)).ceil() as i32;
90        let width = (lon_to_fractional_pixel(deg(max_lon), zoom) - f64::from(west)).ceil() as i32;
91        Self::new(west, north, width, height, zoom)
92    }
93
94    /// A grid of `width` x `height` pixels whose north-west pixel is
95    /// `(west, north)` at `zoom`, with the checks of
96    /// [`for_network`](Self::for_network).
97    pub fn new(
98        west: i32,
99        north: i32,
100        width: i32,
101        height: i32,
102        zoom: i32,
103    ) -> Result<Self, RouteError> {
104        check_zoom(zoom)?;
105        if width < 1 || height < 1 {
106            return Err(RouteError::Invalid(format!(
107                "a grid needs a width and height of at least 1, got {width} x {height}"
108            )));
109        }
110        let cells = i64::from(width) * i64::from(height);
111        if cells > MAX_GRID_CELLS {
112            return Err(RouteError::Invalid(format!(
113                "Requested number of destinations ({cells}) exceeds limit ({MAX_GRID_CELLS}). \
114                 Use a lower zoom level."
115            )));
116        }
117        Ok(WebMercatorGrid {
118            zoom,
119            west,
120            north,
121            width,
122            height,
123        })
124    }
125
126    /// Zoom level, 9 to 12.
127    pub fn zoom(&self) -> i32 {
128        self.zoom
129    }
130
131    /// Pixel number of the westernmost column.
132    pub fn west(&self) -> i32 {
133        self.west
134    }
135
136    /// Pixel number of the northernmost row.
137    pub fn north(&self) -> i32 {
138        self.north
139    }
140
141    /// Number of columns.
142    pub fn width(&self) -> i32 {
143        self.width
144    }
145
146    /// Number of rows.
147    pub fn height(&self) -> i32 {
148        self.height
149    }
150
151    /// Number of cells, `width * height`.
152    pub fn n_cells(&self) -> usize {
153        self.width as usize * self.height as usize
154    }
155
156    /// `(lat, lon)` degrees of the centre of cell `i` (row by row from the
157    /// north-west), the point routed to.
158    pub fn cell_center(&self, i: usize) -> (f64, f64) {
159        let w = self.width as usize;
160        let x = self.west + (i % w) as i32;
161        let y = self.north + (i / w) as i32;
162        (
163            pixel_center_lat(y, self.zoom),
164            pixel_center_lon(x, self.zoom),
165        )
166    }
167
168    /// Every cell centre, in cell order (see [`WebMercatorGrid::cell_center`]).
169    pub fn cell_centers(&self) -> Vec<(f64, f64)> {
170        (0..self.n_cells()).map(|i| self.cell_center(i)).collect()
171    }
172
173    /// The cells R5 considers for points inside an envelope given in
174    /// floating degrees (`WebMercatorGridPointSet.getPointsInEnvelope`):
175    /// every pixel the envelope's edges fall in, inclusive, clipped to the
176    /// grid. Returns the rows and the columns of those cells, `(rows,
177    /// columns)` inclusive, or `None` when the envelope misses the grid.
178    pub(crate) fn cells_in_envelope(
179        &self,
180        (south, north, west, east): (f64, f64, f64, f64),
181    ) -> Option<((usize, usize), (usize, usize))> {
182        let z = self.zoom;
183        let x0 = (lon_to_pixel(west, z) - self.west).max(0);
184        let x1 = (lon_to_pixel(east, z) - self.west).min(self.width - 1);
185        let y0 = (lat_to_pixel(north, z) - self.north).max(0);
186        let y1 = (lat_to_pixel(south, z) - self.north).min(self.height - 1);
187        (x0 <= x1 && y0 <= y1).then_some(((y0 as usize, y1 as usize), (x0 as usize, x1 as usize)))
188    }
189}
190
191fn check_zoom(zoom: i32) -> Result<(), RouteError> {
192    if (MIN_ZOOM..=MAX_ZOOM).contains(&zoom) {
193        Ok(())
194    } else {
195        Err(RouteError::Invalid(format!(
196            "zoom must be from {MIN_ZOOM} to {MAX_ZOOM}, got {zoom}"
197        )))
198    }
199}

201#[cfg(test)]
202mod tests {
203    use super::*;
204
205    #[test]
206    fn pixels_round_trip() {
207        let (lat, lon) = (pixel_center_lat(153981, 10), pixel_center_lon(93740, 10));
208        assert_eq!(lat_to_pixel(lat, 10), 153981);
209        assert_eq!(lon_to_pixel(lon, 10), 93740);
210    }
211
212    #[test]
213    fn zoom_and_size_are_checked() {
214        assert!(WebMercatorGrid::new(0, 0, 10, 10, 8).is_err());
215        assert!(WebMercatorGrid::new(0, 0, 10, 10, 13).is_err());
216        assert!(WebMercatorGrid::new(0, 0, 3000, 3000, 12).is_err());
217        assert!(WebMercatorGrid::new(0, 0, 0, 10, 10).is_err());
218        let g = WebMercatorGrid::new(5, 7, 3, 2, 9).unwrap();
219        assert_eq!(g.n_cells(), 6);
220        let env = |s, n, w, e| {
221            (
222                pixel_center_lat(s, 9),
223                pixel_center_lat(n, 9),
224                pixel_center_lon(w, 9),
225                pixel_center_lon(e, 9),
226            )
227        };
228        // rows 0 and 1, columns 1 and 2: cells 1, 2, 4, 5
229        assert_eq!(g.cells_in_envelope(env(8, 7, 6, 9)), Some(((0, 1), (1, 2))));
230        assert_eq!(g.cells_in_envelope(env(7, 7, 5, 5)), Some(((0, 0), (0, 0))));
231        assert_eq!(g.cells_in_envelope(env(8, 7, 9, 12)), None);
232        assert_eq!(g.cells_in_envelope(env(12, 9, 5, 7)), None);
233    }
