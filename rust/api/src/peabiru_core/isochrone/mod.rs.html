1//! Isochrones.
2//!
3//! Polygon isochrones (`polygon_output = TRUE` in the packages) are made in
4//! two steps, both here: a travel-time *surface* from each origin to the
5//! cells of a Web Mercator grid covering the network (R5's
6//! `TravelTimeSurfaceTask`; [`street_travel_time_surface`],
7//! [`transit_travel_time_surface`]), then the band of each cutoff traced
8//! over the surface by marching squares and grouped into polygons with
9//! holes, as the R package isoband does ([`isochrone_polygons`]).
10//! The surface is R5's, value for value; the rings are isoband's, vertex
11//! for vertex and in the order isoband gives them on the platform (it
12//! depends on the C++ standard library isoband is built with; see
13//! [`isochrone_polygons`]). Repairing the polygons with
14//! `sf::st_make_valid()`, which on longitude/latitude data rebuilds them
15//! with s2 (snapping to 1e-7 degrees), is not done here: the R package
16//! applies it, other callers get isoband's polygons.
17//!
18//! Line isochrones (`polygon_output = FALSE`) are the street edges reached
19//! from each origin, banded by cutoff: [`street_edges`] lists the edges and
20//! the centroid to route to on each, a travel-time matrix to those
21//! centroids gives the times, and [`isochrone_lines`] bands and orders them.

42/// Travel times from each origin to every cell of a Web Mercator grid, in
43/// whole minutes at each percentile:
44/// `i32::MAX` where a cell is not
45/// reached. The minutes are not capped at the trip limit: a street time
46/// includes the partial edges at both ends and may pass it.
47#[derive(Debug, Clone, PartialEq, Eq)]
48pub struct TravelTimeSurface {
49    grid: WebMercatorGrid,
50    percentiles: Vec<i32>,
51    n_origins: usize,
52    /// `minutes[(origin * n_percentiles + k) * n_cells + cell]`.
53    minutes: Vec<i32>,
54}
55
56impl TravelTimeSurface {
57    /// The grid of the cells.
58    pub fn grid(&self) -> &WebMercatorGrid {
59        &self.grid
60    }
61
62    /// The percentiles, ascending.
63    pub fn percentiles(&self) -> &[i32] {
64        &self.percentiles
65    }
66
67    /// Number of origins.
68    pub fn n_origins(&self) -> usize {
69        self.n_origins
70    }
71
72    /// Minutes from `origin` to every cell at the `k`-th percentile, cell by
73    /// cell (row by row from the north-west; see
74    /// [`WebMercatorGrid::cell_center`]); `i32::MAX` where unreached.
75    /// Panics when an index is out of range.
76    pub fn minutes(&self, origin: usize, k: usize) -> &[i32] {
77        assert!(
78            origin < self.n_origins && k < self.percentiles.len(),
79            "({origin}, {k}) is out of range"
80        );
81        let n = self.grid.n_cells();
82        let at = (origin * self.percentiles.len() + k) * n;
83        &self.minutes[at..at + n]
84    }
85}

87/// The work of [`crate::street_travel_time_surface`].
88pub(crate) fn street_travel_time_surface(
89    net: &StreetNetwork,
90    origins: &[(f64, f64)],
91    mode: Mode,
92    params: &RouteParams,
93    percentiles: &[i32],
94    zoom: i32,
95) -> Result<TravelTimeSurface, RouteError> {
96    check_percentiles(percentiles).map_err(RouteError::Invalid)?;
97    params.validate()?;
98    no_scenario(&params.scenario)?;
99    let grid = WebMercatorGrid::for_street(net, zoom)?;
100    let mut percentiles = percentiles.to_vec();
101    percentiles.sort_unstable();
102    let np = percentiles.len();
103    let n = grid.n_cells();
104    let links = linkage::grid_links(net, &grid, mode, params.n_threads, &params.cancel)?;
105    let rows = travel_time_seconds_linked(net, origins, &links, mode, params, |_, times| {
106        let mut row = vec![UNREACHED; np * n];
107        if let Some(times) = times {
108            // TravelTimeReducer.recordUnvaryingTravelTimeAtTarget
109            for (j, &t) in times.iter().enumerate() {
110                let m = if t == UNREACHED { UNREACHED } else { t / 60 };
111                for k in 0..np {
112                    row[k * n + j] = m;
113                }
114            }
115        }
116        row
117    })?;
118    Ok(TravelTimeSurface {
119        grid,
120        percentiles,
121        n_origins: origins.len(),
122        minutes: rows.concat(),
123    })
124}

126/// Travel-time surfaces take no scenario: an error for one that changes
127/// anything.
128fn no_scenario(s: &crate::Scenario) -> Result<(), RouteError> {
129    if s.is_unchanged() {
130        Ok(())
131    } else {
132        Err(RouteError::Invalid(
133            "travel-time surfaces take no scenario: isochrone() has no new_carspeeds, \
134             carspeed_scale or new_lts"
135                .into(),
136        ))
137    }
138}

140/// The work of [`crate::transit_travel_time_surface`].
141pub(crate) fn transit_travel_time_surface(
142    net: &StreetNetwork,
143    layer: &TransitLayer,
144    data: &TransitData,
145    origins: &[(f64, f64)],
146    p: &TransitParams,
147    zoom: i32,
148) -> Result<TravelTimeSurface, RouteError> {
149    no_scenario(&p.scenario)?;
150    let grid = WebMercatorGrid::for_street(net, zoom)?;
151    let q = Query::new_grid(net, layer, data, &grid, p)?;
152    let mut percentiles = p.percentiles.clone();
153    percentiles.sort_unstable();
154    let np = percentiles.len();
155    let n = grid.n_cells();
156    let rows = par_origins(p.n_threads, origins.len(), &p.cancel, |i, scratch| {
157        let mut row = vec![UNREACHED; np * n];
158        q.route_origin(origins[i], scratch, |j, tt| {
159            // TravelTimeReducer.extractTravelTimePercentilesAndRecord
160            tt.sort_unstable();
161            let len = tt.len();
162            for (k, &pc) in percentiles.iter().enumerate() {
163                let v = tt[percentile_index(pc, len).min(len - 1)];
164                row[k * n + j] = if v == UNREACHED { UNREACHED } else { v / 60 };
165            }
166        })?;
167        Ok(row)
168    })?;
169    let rows = rows.into_iter().collect::<Result<Vec<_>, RouteError>>()?;
170    Ok(TravelTimeSurface {
171        grid,
172        percentiles,
173        n_origins: origins.len(),
174        minutes: rows.concat(),
175    })
176}

178/// The cutoffs of isochrones: distinct, ascending and without
179/// the 0. They must be finite and at least 0, and one must be
180/// above 0; [`RouteError::Invalid`] otherwise.
181pub fn isochrone_cutoffs(cutoffs: &[f64]) -> Result<Vec<f64>, RouteError> {
182    if cutoffs.is_empty() {
183        return Err(RouteError::Invalid(
184            "cutoffs must have at least one value".into(),
185        ));
186    }
187    if let Some(c) = cutoffs.iter().find(|c| !c.is_finite() || **c < 0.0) {
188        return Err(RouteError::Invalid(format!(
189            "cutoffs must be finite and at least 0, got {c}"
190        )));
191    }
192    let mut out: Vec<f64> = cutoffs.iter().copied().filter(|&c| c > 0.0).collect();
193    out.sort_by(f64::total_cmp);
194    out.dedup();
195    if out.is_empty() {
196        return Err(RouteError::Invalid(
197            "cutoffs must contain at least one value greater than 0".into(),
198        ));
199    }
200    Ok(out)
201}

203/// The trip limit of isochrones, derived from the cutoffs (not from a
204/// `max_trip_duration`): the largest cutoff, its minutes
205/// truncated. Line isochrones route with it; polygon isochrones with it
206/// plus 10. Errors as [`isochrone_cutoffs`], and when it does not fit an
207/// `i32`.
208pub fn isochrone_trip_limit(cutoffs: &[f64]) -> Result<i32, RouteError> {
209    let top = *isochrone_cutoffs(cutoffs)?
210        .last()
211        .expect("at least one cutoff");
212    if top >= 2_147_483_648.0 {
213        return Err(RouteError::Invalid(format!(
214            "the largest cutoff ({top}) is too large"
215        )));
216    }
217    Ok(top as i32)
218}

220/// One polygon isochrone: the area reached from origin `origin` within
221/// `isochrone` minutes at percentile `percentile`, as a multipolygon in
222/// `[lon, lat]` degrees (empty when nothing is reached).
223#[derive(Debug, Clone, PartialEq)]
224#[non_exhaustive]
225pub struct IsochronePolygon {
226    /// Index of the origin.
227    pub origin: usize,
228    /// The travel-time percentile.
229    pub percentile: i32,
230    /// The cutoff, minutes.
231    pub isochrone: f64,
232    /// The polygons, each a shell followed by its holes, every ring closed;
233    /// as isoband's `separate_polygons()` makes them (shells as traced,
234    /// holes reversed), before any repair (`sf::st_make_valid()`).
235    pub polygons: Vec<Polygon>,
236}

238/// The longitude of a Web Mercator pixel coordinate.
239fn pixel_to_lon(px: f64, zoom: i32) -> f64 {
240    px / 256.0 / 2f64.powi(zoom) * 360.0 - 180.0
241}
242
243/// The latitude of a Web Mercator pixel coordinate.
244fn pixel_to_lat(px: f64, zoom: i32) -> f64 {
245    use std::f64::consts::PI;
246    (PI - px / 256.0 / 2f64.powi(zoom) * 2.0 * PI).sinh().atan() * 180.0 / PI
247}

249/// Polygon isochrones from a travel-time surface: for each origin, each percentile and
250/// each cutoff, the cells reached in less than the cutoff, traced by
251/// isoband's marching squares over the cell centres (`isoband::isobands()`
252/// from 0 to the cutoff), converted to longitude and latitude and grouped
253/// into polygons with holes (`isoband::iso_to_sfg()`).
254///
255/// `cutoffs` are checked by [`isochrone_cutoffs`]. Every (origin,
256/// percentile, cutoff) gets one result, ordered by origin, percentile
257/// (ascending) and cutoff (descending).
258/// [`RouteError::Invalid`] for bad cutoffs, and where isoband stops: two
259/// rings whose relation it cannot decide.
260///
261/// The order of the rings in each band and the vertex each ring starts at
262/// are those isoband gives where this crate runs: isoband walks a C++ hash
263/// table, whose order is that of the C++ standard library it was built
264/// with, libc++ on macOS and libstdc++ on Linux and Windows. This function
265/// follows the target it is compiled for (libc++ on Apple targets and on
266/// WebAssembly, libstdc++ elsewhere), so on each platform the polygons are
267/// those the R package isoband makes there.
268pub fn isochrone_polygons(
269    surface: &TravelTimeSurface,
270    cutoffs: &[f64],
271) -> Result<Vec<IsochronePolygon>, RouteError> {
272    isochrone_polygons_in(surface, cutoffs, CxxLibrary::NATIVE)
273}
274
275/// [`isochrone_polygons`] with the rings in the order isoband gives them
276/// when built with `lib` (exposed through `crate::unstable`).
277pub(crate) fn isochrone_polygons_in(
278    surface: &TravelTimeSurface,
279    cutoffs: &[f64],
280    lib: CxxLibrary,
281) -> Result<Vec<IsochronePolygon>, RouteError> {
282    let cutoffs = isochrone_cutoffs(cutoffs)?;
283    let g = surface.grid;
284    // cell centres: west + 1..=width - 0.5, north + 1..=height - 0.5
285    let x: Vec<f64> = (1..=g.width())
286        .map(|i| f64::from(g.west() + i) - 0.5)
287        .collect();
288    let y: Vec<f64> = (1..=g.height())
289        .map(|i| f64::from(g.north() + i) - 0.5)
290        .collect();
291    let mut out = Vec::new();
292    let mut z = Vec::with_capacity(g.n_cells());
293    for origin in 0..surface.n_origins {
294        for (k, &percentile) in surface.percentiles.iter().enumerate() {
295            z.clear();
296            z.extend(surface.minutes(origin, k).iter().map(|&m| f64::from(m)));
297            let mut bander = isoband::Isobander::new(
298                isoband::Grid {
299                    x: &x,
300                    y: &y,
301                    z: &z,
302                },
303                lib,
304            )
305            .map_err(RouteError::Invalid)?;
306            // isoband computes the bands in the order given (ascending); the
307            // rows go out descending
308            let mut bands = Vec::with_capacity(cutoffs.len());
309            for &c in &cutoffs {
310                let rings: Vec<isoband::Ring> = bander
311                    .band(c)
312                    .into_iter()
313                    .map(|ring| {
314                        ring.into_iter()
315                            .map(|[px, py]| {
316                                [pixel_to_lon(px, g.zoom()), pixel_to_lat(py, g.zoom())]
317                            })
318                            .collect()
319                    })
320                    .collect();
321                let polygons = separate::separate_polygons(rings).map_err(RouteError::Invalid)?;
322                bands.push(IsochronePolygon {
323                    origin,
324                    percentile,
325                    isochrone: c,
326                    polygons,
327                });
328            }
329            bands.reverse();
330            out.extend(bands);
331        }
332    }
333    Ok(out)
334}

340    #[test]
341    fn cutoffs_are_checked() {
342        assert_eq!(
343            isochrone_cutoffs(&[30.0, 0.0, 15.0, 15.0]).unwrap(),
344            vec![15.0, 30.0]
345        );
346        assert_eq!(isochrone_cutoffs(&[7.5]).unwrap(), vec![7.5]);
347        assert!(isochrone_cutoffs(&[]).is_err());
348        assert!(isochrone_cutoffs(&[0.0]).is_err());
349        assert!(isochrone_cutoffs(&[-1.0, 10.0]).is_err());
350        assert!(isochrone_cutoffs(&[f64::NAN]).is_err());
351        assert!(isochrone_cutoffs(&[f64::INFINITY]).is_err());
352        assert_eq!(isochrone_trip_limit(&[0.0, 15.0, 30.9]).unwrap(), 30);
353    }
354
355    #[test]
356    fn pixels_to_degrees() {
357        // the corners of the world at zoom 9
358        let w = 256.0 * 512.0;
359        assert_eq!(pixel_to_lon(0.0, 9), -180.0);
360        assert_eq!(pixel_to_lon(w / 2.0, 9), 0.0);
361        assert!((pixel_to_lat(w / 2.0, 9)).abs() < 1e-12);
362        assert!((pixel_to_lat(0.0, 9) - 85.0511287798066).abs() < 1e-9);
363    }
