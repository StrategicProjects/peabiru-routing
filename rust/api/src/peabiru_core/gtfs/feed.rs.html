11#[derive(Debug)]
12pub enum GtfsError {
13    Zip(ZipError),
14    Csv {
15        file: &'static str,
16        error: CsvError,
17    },
18    /// Why gtfs-lib fails loading the feed (see `validate`).
19    GtfsLibFails(String),
20    /// Why R5 fails building its transit layer from a feed that gtfs-lib
21    /// loads.
22    R5Fails(String),
23    /// Feeds with errors of priority `HIGH`, and how many each has.
24    HighPriority(Vec<(std::path::PathBuf, usize)>),
25    /// A feed peabiru cannot read that gtfs-lib loads with errors of lower
26    /// priority only, so that R5 builds a network from it.
27    NotRead(Box<GtfsError>),
28    /// An error in the feed read from `path`.
29    InFeed {
30        path: std::path::PathBuf,
31        error: Box<GtfsError>,
32    },
33}
34
35impl fmt::Display for GtfsError {
36    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
37        match self {
38            GtfsError::Zip(e) => write!(f, "{e}"),
39            GtfsError::Csv { file, error } => write!(f, "{file} {error}"),
40            GtfsError::GtfsLibFails(why) | GtfsError::R5Fails(why) => f.write_str(why),
41            GtfsError::NotRead(e) => write!(
42                f,
43                "{e} (the GTFS errors of this feed are not of priority HIGH, but peabiru cannot \
44                 read it)"
45            ),
46            GtfsError::HighPriority(feeds) => {
47                let list: Vec<String> = feeds
48                    .iter()
49                    .map(|(p, n)| {
50                        format!(
51                            "{} ({n} error{})",
52                            p.display(),
53                            if *n == 1 { "" } else { "s" }
54                        )
55                    })
56                    .collect();
57                write!(
58                    f,
59                    "high priority GTFS errors found in {}; network build failed",
60                    list.join(", ")
61                )
62            }
63            GtfsError::InFeed { path, error } => write!(f, "{}: {error}", path.display()),
64        }
65    }
66}

68impl std::error::Error for GtfsError {}

70impl GtfsError {
71    /// `error`, in the feed read from `path`.
72    pub(crate) fn in_feed(path: &std::path::Path, error: GtfsError) -> Self {
73        GtfsError::InFeed {
74            path: path.to_path_buf(),
75            error: Box::new(error),
76        }
77    }

205/// gtfs-lib's `getTimeField` on a value that is not blank: `H:M:S` split
206/// as Java's `String.split(":")` splits (trailing empty parts dropped), each
207/// part read by `Integer.parseInt`, and `h * 3600 + m * 60 + s` in Java's
208/// wrapping `int` arithmetic; `None` where gtfs-lib records a
209/// `TimeParseError` and keeps `INT_MISSING`. Out-of-range parts (`25:61:00`)
210/// are kept, as gtfs-lib keeps them (with a `RangeError`).
211pub(crate) fn gtfs_time(s: &str) -> Option<i32> {
212    let mut parts: Vec<&str> = s.split(':').collect();
213    while parts.last() == Some(&"") {
214        parts.pop();
215    }
216    let [h, m, sec] = parts[..] else {
217        return None;
218    };
219    let (h, m, sec) = (parse_int(h)?, parse_int(m)?, parse_int(sec)?);
220    let v = h
221        .wrapping_mul(3600)
222        .wrapping_add(m.wrapping_mul(60))
223        .wrapping_add(sec);
224    // a value that wraps to INT_MISSING is read by R5 as missing
225    (v != INT_MISSING).then_some(v)
226}

228/// The value getters of gtfs-lib's `Entity.Loader` for one record: what a
229/// blank value, a value that does not parse and a missing column (read as
230/// blank by javacsv) become. The errors gtfs-lib records are listed by
231/// `validate`; here only the values matter.
232struct Fields<'a, 'r> {
233    h: &'a Header,
234    r: &'a Row<'r>,
235}
236
237impl<'a> Fields<'a, '_> {
238    fn str(&self, col: &str) -> &'a str {
239        self.r.get(self.h.col(col))
240    }
241
242    fn opt(&self, col: &str) -> Option<&'a str> {
243        Some(self.str(col)).filter(|s| !s.is_empty())
244    }
245
246    fn string(&self, col: &str) -> String {
247        self.str(col).to_string()
248    }
249
250    /// `getIntField(col, _, _, _, default)`.
251    fn int(&self, col: &str, default: i32) -> i32 {
252        self.opt(col)
253            .map_or(default, |s| parse_int(s).unwrap_or(INT_MISSING))
254    }
255
256    /// `getDoubleField`.
257    fn double(&self, col: &str) -> f64 {
258        self.opt(col).and_then(parse_double).unwrap_or(f64::NAN)
259    }
260
261    /// `getTimeField`, `None` for `INT_MISSING`.
262    fn time(&self, col: &str) -> Option<i32> {
263        self.opt(col).and_then(gtfs_time)
264    }
265}

267impl Feed {
268    /// Read a feed from the bytes of a GTFS zip file, with bounds on how
269    /// far the archive may expand, also saying how the archive is laid out
270    /// (for the warnings about feeds with ignored errors). Errors name the
271    /// file, `path`.
272    pub(crate) fn read_zip_checked(
273        path: &std::path::Path,
274        bytes: std::borrow::Cow<'_, [u8]>,
275        limits: Limits,
276    ) -> Result<(Feed, ArchiveLayout), GtfsError> {
277        Archive::with_limits(bytes, limits)
278            .map_err(GtfsError::from)
279            .and_then(|a| Ok((Self::from_archive(&a)?, ArchiveLayout::of(&a))))
280            .map_err(|error| GtfsError::in_feed(path, error))
281    }

283    /// The tables of a feed, every value read as gtfs-lib's loaders read it
284    /// (see [`Fields`]): a blank or unreadable value takes gtfs-lib's
285    /// default rather than failing, so that every feed gtfs-lib loads is
286    /// read. Records gtfs-lib does not keep (no key) are left out. Fails
287    /// only on a missing required table or a file javacsv refuses.
288    pub fn from_archive(a: &Archive) -> Result<Feed, GtfsError> {
289        let mut feed = Feed::default();
290
291        table(a, "agency.txt", |f| {
292            feed.agencies
293                .push((f.string("agency_id"), f.string("agency_name")));
294        })?;
295
296        let mut first_info = true;
297        table(a, "feed_info.txt", |f| {
298            // FeedInfo.Loader: the first record sets the feed id
299            if std::mem::take(&mut first_info) {
300                feed.feed_id = f.opt("feed_id").map(str::to_string);
301            }
302        })?;
303
304        table(a, "stops.txt", |f| {
305            if f.opt("stop_id").is_none() {
306                return; // MissingKeyError: not kept
307            }
308            feed.stops.push(Stop {
309                stop_id: f.string("stop_id"),
310                stop_name: f.string("stop_name"),
311                lat: f.double("stop_lat"),
312                lon: f.double("stop_lon"),
313                location_type: f.int("location_type", 0),
314                parent_station: f.string("parent_station"),
315            });
316        })?;
317
318        table(a, "routes.txt", |f| {
319            if f.opt("route_id").is_none() {
320                return;
321            }
322            feed.routes.push(Route {
323                route_id: f.string("route_id"),
324                route_type: f.int("route_type", 0),
325                agency_id: f.string("agency_id"),
326                short_name: f.string("route_short_name"),
327                long_name: f.string("route_long_name"),
328            });
329        })?;
330
331        table(a, "trips.txt", |f| {
332            if f.opt("trip_id").is_none() {
333                return;
334            }
335            feed.trips.push(Trip {
336                trip_id: f.string("trip_id"),
337                route_id: f.opt("route_id").map(str::to_string),
338                service_id: f.opt("service_id").map(str::to_string),
339                shape_id: f.string("shape_id"),
340            });
341        })?;
342
343        // StopTime.Loader.detectRowIsFlex: a table with a GTFS-Flex column,
344        // a row with a value in one
345        let mut table_has_flex = None;
346        table(a, "stop_times.txt", |f| {
347            if f.opt("trip_id").is_none() {
348                return; // belongs to no trip
349            }
350            let has_flex = *table_has_flex.get_or_insert_with(|| {
351                super::validate::FLEX_FIELDS
352                    .iter()
353                    .any(|c| f.h.col(c).is_some())
354            });
355            if has_flex
356                && super::validate::FLEX_FIELDS
357                    .iter()
358                    .any(|c| f.opt(c).is_some())
359            {
360                feed.flex_trips.insert(f.string("trip_id"));
361            }
362            feed.stop_times.push(StopTime {
363                trip_id: f.string("trip_id"),
364                arrival: f.time("arrival_time"),
365                departure: f.time("departure_time"),
366                stop_id: f.string("stop_id"),
367                stop_sequence: f.int("stop_sequence", 0),
368                pickup_type: f.int("pickup_type", 0),
369                drop_off_type: f.int("drop_off_type", 0),
370                shape_dist: f.double("shape_dist_traveled"),
371            });
372        })?;
373
374        const DAYS: [&str; 7] = [
375            "monday",
376            "tuesday",
377            "wednesday",
378            "thursday",
379            "friday",
380            "saturday",
381            "sunday",
382        ];
383        table(a, "calendar.txt", |f| {
384            let mut days = [false; 7];
385            for (d, name) in days.iter_mut().zip(DAYS) {
386                *d = f.int(name, 0) == 1;
387            }
388            feed.calendars.push(Calendar {
389                service_id: f.string("service_id"),
390                days,
391                start_date: f.int("start_date", 0),
392                end_date: f.int("end_date", 0),
393            });
394        })?;
395
396        table(a, "calendar_dates.txt", |f| {
397            feed.calendar_dates.push(CalendarDate {
398                service_id: f.string("service_id"),
399                date: f.opt("date").and_then(super::validate::basic_iso_date),
400                exception_type: f.int("exception_type", 0),
401            });
402        })?;
403
404        table(a, "frequencies.txt", |f| {
405            feed.frequencies.push(Frequency {
406                trip_id: f.string("trip_id"),
407                start_time: f.time("start_time").unwrap_or(INT_MISSING),
408                end_time: f.time("end_time").unwrap_or(INT_MISSING),
409                headway_secs: f.int("headway_secs", 0),
410                exact_times: f.int("exact_times", 0) == 1,
411            });
412        })?;
413
414        // Shapes are only drawn (transit_network_to_sf()), never
415        // routed on.
416        table(a, "shapes.txt", |f| {
417            let Some(id) = f.opt("shape_id") else {
418                return;
419            };
420            let seq = f.int("shape_pt_sequence", 0);
421            let point = (f.double("shape_pt_lon"), f.double("shape_pt_lat"));
422            let dist = f.double("shape_dist_traveled");
423            feed.shape_dists
424                .entry(id.to_string())
425                .or_default()
426                .insert(seq, dist);
427            match feed.shapes.get_mut(id) {
428                Some(s) => {
429                    s.insert(seq, point);
430                }
431                None => {
432                    feed.shapes
433                        .insert(id.to_string(), BTreeMap::from([(seq, point)]));
434                }
435            }
436        })?;
437        Ok(feed)
438    }
439}

456/// How a feed's archive is laid out, in the two ways that make gtfs-lib
457/// report an error of priority `HIGH` in a feed that peabiru can read.
458#[derive(Debug, Clone, Default, PartialEq, Eq)]
459pub(crate) struct ArchiveLayout {
460    /// Entries of the tables found in a subfolder rather than at the root
461    /// (gtfs-lib's `TableInSubdirectoryError`), in [`TABLES`] order.
462    pub tables_in_subfolder: Vec<String>,
463    /// No `agency.txt` anywhere (gtfs-lib's `NoAgencyInFeedError`).
464    pub no_agency: bool,
465}
466
467impl ArchiveLayout {
468    fn of(a: &Archive) -> Self {
469        ArchiveLayout {
470            tables_in_subfolder: TABLES
471                .iter()
472                .filter_map(|&t| a.find(t).filter(|e| e.name != t))
473                .map(|e| e.name.clone())
474                .collect(),
475            no_agency: a.find("agency.txt").is_none(),
476        }
477    }
478}

480/// Stream the records of one file of the archive into `f`. A missing table
481/// has no records, as for gtfs-lib (which records a `MissingTableError` of
482/// priority `MEDIUM` for a required one and loads the feed).
483fn table(a: &Archive, name: &'static str, mut f: impl FnMut(&Fields)) -> Result<(), GtfsError> {
484    let Some(e) = a.find(name) else {
485        return Ok(());
486    };
487    match for_each_row::<()>(&a.read(e)?, |h, _, r| {
488        f(&Fields { h, r });
489        Ok(())
490    }) {
491        Ok(_) => Ok(()),
492        Err(ReadError::Csv(error)) => Err(GtfsError::Csv { file: name, error }),
493        Err(ReadError::Callback(())) => unreachable!("the callback does not fail"),
494    }
495}

502    #[test]
503    fn times_as_gtfs_lib() {
504        assert_eq!(gtfs_time("08:05:09"), Some(8 * 3600 + 5 * 60 + 9));
505        assert_eq!(gtfs_time("25:30:00"), Some(25 * 3600 + 1800));
506        // Integer.parseInt takes signs; parts out of range are kept
507        assert_eq!(gtfs_time("+8:00:00"), Some(8 * 3600));
508        assert_eq!(gtfs_time("-1:00:00"), Some(-3600));
509        assert_eq!(gtfs_time("8:61:00"), Some(8 * 3600 + 61 * 60));
510        // String.split drops trailing empty parts
511        assert_eq!(gtfs_time("8:00:00:"), Some(8 * 3600));
512        assert_eq!(gtfs_time("8:00:00::"), Some(8 * 3600));
513        for bad in [
514            "8:00",
515            "x:00:00",
516            " 8:00:00",
517            "8::00:00",
518            ":8:00:00",
519            "8:00:00:1",
520            ":::",
521        ] {
522            assert_eq!(gtfs_time(bad), None, "{bad}");
523        }
524        // Java int arithmetic wraps
525        assert_eq!(
526            gtfs_time("596524:00:00"),
527            Some(596_524i32.wrapping_mul(3600))
528        );
529    }

593    fn feed_with(tables: &[(&str, &[u8])]) -> Result<Feed, GtfsError> {
594        let mut files: Vec<(&str, &[u8])> = vec![
595            ("stops.txt", b"stop_id,stop_lat,stop_lon\nA,1,1\n"),
596            ("routes.txt", b"route_id,route_type\nR,3\n"),
597            ("trips.txt", b"route_id,service_id,trip_id\nR,S,T1\n"),
598            (
599                "stop_times.txt",
600                b"trip_id,arrival_time,departure_time,stop_id,stop_sequence\nT1,08:00:00,08:00:00,A,1\n",
601            ),
602        ];
603        for t in tables {
604            files.retain(|f| f.0 != t.0);
605            files.push(*t);
606        }
607        Feed::from_archive(&Archive::from_bytes(make_zip(&files, false)).unwrap())
608    }

610    /// Values gtfs-lib cannot read take its defaults instead of failing:
611    /// a blank `route_type` or `stop_sequence` is 0, a blank calendar date
612    /// 0, a time it cannot parse missing; records without a key are not
613    /// kept.
614    #[test]
615    fn unreadable_values_take_gtfs_lib_defaults() {
616        let f = feed_with(&[
617            ("routes.txt", b"route_id\nR\n\n"),
618            (
619                "stop_times.txt",
620                b"trip_id,arrival_time,departure_time,stop_id\nT1,14:00,x,A\n,08:00:00,08:00:00,A\n",
621            ),
622            ("calendar.txt", b"service_id,monday,tuesday,start_date,end_date\nS,01,+1,,x\n"),
623            ("calendar_dates.txt", b"service_id,date,exception_type\nS,,\nS,20190513Z,+1\n"),
624            ("frequencies.txt", b"trip_id,start_time,end_time\nT1,6:00,07:00:00\n"),
625            ("stops.txt", b"stop_id,stop_lat,stop_lon,location_type\nA,x,1,\n,1,1,0\n"),
626        ])
627        .unwrap();
628        assert_eq!(f.routes[0].route_type, 0);
629        assert_eq!(f.stop_times.len(), 1);
630        let st = &f.stop_times[0];
631        assert_eq!(
632            (st.arrival, st.departure, st.stop_sequence),
633            (None, None, 0)
634        );
635        let c = &f.calendars[0];
636        assert_eq!(&c.days[..3], &[true, true, false]);
637        assert_eq!((c.start_date, c.end_date), (0, INT_MISSING));
638        assert_eq!(f.calendar_dates[0].date, None);
639        assert_eq!(f.calendar_dates[0].exception_type, 0);
640        assert_eq!(
641            (f.calendar_dates[1].date, f.calendar_dates[1].exception_type),
642            (Some(20190513), 1)
643        );
644        let q = &f.frequencies[0];
645        assert_eq!(
646            (q.start_time, q.end_time, q.headway_secs),
647            (INT_MISSING, 7 * 3600, 0)
648        );
649        assert_eq!(f.stops.len(), 1);
650        assert!(f.stops[0].lat.is_nan());
651        assert_eq!(f.stops[0].location_type, 0);
652    }
